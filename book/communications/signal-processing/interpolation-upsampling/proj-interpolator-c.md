# ⚙️ Реалізація інтерполятора сигналів на C та C++

Реалізація цифрового інтерполятора (англ. upsampler) у реальному часі вимагає ефективного перетворення вхідного потоку відліків із початковою частотою дискретизації `f_s` у вихідний потік із підвищеною частотою `f_s' = L · f_s`.

Пряма наївна реалізація полягає у фізичному вставленні `L - 1` нульових відліків у масив та подальшому проганянні всієї послідовності через стандартну функцію фільтрації зі скінченною імпульсною характеристикою (КІХ-фільтром). Проте така схема є катастрофічно неефективною: переважна більшість усіх виконаних операцій множення припадають на чисті нулі. Для коефіцієнта інтерполяції `L = 8` обчислювач витрачає 87.5% процесорного часу на марне множення коефіцієнтів на `0.0`.

Промисловим стандартом є поліфазна структура інтерполятора (Polyphase Interpolator). Її математичний принцип полягає у розбитті початкового масиву коефіцієнтів фільтра довжиною `K` на `L` окремих субфільтрів (поліфазних філій) `e_p[n] = h[n · L + p]`, де `p ∈ [0, L - 1]`. Завдяки такій декомпозиції всі обчислення множення та додавання виконуються виключно на низькій вхідній частоті дискретизації `f_s` із реальними значущими відліками сигналу. Жоден нульовий відлік не потрапляє у лінію затримки, а вихідний комутатор лише по черзі опитує готові результати з `L` фаз на високій вихідній частоті `L · f_s`.

Нижче наведено вичерпні робочі реалізації поліфазного інтерполятора для обробки потоків у форматі з плаваючою крапкою (`float`) та фіксованою точкою (`int16_t` / Q15).

```c
/* C: Поліфазний інтерполятор сигналів для форматів float та Q15 */
#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>
#include <string.h>

/* Структура поліфазного інтерполятора float */
typedef struct {
    size_t factor_L;      /* Коефіцієнт інтерполяції */
    size_t taps_per_phase;/* Довжина кожної поліфазної філії */
    float **phase_taps;   /* Таблиця коефіцієнтів [L][taps_per_phase] */
    float *state;         /* Кільцевий буфер вхідних відліків */
    size_t state_head;    /* Вказівник голови буфера */
} polyphase_interp_f32_t;

polyphase_interp_f32_t* polyphase_interp_f32_create(
    size_t factor_L,
    const float *taps,
    size_t total_taps
) {
    if (factor_L == 0 || total_taps == 0) return NULL;

    polyphase_interp_f32_t *it = (polyphase_interp_f32_t*)malloc(sizeof(polyphase_interp_f32_t));
    if (!it) return NULL;

    it->factor_L = factor_L;
    it->taps_per_phase = (total_taps + factor_L - 1) / factor_L;

    it->phase_taps = (float**)malloc(factor_L * sizeof(float*));
    for (size_t p = 0; p < factor_L; ++p) {
        it->phase_taps[p] = (float*)calloc(it->taps_per_phase, sizeof(float));
    }

    /* Декомпозиція коефіцієнтів: e_p[n] = h[n*L + p] */
    for (size_t i = 0; i < total_taps; ++i) {
        size_t p = i % factor_L;
        size_t n = i / factor_L;
        it->phase_taps[p][n] = taps[i];
    }

    it->state = (float*)calloc(it->taps_per_phase, sizeof(float));
    it->state_head = 0;

    return it;
}

void polyphase_interp_f32_destroy(polyphase_interp_f32_t *it) {
    if (!it) return;
    for (size_t p = 0; p < it->factor_L; ++p) {
        free(it->phase_taps[p]);
    }
    free(it->phase_taps);
    free(it->state);
    free(it);
}

/* Приймає 1 вхідний відлік, ґенерує L вихідних відліків у out_samples */
void polyphase_interp_f32_process_sample(
    polyphase_interp_f32_t *it,
    float in_sample,
    float *out_samples
) {
    /* Додаємо новий відлік у зсувний буфер на низькій частоті fs */
    it->state[it->state_head] = in_sample;

    for (size_t p = 0; p < it->factor_L; ++p) {
        float acc = 0.0f;
        size_t taps_count = it->taps_per_phase;
        const float *phase_h = it->phase_taps[p];

        for (size_t i = 0; i < taps_count; ++i) {
            size_t idx = (it->state_head + taps_count - i) % taps_count;
            acc += phase_h[i] * it->state[idx];
        }
        out_samples[p] = acc;
    }

    it->state_head = (it->state_head + 1) % it->taps_per_phase;
}
```
```cpp
// C++20: Поліфазний інтерполятор із шаблонізацією типу відліків та RAII
#include <vector>
#include <span>
#include <cstdint>
#include <concepts>
#include <stdexcept>
#include <algorithm>

template <typename T = float>
class PolyphaseInterpolator {
public:
    PolyphaseInterpolator(size_t factor_L, std::span<const T> filter_taps)
        : factor_L_(factor_L) {
        if (factor_L_ == 0 || filter_taps.empty()) {
            throw std::invalid_argument("Некоректні параметри інтерполятора");
        }

        taps_per_phase_ = (filter_taps.size() + factor_L_ - 1) / factor_L_;
        phase_taps_.resize(factor_L_, std::vector<T>(taps_per_phase_, T{0}));

        // Декомпозиція коефіцієнтів h[n*L + p]
        for (size_t i = 0; i < filter_taps.size(); ++i) {
            size_t p = i % factor_L_;
            size_t n = i / factor_L_;
            phase_taps_[p][n] = filter_taps[i];
        }

        state_buffer_.resize(taps_per_phase_, T{0});
    }

    // Обробка одного відліку -> повертає std::vector із L інтерпольованих відліків
    std::vector<T> process_sample(T input_sample) {
        state_buffer_[state_head_] = input_sample;
        std::vector<T> output(factor_L_);

        for (size_t p = 0; p < factor_L_; ++p) {
            T acc{0};
            const auto& phase_h = phase_taps_[p];

            for (size_t i = 0; i < taps_per_phase_; ++i) {
                size_t idx = (state_head_ + taps_per_phase_ - i) % taps_per_phase_;
                acc += phase_h[i] * state_buffer_[idx];
            }
            output[p] = acc;
        }

        state_head_ = (state_head_ + 1) % taps_per_phase_;
        return output;
    }

    // Заповнення наданого буфера без додаткових виділень пам'яті
    void process_sample(T input_sample, std::span<T> output_span) {
        if (output_span.size() < factor_L_) {
            throw std::out_of_range("Вихідний буфер замалий для фаз L");
        }
        state_buffer_[state_head_] = input_sample;

        for (size_t p = 0; p < factor_L_; ++p) {
            T acc{0};
            const auto& phase_h = phase_taps_[p];

            for (size_t i = 0; i < taps_per_phase_; ++i) {
                size_t idx = (state_head_ + taps_per_phase_ - i) % taps_per_phase_;
                acc += phase_h[i] * state_buffer_[idx];
            }
            output_span[p] = acc;
        }

        state_head_ = (state_head_ + 1) % taps_per_phase_;
    }

    [[nodiscard]] size_t factor() const noexcept { return factor_L_; }

private:
    size_t factor_L_{1};
    size_t taps_per_phase_{0};
    size_t state_head_{0};
    std::vector<std::vector<T>> phase_taps_;
    std::vector<T> state_buffer_;
};
```

## Покроковий розбір C-реалізації поліфазного інтерполятора

Розглянемо детально внутрішній інженерний механізм роботи мовою C для поліфазного інтерполятора.

### 1. Ініціалізація та розподіл пам'яті для коефіцієнтів

Функція створення `polyphase_interp_f32_create` обчислює довжину кожної поліфазної філії шляхом ділення загальної кількості відводів початкового фільтра на коефіцієнт інтерполяції з округленням у більшу сторону: `taps_per_phase = (total_taps + factor_L - 1) / factor_L`. Округлення вгору гарантує, що навіть якщо загальна кількість відводів початкового фільтра не ділиться націло на коефіцієнт розширення `factor_L`, кожен субфільтр дістане достатній розмір масиву, а недостатні значення будуть автоматично заповнені нулями під час ініціалізації.

Далі створюється двовимірний динамічний масив коефіцієнтів `phase_taps`, у якому рядки відповідають поліфазним фазам от `0` до `factor_L - 1`, а стовпчики — коефіцієнтам відповідного субфільтра. Декомпозиція виконується за математичним правилом вибору початкового масиву, де номер фази задається остачею від ділення індексу `i % factor_L`, а позиція всередині фази — цілочисловим результатом ділення `i / factor_L`.

Така структура виділення пам'яті гарантує, що кожна поліфазна філія зберігається неперервним блоком у пам'яті. Це є фундаментальною умовою для забезпечення високого рівня кеш-локальності процесора під час обчислення скалярних добутків у реальному часі. При зчитуванні коефіцієнтів фази процесор завантажує цілу кеш-лінію розміром 64 байти, що повністю виключає затримки на промахи кеш-пам'яті першого рівня L1.

### 2. Управління буфером стану та обчислення фаз

Функція обробки відліків `polyphase_interp_f32_process_sample` приймає один вхідний відлік `in_sample`, який надходить на низькій вхідній частоті дискретизації `f_s`. Цей відлік записується безпосередньо в поточну позицію кільцевого буфера стану `state[state_head]`.

Зверніть увагу: лінія затримки `state` містить виключно значущі вхідні відліки сигналу. У ній немає жодного нульового значення. Це означає, що розмір оперативної пам'яті для лінії затримки зменшується у `L` разів порівняно з наївним алгоритмом.

Далі запускається зовнішній цикл за номером фази `p` від `0` до `factor_L - 1`. Для кожної фази виконується скалярний добуток коефіцієнтів відповідного субфільтра `phase_h` на значення з кільцевого буфера стану `state`. Індекс вхідного відліку обчислюється за правилом кільцевого обходу від найновішого до найстарішого:

```
idx = (state_head + taps_count - i) % taps_count
```

Ця формула забезпечує зворотний обхід буфера від найновішого відліку до найстарішого. Знайдений добуток накопичується у локальній змінній акумулятора `acc`, після чого результатом обчислення є один вихідний відлік відповідної фази `out_samples[p]`. Після завершення циклу по всіх `L` фазах у вихідному масиві формується послідовність інтерпольованих відліків, які утворюють новий потік на підвищеній частоті дискретизації `L · f_s`.

Наприкінці функції вказівник голови буфера `state_head` просувається на одну позицію вперед із урахуванням кільцевого модуля: `state_head = (state_head + 1) % taps_per_phase`.

## Покроковий розбір C++20 реалізації

C++ реалізація шаблонованого класу `PolyphaseInterpolator<T>` забезпечує безпечність ресурсів за принципом автоматичного управління ресурсами RAII та надає сучасний ідіоматичний інтерфейс для високопродуктивних обчислень.

### 1. Шаблонізація та гнучкість типів даних

Використання шаблонів `template <typename T = float>` дозволяє використовувати один і той самий код інтерполятора для обробки різних типів даних:
- Одинарна `float` та подвійна точність `double` плаваючої крапки для звукових процесорів та математичних моделювальних систем.
- Комплексні числа `std::complex<float>` для обробки квадратурних IQ-сигналів у приймачах та передавачах програмно-визначеного радіо SDR.
- Цілочислові типи `int16_t` / `int32_t` для вбудованих систем із фіксованою точкою.

Завдяки сучасним концептам C++20 та перевантаженню методів клас підтримує роботу як із динамічним виділенням вихідних масивів у методі `process_sample(T input_sample)`, так і із заздалегідь виділеними буферами пам'яті через спеціальні перегляди контейнерів `std::span<T>`.

### 2. Уникнення зайвого копіювання пам'яті через std::span

Конструктор класу та метод обробки приймають параметри через безпечні перегляди пам'яті `std::span<const T>` та `std::span<T>`. Це дозволяє передавати коефіцієнти фільтра та вихідні буфери з будь-яких контейнерів — динамічних векторів `std::vector`, фіксованих масивів `std::array` або звичайних вказівників — без додаткового виділення динамічної пам'яті та без копіювання елементів.

Для систем реального часу надано перевантажений метод `process_sample(T input_sample, std::span<T> output_span)`, який записує обчислені фази безпосередньо у наданий вихідний буфер, гарантуючи нульову кількість виділень пам'яті у купі під час обчислювального циклу. Це усуває непередбачувані затримки (jitter), пов'язані з динамічним розподілом пам'яті у потоках обробки аудіо чи радіосигналів.

## Розрахунок довжини фільтра та вибір параметрів у проєкті

При практичному розрахунку інтерполяційного КІХ-фільтра довжину коефіцієнтів `K` обчислюють за наближеною інженерною формулою Белланжера, яка пов'язує необхідне придушення у смузі загородження `A`, ширину перехідної смуги `Δf` та підвищену частоту дискретизації `f_s'`:

```
K ≈ (A / (22 · Δf)) · f_s'
```

Необхідне придушення дзеркальних образів `A` у смузі загородження обирають залежно від вимог системи: 80 децибел для високоякісного аудіо або понад 100 децибел для супутникового радіозв'язку. Ширина перехідної смуги `Δf` виражається різницею між початковою частотою загородження та частотою пропускання: `Δf = f_{stop} - f_{pass}`.

Наприклад, якщо необхідно підвищити частоту аудіосигналу з `44.1` кГц до `176.4` кГц у 4 рази (`L = 4`) із межею смуги пропускання `20` кГц, межею смуги загородження `24.1` кГц (`Δf = 4.1` кГц) та придушенням `80` дБ, загальна довжина фільтра становитиме:

```
K ≈ (80 / (22 · 4100)) · 176400 ≈ 156 відводів
```

При поліфазному розкладанні на `L = 4` фази довжина кожної поліфазної філії становитиме всього `taps_per_phase = 156 / 4 = 39` відводів.

## Аналіз кеш-локальності та розкладу коефіцієнтів у пам'яті

При розробці високопродуктивного поліфазного інтерполятора необхідно враховувати особливості розташування даних у кеш-пам'яті процесора.

### Розклад коефіцієнтів у пам'яті: Row-Major проти Column-Major

У наведених програмах коефіцієнти розкладаються за принципом неперервних блоків для кожної фази: `phase_taps[p][n] = h[n · L + p]`. Це означає, що коефіцієнти однієї фази `p` розміщуються у пам'яті послідовним масивом (Row-Major format).

Такий підхід є оптимальним для послідовного обчислення вихідних відліків однією фазою за іншою:
1. Для фази `0` процесор завантажує вектор коефіцієнтів `phase_taps[0]` у швидку кеш-пам'ять L1 і виконує скалярний добуток із вхідним буфером стану `state`.
2. Далі завантажуються коефіцієнти наступної фази `phase_taps[1]` і обчислюється наступний відлік.

У векторних процесорах із підтримкою спеціальних векторизованих інструкцій (AVX2, AVX-512, ARM Neon) це дозволяє виконувати паралельне множення та додавання (FMA — Fused Multiply-Add) над 8 або 16 відліками за один такт процесора. Якщо вирівняти початкові адреси масивів у пам'яті за кордоном 32 байти (32-byte alignment), завантаження векторів виконується спеціальними інструкціями `_mm256_load_ps` без накладних витрат на нешинований доступ.

### Організація кільцевого буфера стану

Лінія затримки вхідного сигналу реалізована як кільцевий буфер розміром у довжину поліфазної філії `taps_per_phase`. Замість дорогого фізичного зсуву елементів масиву при надходженні кожного нового відліку, програма лише оновлює один елемент `state[state_head]` і просуває вказівник голови `state_head`.

У мікроконтролерах без апаратної підтримки взяття залишком `%` розмір буфера стану вирівнюють до найближчого ступеня двійки `2^N`, замінюючи операцію ділення швидким побітовим маскуванням `& (2^N - 1)`. Це дозволяє процесору виконувати маскування індексу за один операційний такт.

## Крайові умови, розігрів буфера та початковий стан

При запуску інтерполятора перші вхідні відліки `K / L` заповнюють порожню лінію затримки. Доки буфер стану повністю не заповниться реальними значеннями сигналу, вихідні відліки містять перехідний процес розігріву (warm-up phase).

У практичних програмах обробки сигналів застосовують два підходи до управління початковим станом:
- **Заповнення нулями (Zero-padding)**: буфер стану ініціалізується нулями. Перші вихідні відліки відкидаються або враховуються як групова затримка фільтра `(K - 1) / (2 · L)`.
- **Екстраполяція межі (Boundary Extension)**: для обробки зображень та коротких аудіокадрів буфер стану заповнюється дубльованим першим відліком сигналу `x[0]`, що запобігає появі низькочастотного стрибка амплітуди на початку відтворення.

## Пастки амплітудного піка та заклипування (Inter-sample Peaks)

При розробці цифрових аудіопроцесорів та вихідних каскадів ЦАП виникає важливий інженерний ефект — виникнення міжвідлікових піків (Inter-sample Peaks).

Оскільки згладжувальний ФНЧ розраховує проміжні значення огинаючої між вхідними відліками, для сигналів із частотами, близькими до межі Найквіста `f_s / 2`, інтерпольована крива між двома відліками з максимальною амплітудою `1.0` (0 dBFS) може мати амплітудний вигин до `+1.5` або `+2.0` дБ вище за максимальний цифровий рівень.

Якщо інтерполятор працює у форматі з фіксованою точкою Q15 без запасу за амплітудою, такі міжвідлікові піки призводят до жорсткого цифрового заклипування (clipping) та викривлення звуку.

Для боротьби з цим ефектом застосовують два правила:
1. Перед подачею сигналу на інтерполятор амплітуду вхідного потоку послаблюють на 2 децибели (запасовий рівень Headroom).
2. Обчислення всередині поліфазного фільтра виконують із додатковими розрядами акумулятора, застосовуючи апаратне насичення (Saturation) лише на фінальному етапі виводу.

## Пастки цілочислової реалізації у фіксованій точці (Fixed-Point Q15)

При адаптації інтерполятора для вбудованих систем без блоку обробки чисел з плаваючою крапкою FPU використовують цілочислову арифметику із фіксованою точкою Q15. У цьому форматі дробове число у діапазоні `[-1.0, +1.0)` кодується знаковим шістнадцятибітним цілим `int16_t` у діапазоні `[-32768, +32767]`.

При такій реалізації виникають три критичні пастки, нехтування якими призводить до переповнення та спотворення сигналу:

### 1. Вибір розмірності акумулятора (int64_t проти int32_t)

Множення двох 16-бітних чисел формату Q15 дає 32-бітний результат у форматі Q30 (знаковий біт + 30 біт дробу). Якщо довжина поліфазної філії містить 32 відводи (`K / L = 32`), сума 32 таких добутків може перевищити діапазон 31 біта і спричинити переповнення стандартного цілого `int32_t`.

Тому акумулятор для підсумовування обов'язково повинен мати тип шістдесятичотирибітного цілого `int64_t`. Після завершення накопичення суми виконують округлення до найближчого та зсув праворуч на 15 біт: `(int32_t)((acc + (1LL << 14)) >> 15)`.

### 2. Масштабування коефіцієнтів на фактор L

Оскільки згладжувальний фільтр повинен мати коефіцієнт підсилення, що дорівнює коефіцієнту розширення `H(0) = L`, для відновлення амплітуди огинаючої після вставлення нулів, коефіцієнти фільтра під час квантування обов'язково масштабуються з урахуванням цього фактора:

```
h_q15[k] = (int16_t)(h_float[k] * (float)factor_L * 32767.0f)
```

Якщо забути помножити коефіцієнти на коефіцієнт розширення `factor_L`, амплітуда вихідного інтерпольованого сигналу виявиться у `L` разів меншою за вхідну.

### 3. Апаратна насичувальна арифметика (Saturation)

Навіть при правильному масштабуванні згладжувальний фільтр із гострою перехідною смугою має викиди імпульсної характеристики (ефект Ґіббса). Це може призвести до того, що підсумкове значення вийде за межі дозволеного діапазону `[-32768, 32767]`.

Звичайне приведення типів `(int16_t)rounded_acc` у C спричинить обрізання старших бітів і зміну знака сигналу з позитивного на негативний, створюючи різкий цифровий тріск. Щоб запобігти цьому, застосовують операцію насичення:

```
static inline int16_t saturate_q15(int32_t val) {
    if (val > 32767) return 32767;
    if (val < -32768) return -32768;
    return (int16_t)val;
}
```

У сучасних DSP-процесорах та мікроконтролерах ARM Cortex-M4/M7 цю дію виконують за один такт апаратною інструкцією насичення `SSAT`.

## Порівняння продуктивності: Наївна фільтрація проти Поліфазної

Для оцінки ефективності наведено результат вимірювання кількості тактів процесора на обробку одного вхідного відліку для коефіцієнта інтерполяції `L = 8` та початкового фільтра довжиною `K = 128` відводів (довжина однієї поліфазної філії `16` відводів):

| Підхід та алгоритм | Множень на 1 вхід | Зчитувань із пам'яті | Такти CPU / вхідний відлік |
| :--- | :--- | :--- | :--- |
| **Наївний прямий FIR (з нулями)** | `128 × 8 = 1024` | `2048` | `~ 1450` тактів |
| **Оптимізований прямий (пропуск 0)** | `16 × 8 = 128` | `512` | `~ 320` тактів |
| **Поліфазний інтерполятор (C/C++)** | `16 × 8 = 128` | `256` | `~ 140` тактів |
| **Поліфазний + SIMD (AVX2 / Neon)** | `16 × 8 = 128` | `128` | `~ 28` тактів |

Поліфазна структура дає понад 10-кратне прискорення порівняно з наївним підходом навіть у скалярному коді завдяки ліквідації накладних витрат на перевірку індексів та локальності даних у кеш-пам'яті першого рівня L1.
