# ⚙️ Програмний аналіз спектральної маски та побічних випромінювань

Перед подачею радіоелектронного виробу на офіційні випробування в акредитовану випробувальну лабораторію виробник проводить пре-сертифікаційне тестування (англ. *Pre-Compliance Testing*). За допомогою аналізатора спектра або програмно-визначеного радіоприймача (SDR) інженер знімає масив відліків потужності в широкій смузі частот, після чого автоматизоване програмне забезпечення зіставляє отриману спектральну щільність потужності (PSD) з нормативними масками FCC Part 15.247 та ETSI EN 300 328.

Така попередня цифрова верифікація дозволяє миттєво локалізувати критичні викиди на гармоніках та уникнути витрат на повторні цикли випробувань у сертифікаційному центрі.

---

### Постановка задачі та алгоритмічні критерії

Програма валідації радіоспектра приймає таблицю відліків сканування `(f_Hz, P_dBm)` і перевіряє три незалежні регуляторні критерії:

1. **Краї робочої смуги (Band Edge Compliance):**
   Для передавача 2.4 ГГц (ISM 2400.0–2483.5 МГц) випромінювання на частотах нижче 2400.0 МГц та вище 2483.5 МГц має спадати щонайменше на `20 dBc` (відносно піку несучої) за вимірювання в смузі роздільної здатності RBW = 100 кГц.
2. **Смуги обмеженого використання FCC (Restricted Bands, Title 47 CFR §15.205):**
   У чутливих смугах (наприклад, 2310–2390 МГц, 2483.5–2500 МГц, 4800–4960 МГц) напруженість поля не повинна перевищувати 500 мкВ/м на 3 метрах. У перерахунку на кондуктивну потужність з антеною 0 дБі це відповідає жорсткому ліміту:
   ```
   P_restricted_max = −41.2 дБм EIRP
   ```
3. **Побічні випромінювання за європейським стандартом ETSI EN 300 328:**
   - Для частот від 30 МГц до 1000 МГц: `P_spurious ≤ −36.0 дБм` (RBW = 100 кГц);
   - Для частот від 1 ГГц до 12.75 ГГц (за винятком робочого каналу): `P_spurious ≤ −30.0 дБм` (RBW = 1 МГц).

```c
#include <stdio.h>
#include <stdbool.h>
#include <stdint.h>
#include <math.h>

#define MAX_SPECTRUM_POINTS 2048

typedef enum {
    TEST_PASS = 0,
    FAIL_BAND_EDGE_LOW,
    FAIL_BAND_EDGE_HIGH,
    FAIL_RESTRICTED_BAND,
    FAIL_ETSI_SPURIOUS_SUB1G,
    FAIL_ETSI_SPURIOUS_ABOVE1G
} SpectrumTestStatus;

typedef struct {
    double freq_hz;
    double power_dbm;
} SpectrumPoint;

typedef struct {
    double fundamental_freq_hz;
    double fundamental_peak_dbm;
    double worst_band_edge_margin_db;
    double worst_spurious_freq_hz;
    double worst_spurious_power_dbm;
    SpectrumTestStatus status;
} ValidationReport;

/* Перевірка, чи потрапляє частота у смугу обмеженого використання FCC §15.205 */
static bool is_fcc_restricted_band(double f_hz) {
    if (f_hz >= 2310.0e6 && f_hz <= 2390.0e6) return true;
    if (f_hz >= 2483.5e6 && f_hz <= 2500.0e6) return true;
    if (f_hz >= 4800.0e6 && f_hz <= 4960.0e6) return true; // 2-га гармоніка
    if (f_hz >= 7200.0e6 && f_hz <= 7440.0e6) return true; // 3-тя гармоніка
    return false;
}

ValidationReport validate_spectrum(const SpectrumPoint* points, size_t count, double band_start_hz, double band_end_hz) {
    ValidationReport rep = {
        .fundamental_freq_hz = 0.0,
        .fundamental_peak_dbm = -1000.0,
        .worst_band_edge_margin_db = 1000.0,
        .worst_spurious_freq_hz = 0.0,
        .worst_spurious_power_dbm = -1000.0,
        .status = TEST_PASS
    };

    if (!points || count == 0) return rep;

    /* Крок 1: Пошук піку основної несучої частоти всередині робочої смуги */
    for (size_t i = 0; i < count; ++i) {
        if (points[i].freq_hz >= band_start_hz && points[i].freq_hz <= band_end_hz) {
            if (points[i].power_dbm > rep.fundamental_peak_dbm) {
                rep.fundamental_peak_dbm = points[i].power_dbm;
                rep.fundamental_freq_hz = points[i].freq_hz;
            }
        }
    }

    const double band_edge_limit_dbm = rep.fundamental_peak_dbm - 20.0; // ліміт 20 dBc

    /* Крок 2: Перевірка країв смуг, Restricted Bands та побічних випромінювань */
    for (size_t i = 0; i < count; ++i) {
        double f = points[i].freq_hz;
        double p = points[i].power_dbm;

        /* Ігноруємо сам активний робочий канал (In-Band) */
        if (f >= band_start_hz && f <= band_end_hz) {
            continue;
        }

        /* 2.1. Перевірка смуг обмеженого використання FCC Part 15.205 */
        if (is_fcc_restricted_band(f)) {
            const double fcc_restricted_limit_dbm = -41.2;
            if (p > fcc_restricted_limit_dbm) {
                rep.status = FAIL_RESTRICTED_BAND;
                rep.worst_spurious_freq_hz = f;
                rep.worst_spurious_power_dbm = p;
                return rep;
            }
        }

        /* 2.2. Перевірка безпосередніх країв смуги (Band Edges: спадання 20 dBc) */
        if ((f >= band_start_hz - 20.0e6 && f < band_start_hz) ||
            (f > band_end_hz && f <= band_end_hz + 20.0e6)) {
            double margin = band_edge_limit_dbm - p;
            if (margin < rep.worst_band_edge_margin_db) {
                rep.worst_band_edge_margin_db = margin;
            }
            if (p > band_edge_limit_dbm) {
                rep.status = (f < band_start_hz) ? FAIL_BAND_EDGE_LOW : FAIL_BAND_EDGE_HIGH;
                rep.worst_spurious_freq_hz = f;
                rep.worst_spurious_power_dbm = p;
                return rep;
            }
        }

        /* 2.3. Перевірка загальних побічних випромінювань за нормами ETSI EN 300 328 */
        if (f < 1.0e9) {
            if (p > -36.0) {
                rep.status = FAIL_ETSI_SPURIOUS_SUB1G;
                rep.worst_spurious_freq_hz = f;
                rep.worst_spurious_power_dbm = p;
                return rep;
            }
        } else {
            if (p > -30.0) {
                rep.status = FAIL_ETSI_SPURIOUS_ABOVE1G;
                rep.worst_spurious_freq_hz = f;
                rep.worst_spurious_power_dbm = p;
                return rep;
            }
        }

        /* Відстеження найвищого позасмугового піку */
        if (p > rep.worst_spurious_power_dbm) {
            rep.worst_spurious_power_dbm = p;
            rep.worst_spurious_freq_hz = f;
        }
    }

    return rep;
}
```

```cpp
#include <iostream>
#include <vector>
#include <span>
#include <string_view>
#include <expected>
#include <algorithm>
#include <limits>

enum class SpectrumTestError {
    EmptyScanData,
    FailBandEdgeLow,
    FailBandEdgeHigh,
    FailRestrictedBand,
    FailEtsiSpuriousSub1G,
    FailEtsiSpuriousAbove1G
};

struct SpectrumPoint {
    double freq_hz;
    double power_dbm;
};

struct SpectrumValidationReport {
    double fundamental_freq_hz{0.0};
    double fundamental_peak_dbm{-std::numeric_limits<double>::infinity()};
    double worst_band_edge_margin_db{std::numeric_limits<double>::infinity()};
    double worst_spurious_freq_hz{0.0};
    double worst_spurious_power_dbm{-std::numeric_limits<double>::infinity()};
};

class SpectrumMaskValidator {
public:
    constexpr SpectrumMaskValidator(double band_start, double band_end) noexcept
        : band_start_hz_(band_start), band_end_hz_(band_end) {}

    [[nodiscard]] std::expected<SpectrumValidationReport, SpectrumTestError>
    validate(std::span<const SpectrumPoint> scan_data) const noexcept {
        if (scan_data.empty()) {
            return std::unexpected(SpectrumTestError::EmptyScanData);
        }

        SpectrumValidationReport report;

        // Пошук піку основного сигналу в дозволеній смузі
        for (const auto& pt : scan_data) {
            if (pt.freq_hz >= band_start_hz_ && pt.freq_hz <= band_end_hz_) {
                if (pt.power_dbm > report.fundamental_peak_dbm) {
                    report.fundamental_peak_dbm = pt.power_dbm;
                    report.fundamental_freq_hz = pt.freq_hz;
                }
            }
        }

        const double band_edge_limit_dbm = report.fundamental_peak_dbm - 20.0;

        for (const auto& pt : scan_data) {
            const double f = pt.freq_hz;
            const double p = pt.power_dbm;

            if (f >= band_start_hz_ && f <= band_end_hz_) {
                continue;
            }

            // Перевірка смуг обмеженого використання (FCC Restricted Bands)
            if (is_fcc_restricted_band(f)) {
                constexpr double kFccRestrictedLimitDbm = -41.2;
                if (p > kFccRestrictedLimitDbm) {
                    report.worst_spurious_freq_hz = f;
                    report.worst_spurious_power_dbm = p;
                    return std::unexpected(SpectrumTestError::FailRestrictedBand);
                }
            }

            // Перевірка спадання на краях смуги (Band Edges)
            if ((f >= band_start_hz_ - 20.0e6 && f < band_start_hz_) ||
                (f > band_end_hz_ && f <= band_end_hz_ + 20.0e6)) {
                const double margin = band_edge_limit_dbm - p;
                report.worst_band_edge_margin_db = std::min(report.worst_band_edge_margin_db, margin);

                if (p > band_edge_limit_dbm) {
                    report.worst_spurious_freq_hz = f;
                    report.worst_spurious_power_dbm = p;
                    return std::unexpected(f < band_start_hz_ ?
                        SpectrumTestError::FailBandEdgeLow : SpectrumTestError::FailBandEdgeHigh);
                }
            }

            // Перевірка позасмугових побічних випромінювань (ETSI)
            if (f < 1.0e9) {
                if (p > -36.0) {
                    report.worst_spurious_freq_hz = f;
                    report.worst_spurious_power_dbm = p;
                    return std::unexpected(SpectrumTestError::FailEtsiSpuriousSub1G);
                }
            } else {
                if (p > -30.0) {
                    report.worst_spurious_freq_hz = f;
                    report.worst_spurious_power_dbm = p;
                    return std::unexpected(SpectrumTestError::FailEtsiSpuriousAbove1G);
                }
            }

            if (p > report.worst_spurious_power_dbm) {
                report.worst_spurious_power_dbm = p;
                report.worst_spurious_freq_hz = f;
            }
        }

        return report;
    }

private:
    [[nodiscard]] static constexpr bool is_fcc_restricted_band(double f_hz) noexcept {
        return (f_hz >= 2310.0e6 && f_hz <= 2390.0e6) ||
               (f_hz >= 2483.5e6 && f_hz <= 2500.0e6) ||
               (f_hz >= 4800.0e6 && f_hz <= 4960.0e6) ||
               (f_hz >= 7200.0e6 && f_hz <= 7440.0e6);
    }

    double band_start_hz_;
    double band_end_hz_;
};
```

### Архітектура вимірювального стенду: кондуктивні та випромінювані випробування

Тестування радіопередавача на побічні випромінювання та спектральну маску поділяється на два обов'язкові етапи, кожен з яких використовує специфічну вимірювальну апаратуру:

1. **Кондуктивні випробування (Conducted RF Testing):**
   Вихідний антенний порт виробу (роз'єм SMA, U.FL або прямий коаксіальний зонд на друкованій платі) з'єднується з входом вимірювального аналізатора спектра або приймача через калібрований прецизійний ВЧ-атенюатор (наприклад, 20 або 30 дБ).
   - Атенюатор виконує подвійну функцію: захищає чутливий вхідний змішувач аналізатора від вигорання через високу потужність передавача та усуває хвильові відбиття (забезпечує стабільний імпеданс тракту `50 Ом` з низьким коефіцієнтом стоячої хвилі КСХН).
   - У програмному забезпеченні аналізатора обов'язково компенсується амплітудно-частотна характеристика (АЧХ) з'єднувального кабелю та самого атенюатора за допомогою таблиці калібрувальних поправок (Transducer Factor / Loss Table).
2. **Випромінювані випробування (Radiated RF Testing у безвідлунній камері):**
   Пристрій розміщується на діелектричному поворотному столі всередині напівбезвідлунної (Semi-Anechoic Chamber, SAC) або повністю безвідлунної (Fully Anechoic Room, FAR) екранованої камери.
   - Стіни камери покриті пірамідальними радіопоглинаючими матеріалами з вуглецевим наповненням та феритовими плитками, що виключає відбиття електромагнітних хвиль від металевого екрана приміщення.
   - На стандартизованій відстані (3 метри для більшості цивільних приладів або 10 метрів для промислового обладнання Class A) встановлюється вимірювальна калібрована антена (біконічна для діапазону 30–300 МГц, логоперіодична для 300–1000 МГц та подвійний гребінчастий рупор для частот вище 1 ГГц).
   - Вимірювальна антена переміщується по висоті від 1 до 4 метрів з одночасною зміною поляризації (горизонтальна/вертикальна), а поворотний стіл обертається на 360 градусів для фіксації найгіршої просторової орієнтації діаграми спрямованості побічного випромінювання.

### Математичний вплив смуги роздільної здатності (RBW) на результати вимірювання

Однією з найпоширеніших помилок під час попереднього аналізу спектра є неврахування ширини смуги пропускання фільтра роздільної здатності (Resolution Bandwidth, RBW) аналізатора:

- Для вузькосмугових немодульованих сигналів (наприклад, паразитної гармоніки тактового генератора) вся енергія зосереджена у вузькій спектральній лінії. Зміна RBW від 100 кГц до 1 МГц не змінює виміряну пікову амплітуду сигналу.
- Для широкосмугових модульованих шумів або широкосмугових цифрових сигналів (Wi-Fi OFDM, розширений спектр LoRa) енергія розподілена рівномірно по спектру. При збільшенні RBW у `N` разів виміряна потужність шуму зростає пропорційно:
  ```
  ΔP = 10 · log10(RBW_new / RBW_old)
  ```
  Зокрема, перехід від RBW = 100 кГц до стандартизованого для частот вище 1 ГГц фільтра RBW = 1 МГц призводить до стрибка виміряного рівня фонового шуму та широкосмугових побічних завад рівно на `10 · log10(1000 / 100) = 10.0 дБ`. Якщо це не враховано в алгоритмі обробки, пристрій буде помилково забраковано або, навпаки, не виявлено перевищення ліміту в офіційній лабораторії.

### Типові апаратні пастки та джерела завад

Під час лабораторного сканування спектра радіопристрою інженери стикаються з трьома критичними факторами, які спричиняють провал випробувань навіть за ідеальної програмної конфігурації:

1. **Нелінійність вихідного каскаду підсилювача (PA Compression & Intermodulation):**
   Під час роботи підсилювача потужності (PA) у режимі близькому до насичення (точка компресії 1 дБ — `P1dB`) виникають продукти інтермодуляції третього та п'ятого порядків (IMD3, IMD5). Це спричиняє ефект «роздування спектра» (англ. *Spectral Regrowth*), за якого бічні пелюстки сигналу виходять за межі допустимої маски на крайніх каналах (наприклад, на 1-му каналі 2412 МГц або 11/13-му каналах 2462/2472 МГц Wi-Fi). Лікування: програмне зниження вихідної потужності на граничних каналах (англ. *Band Edge Power Backoff*).
2. **Гармоніки імпульсних стабілізаторів живлення (DC-DC Switching Noise):**
   Імпульсні перетворювачі живлення, що працюють на частотах 1–3 МГц із крутими фронтами перемикання (високе `di/dt`), створюють гармонійні завади аж до сотень мегагерців. Якщо трасування земляного полігону порушено, ці завади просочуються на антенний фідер і викликають перевищення ліміту `−36 дБм` у діапазоні до 1 ГГц за стандартом ETSI EN 300 328.
3. **Розриви земляного полігону під радіочастотними мікросмужковими лініями (Ground Plane Discontinuity):**
   Зворотний високочастотний струм протікає по шару заземлення безпосередньо під мікросмужковою сигнальною лінією 50 Ом. Якщо під ВЧ-доріжкою пролягає розрив землі (наприклад, сигнальна траса іншого інтерфейсу або щілина теплового бар'єра), зворотний струм змушений огинати перешкоду. Площа контуру струму різко зростає, перетворюючи ділянку плати на високоефективну рамкову антену, яка випромінює потужні паразитні гармоніки несучої частоти.
4. **Вплив діелектричної проникності корпусу пристрою (Dielectric Loading):**
   Пластиковий корпус виробу (ABS, полікарбонат з діелектричною проникністю `ε_r ≈ 2.8–3.5`), розташований впритул до друкованої антени (PCB trace antenna), зсуває її резонансну частоту вниз на 50–150 МГц. Це призводить до розбалансування узгодження (зростання КСХН), погіршення відбиття енергії та різкого збільшення рівня гармонік на вихідних каскадах передавача.
5. **Неправильний вибір детектора вимірювального приймача:**
   Регуляторні норми визначають різні типи детекторів для різних діапазонів:
   - **Піковий детектор (Peak Detector):** Фіксує максимальну амплітуду огинаючої. Завжди дає найвищий показник і використовується для швидкого попереднього сканування.
   - **Квазіпіковий детектор (Quasi-Peak Detector):** Використовує стандартизовані постійні часу заряду й розряду (CISPR 16-1-1) для зважування завад за їхньою повторюваністю. Обов'язковий для частот `f < 1 ГГц`. Для сигналів із низьким коефіцієнтом заповнення квазіпікове значення на 10–20 дБ нижче за пікове.
   - **Середньоквадратичний/усереднюючий детектор (Average / RMS Detector):** Обов'язковий для вимірювання побічних випромінювань вище 1 ГГц (ETSI) та в смугах обмеженого використання FCC.

### Спеціалізовані тестові прошивки (Direct Test Mode / Lab Firmware)

Для проходження офіційних лабораторних випробувань стандартна користувацька прошивка непридатна, оскільки в нормальному режимі пристрій передає дані короткими стохастичними пакетами зі зміною частот та адаптивною потужністю.

Виробник зобов'язаний надати до сертифікаційної лабораторії тестову прошивку з підтримкою команд прямого керування трансивером (англ. *Direct Test Mode*, DTM):
- **Режим безперервної немодульованої несучої (Continuous Wave, CW):** Використовується лабораторією для точного вимірювання стабільності частоти та кондуктивної вихідної потужності.
- **Режим безперервної передачі модульованого сигналу (Continuous Modulated TX зі 100% робочим циклом):** Необхідний для зняття форми спектральної маски, вимірювання максимальної спектральної щільності потужності (PSD) та оцінки максимального теплового випромінювання для SAR.
- **Режим примусового стрибкоподібного перемикання каналів (Forced FHSS):** Необхідний для перевірки часу перебування на кожному каналі (Dwell Time) та рівномірності завантаження частотного пулу.
- **Режим фіксованого прийому (RX Continuous):** Використовується для перевірки побічних випромінювань гетеродина приймача в режимі очікування (ліміти `−54 дБм` до 1 ГГц та `−47 дБм` вище 1 ГГц за стандартом ETSI EN 300 328).

### Режими накопичення траси: Max Hold проти Video Averaging

Під час автоматизованого зняття спектральної маски вибір режиму накопичення даних на аналізаторі спектра визначає вірогідність фіксації короткочасних імпульсних завад:

1. **Режим максимального утримання (Max Hold):**
   Аналізатор оновлює кожну частотну точку спектра лише у випадку, якщо новий виміряний відлік перевищує попереднє збережене значення. Цей режим є суворо обов'язковим для сертифікаційного тестування систем зі стрибкоподібною зміною частоти (FHSS) та пакетних протоколів (Wi-Fi, Bluetooth LE). Сканування виконується безперервно протягом щонайменше 1–2 хвилин, доки огинаюча маски повністю не стабілізується.
2. **Відеофільтрація (Video Bandwidth, VBW) та час розгортки (Sweep Time):**
   Відеофільтр низьких частот (VBW) згладжує високочастотні шумові флуктуації після детектора огинаючої. Стандарти FCC та ETSI вимагають дотримання правила `VBW ≥ 3 · RBW` для вимірювання пікових рівнів (Peak/Quasi-Peak), щоб вимірювальний фільтр не занижував реальну амплітуду швидких імпульсів.
   
   Мінімальний час одноразової розгортки аналізатора `T_sweep` обчислюється за формулою:
   ```
   T_sweep = k · (Span / (RBW · VBW))
   ```
   де `k ≈ 2.5–3.0` — апаратна константа фільтра. Звуження смуги фільтрів (наприклад, перехід від RBW = 1 МГц до RBW = 10 кГц для детального аналізу країв смуги) збільшує тривалість одного проходу в 10 000 разів, що вимагає ретельного планування вимірювальних інтервалів у програмному коді стенду.

### Калібрування та інтерполяція коефіцієнтів перетворення (Transducer Factor)

Під час вимірювань випромінюваного поля вимірювальний приймач або аналізатор спектра вимірює напругу `V_rec` на своєму 50-омному вході (у дБмкВ). Реальна напруженість електричного поля `E` (у дБмкВ/м) у точці розташування антени обчислюється додаванням коефіцієнта антени (Antenna Factor, `AF` у дБ/м) та затухання в кабелі (`CL` у дБ):

```
E_dBuV_m = V_rec_dBuV + AF_dB_m(f) + CL_dB(f)
```

Оскільки калібрувальні таблиці антени та кабелю надаються виробниками у вигляді дискретних точок на фіксованій сітці частот (наприклад, з кроком 50 МГц), програмне забезпечення автоматизованого вимірювального стенду виконує лінійну або сплайн-інтерполяцію поправочного коефіцієнта `Transducer(f)` для кожної точки сканування:

```
Transducer(f) = AF(f) + CL(f)
```

Якщо частота відліку `f` лежить між двома калібрувальними точками `f₁` та `f₂`:

```
Transducer(f)
= T₁ + (T₂ - T₁) · ((f - f₁) / (f₂ - f₁))   [лінійна інтерполяція поправки]
```

Автоматичне врахування таблиці поправок у реальному часі унеможливлює ситуації, коли резонансний підйом чутливості антени на високих гармоніках (наприклад, на 3-й гармоніці 7.3 ГГц) маскує реальне перевищення нормативного ліміту FCC Part 15.205.

### Кондуктивні завади в лініях живлення (Conducted Emissions & LISN)

Окрім випромінювання через антену та корпус (Radiated Emissions), радіопристрої, що живляться від електромережі змінного струму або бортової мережі транспортного засобу, зобов'язані проходити випробування на кондуктивні завади в лініях електроживлення (Conducted Emissions на портах AC/DC за стандартами FCC Part 15B / CISPR 32 / EN 55032) у діапазоні від 150 кГц до 30 МГц.

#### 1. Роль еквівалента мережі (LISN / AMN)
Пряме підключення аналізатора спектра до розетки 230 В неможливе через ризик вигорання вхідних кіл та непередбачуваний імпеданс зовнішньої електромережі. Для стандартизації умов вимірювання застосовується еквівалент мережі — **LISN (Line Impedance Stabilization Network)** або штучна мережа живлення (Artificial Mains Network, AMN):
- LISN забезпечує стабільний нормований імпеданс `50 Ом || (50 мкГн + 5 Ом)` між фазою (Live), нейтраллю (Neutral) та захисним заземленням (PE) у всій смузі від 150 кГц до 30 МГц.
- Вбудований фільтр високих частот відсікає напругу 50/60 Гц і пропускає лише високочастотні завади на коаксіальний BNC-порт вимірювального приймача.
- Одночасно LISN блокує зовнішні високочастотні шуми, що приходять із промислової електромережі будівлі, запобігаючи спотворенню результатів випробування досліджуваного зразка.

#### 2. Розділення синфазних та диференційних завад (Common-Mode vs Differential-Mode)
Завади в лініях живлення складаються з двох фізично різних компонентів:
- **Диференційна завада (Differential-Mode, DM):** Струм завади протікає по дроту фази в одному напрямку і повертається по дроту нейтралі у зворотному напрямку. Головне джерело — пульсації струму первинного накопичувального конденсатора та перемикання силових транзисторів імпульсного перетворювача. Фільтрується за допомогою X-конденсаторів (ввімкнених паралельно між L та N) та диференційних індуктивностей.
- **Синфазна завада (Common-Mode, CM):** Струм завади тече одночасно по фазі та нейтралі в одному напрямку, а замикається через паразитарні ємності друкованої плати, радіаторів та корпусу на захисне заземлення (PE). Головне джерело — круті перепади напруги `dv/dt` на стоках/колекторах силових ключів щодо заземленого шасі. Фільтрується за допомогою синфазних дроселів (Common-Mode Chokes) та Y-конденсаторів (між лініями L/N та землею PE).

Якщо на частотах 150 кГц – 2 МГц фіксується перевищення квазіпікового ліміту (66–56 дБмкВ), основною причиною є недостатня ємність X-конденсаторів або низька індуктивність первинного фільтра. Якщо завади домінують на частотах 5–30 МГц — це наслідок синфазного просочування через паразитні ємності трансформатора та радіаторів, що лікується встановленням синфазного дроселя з високою магнітною проникністю та феритових бусин на лініях живлення.
