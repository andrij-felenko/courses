# 📋 Інтерфейс та специфікація модулів верифікації формальних доведень PA

Цей довідник описує публічний інтерфейс, контракти системних структур даних, алгоритми перевірки та специфікацію кодів помилок модуля верифікації формальних доведень для арифметики Пеано (PA Proof Checker API).

Модуль верифікації формальних доведень призначений для автоматичної перевірки коректності послідовностей логічних виведень у численні предикатів першого порядку з аксіоматикою Пеано. Програма перевіряє, чи є кожен крок доведення або аксіомою `PA`, або наслідком попередніх кроків за правилами *Modus Ponens*, узагальнення (Generalization) чи застосуванням схеми індукції.

## 1. Загальна архітектура та модель інтеграції

Модуль верифікатора розроблено у вигляді автономної системної бібліотеки без зовнішніх залежностей. Вона забезпечує суворий синтаксичний перевіряльник для формальних систем виведення у стилі Гільберта (Hilbert-style deduction systems).

Формальне доведення у системі Гільберта є впорядкованою скінченною послідовністю формул `φ₁, φ₂, ..., φ_n`, де кожна формула `φ_i` або є аксіомою теорії, або виводиться з попередніх формул `φ_j, φ_k` (де `j, k < i`) за допомогою логічних правил виведення.

Основними компонентами специфікації інтерфейсу є:
1. **Перераховувальний тип правил виведення (`PaRuleType`):** Фіксує точний перелік дозволених аксіом Пеано, аксіом числення предикатів та логічних правил виведення. Кожна аксіома має унікальний числовий кодовий ідентифікатор.
2. **Структура кроку доведення (`PaProofStep`):** Представляє один елементарний крок у послідовності виведення. Вона містить порядковий номер кроку, вказівник на синтаксичне дерево формули (AST), вибране правило виведення та посилання на номери попередніх кроків-передумов, на які спирається виведення.
3. **Структура контексту доведення (`PaProofContext`):** Агрегує впорядкований масив усіх кроків доведення та налаштування рівня суворості перевірки.
4. **Коди статусів та помилок (`PaStatusCode`):** Визначений перелік числових кодів повернення для ідентифікації точної причини відхилення доведення.

Приклад оголошення типів даних наведено у двох варіантах мовного інтерфейсу: для мови C та ідіоматичного модуля C++.

```c
#include <stdint.h>
#include <stddef.h>
#include <stdbool.h>

/* Ідентифікатори аксіом та правил виведення PA */
typedef enum {
    PA_RULE_AXIOM_EQUALITY = 101,  /* Аксіоми рівності (=) */
    PA_RULE_AXIOM_PEANO_1  = 102,  /* ¬(S(x) = 0) */
    PA_RULE_AXIOM_PEANO_2  = 103,  /* S(x) = S(y) → x = y */
    PA_RULE_AXIOM_ADD_BASE = 104,  /* x + 0 = x */
    PA_RULE_AXIOM_ADD_SUCC = 105,  /* x + S(y) = S(x + y) */
    PA_RULE_AXIOM_MUL_BASE = 106,  /* x · 0 = 0 */
    PA_RULE_AXIOM_MUL_SUCC = 107,  /* x · S(y) = (x · y) + x */
    PA_RULE_SCHEME_INDUCTION = 108,/* Схема індукції */
    
    PA_RULE_MODUS_PONENS   = 201,  /* З A та A → B виводиться B */
    PA_RULE_GENERALIZATION = 202   /* З A(x) виводиться ∀x A(x) */
} PaRuleType;

/* Коди повернення та статусів верифікації */
typedef enum {
    PA_SUCCESS               = 0,
    PA_ERR_INVALID_STEP_ID   = 1001,
    PA_ERR_NOT_AN_AXIOM      = 1002,
    PA_ERR_MP_MISMATCH       = 1003,
    PA_ERR_INDUCTION_SCHEMA  = 1004,
    PA_ERR_UNBOUND_VARIABLE  = 1005,
    PA_ERR_CIRCULAR_PROOF    = 1006
} PaStatusCode;

/* Структура одного кроку доведення */
typedef struct PaProofStep {
    uint32_t step_id;           /* Унікальний номер кроку (1-indexed) */
    const void* formula_ast;    /* Указник на синтаксичне дерево формули */
    PaRuleType rule;            /* Обґрунтування кроку */
    uint32_t ref_step_1;        /* Посилання на 1-й крок (0, якщо немає) */
    uint32_t ref_step_2;        /* Посилання на 2-й крок (0, якщо немає) */
} PaProofStep;

/* Контекст верифікації формального доведення */
typedef struct PaProofContext {
    const PaProofStep* steps;   /* Масив кроків доведення */
    size_t step_count;          /* Загальна кількість кроків */
    uint32_t flags;             /* Прапорці налаштувань */
} PaProofContext;
```

```cpp
#include <cstdint>
#include <cstddef>
#include <vector>
#include <memory>
#include <string_view>
#include <expected>

namespace peano::verifier {

enum class RuleType : uint32_t {
    AxiomEquality       = 101,  // Аксіоми рівності (=)
    AxiomPeano1         = 102,  // ¬(S(x) = 0)
    AxiomPeano2         = 103,  // S(x) = S(y) → x = y
    AxiomAddBase        = 104,  // x + 0 = x
    AxiomAddSucc        = 105,  // x + S(y) = S(x + y)
    AxiomMulBase        = 106,  // x · 0 = 0
    AxiomMulSucc        = 107,  // x · S(y) = (x · y) + x
    SchemeInduction     = 108,  // Схема індукції
    
    ModusPonens         = 201,  // З A та A → B виводиться B
    Generalization      = 202   // З A(x) виводиться ∀x A(x)
};

enum class StatusCode : uint32_t {
    Success             = 0,
    InvalidStepId       = 1001,
    NotAnAxiom          = 1002,
    MpMismatch          = 1003,
    InductionSchema     = 1004,
    UnboundVariable     = 1005,
    CircularProof       = 1006
};

struct ProofStep {
    uint32_t step_id;           // Номер кроку (1-indexed)
    const void* formula_ast;    // Вказник на AST формули
    RuleType rule;              // Правило виведення
    uint32_t ref_step_1;        // Посилання на 1-шу передумову
    uint32_t ref_step_2;        // Посилання на 2-гу передумову
};

struct ProofContext {
    std::vector<ProofStep> steps;
    uint32_t flags{0};
};

} // namespace peano::verifier
```

## 3. Детальна розфільтровка кодів статусів та діагностика помилок

Під час проходження алгоритму перевірки кожен крок випробовується на відповідність логічним вимогам. Нижче наведено розширений перелік усіх статусів та детальні умови їх активації:

1. **`PA_SUCCESS` (`0`):**
   Крок або весь ланцюжок виведення повністю відповідає формальним правилам. Усі посилання на попередні кроки знаходяться у межах діапазону `1 <= ref_step < step_id`, синтаксична форма формули відповідає задекларованому правилу `rule`.

2. **`PA_ERR_INVALID_STEP_ID` (`1001`):**
   Виникає при виявленні порушення монотонності нумерації кроків `step_id`. Кроки у масиві повинні бути впорядковані строго за зростанням `1, 2, 3...`. Якщо крок має номер `0` або номер, що перевищує розмір масиву, верифікатор негайно відхиляє доведення.

3. **`PA_ERR_NOT_AN_AXIOM` (`1002`):**
   Помилка синтаксичного зіставлення шаблону (Pattern Matching). Якщо крок обґрунтовано як аксіома (наприклад `PA_RULE_AXIOM_ADD_BASE`), але вказане дерево AST не є уніфікацією шаблону `t + 0 = t` для довільного терма `t`, верифікатор повертає цей код.

4. **`PA_ERR_MP_MISMATCH` (`1003`):**
   Найпоширеніша помилка при формуванні кроків *Modus Ponens*. Якщо поточний крок стверджує формулу `B` з посиланням на крок `i` (формула `A`) та крок `j` (формула `C`), але формула `C` синтаксично не є імплікацією `A → B`, верифікатор сигналізує про розрив у логічному ланцюжку.

5. **`PA_ERR_INDUCTION_SCHEMA` (`1004`):**
   Виникає при спробі використати схему індукції, якщо підключена формула синтаксично не розкладається на структуру `(B(0) ∧ ∀x (B(x) → B(S(x)))) → ∀x B(x)`. Верифікатор перевіряє, щоб формула в базі індукції `B(0)` збігалася з формулою `B(x)` під квантором, у якій змінну `x` замінено на `0`.

6. **`PA_ERR_UNBOUND_VARIABLE` (`1005`):**
   Помилка правила узагальнення (Generalization). Якщо крок отримує формулу `∀x A(x)` з попереднього кроку `A(x)`, але змінна `x` присутня як вільна змінна у нескасованих гіпотезах (допущеннях) даної гілки виведення, правило узагальнення застосовувати заборонено.

7. **`PA_ERR_CIRCULAR_PROOF` (`1006`):**
   Спроба створити циклічну аргументацію. Виникає, якщо крок посилається на самого себе (`ref_step_1 == step_id`) або на крок, який знаходиться нижче у масиві (`ref_step_1 > step_id`).

| Код статусу / Помилка | Числове значення | Критичність | Область виникнення |
| :--- | :--- | :--- | :--- |
| `PA_SUCCESS` | `0` | Норма | Усі перевірки пройшли успішно |
| `PA_ERR_INVALID_STEP_ID` | `1001` | Висока | Структура масиву кроків |
| `PA_ERR_NOT_AN_AXIOM` | `1002` | Середня | Зіставлення AST з аксіомами |
| `PA_ERR_MP_MISMATCH` | `1003` | Висока | Логічне виведення (Modus Ponens) |
| `PA_ERR_INDUCTION_SCHEMA` | `1004` | Висока | Перевірка індукційного шаблону |
| `PA_ERR_UNBOUND_VARIABLE` | `1005` | Середня | Аналіз вільних змінних у кванторах |
| `PA_ERR_CIRCULAR_PROOF` | `1006` | Критична | Циклічні посилання у графі доведення |

## 4. Специфікація функцій верифікації

```c
#include <stdint.h>
#include <stdbool.h>

/* Перевірка одного кроку доведення */
PaStatusCode pa_verify_step(
    const PaProofContext* ctx,
    size_t current_step_idx

/* Повна перевірка всього масиву кроків доведення */
PaStatusCode pa_verify_proof(
    const PaProofContext* ctx,
    uint32_t* failed_step_id

/* Синтаксичний аналізатор відповідності схемі індукції */
bool pa_is_induction_instance(

    const char* var_name
);
```

```cpp
#include <expected>
#include <string_view>

namespace peano::verifier {

// Перевірка одного кроку доведення
StatusCode verify_step(const ProofContext& ctx, size_t current_step_idx) noexcept;

std::expected<void, std::pair<uint32_t, StatusCode>> verify_proof(const ProofContext& ctx) noexcept;
```

### 4.1. Детальний алгоритм верифікації кроку `pa_verify_step`

Під час виклику `pa_verify_step` верифікатор виконує такі алгоритмічні дії:

1. **Перевірка меж:** Отримує об'єкт `PaProofStep` за індексом `current_step_idx`. Перевіряє, що `step_id == current_step_idx + 1`.
2. **Аналіз правила `rule`:**
   - Якщо `rule` належить до групи аксіом (`101..108`), перевіряє, що `ref_step_1 == 0` та `ref_step_2 == 0`. Викликає внутрішній солівер уніфікації AST для перевірки, чи дійсно формула є екземпляром аксіоми.
   - Якщо `rule == PA_RULE_MODUS_PONENS`, перевіряє, що `1 <= ref_step_1 < step_id` та `1 <= ref_step_2 < step_id`. Дістає AST формули `A` з кроку `ref_step_1` та AST формули `C` з кроку `ref_step_2`. Перевіряє, що `C` має вигляд `Implication(left, right)`, де `left` еквівалентна `A`, а `right` еквівалентна поточній формулі.
   - Якщо `rule == PA_RULE_GENERALIZATION`, перевіряє, що поточною формулою є `∀x A(x)`, а формула з `ref_step_1` збігається з `A(x)`. Здійснює обхід дерева контексту гіпотез для перевірки вільності змінної `x`.
3. **Повернення результату:** У разі успішного проходження всіх перевірок повертається `PA_SUCCESS`, інакше повертається відповідний числовий код помилки.

### 4.2. Алгоритм повної перевірки доведення `pa_verify_proof`

Функція `pa_verify_proof` реалізує циклічний прохід по всіх кроках контексту `ctx->steps`:
- Ініціалізує лічильник `current_step_idx = 0`.
- У циклі викликає `pa_verify_step(ctx, current_step_idx)`.
- Якщо повертається код, відмінний від `PA_SUCCESS`, записує `steps[current_step_idx].step_id` у змінну `*failed_step_id` і негайно повертає отриманий код помилки.
- Якщо цикл доходить до кінця (`current_step_idx == ctx->step_count`), повертає `PA_SUCCESS`.

## 5. Правила уніфікації синтаксичних дерев (AST Pattern Matching)

Уніфікація AST є серцем верифікатора `PA`. Під час перевірки того, чи є формула `F` екземпляром аксіоми `PA_RULE_AXIOM_ADD_BASE`, верифікатор не просто порівнює рядки, а будує карта уніфікації змінних.

### 5.1. Уніфікація аксіоми додавання `t + 0 = t`
1. Перевіряється, чи є верхній вузол формули `F` оператором рівності `=`.
2. Ліва частина `F.left` повинна бути оператором додавання `+`.
3. Права частина додавання `F.left.right` повинна бути константою `0`.
4. Ліва частина додавання `F.left.left` порівнюється на синтаксичну ізоморфність із правою частиною рівності `F.right`.
5. Якщо терм `t = F.left.left` ідентичний `F.right`, уніфікація вважається успішною.

### 5.2. Уніфікація аксіоми індукції
Для схеми індукції верифікатор виконує трикомпонентне розкладання формули `F`:
- Формула повинен бути головною імплікацією `Hypothesis → Conclusion`.
- Посилка `Hypothesis` повинна бути кон'юнкцією `Base ∧ Step`.
- `Base` повинна мати вигляд `B[x ↦ 0]`.
- `Step` повинен мати вигляд `∀x (B(x) → B(S(x)))`.
- `Conclusion` повинна мати вигляд `∀x B(x)`.

Якщо всі п'ять умов виконуються для однієї і тієї ж підформули `B`, крок визнається суворо валідним.

## 6. Контракти викликів та приклад використання API

Нижче наведено повністю робочий приклад виклику API верифікатора для перевірки доведення аксіоматичного кроку `PA`.

:::tabs
```c
#include <stdio.h>
#include <stdint.h>
#include <stdbool.h>

int main(void) {
    printf("=== Перевірка формального доведення через PA Proof Checker API (C) ===\n");

    /* Створення масиву кроків для доведення: ∀x (x + 0 = x) */
    PaProofStep steps[2];

    /* Крок 1: Аксіома додавання бази: x + 0 = x */
    steps[0].step_id = 1;
    steps[0].formula_ast = (const void*)0x1000; /* Умовна адреса AST */
    steps[0].rule = PA_RULE_AXIOM_ADD_BASE;
    steps[0].ref_step_1 = 0;
    steps[0].ref_step_2 = 0;

    /* Крок 2: Узагальнення за x: ∀x (x + 0 = x) */
    steps[1].step_id = 2;
    steps[1].formula_ast = (const void*)0x2000;
    steps[1].rule = PA_RULE_GENERALIZATION;
    steps[1].ref_step_1 = 1;
    steps[1].ref_step_2 = 0;

    PaProofContext ctx;
    ctx.steps = steps;
    ctx.step_count = 2;
    ctx.flags = 0;

    uint32_t failed_step = 0;
    PaStatusCode status = pa_verify_proof(&ctx, &failed_step);

    if (status == PA_SUCCESS) {
        printf("Результат: Формальне доведення УСПІШНО ВЕРИФІКОВАНО (PA_SUCCESS)\n");
    } else {
        printf("Результат: ПОМИЛКА ВЕРИФІКАЦІЇ у кроці %u (Код помилки: %d)\n", failed_step, status);
    }

    return 0;
}
```

```cpp
#include <iostream>
#include <vector>

int main() {
    using namespace peano::verifier;

    std::cout << "=== Перевірка формального доведення через PA Proof Checker API (C++20) ===\n";

    ProofContext ctx;
    ctx.steps.push_back(ProofStep{
        .step_id = 1,
        .formula_ast = reinterpret_cast<const void*>(0x1000),
        .rule = RuleType::AxiomAddBase,
        .ref_step_1 = 0,
        .ref_step_2 = 0
    });

    ctx.steps.push_back(ProofStep{
        .step_id = 2,
        .formula_ast = reinterpret_cast<const void*>(0x2000),
        .rule = RuleType::Generalization,
        .ref_step_1 = 1,
        .ref_step_2 = 0
    });

    auto result = verify_proof(ctx);

    if (result.has_value()) {
        std::cout << "Результат: Формальне доведення УСПІШНО ВЕРИФІКОВАНО (Success)\n";
    } else {
        auto [failed_step, status] = result.error();
        std::cout << "Результат: ПОМИЛКА ВЕРИФІКАЦІЇ у кроці " << failed_step 
                  << " (Код помилки: " << static_cast<uint32_t>(status) << ")\n";
    }

    return 0;
}
```

Використання цього API дозволяє легко інтегрувати автоматичний модуль перевірки доведень у SMT-солівери, навчальні системи символьної логіки та програмні комплекси формальної верифікації.
