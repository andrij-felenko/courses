# Проект: Симулятор 8-бітного АЛУ на C++

Щоб глибоко зрозуміти принципи двійкової арифметики та апаратні прапорці стану, найкраще написати програмний симулятор арифметико-логічного пристрою (АЛУ). Ми реалізуємо спрощену модель 8-бітного АЛУ мовою C++17. Наша модель складатиметься з 5 логічних компонентів, що імітують поведінку реального заліза, і буде обчислювати не лише сам результат операції, але й чотири класичні прапорці стану: Carry (перенос), Zero (нуль), Negative (від'ємне число) та Overflow (переповнення).

На відміну від звичайного використання оператора `+` у програмуванні, наш симулятор побітово аналізуватиме вхідні дані, що дозволить побачити внутрішню кухню процесора.

Структура `ALUFlags` зберігатиме поточний стан регістра прапорців:
```cpp
struct ALUFlags {
    bool C; // Carry (Перенос: вихід за межі беззнакового діапазону)
    bool Z; // Zero (Нуль: результат дорівнює 0)
    bool N; // Negative (Від'ємне: старший біт результату дорівнює 1)
    bool V; // Overflow (Переповнення: вихід за межі знакового діапазону)
};
```

Головна функція нашого симулятора — це двійкове додавання двох 8-бітних чисел. Ми свідомо використовуватимемо 16-бітні змінні під час проміжних обчислень, щоб легко захопити 9-й біт, який стане нашим прапорцем переносу.

```cpp
#include <iostream>
#include <cstdint>
#include <iomanip>

struct ALUFlags {
    bool C = false;
    bool Z = false;
    bool N = false;
    bool V = false;
};

class SimpleALU {
public:
    ALUFlags flags;

    // Компонент 1: Базовий 8-бітний суматор із прапорцями
    uint8_t add(uint8_t a, uint8_t b) {
        // Компонент 2: Обчислення сирого результату
        // Кастуємо до 16 біт для захоплення переносу
        uint16_t result_16 = static_cast<uint16_t>(a) + static_cast<uint16_t>(b);
        uint8_t result_8 = static_cast<uint8_t>(result_16);

        // Компонент 3: Прапорець Carry (C) - перевірка 9-го біту
        flags.C = (result_16 > 0xFF);

        // Компонент 4: Прапорці Zero (Z) та Negative (N)
        flags.Z = (result_8 == 0);
        flags.N = ((result_8 & 0x80) != 0); // Перевіряємо 7-й (старший) біт

        // Компонент 5: Прапорець Overflow (V)
        // Переповнення знакового числа стається, коли обидва операнди 
        // мали однаковий знак, але результат отримав протилежний знак.
        bool a_sign = (a & 0x80) != 0;
        bool b_sign = (b & 0x80) != 0;
        bool r_sign = (result_8 & 0x80) != 0;
        flags.V = (a_sign == b_sign) && (a_sign != r_sign);

        return result_8;
    }

    // Додаткова операція віднімання через додавання в доповняльному коді
    uint8_t sub(uint8_t a, uint8_t b) {
        // Віднімання a - b еквівалентне a + (~b + 1)
        uint8_t not_b = ~b;
        
        // Магічний трюк процесорів: щоб не робити два додавання,
        // вхідний carry-in суматора просто встановлюється в 1.
        uint16_t result_16 = static_cast<uint16_t>(a) + static_cast<uint16_t>(not_b) + 1;
        uint8_t result_8 = static_cast<uint8_t>(result_16);

        flags.C = (result_16 > 0xFF);
        flags.Z = (result_8 == 0);
        flags.N = ((result_8 & 0x80) != 0);
        
        bool a_sign = (a & 0x80) != 0;
        bool b_sign = (not_b & 0x80) != 0;
        bool r_sign = (result_8 & 0x80) != 0;
        flags.V = (a_sign == b_sign) && (a_sign != r_sign);

        return result_8;
    }
};

void print_result(const std::string& op, uint8_t a, uint8_t b, uint8_t res, const ALUFlags& f) {
    std::cout << op << " 0x" << std::hex << (int)a << " і 0x" << (int)b 
              << " = 0x" << (int)res << " | Прапорці: "
              << "C=" << f.C << " Z=" << f.Z << " N=" << f.N << " V=" << f.V << "\n";
}

int main() {
    SimpleALU alu;
    
    // Приклад 1: Звичайне додавання без прапорців
    uint8_t res = alu.add(0x10, 0x20); 
    print_result("Додавання", 0x10, 0x20, res, alu.flags); // C=0, Z=0, N=0, V=0

    // Приклад 2: Беззнакове переповнення (Carry)
    res = alu.add(0xFF, 0x02); 
    print_result("Додавання", 0xFF, 0x02, res, alu.flags); // 255 + 2 = 1. C=1

    // Приклад 3: Знакове переповнення (Overflow)
    // 127 (0x7F) + 1 = 128 (0x80, що є -128 у знаковому форматі)
    res = alu.add(0x7F, 0x01); 
    print_result("Додавання", 0x7F, 0x01, res, alu.flags); // N=1, V=1, C=0

    // Приклад 4: Віднімання 5 - 5 = 0
    res = alu.sub(0x05, 0x05); 
    print_result("Віднімання", 0x05, 0x05, res, alu.flags); // Z=1

    return 0;
}
```

У цьому коді ми чітко бачимо фундаментальну властивість процесорної архітектури: АЛУ не знає, які числа ви йому даєте — знакові (int8_t) чи беззнакові (uint8_t). Воно завжди виконує ідентичне бітове додавання і завжди обчислює обидва прапорці виходу за межі: C (для беззнакової інтерпретації) та V (для знакової). Програміст (або компілятор) сам обирає, на який прапорець звертати увагу після виконання інструкції, використовуючи відповідні умовні переходи (наприклад, `JC` — Jump if Carry проти `JO` — Jump if Overflow в асемблері x86).
