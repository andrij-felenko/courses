# Аналізатор мультиплікативної групи (Z/nZ)* мовою C++

Дослідження властивостей мультиплікативної групи залишків вручну може бути складним навіть для невеликих n. Щоб автоматизувати цей процес, створимо аналізатор мовою C++17. Скрипт виводить елементи групи (Z/nZ)*, обчислює її порядок (за допомогою функції Ейлера) та визначає максимальний порядок елементів групи. Якщо цей максимальний порядок збігається з порядком групи, вона є циклічною, а відповідний елемент — генератором (первісним коренем).

## Програмний код

```cpp
#include <iostream>
#include <numeric>
#include <vector>
#include <algorithm>

// 1. Функція обчислення найбільшого спільного дільника
int gcd(int a, int b) {
    while (b) {
        a %= b;
        std::swap(a, b);
    }
    return a;
}

// 2. Обчислення a^b (mod n)
int power_mod(int base, int exp, int mod) {
    int res = 1;
    base %= mod;
    while (exp > 0) {
        if (exp % 2 == 1) res = (res * base) % mod;
        base = (base * base) % mod;
        exp /= 2;
    }
    return res;
}

// 3. Знаходження порядку елемента a в (Z/nZ)*
int find_element_order(int a, int n, int phi) {
    // Порядок повинен бути дільником phi(n) (теорема Лагранжа)
    int order = phi;
    for (int i = 1; i * i <= phi; ++i) {
        if (phi % i == 0) {
            // Перевіряємо i
            if (power_mod(a, i, n) == 1) {
                order = std::min(order, i);
            }
            // Перевіряємо phi/i
            if (power_mod(a, phi / i, n) == 1) {
                order = std::min(order, phi / i);
            }
        }
    }
    return order;
}

// 4. Головна логіка аналізатора
void analyze_group(int n) {
    std::vector<int> elements;
    
    // Знаходимо всі оборотні елементи
    for (int i = 1; i < n; ++i) {
        if (gcd(i, n) == 1) {
            elements.push_back(i);
        }
    }
    
    int phi = elements.size();
    std::cout << "Аналіз мультиплікативної групи (Z/" << n << "Z)*\n";
    std::cout << "Порядок групи (phi(" << n << ")): " << phi << "\n";
    std::cout << "Елементи: ";
    for (int e : elements) std::cout << e << " ";
    std::cout << "\n\n";
    
    // Аналізуємо порядки
    int max_order = 0;
    std::vector<int> generators;
    
    std::cout << "Порядки елементів:\n";
    for (int e : elements) {
        int order = find_element_order(e, n, phi);
        std::cout << e << "^" << order << " = 1\n";
        max_order = std::max(max_order, order);
        if (order == phi) {
            generators.push_back(e);
        }
    }
    
    std::cout << "\nМаксимальний порядок елемента: " << max_order << "\n";
    
    if (!generators.empty()) {
        std::cout << "Група є циклічною!\nПервісні корені: ";
        for (int g : generators) std::cout << g << " ";
        std::cout << "\n";
    } else {
        std::cout << "Група НЕ є циклічною (максимальний порядок < phi(" << n << ")).\n";
    }
}

// 5. Точка входу
int main() {
    int n;
    std::cout << "Введіть модуль n: ";
    if (std::cin >> n && n > 1) {
        analyze_group(n);
    } else {
        std::cout << "Помилка вводу. Модуль має бути цілим числом > 1.\n";
    }
    return 0;
}
```

## Розбір коду за компонентами

1. **Функція `gcd(a, b)`:** Стандартний алгоритм Евкліда для знаходження найбільшого спільного дільника, що використовується для перевірки оборотності.
2. **Алгоритм `power_mod`:** Реалізує швидке піднесення до степеня за модулем (експоненціювання квадратами). Це критично для перевірки великих показників степеня без переповнення.
3. **Функція `find_element_order`:** Знаходить найменше k таке, що `a^k ≡ 1 (mod n)`. Замість того щоб перевіряти всі k підряд, ми знаємо (з теореми Лагранжа), що k обов'язково є дільником порядку групи `phi(n)`. Це різко скорочує кількість обчислень.
4. **Функція `analyze_group`:** Формує множину елементів (Z/nZ)*, після чого ітерує по кожному з них і шукає його порядок. Також вона відслідковує максимальний порядок. Якщо знайдено елемент з порядком, що дорівнює `phi(n)`, група вважається циклічною.
5. **Точка входу `main`:** Простий користувацький інтерфейс.

Цей скрипт дозволяє швидко візуалізувати відмінність між модулем, що має циклічну мультиплікативну групу (наприклад, 7 чи 9), і модулем, що не має (наприклад, 8 чи 15).
