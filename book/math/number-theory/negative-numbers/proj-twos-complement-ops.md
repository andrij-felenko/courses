# ⚙️ Обробка знака, переповнення та евклідового ділення від'ємних чисел

Стандартні оператори ділення та остачі у більшості сучасних мов програмування (зокрема C, C++, Java) реалізують так зване усічене ділення (truncated division), у якому остача завжди успадковує знак діленого. Це створює серйозні пастки при обробці від'ємних чисел у критичних алгоритмах — криптографії, системному програмуванні, обчисленні цикличності за модулем, обробці дат і часу та реалізації арифметики довільної точності (BigInt). У цій вставці наведено практичну реалізацію алгоритмів математично коректного (евклідового) ділення з невід'ємною остачею, алгоритми безпечного знакового додавання, віднімання й множення з детективуванням переповнення, а також бітові операції додаткового коду мовами C та C++.

## 1. Архітектурні вимоги та відмінності ділення

Перед написанням коду визначимо три ключові алгоритмічні задачі, які виникають при роботі зі знаковими цілими числами на системному рівні:

1. **Евклідове ділення з остачею (Euclidean Division)**:
   Необхідно обчислити частку `q` та остачу `r` для будь-яких цілих чисел `a` (ділене) та `b ≠ 0` (дільник) такі, що виконується тотожність `a = q · b + r` за обов'язкової умови `0 ≤ r < |b|`. На відміну від стандартного оператора `%` у мовах C та C++, які повертають від'ємну остачу при від'ємному діленому (наприклад, `-13 % 5 == -3`), евклідова остача для `-13 ÷ 5` зобов'язана дорівнювати `+2`, а частка — `-3`.

2. **Детекція знакового переповнення (Signed Overflow Detection)**:
   Згідно зі стандартами C99 та C++11/17/20, знакове переповнення (*signed integer overflow*) викликає невизначену поведінку (Undefined Behavior). Компілятори з увімкненою оптимізацією (`-O2`, `-O3`) мають право вилучати пост-фактум перевірки вигляду `if (a + b < a)`, оскільки вважають, що переповнення в коректній програмі виникнути не може. Тому перевірку меж необхідно виконувати строго **до** здійснення потенційно небезпечної арифметичної операції.

3. **Бітові маніпуляції додаткового коду (Two's Complement)**:
   Необхідно реалізувати низькорівневі функції витягнення знакового біта, перевірки прапорця від'ємності та апаратного інвертування за правилом двохпозиційного доповнення `~x + 1` без використання знакового мінуса, а також знакового розширення (sign extension).

```c
#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>
#include <stdint.h>
#include <limits.h>

/* Структура для збереження результату Евклідового ділення */
typedef struct {
    int64_t quot; /* Частка q */
    int64_t rem;  /* Остача r, гарантовано 0 <= rem < |b| */
} EuclideanResult;

/*
 * Евклідове ділення: повертає q та r такі, що a = q*b + r та 0 <= r < |b|.
 * Коректно обробляє будь-які комбінації знаків a та b.
 */
EuclideanResult euclidean_div(int64_t a, int64_t b) {
    if (b == 0) {
        fprintf(stderr, "Критична помилка: ділення на нуль!\n");
        exit(EXIT_FAILURE);
    }

    int64_t q = a / b;
    int64_t r = a % b;

    /*
     * Якщо остача r від'ємна, коригуємо частку та остачу:
     * - Якщо d>0: q = q - 1, r = r + b
     * - Якщо d<0: q = q + 1, r = r - b
     */
    if (r < 0) {
        if (b > 0) {
            q -= 1;
            r += b;
        } else {
            q += 1;
            r -= b;
        }
    }

    EuclideanResult res = {q, r};
    return res;
}

/*
 * Безпечне додавання знакових 64-бітних чисел із детекцією переповнення.
 * Повертає false у разі виникнення переповнення.
 */
bool safe_add_int64(int64_t a, int64_t b, int64_t *out_result) {
    if ((b > 0 && a > INT64_MAX - b) || (b < 0 && a < INT64_MIN - b)) {
        return false; /* Переповнення виявлено */
    }
    *out_result = a + b;
    return true;
}

/*
 * Безпечне віднімання знакових 64-бітних чисел із детекцією переповнення.
 */
bool safe_sub_int64(int64_t a, int64_t b, int64_t *out_result) {
    if ((b > 0 && a < INT64_MIN + b) || (b < 0 && a > INT64_MAX + b)) {
        return false; /* Переповнення виявлено */
    }
    *out_result = a - b;
    return true;
}

/*
 * Безпечне множення знакових 64-бітних чисел із детекцією переповнення.
 */
bool safe_mul_int64(int64_t a, int64_t b, int64_t *out_result) {
    if (a == 0 || b == 0) {
        *out_result = 0;
        return true;
    }
    if (a > 0) {
        if (b > 0 && a > INT64_MAX / b) return false;
        if (b < 0 && b < INT64_MIN / a) return false;
    } else {
        if (b > 0 && a < INT64_MIN / b) return false;
        if (b < 0 && a < INT64_MAX / b) return false;
    }
    *out_result = a * b;
    return true;
}

/*
 * Апаратне бітове оперування: обчислення додаткового коду (~val + 1)
 * з попереднім зведенням до uint64_t для уникнення Undefined Behavior.
 */
uint64_t negate_twos_complement(int64_t val) {
    uint64_t uval = (uint64_t)val;
    return (~uval) + 1U;
}

/* Перевірка знакового біта (старший біт 63) */
bool is_sign_bit_set(int64_t val) {
    return ((uint64_t)val >> 63) != 0;
}

int main(void) {
    int64_t dividend = -13;
    int64_t divisor = 5;

    /* Демонстрація відмінності стандартного ділення C від Евклідового */
    int64_t c_quot = dividend / divisor;
    int64_t c_rem = dividend % divisor;
    printf("Стандартне ділення C (-13 / 5): q = %lld, r = %lld\n",
           (long long)c_quot, (long long)c_rem);

    EuclideanResult eucl = euclidean_div(dividend, divisor);
    printf("Математичне Евклідове (-13 / 5): q = %lld, r = %lld (0 <= r < %lld)\n",
           (long long)eucl.quot, (long long)eucl.rem, (long long)llabs(divisor));

    /* Демонстрація безпечного додавання та множення */
    int64_t res = 0;
    if (safe_add_int64(INT64_MAX - 10, 20, &res)) {
        printf("Результат додавання: %lld\n", (long long)res);
    } else {
        printf("Виявлено знакове переповнення при додаванні!\n");
    }

    if (safe_mul_int64(-1000000000LL, 10000000000LL, &res)) {
        printf("Результат множення: %lld\n", (long long)res);
    } else {
        printf("Виявлено знакове переповнення при множенні!\n");
    }

    /* Демонстрація бітових операцій додаткового коду */
    int64_t neg_val = -42;
    uint64_t raw_twos = (uint64_t)neg_val;
    printf("Число -42 в шестнадцятковому додатковому коді: 0x%016llX\n",
           (unsigned long long)raw_twos);
    printf("Знаковий біт встановлений: %s\n", is_sign_bit_set(neg_val) ? "так" : "ні");

    return 0;
}
```

```cpp
#include <iostream>
#include <cstdint>
#include <cmath>
#include <limits>
#include <utility>
#include <optional>
#include <stdexcept>
#include <iomanip>

namespace math {

// Структура результату Евклідового ділення
struct EuclideanResult {
    std::int64_t quot; // Частка q
    std::int64_t rem;  // Остача r, 0 <= r < |b|
};

// Евклідове ділення: a = q*b + r, 0 <= r < |b|
constexpr EuclideanResult euclidean_div(std::int64_t a, std::int64_t b) {
    if (b == 0) {
        throw std::invalid_argument("Ділення на нуль у euclidean_div");
    }

    std::int64_t q = a / b;
    std::int64_t r = a % b;

    if (r < 0) {
        if (b > 0) {
            q -= 1;
            r += b;
        } else {
            q += 1;
            r -= b;
        }
    }
    return {q, r};
}

// Безпечне знакове додавання з використанням std::optional
constexpr std::optional<std::int64_t> safe_add(std::int64_t a, std::int64_t b) noexcept {
    if ((b > 0 && a > std::numeric_limits<std::int64_t>::max() - b) ||
        (b < 0 && a < std::numeric_limits<std::int64_t>::min() - b)) {
        return std::nullopt; // Переповнення
    }
    return a + b;
}

// Безпечне знакове множення з використанням std::optional
constexpr std::optional<std::int64_t> safe_mul(std::int64_t a, std::int64_t b) noexcept {
    if (a == 0 || b == 0) return 0;
    if (a > 0) {
        if (b > 0 && a > std::numeric_limits<std::int64_t>::max() / b) return std::nullopt;
        if (b < 0 && b < std::numeric_limits<std::int64_t>::min() / a) return std::nullopt;
    } else {
        if (b > 0 && a < std::numeric_limits<std::int64_t>::min() / b) return std::nullopt;
        if (b < 0 && a < std::numeric_limits<std::int64_t>::max() / b) return std::nullopt;
    }
    return a * b;
}

// Перевірка знакового біта та обчислення додаткового коду
constexpr bool is_negative(std::int64_t val) noexcept {
    return val < 0;
}

constexpr std::uint64_t twos_complement_negate(std::int64_t val) noexcept {
    return static_cast<std::uint64_t>(~static_cast<std::uint64_t>(val)) + 1U;
}

} // namespace math

int main() {
    constexpr std::int64_t a = -13;
    constexpr std::int64_t b = 5;

    // Вбудовані оператори C++
    std::cout << "Стандартний C++ operator%: -13 % 5 = " << (a % b) << "\n";

    // Евклідове ділення
    const auto [q, r] = math::euclidean_div(a, b);
    std::cout << "Евклідове ділення: -13 = (" << q << ") * 5 + " << r << "\n";

    // Детекція переповнення
    constexpr std::int64_t max_val = std::numeric_limits<std::int64_t>::max();
    if (auto res = math::safe_add(max_val, 100)) {
        std::cout << "Сума: " << *res << "\n";
    } else {
        std::cout << "Виявлено знакове переповнення (Overflow) при додаванні!\n";
    }

    if (auto res = math::safe_mul(-1000000000LL, 10000000000LL)) {
        std::cout << "Добуток: " << *res << "\n";
    } else {
        std::cout << "Виявлено знакове переповнення (Overflow) при множенні!\n";
    }

    // Додатковий код
    constexpr std::int64_t neg_val = -42;
    std::cout << "Число -42 у додатковому коді: 0x"
              << std::hex << std::setw(16) << std::setfill('0')
              << static_cast<std::uint64_t>(neg_val) << std::dec << "\n";

    return 0;
}
```

## 2. Поглиблений аналіз пасток та системних ризиків

### 1. Пастка несиметричності діапазону (Асиметрія додаткового коду)
У N-бітному знаковому представленні додатковий код має асиметричний діапазон відносно нуля. Розглянемо причинно-наслідковий зв'язок цієї асиметрії:
- Загальна кількість комбінацій для N бітів дорівнює `2ⁿ`.
- Одне значення витрачається на представлення нуля `0` (всі біти рівні `0`).
- Позитивний діапазон містить `2ⁿ⁻¹ - 1` чисел (наприклад, від `1` до `9223372036854775807` для 64-бітних чисел).
- Негативний діапазон містить `2ⁿ⁻¹` чисел (наприклад, від `-9223372036854775808` до `-1`).

Звідси випливає критична системна пастка: **найменше від'ємне число `INT64_MIN` не має відповідного додатного аналога у своєму типі `int64_t`**.
Спроба виконати операцію `-INT64_MIN` або виклик `std::abs(INT64_MIN)` викликає миттєве знакове переповнення та Undefined Behavior.

Для безпечного обчислення абсолютної величини без виходу за межі типів число слід спочатку звести до беззнакового типу `uint64_t`, де взяття протилежного виконується у кільці лишків за модулем `2⁶⁴`:

:::tabs
```c
uint64_t safe_abs_c(int64_t val) {
    if (val >= 0) return (uint64_t)val;
    return ((uint64_t)(-(val + 1))) + 1U; /* Безпечно працює навіть для INT64_MIN */
}
```
```cpp
constexpr std::uint64_t safe_abs_cpp(std::int64_t val) noexcept {
    if (val >= 0) return static_cast<std::uint64_t>(val);
    return static_cast<std::uint64_t>(-(val + 1)) + 1U;
}
```

### 2. Пастка знакового ділення `INT64_MIN / -1`
Операція ділення `a / b` при значеннях `a = INT64_MIN` та `b = -1` є єдиним випадком знакового ділення, який викликає апаратне виключення процесора (на архітектурі x86/x64 це породжує переривання `#DE` — *Divide Error Exception*, що призводить до аварійного завершення програми із сигналом `SIGFPE`).

Це відбувається тому, що точний математичний результат `+9223372036854775808` перевищує максимально можливе значення `INT64_MAX` на одиницю і не може бути записаний у 64-бітний знаковий регістр `RAX`.

У будь-якому промисловому коді ділення знакових чисел зобов me перевіряти цей крайній випадок перед викликом апаратного ділення:

:::tabs
```c
if (a == INT64_MIN && b == -1) {
    /* Обробка переповнення ділення у C */
}
```
```cpp
if (a == std::numeric_limits<std::int64_t>::min() && b == -1) {
    /* Обробка переповнення ділення у C++ */
}
```

### 3. Оптимізаційні агресії компілятора при перевірках переповнення
Сучасні компілятори (GCC, Clang) активно спираються на факт відсутності знакового переповнення для здійснення векторизації та оптимізації циклів. Якщо програміст пише небезпечний код вигляду:

:::tabs
```c
int32_t c = a + b;
if (c < a) {
    /* Перевірка на переповнення у C (небезпечно!) */
}
```
```cpp
std::int32_t c = a + b;
if (c < a) {
    /* Перевірка на переповнення у C++ (небезпечно!) */
}
```
:::

Компілятор при аналізі зауважить: «оскільки знакове переповнення є Undefined Behavior, вважаємо, що `a + b` ніколи не переповнюється, отже `a + b >= a` при `b >= 0` завжди істинне». В результаті блок `if (c < a)` просто вилучається з відкомпільованого машинного коду!

Для уникнення ручних перевірок сучасні компілятори надають спеціальні вбудовані функції (*built-in functions*), які транслюються безпосередньо в апаратну перевірку прапорця `Overflow Flag (V)` у регістрі процесора:

:::tabs
```c
int64_t result;
if (__builtin_add_overflow(a, b, &result)) {
    /* Переповнення виявлено за прапорцем процесора у C */
}
```
```cpp
std::int64_t result;
if (__builtin_add_overflow(a, b, &result)) {
    /* Переповнення виявлено за прапорцем процесора у C++ */
}
```

### 4. Циклічна арифметика та Евклідове ділення у практичних задачах
У практичному програмуванні Евклідове ділення є незамінним у таких сценаріях:

- **Обчислення хеш-таблиць з від'ємними ключами**: індекс кошика `bucket = hash(key) % capacity` зобов'язаний завжди знаходитися в діапазоні `[0, capacity - 1]`. При стандартному C-діленні від'ємні хеші дадуть від'ємний індекс кошика, що спричинить вихід за межі масиву (Out-of-Bounds Memory Access).
- **Робота з днями тижня та часовими поясами**: якщо поточний день тижня `3` (середа), то який день тижня був 10 днів тому? Формула `(3 - 10) % 7` у C дасть `-7 % 7 == 0` (або `-10 % 7 == -3`), що потребує коригування. Евклідове ділення `(3 - 10) mod 7` дає `+3 - 10 = -7 = (-1)·7 + 0`, чи `(3 - 12) mod 7 = -9 = (-2)·7 + 5` (п'ятниця), що є логічно правильним.
- **Обробка циклічних кутів у тригонометрії та 2D-графіці**: зсув кута на від'ємну величину (наприклад, `-450°`) повинен приведення до стандартного діапазону `[0°, 360°)`. Евклідове ділення `-450 mod 360` дає остачу `270°`, що точно відповідає геометричному розташуванню вектора на площині.
