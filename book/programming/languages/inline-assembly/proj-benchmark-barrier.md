# ⚙️ Практикум: таймери процесора, бар'єри компілятора та пастки оптимізації

Цей практикум демонструє практичне застосування вбудованого асемблера для прямої взаємодії з апаратними регістрами сучасних мікропроцесорів, глибоке дослідження впливу бар'єра компілятора на оптимізацію пам'яті, виправлення критичних дефектів розподілу регістрів та перехід на вбудовані функції (intrinsics).

У ході лабораторних завдань ми розглянемо чотири фундаментальні сценарії: від вимірювання тривалості виконання коду з точністю до окремих тактів генератора до усунення підступних помилок розподілювача регістрів при агресивних оптимізаціях.

## Завдання 1: Високоточне вимірювання тактів процесора (RDTSC / CNTVCT)

Для точного профілювання критичних ділянок коду, алгоритмів шифрування чи драйверів пристроїв стандартні таймери операційної системи (такі як `clock_gettime` або `std::chrono::high_resolution_clock`) створюють завеликі накладні витрати через системні виклики та низьку дискретність. Вирішення полягає в прямому зчитуванні лічильника тактів процесора.

На платформі x86-64 для цього призначена інструкція `rdtsc` (Read Time-Stamp Counter), яка записує 64-бітне значення лічильника в пару регістрів `EDX:EAX`. Проте сучасні суперскалярні процесори з позачерговим виконанням (Out-of-Order Execution) можуть виконати `rdtsc` раніше або пізніше вимірюваного коду. Щоб уникнути викривлення вимірювань, перед зчитуванням лічильника необхідно вставити інструкцію серіалізації конвеєра `lfence` (або використати інструкцію `rdtscp`, яка додатково зчитує ідентифікатор ядра `IA32_TSC_AUX`).

На платформі ARM64 (AArch64) аналогічну функцію виконує системний регістр `CNTVCT_EL0` (Counter-timer Virtual Count register), доступ до якого здійснюється через інструкцію `mrs` з обов'язковою серіалізацією конвеєра інструкцією `isb` (Instruction Synchronization Barrier).

### Реалізація на x86-64 та AArch64

:::tabs
```c
#include <stdint.h>
#include <stdio.h>

/* Зчитування TSC на x86-64 із серіалізацією конвеєра через lfence */
static inline uint64_t cpu_cycles_x86(void) {
    uint32_t lo, hi;
    /* lfence запобігає випереджальному виконанню rdtsc процесором */
    __asm__ __volatile__ (
        "lfence\n\t"
        "rdtsc"
        : "=a" (lo), "=d" (hi)
        :
        : "memory"
    );
    return ((uint64_t)hi << 32) | lo;
}

/* Зчитування віртуального лічильника CNTVCT_EL0 на ARM64 */
static inline uint64_t cpu_cycles_arm64(void) {
    uint64_t val;
    __asm__ __volatile__ (
        "isb\n\t"
        "mrs %0, cntvct_el0"
        : "=r" (val)
        :
        : "memory"
    );
    return val;
}

int main(void) {
    uint64_t start = cpu_cycles_x86();
    
    /* Вимірювана ділянка коду: простий накопичувальний цикл */
    volatile int sum = 0;
    for (int i = 0; i < 1000; ++i) {
        sum += i;
    }
    
    uint64_t end = cpu_cycles_x86();
    printf("Витрачено тактів процесора: %lu\n", (unsigned long)(end - start));
    return 0;
}
```
```cpp
#include <cstdint>
#include <iostream>

/* Ідіоматична C++ обгортка над апаратним таймером процесора */
class CpuCycleTimer {
public:
    static uint64_t now() noexcept {
#if defined(__x86_64__) || defined(_M_X64)
        uint32_t lo = 0, hi = 0;
        asm volatile (
            "lfence\n\t"
            "rdtsc"
            : "=a" (lo), "=d" (hi)
            :
            : "memory"
        );
        return (static_cast<uint64_t>(hi) << 32) | lo;
#elif defined(__aarch64__)
        uint64_t val = 0;
        asm volatile (
            "isb\n\t"
            "mrs %0, cntvct_el0"
            : "=r" (val)
            :
            : "memory"
        );
        return val;
#else
        return 0;
#endif
    }
};

int main() {
    const uint64_t start = CpuCycleTimer::now();
    
    volatile int sum = 0;
    for (int i = 0; i < 1000; ++i) {
        sum += i;
    }
    
    const uint64_t end = CpuCycleTimer::now();
    std::cout << "Витрачено тактів процесора: " << (end - start) << '\n';
}
```
:::

Зверніть увагу на обмеження `"=a"` та `"=d"` у варіанті для x86: вони жорстко наказують розподілювачу регістрів пов'язати змінну `lo` з регістром `EAX`, а `hi` — з `EDX`, що точно відповідає апаратній специфікації інструкції `rdtsc`. Для забезпечення стабільності результатів вимірювання у реальних тестах потік рекомендується прив'язувати до одного ядра процесора через `sched_setaffinity`, щоб уникнути міграції між ядрами з різними базовими значеннями лічильників.

## Завдання 2: Дослідження бар'єра компілятора (`"memory"` clobber)

Програмний бар'єр компілятора `asm volatile("" ::: "memory")` є унікальною конструкцією: він не породжує жодної машинної інструкції в бінарному файлі, проте кардинально змінює поведінку проміжних фаз оптимізації.

Коли компілятор обробляє цикл без бар'єра, фаза розфарбовування графа інтерференції розміщує змінні у регістрах процесора, а оптимізатор циклів (Loop Vectorizer) може перетворити послідовні записи в один широкий векторний запис AVX або об'єднати операції. Присутність бар'єра повідомляє оптимізатору, що стан пам'яті міг довільно змінитися, що змушує компілятор скидати значення в пам'ять після кожної ітерації.

### Експериментальний стенд

:::tabs
```c
#include <stdint.h>
#include <stdbool.h>

/* Функція обробки буфера з явним бар'єром компілятора */
void process_with_barrier(int *shared_buffer, int count) {
    for (int i = 0; i < count; ++i) {
        shared_buffer[i] = i * 2;
        
        /* Програмний бар'єр забороняє переносити або об'єднувати записи */
        __asm__ __volatile__ ("" ::: "memory");
    }
}

/* Функція обробки буфера без бар'єра (піддається агресивній оптимізації) */
void process_without_barrier(int *shared_buffer, int count) {
    for (int i = 0; i < count; ++i) {
        shared_buffer[i] = i * 2;
    }
}
```
```cpp
#include <span>
#include <cstdint>

/* C++ варіант з безпечним інтерфейсом std::span */
void process_with_barrier(std::span<int> buffer) noexcept {
    int val = 0;
    for (int &elem : buffer) {
        elem = val++ * 2;
        /* Програмний бар'єр інвалідує кешовані значення в регістрах */
        asm volatile ("" ::: "memory");
    }
}

void process_without_barrier(std::span<int> buffer) noexcept {
    int val = 0;
    for (int &elem : buffer) {
        elem = val++ * 2;
    }
}
```
:::

### Аналіз згенерованого асемблерного коду

Якщо скомпілювати обидві функції через GCC або Clang з прапорцем `-O3`:

1. У функції `process_without_barrier` оптимізатор повністю розгортає цикл і генерує векторні команди `vmovdqu` / `vpaddd`, записуючи по 8 чи 16 елементів одночасно.
2. У функції `process_with_barrier` оптимізатор не може застосувати векторизацію. Він вимушений згенерувати класичний цикл зі скалярною інструкцією `movl %edx, (%rdi,%rax,4)` на кожній ітерації, оскільки `"memory"` clobber формально вказує на можливість модифікації масиву всередині тіла циклу.

Цей експеримент наочно демонструє ціну використання програмного бар'єра: надмірне застосування `"memory"` clobber блокує автовекторизацію, конвеєризацію циклів та інваріантне винесення коду.

## Завдання 3: Пастка раннього перезапису (Earlyclobber Bug)

Однією з найнебезпечніших та найважчих для діагностики помилок при написанні вбудованого асемблера є відсутність модифікатора `&` (Earlyclobber) у багаторядкових вставках.

Якщо блок асемблера містить послідовність із кількох команд, де вихідний регістр перезаписується на першому кроці, розподілювач регістрів компілятора (який вважає, що входи зчитуються лише до початку блоку) може призначити виходу та одному з входів один і той самий фізичний регістр процесора.

### Відтворення помилки

:::tabs
```c
#include <stdint.h>
#include <stdio.h>

/* ДЕФЕКТ: при оптимізації -O2 змінна multiplier буде пошкоджена */
uint64_t unsafe_accumulate(uint64_t base, uint64_t multiplier) {
    uint64_t result;
    __asm__ (
        "movq %1, %0\n\t"       /* Крок 1: запис у %0 руйнує %2, якщо вони в одному регістрі! */
        "imulq %2, %0\n\t"      /* Крок 2: множення на вже пошкоджений %2 */
        "addq %1, %0"           /* Крок 3: додавання base */
        : "=r" (result)         /* ПОМИЛКА: пропущено модифікатор & */
        : "r" (base), "r" (multiplier)
        : "cc"
    );
    return result;
}
```
```cpp
#include <cstdint>
#include <iostream>

/* ДЕФЕКТ: некоректний розподіл регістрів під оптимізацією */
uint64_t unsafe_accumulate(uint64_t base, uint64_t multiplier) noexcept {
    uint64_t result = 0;
    asm (
        "movq %1, %0\n\t"
        "imulq %2, %0\n\t"
        "addq %1, %0"
        : "=r" (result)         /* ПОМИЛКА: пропущено модифікатор & */
        : "r" (base), "r" (multiplier)
        : "cc"
    );
    return result;
}
```
:::

Під час компіляції з `-O2` розподілювач регістрів бачить, що вхідний аргумент `multiplier` більше не потрібен після завершення вставки, і призначає вихідному операнду `result` та вхідному `multiplier` спільний регістр `%rax`. У результаті перша ж інструкція `movq %rdi, %rax` затирає значення `multiplier`, а наступна команда `imulq %rax, %rax` множить число саме на себе замість потрібного коефіцієнта.

### Виправлення за допомогою модифікатора `=&r`

:::tabs
```c
#include <stdint.h>

/* КОРЕКТНО: модифікатор & гарантує унікальність фізичного регістра */
uint64_t safe_accumulate(uint64_t base, uint64_t multiplier) {
    uint64_t result;
    __asm__ (
        "movq %1, %0\n\t"
        "imulq %2, %0\n\t"
        "addq %1, %0"
        : "=&r" (result)        /* ВИПРАВЛЕНО: earlyclobber захищає від злиття регістрів */
        : "r" (base), "r" (multiplier)
        : "cc"
    );
    return result;
}
```
```cpp
#include <cstdint>

/* КОРЕКТНИЙ ідіоматичний C++ код */
uint64_t safe_accumulate(uint64_t base, uint64_t multiplier) noexcept {
    uint64_t result = 0;
    asm (
        "movq %1, %0\n\t"
        "imulq %2, %0\n\t"
        "addq %1, %0"
        : "=&r" (result)        /* ВИПРАВЛЕНО: earlyclobber */
        : "r" (base), "r" (multiplier)
        : "cc"
    );
    return result;
}
```
:::

Модифікатор `&` змушує планувальник графів інтерференції вважати інтервал життя вихідного регістра перекритим з інтервалами життя всіх вхідних регістрів, що гарантує виділення абсолютно незалежного фізичного регістра.

## Завдання 4: Заміна вбудованого асемблера на Intrinsics

Для обчислювальних задач (бітових маніпуляцій, векторних розрахунків SIMD) використання вбудованого асемблера є застарілою практикою, яка блокує оптимізацію констант та автовекторизацію. Сучасним стандартом є застосування вбудованих функцій компілятора (Compiler Intrinsics).

### Порівняльний аналіз: підрахунок провідних нулів та SIMD-додавання

:::tabs
```c
#include <stdint.h>
#include <immintrin.h>

/* Варіант 1: Вбудований асемблер (не підтримує згортання констант) */
int count_leading_zeros_asm(uint32_t val) {
    if (val == 0) return 32;
    int res;
    __asm__ ("bsrl %1, %0" : "=r" (res) : "r" (val) : "cc");
    return 31 - res;
}

/* Варіант 2: Вбудована функція компілятора (повна оптимізація в IR) */
int count_leading_zeros_intrinsic(uint32_t val) {
    if (val == 0) return 32;
    return __builtin_clz(val); /* Автоматично генерує LZCNT або BSR */
}

/* Паралельне векторне додавання через SSE Intrinsics */
void add_vectors_sse(const float *a, const float *b, float *result) {
    __m128 va = _mm_loadu_ps(a);
    __m128 vb = _mm_loadu_ps(b);
    __m128 vres = _mm_add_ps(va, vb);
    _mm_storeu_ps(result, vres);
}
```
```cpp
#include <cstdint>
#include <bit>
#include <immintrin>
#include <span>

/* Варіант C++20: стандартний алгоритм транслюється прямо в intrinsic */
constexpr int count_leading_zeros_modern(uint32_t val) noexcept {
    return std::countl_zero(val); /* Може обчислюватися під час компіляції (constexpr) */
}

/* Ідіоматична робота з SIMD через intrinsics у C++ */
void add_vectors_sse(std::span<const float, 4> a, 
                     std::span<const float, 4> b, 
                     std::span<float, 4> result) noexcept {
    const __m128 va = _mm_loadu_ps(a.data());
    const __m128 vb = _mm_loadu_ps(b.data());
    const __m128 vres = _mm_add_ps(va, vb);
    _mm_storeu_ps(result.data(), vres);
}
```
:::

Якщо передати константу `count_leading_zeros_intrinsic(0x00FF0000)` у варіант з intrinsics, компілятор виконає операцію `__builtin_clz` безпосередньо під час аналізу синтаксичного дерева, повністю замінивши виклик функції на число `8`. Варіант на вбудованому асемблері змушений щоразу генерувати інструкцію `bsrl` під час виконання у процесорі.

Крім того, при використанні intrinsics компілятор має повну свободу в розташуванні даних: він може поєднати векторне завантаження `_mm_loadu_ps` та додавання в одну комбіновану інструкцію `addps (%rdi), %xmm0`, тоді як у варіанті з `asm` програміст змушений кодувати кожну мікроінструкцію окремо.
