# ⚙️ Алгоритм розрахунку варіантності та фазової рівноваги

Чисельний аналіз фазових перетворень та варіантності багатокомпонентних реагуючих систем вимагає автоматизованого підрахунку числа незалежних компонентів, виявлення стехіометрично незалежних хімічних реакцій та перевірки термодинамічних обмежень. У промислових термодинамічних симуляторах (таких як Aspen Plus, FactSage або Thermo-Calc) аналіз варіантності за Гіббсом передує безпосередньому чисельному розрахунку фазового складу та мінімізації вільної енергії Гіббса. Він дозволяє встановити, чи є задана термодинамічна постановка коректно визначеною, чи містить вона надлишкові обмеження, або чи описує неможливий стан співіснування фаз.

Нижче наведено побудову математичного алгоритму розрахунку ступенів вільності за Гіббсом, аналіз структури даних, алгоритм фазового розщеплення (Flash Calculation) за рівнянням Рачфорда-Райса, підходи системи CALPHAD, метод обчислення координат трикутника Гіббса, аналіз чисельної стійкості матриці Гессе, оцінку часової складності та реалізовано готовий до використання програмний модуль двома мовами — C та C++.

## 1. Постановка задачі та математична структура системи

Розглядається гетерогенна фізико-хімічна система, задана такими вхідними параметрами:
1. Переліком з `N[spec]` хімічних речовин (молекулярних продуктів, іонів, твердих модифікацій або варіантів фазового стану);
2. Набором з `R[tot]` можливих хімічних та фазових реакцій, описаних матрицею стехіометричних коефіцієнтів `A` розміром `R[tot] × N[spec]`;
3. Кількістю `q` додаткових початкових стехіометричних, електронетральних або концентраційних обмежень;
4. Поточним числом співіснуючих у системі фаз `P`;
5. Прапором типу системи (загальна термодинамічна система чи конденсована фаза при постійному тиску).

Математична задача зводиться до виконання чотирьох послідовних кроків:
- Обчислення рангу стехіометричної матриці `R = rank(A)` методом Гаусса з частковим вибором головного елемента по стовпчиках, що визначає кількість термодинамічно незалежних реакцій;
- Визначення числа незалежних компонентів Гіббса за формулою `K = N[spec] - R - q`;
- Розрахунку варіантності (числа ступенів вільності) `F = K - P + 2` (або `F = K - P + 1` для конденсованого стану при `P = const`);
- Автоматичної класифікації термодинамічного стану системи (нонваріантна точка, моноваріантна лінія, диваріантна область або нестійкий/перевизначений стан при `F < 0`).

```c
#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>
#include <math.h>

#define MAX_SPECIES 16
#define MAX_REACTIONS 16
#define EPSILON 1e-9

typedef struct {
    int num_species;
    int num_reactions;
    int num_constraints;
    int num_phases;
    bool is_condensed;
    double stoich_matrix[MAX_REACTIONS][MAX_SPECIES];
} PhaseSystemC;

typedef struct {
    int rank_reactions;
    int gibbs_components;
    int degrees_of_freedom;
    bool is_valid_equilibrium;
} PhaseResultC;

static int compute_matrix_rank(double matrix[MAX_REACTIONS][MAX_SPECIES], int rows, int cols) {
    double temp[MAX_REACTIONS][MAX_SPECIES];
    for (int i = 0; i < rows; ++i) {
        for (int j = 0; j < cols; ++j) {
            temp[i][j] = matrix[i][j];
        }
    }

    int rank = 0;
    bool row_selected[MAX_REACTIONS] = { false };

    for (int col = 0; col < cols; ++col) {
        int pivot_row = -1;
        double max_val = EPSILON;

        for (int row = 0; row < rows; ++row) {
            if (!row_selected[row] && fabs(temp[row][col]) > max_val) {
                max_val = fabs(temp[row][col]);
                pivot_row = row;
            }
        }

        if (pivot_row != -1) {
            row_selected[pivot_row] = true;
            rank++;

            for (int row = 0; row < rows; ++row) {
                if (row != pivot_row && fabs(temp[row][col]) > EPSILON) {
                    double factor = temp[row][col] / temp[pivot_row][col];
                    for (int c = col; c < cols; ++c) {
                        temp[row][c] -= factor * temp[pivot_row][c];
                    }
                }
            }
        }
    }
    return rank;
}

PhaseResultC analyze_phase_system_c(const PhaseSystemC* sys) {
    PhaseResultC res;
    if (sys == NULL || sys->num_species <= 0) {
        res.is_valid_equilibrium = false;
        return res;
    }

    res.rank_reactions = compute_matrix_rank(
        (double(*)[MAX_SPECIES])sys->stoich_matrix, 
        sys->num_reactions, 
        sys->num_species
    );

    res.gibbs_components = sys->num_species - res.rank_reactions - sys->num_constraints;
    if (res.gibbs_components < 1) {
        res.gibbs_components = 1;
    }

    int env_params = sys->is_condensed ? 1 : 2;
    res.degrees_of_freedom = res.gibbs_components - sys->num_phases + env_params;
    res.is_valid_equilibrium = (res.degrees_of_freedom >= 0);

    return res;
}
```

```cpp
#include <vector>
#include <string>
#include <cmath>
#include <optional>
#include <algorithm>
#include <stdexcept>

namespace thermo {

struct PhaseSystem {
    std::vector<std::string> species_names;
    std::vector<std::vector<double>> stoich_matrix;
    int num_constraints = 0;
    int num_phases = 1;
    bool is_condensed = false;
};

struct PhaseResult {
    int independent_reactions = 0;
    int gibbs_components = 0;
    int degrees_of_freedom = 0;
    bool is_valid_equilibrium = false;
    std::string equilibrium_type;
};

class PhaseEquilibriumSolver {
public:
    static std::optional<PhaseResult> analyze(const PhaseSystem& sys) {
        if (sys.species_names.empty() || sys.num_phases < 1) {
            return std::nullopt;
        }

        const int num_species = static_cast<int>(sys.species_names.size());
        const int num_reactions = static_cast<int>(sys.stoich_matrix.size());

        int rank = calculate_rank(sys.stoich_matrix, num_species);

        PhaseResult res;
        res.independent_reactions = rank;
        res.gibbs_components = std::max(1, num_species - rank - sys.num_constraints);

        const int env_params = sys.is_condensed ? 1 : 2;
        res.degrees_of_freedom = res.gibbs_components - sys.num_phases + env_params;
        res.is_valid_equilibrium = (res.degrees_of_freedom >= 0);

        if (res.degrees_of_freedom < 0) {
            res.equilibrium_type = "Перевизначена (термодинамічно неможлива рівновага)";
        } else if (res.degrees_of_freedom == 0) {
            res.equilibrium_type = "Нонваріантна (F = 0, інваріантна точка)";
        } else if (res.degrees_of_freedom == 1) {
            res.equilibrium_type = "Моноваріантна (F = 1, моноваріантна лінія)";
        } else if (res.degrees_of_freedom == 2) {
            res.equilibrium_type = "Диваріантна (F = 2, двовимірна область)";
        } else {
            res.equilibrium_type = "Поліваріантна (F > 2)";
        }

        return res;
    }

private:
    static constexpr double EPSILON = 1e-9;

    static int calculate_rank(std::vector<std::vector<double>> matrix, int num_cols) {
        if (matrix.empty()) return 0;
        const int num_rows = static_cast<int>(matrix.size());

        int rank = 0;
        std::vector<bool> row_selected(num_rows, false);

        for (int col = 0; col < num_cols; ++col) {
            int pivot_row = -1;
            double max_val = EPSILON;

            for (int row = 0; row < num_rows; ++row) {
                if (!row_selected[row] && std::abs(matrix[row][col]) > max_val) {
                    max_val = std::abs(matrix[row][col]);
                    pivot_row = row;
                }
            }

            if (pivot_row != -1) {
                row_selected[pivot_row] = true;
                rank++;

                for (int row = 0; row < num_rows; ++row) {
                    if (row != pivot_row && std::abs(matrix[row][col]) > EPSILON) {
                        double factor = matrix[row][col] / matrix[pivot_row][col];
                        for (int c = col; c < num_cols; ++c) {
                            matrix[row][c] -= factor * matrix[pivot_row][c];
                        }
                    }
                }
            }
        }
        return rank;
    }
};

} // namespace thermo
```

## 3. Детальний аналіз алгоритму зведення матриці та вибору опорних елементів

При розрахунку незалежних хімічних реакцій ключовим математичним кроком є знаходження лінійної незалежності рядків стехіометричної матриці. Кожний рядок описує хімічне рівняння, у якому реагенти мають від'ємні коефіцієнти, а продукти — додатні.

Застосований метод Гаусса працює за такими чіткими кроками:
1. Для кожного стовпчика (що відповідає певній хімічній речовині) шукається рядок із найбільшим за модулем значенням коефіцієнта (пошук головного елемента по стовпчику). Це гарантує максимальну чисельну стійкість проти накопичення похибок округлення плаваючої коми.
2. Якщо найбільший елемент перевищує порівняльний поріг `EPSILON = 1e-9`, обраний рядок фіксується як опорна лінійно незалежна реакція, а ранг матриці збільшується на одиницю.
3. За допомогою знайдененого опорного рядка з усіх інших невибраних рядків виключається поточна змінна шляхом віднімання відповідного кратне значення.
4. Після завершення проходу по всіх стовпчиках підсумковий ранг `rank` дорівнює точній кількості стехіометрично незалежних хімічних реакцій у системі.

Цей підхід дозволяє автоматично виявляти приховані лінійні залежності у великих базах даних хімічних реакцій (наприклад, коли у списку задано десятки підреакцій розкладу й відновлення).

## 4. Оцінка продуктивності, складності та потокобезпечності

З точки зору обчислювальної складності, алгоритм Гауссового виключення для матриці розміром `R × N` вимагає часу `O(N · R²)`, де `R` — кількість реакцій, а `N` — кількість хімічних сполук. Для типових технологічних задач (`N <= 50`, `R <= 30`) час виконання становить менше 5 мікросекунд на один розрахунковий вузол.

Оцінка пам'яті:
- У версії мовою C модуль використовує фіксований стек розміром близько `3 КБ` без динамічного виділення пам'яті через `malloc()`, що робить його придатним для роботи у системах реального часу та вбудованих контролерах (Embedded Controllers).
- У версії мовою C++ використовується чисто функціональний стиль без збереження внутрішнього стану (stateless design). Усі методи є `static` та `const`-коректними. Це гарантує повну потокобезпечність (Thread-Safety), дозволяючи паралельно аналізувати тисячі просторових комірок у гідродинамічних симуляторах (CFD) без блокувань та м'ютексів.

Завдяки відсутності проміжних алокацій динамічної пам'яті у внутрішніх циклах обчислення рангу матриці, алгоритм володіє чудовою лок-фрі скалярністю та мінімальним кеш-міссом на сучасних процесорних архітектурах.

## 5. Графічні обчислення у трикомпонентних системах: Трикутник Гіббса

Для трикомпонентних систем (`K = 3`, наприклад, `H2O - ethanol - benzene`) склад будь-якої фази виражається трьома мольними частками `x[A]`, `x[B]`, `x[C]`, чия сума дорівнює одиниці (`x[A] + x[B] + x[C] = 1`). 

Графічне перетворення мольних часток у декартові координати `(X, Y)` на площині трикутника Гіббса-Розебома з довжиною сторони `L = 1` здійснюється за афінними формулами:

```
X = x[B] + 0.5 · x[C]                   [координата X у трикутнику Гіббса]

Y = (√3 / 2) · x[C]                      [координата Y у трикутнику Гіббса]
```

Обернене перетворення координат `(X, Y)` точки на графічній діаграмі у мольні частки компонентів виконується алгебраїчно:

```
x[C] = (2 / √3) · Y                      [мольна частка третього компонента C]

x[B] = X - 0.5 · x[C]                   [мольна частка другого компонента B]

x[A] = 1 - x[B] - x[C]                  [мольна частка першого компонента A]
```

Ці графічні перетворення використовуються в алгоритмах генерації векторних діаграм розчинності, побудови нод (конод) та визначення кривавих меж фазового розшарування (бінодалей).

## 6. Методологія CALPHAD та прямої мінімізації вільної енергії

Сучасний комп'ютерний розрахунок двовимірних та тривимірних діаграм стану матеріалів базується на підході CALPHAD (Calculation of Phase Diagrams). Метод CALPHAD поєднує експериментальні термодинамічні дані з чисельною мінімізацією енергії Гіббса.

Загальна термодинамічна функція енергії Гіббса системи `G[sys]` визначається сумою внесків усіх фаз:

```
G[sys](T, P, n) = ∑ [ n^(j) · g^(j)(T, P, x^(j)) ]  [сумарна енергія Гіббса системи]
```

де `n^(j)` — мольна кількість фази `j`, а `g^(j)` — мольна вільна енергія Гіббса відповідної фази.

Оптимізаційний модуль розв'язує задачу опуклого програмування з обмеженнями збереження кількості елементів:

```
min G[sys]  при умовах:  ∑ [ n^(j) · x[i]^(j) ] = N[i]^(0)  та  n^(j) >= 0
```

Перед початком чисельної мінімізації модуль оцінки варіантності на основі правила фаз Гіббса перевіряє, скільки фаз одночасно можуть бути активними у поточній векторизованій точці. Якщо алгоритм виявляє, що оптимізаційний крок спробував створити додаткову фазу при `F = 0`, система фіксує нонваріантну точку (наприклад, евтектичну ізотерму) та переходить до розрахунку трифазного балансу.

При побудові двофазних нод (tie-lines) чисельний алгоритм визначає кінцеві точки співіснування фаз `x[L]` та `x[α]`, розраховуючи їхні відносні мольні маси за правилом важеля `n[L] = (x[0] - x[α]) / (x[L] - x[α])`. Це забезпечує точне збереження маси між співіснуючими фазами.

## 7. Алгоритм розрахунку фазового розщеплення (Flash Calculation)

Після підтвердження коректності варіантності системи (`F >= 0`) термодинамічний симулятор виконує обчислення фазових часток та складів співіснуючих фаз. Для двофазної суміші рідина-пара застосовують алгоритм розщеплення (Flash Calculation), що базується на рівнянні Рачфорда-Райса (Rachford-Rice Equation).

Нехай `z[i]` — валовий (загальний) склад компонента `i` у вихідній суміші, `x[i]` — мольна частка у рідкій фазі, `y[i]` — мольна частка у газовій фазі, а `V` — відносна мольна частка газової фази (`0 <= V <= 1`).

Коефіцієнти фазового розподілу (K-values) визначаються як `K[i] = y[i] / x[i]`. Зі збереження маси та умов нормування складів `∑ x[i] = 1` та `∑ y[i] = 1` випливає фундаментальне рівняння Рачфорда-Райса:

```
f(V) = ∑ [ z[i] · (K[i] - 1) / (1 + V · (K[i] - 1)) ] = 0 [рівняння Рачфорда-Райса]
```

Функція `f(V)` є монотонно спадною по `V` у фізичному інтервалі `[0, 1]`. Чисельне розв'язання цього рівняння здійснюється методом Ньютона-Рафсона.

Схема чисельного кроку Ньютона-Рафсона має вигляд:

```
V[k+1] = V[k] - f(V[k]) / f'(V[k])      [ітераційний крок Ньютона-Рафсона]
```

де похідна функції `f'(V)` обчислюється аналітично:

```
f'(V) = - ∑ [ z[i] · (K[i] - 1)² / (1 + V · (K[i] - 1))² ] [похідна функції Рачфорда-Райса]
```

Знаючи розраховану частку парової фази `V`, мольні частки компонентів у рідкій та газовій фазах відновлюються алгебраїчно:

```
x[i] = z[i] / (1 + V · (K[i] - 1))        [склад рідкої фази]

y[i] = K[i] · x[i]                        [склад газової фази]
```

Цей чисельний алгоритм володіє квадратичною швидкістю збіжності і вимагає зазвичай від 4 до 8 ітерацій для досягнення точності `1e-12` по масовому балансу.

## 8. Чисельний контроль стійкості матриці Гессе

У ході ітераційної мінімізації вільної енергії Гіббса виникають випадки, коли розчин наближається до межі термодинамічної стійкості (спінодалі). Для чисельної перевірки додатної визначеності матриці Гессе `H[i, j] = (∂²G / ∂N[i] ∂N[j])` застосовують розклад Чолеського (Cholesky Decomposition) `H = L · Lᵀ`.

Якщо на певній ітерації елемент на головній діагоналі матриці `L` стає від'ємним або близьким до нуля (`L[k, k] < 1e-12`), це однозначно сигналізує про втрату додатної визначеності гесіана. Наслідком є фазовий розпад однорідної фази на дві співіснуючі області. Алгоритм у цій точці автоматично збільшує значення числа фаз `P -> P + 1` та перераховує число ступенів вільності `F`, запобігаючи зацикленню нелінійного оптимізатора.

## 9. Покрокове простеження виконання програмного коду

Простежимо виконання розрахунку у класі `PhaseEquilibriumSolver` при виклику статичного методу `analyze()`:

1. **Валідація вхідних даних**: Вхідний об'єкт `PhaseSystem` перевіряється на порожнечу списку речовин `species_names.empty()` та коректність числа фаз `num_phases >= 1`. У разі невілідних даних метод повертає порожній `std::nullopt`.
2. **Копіювання стехіометричної матриці**: Вектор векторів `stoich_matrix` передається за значенням у приватний метод `calculate_rank()`. Це створює локальну робочу копію матриці, що запобігає модифікації оригінальних даних користувача під час виконання Гауссових перетворень.
3. **Обчислення рангу**: Внутрішній цикл здійснює пошук максимального за модулем елемента серед невибраних рядків `row_selected[row] == false`. Знайдений опорний рядок маркується прапором `true`, а інші рядки обнуляються по поточному стовпчику.
4. **Обчислення компонентів Гіббса**: Кількість незалежних компонентів розраховується як `res.gibbs_components = std::max(1, num_species - rank - sys.num_constraints)`. Захисний виклик `std::max(1, ...)` запобігає виникненню некоректних нульових значення компонентів.
5. **Розрахунок варіантності та формування результату**: Розраховується `degrees_of_freedom` з урахуванням прапора `is_condensed` та формується текстовий рядок `equilibrium_type` для відображення термодинамічного статусу у користувацькому інтерфейсі.

## 10. Детальний аналіз чотирьох прикладних фізико-хімічних систем

Для демонстрації універсальності алгоритму розглянемо чотири практичні приклади з хімічної інженерії, металургії, електрохімії та геохімії.

### Приклад А: Паровий реформінг метану з утворенням сажі

Розглянемо хімічний реактор парової конверсії метану, у якому синтезується водневе паливо. 

У газовій фазі містяться `N[spec] = 5` речовин: `CH4`, `H2O`, `CO`, `CO2`, `H2`.
Сформулюємо потенційні хімічні реакції:
1. `CH4 + H2O ⇌ CO + 3 H2` (паровий реформінг);
2. `CO + H2O ⇌ CO2 + H2` (реакція зсуву водяного газу);
3. `CH4 + 2 H2O ⇌ CO2 + 4 H2` (сумарний реформінг).

Запишемо стехіометричну матрицю `A` розміром `3 × 5` для порядку компонентів `[CH4, H2O, CO, CO2, H2]`:
- Рядок 1: `[-1, -1, +1,  0, +3]`
- Рядок 2: `[ 0, -1, -1, +1, +1]`
- Рядок 3: `[-1, -2,  0, +1, +4]`

Застосування Гауссового алгоритму зведення матриці показує, що третій рядок є точним сумарним результатом перших двох (`Рядок 3 = Рядок 1 + Рядок 2`). Отже, ранг матриці дорівнює `R = rank(A) = 2`. У системі є лише дві незалежні реакції.

Оскільки початкових концентраційних обмежень немає (`q = 0`), число незалежних компонентів Гіббса становить:
```
K = 5 - 2 - 0 = 3 компоненти (незалежні елементи: C, H, O)
```

За наявності тільки однієї газової фази (`P = 1`) число ступенів вільності дорівнює:
```
F = 3 - 1 + 2 = 4 (температура T, тиск P та мольні частки двох газових компонентів)
```

Якщо при зниженні температури починає випадати твердий вуглець (сажа `C(s)`), з'являється друга фаза (`P = 2`) та нова хімічна реакція `C + H2O ⇌ CO + H2`. Ранг матриці збільшується до `R = 3`, число компонентів залишається `K = 6 - 3 = 3`, а варіантність зменшується до `F = 3 - 2 + 2 = 3`.

### Приклад Б: Металургійна система залізо-вуглець (Fe-C)

Розглянемо евтектоїдне перетворення у сталі при температурі `727 °C`.

У системі присутні дві елементні речовини (`Fe` та `C`), які утворюють `K = 2` незалежні компоненти. При евтектоїдній температурі у твердому стані співіснують три фази (`P = 3`):
1. Аустеніт `γ-Fe(C)` (твердий розчин вуглецю в гранецентрованому залізі);
2. Феррит `α-Fe` (об'ємноцентроване залізо з низькою розчинністю вуглецю);
3. Цементит `Fe3C` (карбід заліза).

Оскільки процес кристалізації та термообробки сталі відбувається при атмосферному тиску (`P = const`), застосовуємо приведене правило фаз для конденсованих систем (`F[cond] = K - P + 1`):
```
F[cond] = 2 - 3 + 1 = 0 (нонваріантна евтектоїдна точка)
```

Значення `F[cond] = 0` означає, що евтектоїдне перетворення аустеніту на перліт (суміш ферриту та цементиту) перебігає при строго фіксованій температурі `727 °C` та строго визначеному складі аустеніту (`0.76 % C`). Доки не зникне остання гранула аустеніту, температура сплаву не може змінитися.

### Приклад В: Синтез аміаку за методом Габера-Боша

У промисловому реакторі синтезу аміаку під тиском `200 атм` перебуває суміш `N[spec] = 4` речовин: `N2`, `H2`, `NH3` та інертного аргону `Ar`.

Перебігає одна хімічна реакція: `N2 + 3 H2 ⇌ 2 NH3` (`R = 1`).

Якщо вихідний газ подається у суворо стехіометричному співвідношенні азоту та водню 1:3 (`N[H2] = 3 · N[N2]`), виникає одне початкове концентраційне обмеження (`q = 1`).

Розраховуємо число незалежних компонентів Гіббса:
```
K = 4 - 1 - 1 = 2 компоненти (суміш [N2+3H2] та інертний Ar)
```

При однофазному газовому стані (`P = 1`) варіантність системи становить:
```
F = 2 - 1 + 2 = 3 (температура T, тиск P та концентрація інертного аргону)
```

Якщо аміак охолоджується і починає конденсуватися у рідку фазу (`P = 2`), число ступенів вільності зменшується до `F = 2 - 2 + 2 = 2`. Це дозволяє технологічному контролеру точно управляти виходом рідкого аміаку шляхом регулювання температури конденсатора та робочого тиску.

### Приклад Г: Електрохімічна гетерогенна система розчинення вапняку

Розглянемо систему взаємодії вапняку з вуглекислою водою в умовах замкненого карстового пласта.

У системі присутні `N[spec] = 7` речовин: `CaCO3(s)`, `H2O(l)`, `CO2(g)`, `Ca²⁺(aq)`, `HCO3⁻(aq)`, `H⁺(aq)`, `OH⁻(aq)`.

У системі перебігають 3 незалежні хімічні рівноваги (`R = 3`):
1. `CaCO3(s) + H⁺ ⇌ Ca²⁺ + HCO3⁻`
2. `CO2(g) + H2O ⇌ H⁺ + HCO3⁻`
3. `H2O ⇌ H⁺ + OH⁻`

Внаслідок вимоги електронетральності рідкого розчину виникає одне обов'язкове обмеження (`q = 1`):
`2 · [Ca²⁺] + [H⁺] = [HCO3⁻] + [OH⁻]`

Число незалежних компонентів Гіббса дорівнює:
```
K = 7 - 3 - 1 = 3 компоненти (CaCO3, H2O, CO2)
```

При співіснуванні трьох фаз (`P = 3`: твердий `CaCO3`, рідкий розчин, газова фаза `CO2`) число ступенів вільності дорівнює:
```
F = 3 - 3 + 2 = 2 (температура T та тиск пари CO2)
```

Задавши температуру та парціальний тиск вуглекислого газу, хімічний склад водного розчину та розчинність вапняку стають строго визначеними.

## 11. Інтеграція у динамічні симулятори та обробка фазових меж

У динамічних симуляторах хіміко-технологічних процесів аналіз варіантності інтегрується у загальний цикл інтегрування систем диференціально-алгебраїчних рівнянь (DAE).

Процес симуляції виконується за такими етапами:
1. **Ініціалізація векторів стану**: Отримання поточних температур, тисків та мольних кількостей речовин з попереднього часового кроку.
2. **Перевірка наявних фаз**: Сканування фазового складу та виявлення фаз, чиї мольні маси перевищують поріг існування (`N[phase] > 1e-12`).
3. **Виклики модуля `analyze()`**: Отримання значення `F`. Якщо `F < 0`, видається попередження про перевизначеність та виконується автоматична процедура схлопування фази (Phase Disappearance Algorithm).
4. **Розрахунок нових фазових рівноваг**: При `F >= 0` розраховуються коефіцієнти активності `γ[i]` та фугітивності `ϕ[i]` з наступною корекцією складу фаз.

## 12. Чисельні застереження та обробка крайових випадків

Під час практичної розробки термодинамічного програмного забезпечення слід враховувати такі чисельні особливості:

1. **Поріг машинного нуля `EPSILON`**: Для визначення рангу стехіометричної матриці слід використовувати абсолютне значення `EPSILON = 1e-9`. Занадто мале значення (наприклад `1e-16`) може призвести до того, що чисельні похибки округлення плаваючої коми перетворять лінійно залежну реакцію на незалежну, хибно збільшивши ранг матриці `R`.
2. **Електронетральність розчинів електролітів**: У водних розчинах солей, кислот та основ сума позитивних і негативних електричних зарядів повинна дорівнювати нулю (`∑ z[i] · C[i] = 0`). Закон збереження заряду додає одне обов'язкове початкове обмеження `q`, яке зменшує число компонентів Гіббса `K` на одиницю.
3. **Зародження та зникнення фаз**: У динамічних симуляція хімічних реакторів число фаз `P` може змінюватися у часі. Якщо внаслідок розрахунку виникає стан `F < 0`, алгоритм повинен зафіксувати термодинамічну неможливість співіснування такої кількості фаз та вилучити виснажену фазу з розрахункової схеми.
4. **Критичні точки та азеотропи**: У точках азеотропів та критичних точках змішуваності виникають додаткові термодинамічні рівності (`x[V] = x[L]`), що зменшує ефективне число інтенсивних змінних на одиницю та вимагає спеціальної обробки у модулі аналізу варіантності.
