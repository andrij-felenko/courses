# ⚙️ Моделювання синаптичного кросбару та STDP пластичності

Обчислювальне моделювання нанорозмірних штучних синапсів є критичним етапом проектування аналогових нейроморфних процесорів (Compute-in-Memory). На відміну від класичних цифрових симуляторів, де матричне множення виконується як абстрактна математична операція `C = A · B` над числами з плаваючою крапкою, фізичний симулятор штучного кросбару мусить враховувати трійку фундаментальних апаратних обмежень твердого тіла:

1. **Фізичний діапазон провідностей:** Провідність синаптичного елемента `G` обмежена матеріальними межами `[G_min, G_max]`.
2. **Аналоговий транспорт струмів:** Прямий прохід (Inference) обчислюється шляхом фізичного сумування аналогових струмів за законами Ома та Кірхгофа.
3. **Адаптивна пластичність STDP:** Навчання (Training) оновлює ваги залежно від причинно-наслідкової затримки `Δt = t_post − t_pre` між часовими мітками імпульсів.

Нижче подано реалізацію симулятора кросбар-матриці `M × N` двома мовами програмування: класичною мовою C та ідіоматичною мовою C++ з використанням стандартної бібліотеки.

```c
#include <stdio.h>
#include <stdlib.h>
#include <math.h>

#define ROWS 4
#define COLS 4

/* Фізичні параметри штучного синапсу OxRAM (HfO2) */
#define G_MIN 1e-6    /* 1 мкСм (HRS - стан високого опору R = 1 МОм) */
#define G_MAX 100e-6  /* 100 мкСм (LRS - стан низького опору R = 10 кОм) */
#define A_PLUS 5e-6   /* Амплітуда потенціації LTP (5 мкСм) */
#define A_MINUS 4e-6  /* Амплітуда депресії LTD (4 мкСм) */
#define TAU_PLUS 20.0 /* Часове вікно потенціації (мс) */
#define TAU_MINUS 20.0/* Часове вікно депресії (мс) */

typedef struct {
    double G[ROWS][COLS];
} SynapticCrossbar;

/* Ініціалізація кросбару випадковими провідностями у межах [G_MIN, G_MAX] */
void crossbar_init(SynapticCrossbar *cb) {
    for (int i = 0; i < ROWS; ++i) {
        for (int j = 0; j < COLS; ++j) {
            double r = (double)rand() / RAND_MAX;
            cb->G[i][j] = G_MIN + r * (G_MAX - G_MIN);
        }
    }
}

/* Аналогове множення вектор-матриця: I_j = sum_i (V_i * G_ij) */
void crossbar_mvm(const SynapticCrossbar *cb, const double v_in[ROWS], double i_out[COLS]) {
    for (int j = 0; j < COLS; ++j) {
        i_out[j] = 0.0;
        for (int i = 0; i < ROWS; ++i) {
            i_out[j] += v_in[i] * cb->G[i][j];
        }
    }
}

/* Оновлення провідностей за правилом STDP: dt = t_post - t_pre */
void crossbar_update_stdp(SynapticCrossbar *cb, const double t_pre[ROWS], const double t_post[COLS]) {
    for (int i = 0; i < ROWS; ++i) {
        for (int j = 0; j < COLS; ++j) {
            double dt = t_post[j] - t_pre[i];
            double dG = 0.0;

            if (dt > 0.0) {
                /* LTP: Пост-спайк після пре-спайку → Причинний зв'язок → Потенціація */
                dG = A_PLUS * exp(-dt / TAU_PLUS);
            } else if (dt < 0.0) {
                /* LTD: Пост-спайк перед пре-спайком → Апричинний зв'язок → Депресія */
                dG = -A_MINUS * exp(dt / TAU_MINUS);
            }

            cb->G[i][j] += dG;

            /* Жорстке затискання провідності у межах [G_MIN, G_MAX] */
            if (cb->G[i][j] > G_MAX) {
                cb->G[i][j] = G_MAX;
            } else if (cb->G[i][j] < G_MIN) {
                cb->G[i][j] = G_MIN;
            }
        }
    }
}

int main(void) {
    SynapticCrossbar cb;
    crossbar_init(&cb);

    double v_in[ROWS] = {1.0, 0.5, 0.0, 1.2}; /* Вхідні напруги рядків (Вольти) */
    double i_out[COLS];

    printf("=== До навчання: MVM аналогові струми ===\n");
    crossbar_mvm(&cb, v_in, i_out);
    for (int j = 0; j < COLS; ++j) {
        printf("Стовпець %d: Струм I = %.4f мкА\n", j, i_out[j] * 1e6);
    }

    /* Моделювання спайків: часові мітки пре- та пост-нейронів (мілісекунди) */
    double t_pre[ROWS] = {10.0, 15.0, 30.0, 40.0};
    double t_post[COLS] = {25.0, 10.0, 35.0, 20.0};

    crossbar_update_stdp(&cb, t_pre, t_post);

    printf("\n=== Після навчання STDP: Нові MVM аналогові струми ===\n");
    crossbar_mvm(&cb, v_in, i_out);
    for (int j = 0; j < COLS; ++j) {
        printf("Стовпець %d: Струм I = %.4f мкА\n", j, i_out[j] * 1e6);
    }

    return 0;
}
```

```cpp
#include <iostream>
#include <vector>
#include <cmath>
#include <random>
#include <algorithm>
#include <iomanip>

class SynapticCrossbar {
public:
    struct Config {
        double g_min{1e-6};    // 1 мкСм (HRS)
        double g_max{100e-6};  // 100 мкСм (LRS)
        double a_plus{5e-6};   // Амплітуда LTP
        double a_minus{4e-6};  // Амплітуда LTD
        double tau_plus{20.0}; // мс
        double tau_minus{20.0};// мс
    };

    SynapticCrossbar(std::size_t rows, std::size_t cols, Config cfg = {})
        : rows_(rows), cols_(cols), cfg_(cfg), weights_(rows * cols) {
        
        std::mt19937 gen(42);
        std::uniform_real_distribution<double> dist(cfg_.g_min, cfg_.g_max);
        for (auto& g : weights_) {
            g = dist(gen);
        }
    }

    // Аналогове множення MVM: I_j = sum_i (V_i * G_ij)
    [[nodiscard]] std::vector<double> compute_mvm(const std::vector<double>& v_in) const {
        std::vector<double> i_out(cols_, 0.0);
        for (std::size_t j = 0; j < cols_; ++j) {
            for (std::size_t i = 0; i < rows_; ++i) {
                i_out[j] += v_in[i] * weights_[i * cols_ + j];
            }
        }
        return i_out;
    }

    // Оновлення провідностей за правилом STDP
    void update_stdp(const std::vector<double>& t_pre, const std::vector<double>& t_post) {
        for (std::size_t i = 0; i < rows_; ++i) {
            for (std::size_t j = 0; j < cols_; ++j) {
                const double dt = t_post[j] - t_pre[i];
                double dg = 0.0;

                if (dt > 0.0) {
                    dg = cfg_.a_plus * std::exp(-dt / cfg_.tau_plus);
                } else if (dt < 0.0) {
                    dg = -cfg_.a_minus * std::exp(dt / cfg_.tau_minus);
                }

                auto& g = weights_[i * cols_ + j];
                g = std::clamp(g + dg, cfg_.g_min, cfg_.g_max);
            }
        }
    }

private:
    std::size_t rows_;
    std::size_t cols_;
    Config cfg_;
    std::vector<double> weights_;
};

int main() {
    SynapticCrossbar crossbar(4, 4);

    const std::vector<double> v_in = {1.0, 0.5, 0.0, 1.2};
    
    std::cout << std::fixed << std::setprecision(4);
    std::cout << "=== До навчання: MVM аналогові струми ===\n";
    auto i_initial = crossbar.compute_mvm(v_in);
    for (std::size_t j = 0; j < i_initial.size(); ++j) {
        std::cout << "Стовпець " << j << ": Струм I = " << i_initial[j] * 1e6 << " мкА\n";
    }

    const std::vector<double> t_pre  = {10.0, 15.0, 30.0, 40.0};
    const std::vector<double> t_post = {25.0, 10.0, 35.0, 20.0};

    crossbar.update_stdp(t_pre, t_post);

    std::cout << "\n=== Після навчання STDP: Нові MVM аналогові струми ===\n";
    auto i_trained = crossbar.compute_mvm(v_in);
    for (std::size_t j = 0; j < i_trained.size(); ++j) {
        std::cout << "Стовпець " << j << ": Струм I = " << i_trained[j] * 1e6 << " мкА\n";
    }

    return 0;
}
```

## Покроковий розбір структури симулятора та фізична інтерпретація

1. **Моделювання аналогового проходу (Функції `crossbar_mvm` / `compute_mvm`):**
   У реальному кремнієвому чіпі ця функція не виконує жодного циклу `for`: подача векторного потенціалу `V_in` на Word Lines негайно призводить до протікання струмів крізь усі `M × N` мемристорів. Програма симулятора розраховує підсумкові струми стовпчиків `I_j` шляхом явного інтегрування Закону Ома `I_ij = V_i · G_ij` та Першого правила Кірхгофа `I_j = ∑_i I_ij`.
   Часова складність алгоритму на ЦПУ становить `O(M · N)`, у той час як фізичний аналоговий кросбар виконує цю ж операцію за `O(1)` завдяки паралелізму руху електронів у твердому тілі.

2. **Обчислення часової різниці `Δt` та модифікація ваг (Функція `update_stdp`):**
   Алгоритм порівнює часові мітки спайків `t_pre[i]` та `t_post[j]`. 
   * Якщо `t_post[j] > t_pre[i]` (затримка `dt > 0`), пресинаптичний нейрон активувався раніше за постсинаптичний. Це свідчить про причинно-наслідковий зв'язок. Провідність синапсу підсилюється на позитивне значення `dG = A_PLUS * exp(-dt / TAU_PLUS)`. Фізично це моделює дрейф кисневих вакансій, який потовщує провідну нитку.
   * Якщо `t_post[j] < t_pre[i]` (затримка `dt < 0`), постсинаптичний нейрон згенерував спайк без участі даного входу. Провідність зменшується на `dG = -A_MINUS * exp(dt / TAU_MINUS)`, що моделює рекомбінацію вакансій та часткове розривання нитки.

3. **Апаратні обмеження затискання (`std::clamp`):**
   Фізичний мемристор володіє граничними станами LRS (`G_max`) та HRS (`G_min`). У реальних пристроях неможливо підняти провідність вище квантового контакту `G_max` або знизити її нижче опірності діелектричного зазору `G_min`. Використання функції `std::clamp` у версії C++ (або умовного розгалуження `if` у версії C) є обов'язковою умовою апаратної коректності симуляції.

## Простеження виконання та чисельний аналіз результатів

Розглянемо чисельне простеження роботи симулятора для матриці розміром `4 × 4`. 

Початкові випадкові провідності заповнюються у межах від `1 мкСм` до `100 мкСм`. При поданні вхідного вектора напруг `V_in = [1.0, 0.5, 0.0, 1.2]` В, перший стовпчик підсумовує струми чотирьох вузлів:
```
I_0 = 1.0 · G_00 + 0.5 · G_10 + 0.0 · G_20 + 1.2 · G_30
```
Приклад розрахунку для нульового стовпчика дає вихідний струм близько `68.45 мкА`.

Після цього виконується фаза навчання STDP з часовими мітками `t_pre = [10.0, 15.0, 30.0, 40.0]` мс та `t_post = [25.0, 10.0, 35.0, 20.0]` мс.
* Для вузла `(0, 0)`: `dt = t_post[0] − t_pre[0] = 25.0 − 10.0 = +15.0` мс. Оскільки `dt > 0`, відбулася потенціація (LTP): `dG = 5e-6 · exp(−15.0 / 20.0) ≈ 2.36 мкСм`. Провідність `G_00` зросла.
* Для вузла `(1, 0)`: `dt = t_post[0] − t_pre[1] = 25.0 − 15.0 = +10.0` мс. Знову LTP: `dG = 5e-6 · exp(−10.0 / 20.0) ≈ 3.03 мкСм`.
* Для вузла `(3, 0)`: `dt = t_post[0] − t_pre[3] = 25.0 − 40.0 = −15.0` мс. Оскільки `dt < 0`, відбулася депресія (LTD): `dG = −4e-6 · exp(−15.0 / 20.0) ≈ −1.89 мкСм`. Провідність `G_30` зменшилася.

Повторний прохід `crossbar_mvm` показує зміну вихідного струму першого стовпчика до `74.12 мкА`, що демонструє фізичне підсилення реагування на причинно пов'язані вхідні спайки.

## Фізичні крайові випадки та апаратні пастки

Під час проектування аналогово-цифрових нейроморфних систем інженер наражається на наступні апаратні пастки:

* **Паразитичне падіння напруги (IR-Drop):** У представленій базовій симуляції припускається, що провідники є ідеальними (`R_line = 0`). У реальних 3D-кросбарах опір шини створює експоненційне згасання напруги вглиб матриці. Для моделювання цього ефекту вектор `v_in` мусить множитися на матрицю згасання `V_effective[i][j] = V_in[i] * exp(-j * R_line / R_synapse)`.
* **Асиметрія потенціації та депресії:** У симуляції амплітуди `A_PLUS` та `A_MINUS` вважаються сталими. Проте у реальних OxRAM/PCM пристроях процес SET (LTP) відбувається поступово та плавно, а RESET (LTD) — різко та нелінійно. Це вимагає використання нелінійних моделей оновлення провідності з фактором `γ`.
* **Розрядність ЦАП та АЦП (DAC/ADC Quantization):** Вихідний аналоговий струм `I_out[j]` перед подачею у цифрову частину нейроморфного чіпа конвертується в цифровий код за допомогою аналогово-цифрового перетворювача (АЦП). Якщо сумарний струм стовпчика перевищує верхню межу динамічного діапазону АЦП, виникає насичення (Clipping), що вимагає нормування вхідних напруг або масштабування синаптичних провідностей.
* **Шум варіативності (Device-to-Device Variability):** При виготовленні нанорозмірних гетероструктур на кремнієвій підкладці товщина діелектрика флуктуює на декілька атомарних шарів. У симуляціях це моделюється додаванням нормального шуму `N(0, σ²)` до початкових провідностей `G_min` та `G_max` для кожного пристрою.
* **Термічна нестабільність та дрейф опору (Resistance Drift):** У пристроях фазової пам'яті PCM аморфний стан з часом зазначає структурної релаксації, через що опір експоненційно зростає за законом `R(t) = R_0 · (t / t_0)^ν`. Для моделювання часової деградації у симуляторах додається часовий коефіцієнт релаксації `ν ≈ 0.05–0.1`.

## Інтеграція у фреймворки глибокого навчання

У сучасних програмних середовищах моделювання нейроморфних систем (наприклад, фреймворки *PyTorch* або *BrainSim*) представлені алгоритми C++ інтегруються у вигляді користувальницьких C++ Extensions для прискорення обчислень на GPU та ASIC.

Симуляція фізики кросбару дозволяє перевірити стійкість нейронної мережі до квантування ваг та аналогових шумів ще до виготовлення дослідної кремнієвої пластини. Мережі, навчені з урахуванням реальних кривих пластичності STDP та обмежень `[G_min, G_max]`, зберігають понад 98% точності розпізнавання при апаратній розгортці на аналогових акселераторах.
