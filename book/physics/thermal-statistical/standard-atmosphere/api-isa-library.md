# 📑 Інтерфейс бібліотеки ISA: Довідник API C та C++

Бібліотечний інтерфейс системи Стандартної атмосфери ISA надає стандартизований набір типів даних, фундаментальних констант та обчислювальних функцій для розрахунку фізичних параметрів повітряного океану в діапазоні висот від 0 до 86 кілометрів.

Цей довідник описує повну поверхню інтерфейсу бібліотеки (Application Programming Interface — API) мовами C (C99/C11) та C++ (C++20), включаючи семантику викликів, одиниці вимірювання, граничні умови, обробку помилок та гарантії продуктивності.

### Модульна структура та підключення бібліотеки

Бібліотека поставляється у вигляді заголовочних файлів `isa.h` (для C) та `isa.hpp` (для C++), а також єдиного реалізаційного файла `isa.c` / `isa.cpp`. Модуль є повністю самодостатнім, не вимагає зовнішніх сторонніх залежностей окрім стандартної математичної бібліотеки `libm` (`<math.h>` / `<cmath>`) і призначений для безпосереднього вбудовування в бортові системи авіоніки, безпілотні літальні апарати (БПЛА), обчислювачі повітряних сигналів (Air Data Computers) та графічні фізичні рушії.

Основні фізичні принципи побудови бібліотеки:
- **Нульове динамічне виділення пам'яті:** Усі структури даних мають суворо визначений розмір на етапі компіляції і розміщуються у стек-фреймі виклику або статичній пам'яті. Виклики `malloc`, `free`, `new` або `delete` повністю відсутні.
- **Абсолютна потокобезпечність та реентерабельність:** Модуль не має глобального змінюваного стану (mutable global state). Усі функції є чистими (pure functions), що дозволяє безпечно викликати їх одночасно з багатьох паралельних обчислювальних потоків або переривань таймера без блокувань і м'ютексів.
- **Детермінованість часу виконання:** Алгоритм локалізації атмосферного шару використовує послідовний або бінарний пошук у фіксованому масиві з 7 елементів, що гарантує суворо обмежений час виконання за фіксовану кількість тактів процесора (підтверджено для стандартів авіаційної безпеки DO-178C / MISRA C:2012).

:::tabs
```c
/*
 * isa.h — C API для роботи з Міжнародною стандартною атмосферою (ISA ISO 2533)
 */
#ifndef ISA_LIBRARY_H
#define ISA_LIBRARY_H

#include <stdbool.h>
#include <stddef.h>

#ifdef __cplusplus
extern "C" {
#endif

/* ── Константи ISA ISO 2533:1975 ────────────────────────────────────────── */
#define ISA_G0            9.80665      /* Стандартне прискорення вільного падіння, м/с² */
#define ISA_R_SPEC        287.05287    /* Питома газова стала сухого повітря, Дж/(кг·К) */
#define ISA_EARTH_RADIUS  6356766.0    /* Ефективний геопотенціальний радіус Землі, м */
#define ISA_GAMMA_AIR     1.40         /* Показник адіабати сухого повітря */
#define ISA_SUTHERLAND_S  110.4        /* Константа Сазерленда, K */
#define ISA_SUTHERLAND_B  1.458e-6     /* Коефіцієнт в'язкості Сазерленда, кг/(м·с·K¹/²) */

#define ISA_P0            101325.0     /* Тиск на рівні моря, Па */
#define ISA_T0            288.15       /* Температура на рівні моря, K */
#define ISA_RHO0          1.2250       /* Густина на рівні моря, кг/м³ */
#define ISA_MAX_H_METERS  84852.0      /* Максимальна геопотенціальна висота, м */

/* ── Перелік кодових статусах помилок ────────────────────────────────────── */
typedef enum {
    ISA_SUCCESS = 0,               /* Успішне виконання */
    ISA_ERROR_NULL_POINTER = -1,   /* Передано нульовий вказівник */
    ISA_ERROR_OUT_OF_BOUNDS = -2,  /* Висота поза допустимим діапазоном 0..84852 м */
    ISA_ERROR_INVALID_PARAM = -3   /* Невалідний фізичний параметр */
} isa_error_t;

/* ── Повний термодинамічний стан атмосфери ──────────────────────────────── */
typedef struct {
    double h_geopotential; /* Геопотенціальна висота h, м */
    double z_geometric;    /* Реальна геометрична висота Z, м */
    double temperature;    /* Абсолютна температура T, K */
    double temp_celsius;   /* Температура, °C */
    double pressure;       /* Статичний тиск P, Па */
    double density;        /* Густина повітря ρ, кг/м³ */
    double sound_speed;    /* Місцева швидкість звуку a, м/с */
    double dynamic_visc;   /* Динамічна в'язкість μ, кг/(м·с) */
    double kinematic_visc; /* Кінематична в'язкість ν, м²/с */
} isa_state_t;

/* ── Публічна поверхня C API ────────────────────────────────────────────── */

/* Перетворення геометричної висоти Z у геопотенціальну h */
double isa_z_to_h(double z_geometric_m);

/* Перетворення геопотенціальної висоти h у геометричну Z */
double isa_h_to_z(double h_geopotential_m);

/* Основний розрахунок стану атмосфери за геопотенціальною висотою h */
isa_error_t isa_evaluate(double h_geopotential_m, isa_state_t *result);

/* Обчислення барометричної висоти h_p у тропосфері за тиском P */
isa_error_t isa_get_pressure_altitude(double pressure_pa, double *h_p_m);

/* Обчислення висоти за густиною h_d за фактичним тиском та температурою */
isa_error_t isa_get_density_altitude(double pressure_pa, double temp_k, double *h_d_m);

/* Перерахунок індикаторної швидкості CAS у істинну TAS */
double isa_cas_to_tas(double cas_ms, double density_kg_m3);

/* Перерахунок істинної швидкості TAS у число Маха */
double isa_tas_to_mach(double tas_ms, double sound_speed_ms);

#ifdef __cplusplus
}
#endif

#endif /* ISA_LIBRARY_H */
```
```cpp
// isa.hpp — Modern C++20 API для Міжнародної стандартної атмосфери ISA
#ifndef ISA_LIBRARY_HPP
#define ISA_LIBRARY_HPP

#include <array>
#include <cmath>
#include <optional>
#include <string_view>
#include <concepts>

namespace isa {

// ── Фундаментальні фізичні константи (constexpr) ─────────────────────────
constexpr double g0            = 9.80665;
constexpr double r_spec        = 287.05287;
constexpr double r_earth       = 6356766.0;
constexpr double gamma_air     = 1.40;
constexpr double sutherland_s  = 110.4;
constexpr double sutherland_b  = 1.458e-6;

constexpr double p0            = 101325.0;
constexpr double t0            = 288.15;
constexpr double rho0          = 1.2250;
constexpr double max_h_meters  = 84852.0;

// ── Повний термодинамічний стан атмосфери ────────────────────────────────
struct State {
    double h_geopotential{0.0};
    double z_geometric{0.0};
    double temperature{0.0};
    double temp_celsius{0.0};
    double pressure{0.0};
    double density{0.0};
    double sound_speed{0.0};
    double dynamic_visc{0.0};
    double kinematic_visc{0.0};

    [[nodiscard]] constexpr double pressure_hpa() const noexcept {
        return pressure / 100.0;
    }
};

// ── C++20 Клас обчислювача атмосфери ISA ─────────────────────────────────
class AtmosphereCalculator {
public:
    AtmosphereCalculator() = delete; // Статичний модуль без стану

    [[nodiscard]] static constexpr double z_to_h(double z_m) noexcept {
        return (r_earth * z_m) / (r_earth + z_m);
    }

    [[nodiscard]] static constexpr double h_to_z(double h_m) noexcept {
        if (h_m >= r_earth) return r_earth * 100.0;
        return (r_earth * h_m) / (r_earth - h_m);
    }

    // Розрахунок стану за висотою (повертає std::optional замість винятків)
    [[nodiscard]] static std::optional<State> evaluate(double h_geopotential_m) noexcept;

    // Обернені навігаційні обчислювачі
    [[nodiscard]] static std::optional<double> pressure_altitude(double pressure_pa) noexcept;
    [[nodiscard]] static std::optional<double> density_altitude(double pressure_pa, double temp_k) noexcept;

    // Аеродинамічні перетворення швидкостей
    [[nodiscard]] static constexpr double cas_to_tas(double cas_ms, double density_kg_m3) noexcept {
        if (density_kg_m3 <= 0.0) return cas_ms;
        return cas_ms * std::sqrt(rho0 / density_kg_m3);
    }

    [[nodiscard]] static constexpr double tas_to_mach(double tas_ms, double sound_speed_ms) noexcept {
        if (sound_speed_ms <= 0.0) return 0.0;
        return tas_ms / sound_speed_ms;
    }
};

} // namespace isa

#endif // ISA_LIBRARY_HPP
```

### Детальний опис типів даних та структур

#### 1. Структура стан атмосфери `isa_state_t` / `isa::State`

Структура містить повний набір локальних фізичних та термодинамічних параметрів атмосфери, розрахованих для вказаної геопотенціальної висоти:

- `h_geopotential` (тип `double`): Вхідна геопотенціальна висота `h` у метрах від рівня моря. Повинна знаходитися в діапазоні `0.0 <= h <= 84852.0`.
- `z_geometric` (тип `double`): Перерахована реальна фізична геометрична висота `Z` у метрах, обчислена за дробово-раціональним співвідношенням `Z = (R_E · h) / (R_E - h)`.
- `temperature` (тип `double`): Абсолютна температура повітря `T` у kelvin (K).
- `temp_celsius` (тип `double`): Температура повітря у градусах Цельсія (°C), де `T_C = T - 273.15`.
- `pressure` (тип `double`): Статичний атмосферний тиск `P` у Паскалях (Па). Для переведення в гектопаскалі (гПа / мбар) ділиться на 100.
- `density` (тип `double`): Масова густина повітря `ρ` у кілограмах на кубічний метр (кг/м³).
- `sound_speed` (тип `double`): Адіабатична швидкість поширення звуку `a` у метрах на секунду (м/с).
- `dynamic_visc` (тип `double`): Динамічна в'язкість Сазерленда `μ` у кілограмах на метр-секунду (кг/(м·с) або Па·с).
- `kinematic_visc` (тип `double`): Кінематична в'язкість `ν` у квадратних метрах на секунду (м²/с), розрахована як `ν = μ / ρ`.

#### 2. Перелік помилок `isa_error_t` (C API)

У C API виконання розрахунків повертає числовий код статусу:
- `ISA_SUCCESS` (0): Операція виконана успішно, вихідна структура проініціалізована коректними значеннями.
- `ISA_ERROR_NULL_POINTER` (-1): У функцію передано нульовий вказівник (`NULL`) замість вихідної структури.
- `ISA_ERROR_OUT_OF_BOUNDS` (-2): Вхідне значення геопотенціальної висоти виходить за межі визначеності моделі (`h < 0.0` або `h > 84852.0 м`).
- `ISA_ERROR_INVALID_PARAM` (-3): Передано від'ємний тиск або від'ємну абсолютну температуру при обернених розрахунках.

### Детальний аналіз та семантика функцій бібліотеки

#### `isa_evaluate` / `isa::AtmosphereCalculator::evaluate`

Головна розрахункова функція бібліотеки. Вона обчислює повний термодинамічний стан атмосфери для заданої геопотенціальної висоти `h`.

:::tabs
```c
isa_error_t isa_evaluate(double h_geopotential_m, isa_state_t *result);
```
```cpp
std::optional<isa::State> isa::AtmosphereCalculator::evaluate(double h_geopotential_m) noexcept;
```
:::

- **Вхідні параметри:**
  - `h_geopotential_m` (тип `double`): Геопотенціальна висота `h` у метрах від рівня моря. Необхідна умова: `0.0 <= h_geopotential_m <= 84852.0`.
  - `result` (тип `isa_state_t*`): Вказівник на структуру для запису розрахованих параметрів (тільки у C API).
- **Повертане значення:**
  - У C API повертає `ISA_SUCCESS` (0) у разі успіху. Якщо `result == NULL`, повертає `ISA_ERROR_NULL_POINTER`. Якщо висота виходить за дозволені межі, повертає `ISA_ERROR_OUT_OF_BOUNDS`.
  - У C++ API повертає об'єкт `std::optional<isa::State>`. Якщо вхідні дані знаходяться у допустимому діапазоні, об'єкт містить заповнену структуру `isa::State`. Якщо значення невалідне, повертається `std::nullopt` без генерування винятків.
- **Внутрішній алгоритм:**
  1. Перевіряє вхідний діапазон висоти.
  2. Знаходить відповідний атмосферний шар у таблиці меж 7 шарів ISA.
  3. Розраховує температуру `T(h) = T_b + L · (h - h_b)`.
  4. Обчислює тиск за градієнтною степеневою або ізотермічною експоненційною формулою.
  5. Обчислює густину `ρ = P / (R_spec · T)`.
  6. Обчислює швидкість звуку `a = √(γ · R_spec · T)` та в'язкість за Сазерлендом.
- **Часова та просторова складність:**
  - Часова складність: `O(1)` — алгоритм виконує фіксовану кількість операцій (не більше 7 порівнянь шарів).
  - Простір пам'яті: `O(1)` — використовує лише декілька регістрів ЦП або змінні у стек-фреймі.

#### `isa_z_to_h` та `isa_h_to_z`

Конвертери висоти виконують взаємне перетворення між геометричною висотою `Z` (реальна фізична відстань) та геопотенціальною висотою `h` (математична шкала постійного `g_0`).

:::tabs
```c
double isa_z_to_h(double z_geometric_m);
double isa_h_to_z(double h_geopotential_m);
```
```cpp
constexpr double isa::AtmosphereCalculator::z_to_h(double z_m) noexcept;
constexpr double isa::AtmosphereCalculator::h_to_z(double h_m) noexcept;
```
:::

- **Математичні вирази:**
  - `h = (R_E · Z) / (R_E + Z)`
  - `Z = (R_E · h) / (R_E - h)`
- **Крайові випадки та безпека:**
  - Якщо у функцію `h_to_z` передати висоту `h >= R_E` (що фізично дорівнює радіусу Землі), функція запобігає діленню на нуль і повертає граничне безпечне значення `R_E * 100.0`.
  - У C++14/C++20 обидва методи позначені специфікатором `constexpr`, що дозволяє компілятору обчислювати геометричні перетворення висот безпосередньо під час компіляції програми для константних виразів.

#### `isa_get_pressure_altitude` / `isa::AtmosphereCalculator::pressure_altitude`

Обчислює геопотенціальну барометричну висоту `h_p` (Pressure Altitude) у тропосферному діапазоні (від 0 до 11 000 метрів) за відомим виміряним статичним тиском `P`.

:::tabs
```c
isa_error_t isa_get_pressure_altitude(double pressure_pa, double *h_p_m);
```
```cpp
std::optional<double> isa::AtmosphereCalculator::pressure_altitude(double pressure_pa) noexcept;
```
:::

- **Параметри:**
  - `pressure_pa`: Виміряний статичний атмосферний тиск у Паскалях. Повинен бути суворо додатним `pressure_pa > 0.0`.
  - `h_p_m`: Вказівник на змінну для зберігання обчисленої барометричної висоти в метрах.
- **Математична формула:**
  - `h_p = (T_0 / 0.0065) · (1.0 - (P / P_0)^(1 / 5.25588))`
  - Числовий коефіцієнт `T_0 / 0.0065 = 288.15 / 0.0065 ≈ 44330.77 м`.
- **Галузь застосування:**
  - Функція є серцем цифрових барометричних висотомірів (Air Data Computers) авіалайнерів, цивільних літаків та автопілотів БПЛА.

#### `isa_get_density_altitude` / `isa::AtmosphereCalculator::density_altitude`

Обчислює висоту за густиною `h_d` (Density Altitude) за фактичним виміряним тиском `P` та температурою повітря `T`.

:::tabs
```c
isa_error_t isa_get_density_altitude(double pressure_pa, double temp_k, double *h_d_m);
```
```cpp
std::optional<double> isa::AtmosphereCalculator::density_altitude(double pressure_pa, double temp_k) noexcept;
```
:::

- **Параметри:**
  - `pressure_pa`: Фактичний статичний тиск у Паскалях (`pressure_pa > 0.0`).
  - `temp_k`: Фактична виміряна температура повітря в kelvin (`temp_k > 0.0`).
- **Алгоритм обчислення:**
  1. Розраховується фактична густина повітря за рівнянням стану: `ρ_actual = P / (R_spec · T)`.
  2. Використовується обернена тропосферна формула густини:
     `h_d = (T_0 / 0.0065) · (1.0 - (ρ_actual / ρ_0)^(1 / 4.25588))`
- **Практичний зміст:**
  - Дозволяє екіпажу літака розрахувати реальну підйомну силу крила та довжину розгону під час зльоту в умовах підвищеної температури навколишнього середовища.

#### `isa_cas_to_tas` та `isa_tas_to_mach`

Допоміжні аеродинамічні функції перерахунку швидкостей польоту.

:::tabs
```c
double isa_cas_to_tas(double cas_ms, double density_kg_m3);
double isa_tas_to_mach(double tas_ms, double sound_speed_ms);
```
```cpp
constexpr double isa::AtmosphereCalculator::cas_to_tas(double cas_ms, double density_kg_m3) noexcept;
constexpr double isa::AtmosphereCalculator::tas_to_mach(double tas_ms, double sound_speed_ms) noexcept;
```
:::

- **Параметри та формули:**
  - `cas_ms`: Приладова індикаторна швидкість у м/с.
  - `density_kg_m3`: Густина повітря `ρ` у кг/м³.Формула: `TAS = CAS · √(ρ_0 / ρ)`.
  - `sound_speed_ms`: Локальна швидкість звуку `a` у м/с. Формула: `M = TAS / a`.
- **Захист від некоректних даних:**
  - Якщо `density_kg_m3 <= 0.0`, функція `cas_to_tas` повертає вихідну швидкість `cas_ms`.
  - Якщо `sound_speed_ms <= 0.0`, функція `tas_to_mach` повертає `0.0`.

### Апаратна бінарна сумісність (ABI) та підключення між мовами

Оскільки C є базовим стандартом сумісності бінарного коду (ABI — Application Binary Interface), бібліотека ISA побудована з урахуванням прямої інтеграції в багатомовні програмні комплекси.

1. **Сумісність C та C++ (`extern "C"`):**
   Заголовочний файл `isa.h` містить препроцесорні директиви `#ifdef __cplusplus extern "C" { #endif`. Це гарантує, що при виклику функцій з C++ компілятор не застосовує спотворення імен (name mangling), зберігаючи чистову C-символіку у зкомпільованому об'єктному файлі чи динамічній бібліотеці (`.so` / `.dll`).
2. **Інтеграція з Python (через `ctypes` або `CFFI`):**
   C-функції бібліотеки можуть бути викликані безпосередньо з Python скриптів без написання C-розширень за допомогою стандартного модуля `ctypes`. Це особливо зручно для пост-обробки телеметрії польотів та підготовки наукових графіків.
3. **Виклики з Rust, Go та Ada:**
   Завдяки стандартній соглашенню про виклики C (cdecl), C-інтерфейс бібліотеки безперешкодно імпортується в Rust (через `extern "C"` блок), Go (через `cgo`) та Ada (через `pragma Import(C)`), що дає змогу використовувати єдине перевірене обчислювальне ядро у сучасному безпечному ПЗ.

### Одиниці вимірювання та перетворення позасистемних авіаційних величин

Усі внутрішні обчислення бібліотеки виконуються строго в міжнародній системі одиниць SI (метри, Паскалі, кілограми, градуси Кельвіна, секунди). Проте у реальній цивільній авіації традиційно застосовуються позасистемні одиниці вимірювання (фути, вузли, дюйми ртутного стовпа, міліметри ртутного стовпа, гектопаскалі).

Для зручності розробників бортових інтерфейсів нижче наведено співвідношення та коефіціенти конверсії:

1. **Висота:**
   - 1 фут (`ft`) = `0.3048` метра.
   - 1 метр (`m`) = `3.28084` фута.
   - Ешелон польоту (Flight Level — FL): `FL100 = 10 000 футів = 3048 метрів`.
2. **Тиск:**
   - 1 Паскаль (`Pa`) = `0.01` гектопаскаля (`hPa`) / мілібара (`mbar`).
   - 1 мм рт. ст. (`mmHg` / torr) = `133.322` Паскаля.
   - 1 дюйм рт. ст. (`inHg`) = `3386.388` Паскаля.
   - Стандартний приземний тиск `P_0 = 1013.25 hPa = 760.0 mmHg = 29.92 inHg`.
3. **Швидкість:**
   - 1 вузол (`knot` / `kt`) = `1 морська миля/год = 0.514444 м/с = 1.852 км/год`.
   - 1 м/с = `1.94384` вузла = `3.6` км/год.

### Портування на мікроконтролери без підтримки FPU та підсистеми RTOS

При портуванні бібліотеки на вбудовані мікроконтролери (Embedded MCUs) без апаратного модуля FPU (Cortex-M0, ARM7TDMI, AVR 8-bit) рекомендується враховувати наступні системні аспекти:

1. **Використання 32-бітного `float` замість 64-бітного `double`:**
   У заголовочному файлі можна визначити макрос `#define ISA_USE_SINGLE_PRECISION`. У цьому випадку всі змінні та параметри функцій конвертуються в тип `float`, що зменшує обсяг споживаної оперативної пам'яті (RAM) та прискорює виконання розрахунків у програмному емуляторі `soft-float` у 5–8 разів при збереженні точності висоти до `0.1 м`.
2. **Інтеграція в операційні системи реального часу (FreeRTOS, VxWorks, RT-Thread):**
   Оскільки розрахункові функції не використовують глобальних змінних, статичного залишку чи викликів операційної системи, вони є суворо реентерабельними. Їх можна викликати з контексту будь-якої задачі RTOS (Task context) або безпосередньо з обробника переривання (ISR context) без необхідності захоплення взаємовиключних семафорів (Mutexes).

### Налаштування прапорців компілятора та оптимізація

Для досягнення максимальної швидкодії та гарантії безпеки при збірці бібліотеки recommended використовувати наступні рекомендовані конфігурації компіляції:

- **Для GCC / Clang:**
  - `-O3`: Максимальний рівень автоматичної векторнізації та оптимізації математичних виразів.
  - `-fno-exceptions -fno-rtti`: Вимикає підтримку винятків та RTTI для зменшення розміру бінарного коду у вбудованих прошивках (embedded firmware).
  - `-Wdouble-promotion`: Видає попередження, якщо числа типу `float` випадково приводиться до `double`, запобігаючи невидимим втратам швидкодії на мікроконтролерах.
- **Для MSVC (Microsoft Visual C++):**
  - `/O2`: Оптимізація за швидкістю виконання.
  - `/fp:fast` або `/fp:precise`: Для авіаційних розрахунків рекомендується залишати `/fp:precise` для суворої відповідності стандарту IEEE 754.

### Версіонування та сумісність заголовочних файлів

Заголовочні файли `isa.h` та `isa.hpp` використовують препроцесорні макроси версіонування:
:::tabs
```c
#define ISA_VERSION_MAJOR 1
#define ISA_VERSION_MINOR 0
#define ISA_VERSION_PATCH 0
```
```cpp
constexpr uint32_t version_major = 1;
constexpr uint32_t version_minor = 0;
constexpr uint32_t version_patch = 0;
```
:::
Це забезпечує зворотну сумісність за заголовками для майбутніх розширень (наприклад, включення шарів термосфери 86–100 км або моделей вологості).

### Межі чисельної точності та зауваження до IEEE 754

Обчислення тиску та густини виконуються за формулами степеня `(T/T_b)^n_p`. Оскільки показник степеня `n_p` у тропосфері дорівнює `5.25588`, малі похибки у представленні температури `T` масштабуються у 5.25 рази у вихідному значення тиску.

При використанні типу `double` (53 біти мантиси в IEEE 754, близько 15–17 значущих десяткових цифр):
- Абсолютна похибка тиску на рівні моря (101 325 Па) становить менше `10⁻⁸ Па`.
- На висоті 80 км (тиск ~0.37 Па) відносна похибка обчислення становить менше `10⁻¹³`.

Це перевищує вимоги будь-яких фізичних вимірювальних приладів і робить бібліотеку фундаментальним еталоном для перевірки бортового ПЗ.

### Структура вирівнювання пам'яті (Memory Alignment & Standard Layout)

Для передачі структури `isa_state_t` через міжпроцесну взаємодію (IPC — Inter-Process Communication) або мережеві UDP-сокети авіоніки значення вирівнювання полів має значення.

У C++20 гарантується, що структура `isa::State` є стандартним розкладка-типом (`std::is_standard_layout_v<isa::State> == true`) та є тривіально копійованою (`std::is_trivially_copyable_v<isa::State> == true`). Це означає, що об'єкт `isa::State` можна безпосередньо копіювати через `memcpy` або серіалізувати в побітовий буфер без проміжних перетворень, що забезпечує найвищу швидкодію в системних драйверах.

### Приклад використання API у реальному додатку C++

Нижче наведено повноцінний приклад розробки авіаційного модуля телеметрії, який використовує розроблений інтерфейс бібліотеки ISA C++20:

```cpp
#include <iostream>
#include <iomanip>
#include "isa.hpp"

void process_flight_telemetry(double gps_altitude_z_meters, double static_pressure_pa, double ias_knots) {
    // 1. Конвертуємо геометричну висоту GPS у геопотенціальну
    const double h_geo = isa::AtmosphereCalculator::z_to_h(gps_altitude_z_meters);

    // 2. Обчислюємо теоретичний стан атмосфери ISA
    const auto atmosphere = isa::AtmosphereCalculator::evaluate(h_geo);
    if (!atmosphere.has_value()) {
        std::cerr << "Помилка: Висота виходить за межі моделі ISA!" << std::endl;
        return;
    }

    const auto& env = atmosphere.value();

    // 3. Переводимо приладову швидкість із вузлів у м/с (1 вузол = 0.514444 м/с)
    const double cas_ms = ias_knots * 0.514444;

    // 4. Розраховуємо істинну повітряну швидкість (TAS) та число Маха
    const double tas_ms = isa::AtmosphereCalculator::cas_to_tas(cas_ms, env.density);
    const double mach = isa::AtmosphereCalculator::tas_to_mach(tas_ms, env.sound_speed);

    // 5. Розраховуємо барометричну висоту за тиском
    const auto pressure_alt = isa::AtmosphereCalculator::pressure_altitude(static_pressure_pa);

    std::cout << std::fixed << std::setprecision(2);
    std::cout << "=== ПОЛЬОТНА ТЕЛЕМЕТРІЯ ===" << std::endl;
    std::cout << "Геометрична висота (GPS): " << gps_altitude_z_meters << " м" << std::endl;
    std::cout << "Геопотенціальна висота:   " << h_geo << " м" << std::endl;
    std::cout << "Барометрична висота:      " << (pressure_alt.value_or(0.0)) << " м" << std::endl;
    std::cout << "Температура ISA:          " << env.temp_celsius << " °C" << std::endl;
    std::cout << "Статичний тиск:           " << env.pressure_hpa() << " гПа" << std::endl;
    std::cout << "Приладова швидкість (CAS):" << ias_knots << " вузлів (" << cas_ms << " м/с)" << std::endl;
    std::cout << "Істинна швидкість (TAS):  " << (tas_ms / 0.514444) << " вузлів (" << tas_ms << " м/с)" << std::endl;
    std::cout << "Число Маха:               " << mach << " M" << std::endl;
}
```

Покроковий розбір виконання функції:
- На кроці 1 функція перераховує висоту GPS у геопотенціал. На висоті 10 000 метрів це дає `h = 9984.3 м`.
- На кроці 2 метод `evaluate` розраховує атмосферний стан для геопотенціалу 9984.3 м (тропосферний шар).
- На кроках 3-4 обчислювач перераховує приладові вузли в м/с та розраховує `TAS` і число Маха `M`.
- На кроці 5 обчислюється барометрична висота для перевірки відповідності показань барометра та GPS.

Цей довідник API надає вичерпне керівництво для розробників авіаційного ПЗ, дозволяючи легко інтегрувати математичну модель ISA ISO 2533 у будь-який C або C++ проєкт з високими вимогами до надійності та обчислювальної швидкодії.
