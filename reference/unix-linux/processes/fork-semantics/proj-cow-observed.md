# ⚙️ Скільки коштує fork на цій машині: програма, яка це показує

Обидві статті витрат розгалуження — негайну копію таблиць сторінок і розтягнуті в часі копіювання при записі — можна не переказувати, а зміряти власноруч: ця програма на C будує залежність затримки `fork` від обсягу відображеної пам'яті й окремо рахує, скільки сторінок ядро справді скопіювало після розгалуження. Перше число має лягти на пряму з відомим наперед нахилом, друге — збігтися з кількістю зачеплених сторінок точно, до одиниць. Коли не збігається, винна одна з небагатьох добре відомих причин — і кожну з них видно тією ж програмою.

Усе нижче — Linux: `/proc/self/smaps_rollup` і прозорі великі сторінки є тільки тут. Виміри самі по собі (`clock_gettime`, `getrusage`) — POSIX і працюють будь-де.

## Два передбачення, які треба перевірити

Перед виміром варто записати, що саме має вийти, — інакше буде спокуса пояснити будь-яке число заднім числом.

**Перше.** Під час `fork` ядро копіює таблиці сторінок батька. На x86-64 зі сторінками 4 КіБ один запис нижнього рівня займає 8 байтів і описує одну сторінку. Отже, обсяг роботи прямо пропорційний обсягу **зачепленої** пам'яті, з коефіцієнтом 8/4096 = 1/512:

```
записів таблиці на 1 ГіБ  =  1 ГіБ / 4 КіБ         =  262144
байтів таблиці на 1 ГіБ   =  262144 · 8 Б          =  2 МіБ        (1/512 обсягу)
```

Отже, залежність затримки від обсягу має бути **прямою**, і її нахил — стала машини, а не програми.

**Друге.** Після повернення з `fork` усі сторінки батька позначено лише для читання, і кожен його запис у власні дані спричиняє [сторінковий виняток](book:unix-linux/page-fault) з копіюванням однієї сторінки. Отже, число таких винятків має дорівнювати числу **зачеплених** сторінок — не всій пам'яті й не нулю. Механізм цілком — у статті про [копіювання при записі](book:unix-linux/copy-on-write); тут потрібен лише його наслідок, який видно ззовні.

## Програма

Один файл, ніяких прав, ніяких залежностей. Збірка: `cc -O2 -Wall -Wextra -o forkcost forkcost.c`.

```c
/* forkcost.c — дві статті витрат fork, які видно на власній машині. */
#define _GNU_SOURCE
#include <fcntl.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/mman.h>
#include <sys/resource.h>
#include <sys/wait.h>
#include <time.h>
#include <unistd.h>

static size_t PAGE;              /* розмір сторінки цієї машини */

/* Різниця двох моментів у мілісекундах. Віднімаємо ЦІЛІ поля й лише потім
   переводимо в double: секунди від старту системи, помножені на 10⁶, з'їли б
   усю точність мантиси, і наносекунди в ній потонули б. */
static double ms_between(struct timespec a, struct timespec b)
{
    return (double)(b.tv_sec - a.tv_sec) * 1e3
         + (double)(b.tv_nsec - a.tv_nsec) / 1e6;
}

/* Одне поле з /proc/self/smaps_rollup, у кілобайтах; −1, якщо поля нема.
   Файл малий і віддається одним читанням, тож stdio тут зайве: воно саме
   виділило б буфер і додало винятків рівно туди, де ми їх рахуємо. */
static long rollup(const char *key)          /* key — напр. "Private_Dirty:" */
{
    static char buf[8192];
    char pat[64];

    int fd = open("/proc/self/smaps_rollup", O_RDONLY);
    if (fd < 0)
        return -1;
    buf[0] = '\n';                           /* щоб шукати початок РЯДКА */
    ssize_t n = read(fd, buf + 1, sizeof(buf) - 2);
    close(fd);
    if (n <= 0)
        return -1;
    buf[n + 1] = '\0';

    snprintf(pat, sizeof(pat), "\n%s", key); /* «\nPss:» не збіжиться з «Pss_Anon:» */
    char *s = strstr(buf, pat);
    return s ? strtol(s + strlen(pat), NULL, 10) : -1;
}

/* Ділянка потрібного обсягу; touch=1 — записати по байту в кожну сторінку. */
static void *area(size_t bytes, int touch, int allow_thp)
{
    void *p = mmap(NULL, bytes, PROT_READ | PROT_WRITE,
                   MAP_PRIVATE | MAP_ANONYMOUS, -1, 0);
    if (p == MAP_FAILED) {
        perror("mmap");
        exit(1);
    }
    if (!allow_thp)                          /* ДО дотику, інакше вже пізно */
        (void)madvise(p, bytes, MADV_NOHUGEPAGE);
    if (touch) {
        volatile unsigned char *q = p;       /* volatile — щоб не викинув компілятор */
        for (size_t off = 0; off < bytes; off += PAGE)
            q[off] = 1;
    }
    return p;
}

/* ── Частина 1: затримка fork ─────────────────────────────────────────────── */

/* Час, за який fork повертається В БАТЬКА. Дитина не робить нічого. */
static double fork_once(void)
{
    struct timespec t0, t1;

    clock_gettime(CLOCK_MONOTONIC, &t0);
    pid_t pid = fork();
    if (pid == 0)
        _exit(0);                            /* НЕ exit: ні atexit, ні буферів */
    clock_gettime(CLOCK_MONOTONIC, &t1);
    if (pid < 0) {
        perror("fork");
        exit(1);
    }
    waitpid(pid, NULL, 0);
    return ms_between(t0, t1);
}

static void part1(size_t max_mib, int reps, int touch, int allow_thp)
{
    double prev_ms = 0.0;
    size_t prev_mib = 0;

    printf("  МіБ   таблиці, КіБ    fork, мс   нахил, мс/ГіБ\n");
    for (size_t mib = 64; mib <= max_mib; mib *= 2) {
        size_t bytes = mib << 20;
        void *p = area(bytes, touch, allow_thp);

        double best = 1e9;                   /* мінімум, а не середнє: шум лише додає */
        for (int i = 0; i < reps; i++) {
            double d = fork_once();
            if (d < best)
                best = d;
        }
        munmap(p, bytes);

        char slope[32];
        if (prev_mib)
            snprintf(slope, sizeof(slope), "%.2f",
                     (best - prev_ms) * 1024.0 / (double)(mib - prev_mib));
        else
            snprintf(slope, sizeof(slope), "%s", "—");
        printf("%5zu   %12zu   %9.3f   %12s\n",
               mib, bytes / PAGE * 8 / 1024, best, slope);

        prev_ms = best;
        prev_mib = mib;
    }
}

/* ── Частина 2: скільки сторінок справді скопійовано ──────────────────────── */

static void part2(size_t mib, size_t touch_mib, int keep_child, int allow_thp)
{
    size_t bytes = mib << 20;
    void *p = area(bytes, 1, allow_thp);
    int sync_pipe[2];
    struct rusage r0, r1;

    if (pipe(sync_pipe) < 0) {
        perror("pipe");
        exit(1);
    }

    /* Прогрів: перший похід у /proc сам коштує кількох винятків. Хай вони
       трапляться ДО того, як почнемо рахувати. */
    rollup("Rss:");
    getrusage(RUSAGE_SELF, &r0);

    pid_t pid = fork();
    if (pid < 0) {
        perror("fork");
        exit(1);
    }
    if (pid == 0) {
        close(sync_pipe[1]);
        char c;
        ssize_t n = read(sync_pipe[0], &c, 1); /* спить, доки батько пише */
        (void)n;
        _exit(0);
    }
    close(sync_pipe[0]);

    if (!keep_child) {                       /* варіант «дитина вже мертва» */
        close(sync_pipe[1]);
        waitpid(pid, NULL, 0);
    }

    long thp  = rollup("AnonHugePages:");
    long rss0 = rollup("Rss:"),           pss0 = rollup("Pss:");
    long pd0  = rollup("Private_Dirty:"), sd0  = rollup("Shared_Dirty:");
    getrusage(RUSAGE_SELF, &r0);             /* оце вже справжня основа виміру */

    size_t pages = (touch_mib << 20) / PAGE;
    volatile unsigned char *q = p;
    for (size_t i = 0; i < pages; i++)
        q[i * PAGE] += 1;                    /* читання-запис: виняток гарантовано */

    getrusage(RUSAGE_SELF, &r1);
    long rss1 = rollup("Rss:"),           pss1 = rollup("Pss:");
    long pd1  = rollup("Private_Dirty:"), sd1  = rollup("Shared_Dirty:");

    printf("  ділянка               : %zu МіБ, сторінка %zu Б%s\n", mib, PAGE,
           keep_child ? "" : "   [дитина вже завершилась]");
    printf("  AnonHugePages         : %ld КіБ%s\n", thp,
           thp > 0 ? "   ⚠ великі сторінки живі" : "");
    printf("  зачеплено сторінок    : %zu  (%zu МіБ)\n", pages, touch_mib);
    printf("  ru_minflt приріст     : %ld\n", r1.ru_minflt - r0.ru_minflt);
    printf("  Private_Dirty         : %ld → %ld КіБ  (%+ld)\n", pd0, pd1, pd1 - pd0);
    printf("  Shared_Dirty          : %ld → %ld КіБ  (%+ld)\n", sd0, sd1, sd1 - sd0);
    printf("  Pss                   : %ld → %ld КіБ  (%+ld)\n", pss0, pss1, pss1 - pss0);
    printf("  Rss                   : %ld → %ld КіБ  (%+ld)\n", rss0, rss1, rss1 - rss0);
    printf("  скопійовано сторінок  : %ld  (з приросту Private_Dirty)\n",
           (pd1 - pd0) * 1024 / (long)PAGE);

    if (keep_child) {
        close(sync_pipe[1]);                 /* дитина прокидається й виходить */
        waitpid(pid, NULL, 0);
    }
    munmap(p, bytes);
}

int main(int argc, char **argv)
{
    PAGE = (size_t)sysconf(_SC_PAGESIZE);
    const char *mode = argc > 1 ? argv[1] : "lat";
    size_t a = argc > 2 ? strtoul(argv[2], NULL, 10) : 0;
    size_t b = argc > 3 ? strtoul(argv[3], NULL, 10) : 0;

    if (!strcmp(mode, "lat"))            part1(a ? a : 2048, 7, 1, 0);
    else if (!strcmp(mode, "lat-cold"))  part1(a ? a : 2048, 7, 0, 0);
    else if (!strcmp(mode, "lat-thp"))   part1(a ? a : 2048, 7, 1, 1);
    else if (!strcmp(mode, "cow"))       part2(a ? a : 512, b ? b : 64, 1, 0);
    else if (!strcmp(mode, "cow-dead"))  part2(a ? a : 512, b ? b : 64, 0, 0);
    else {
        fprintf(stderr, "режими: lat | lat-cold | lat-thp | cow | cow-dead\n");
        return 2;
    }
    return 0;
}
```

Одна вимога до запуску: режим `lat 2048` справді **займає** два гігабайти оперативної пам'яті, бо зачіпає кожну сторінку. Не став обсяг більший за вільну пам'ять — інакше зміряєш не `fork`, а свопінг.

## Частина перша: пряма з нахилом 1/512

```
$ ./forkcost lat 2048
  МіБ   таблиці, КіБ    fork, мс   нахил, мс/ГіБ
   64            128       0.147              —
  128            256       0.201           0.86
  256            512       0.318           0.94
  512           1024       0.536           0.87
 1024           2048       0.994           0.92
 2048           4096       1.882           0.89
```

Стовпчик нахилу — це похідна, порахована між сусідніми рядками; саме він, а не абсолютна затримка, є величиною машини. Стала складова (тут близько 0.09 мс) — це створення облікового запису задачі, нового PID, копія таблиці дескрипторів і решта роботи, яка не залежить від обсягу пам'яті; у похідній вона зникає.

Тепер зіставмо нахил із передбаченням:

```
байтів таблиці на 1 ГіБ   =  2 МіБ                    (1 ГіБ / 4 КіБ · 8 Б)
виміряний нахил           ≈  0.90 мс на ГіБ
темп заповнення таблиць   =  2 МіБ / 0.90 мс          ≈  2.2 ГіБ/с
```

Трохи більше за два гігабайти за секунду — це в рази менше за швидкість звичайного копіювання пам'яті, і так і має бути: ядро не переписує таблиці блоком, а обробляє **кожен запис окремо** — знаходить сторінку, на яку той указує, збільшує їй лічильник посилань і лічильник відображень, знімає право запису в обох таблицях. Копіювання восьми байтів тут — найдешевша частина роботи.

![Три прямі: крута для зачепленої пам'яті зі звичайними сторінками і дві майже нульові — для незачепленої пам'яті й для великих сторінок](/reference/unix-linux/processes/fork-semantics/img/fork-latency-vs-size.svg)

*Нахил прямої — це стала машини: 8 байтів таблиці на кожні 4 КіБ пам'яті. Дві плоскі прямі внизу здобуто тією ж програмою; обидві означають «таблиць майже нема», але причини в них різні.*

### Прямої може не бути — і це теж вимір

Запусти той самий вимір режимом `lat-cold`, який ділянку відображає, але не зачіпає:

```
$ ./forkcost lat-cold 2048
  МіБ   таблиці, КіБ    fork, мс   нахил, мс/ГіБ
   64            128       0.061              —
    …              …           …               …
 2048           4096       0.062           0.00
```

Пряма зникла. Два гігабайти адресного простору обходяться так само, як шістдесят чотири мегабайти, — і це не похибка виміру, а задокументована поведінка ядра. Стовпчик «таблиці» тут бреше: він рахує, скільки записів **було б**, а їх нема. Приватну анонімну ділянку, у яку ще ніхто не писав, ядро при `fork` **пропускає цілком** — просто не заходить у неї. У функції `vma_needs_copy` (файл `mm/memory.c`) це видно дослівно:

```c
    if (src_vma->anon_vma)
        return true;
    /*
     * Don't copy ptes where a page fault will fill them correctly.  Fork
     * becomes much lighter when there are big shared or private readonly
     * mappings. ...
     */
    return false;
```

Поле `anon_vma` з'являється в ділянки лише тоді, коли в неї вперше щось записали. Немає його — немає жодної сторінки, яку варто описувати, і сторінковий виняток однаково заповнить запис правильно, коли до нього дійде черга. Тому **прогрів обов'язковий**: без нього програма міряє не ціну копіювання таблиць, а ціну відмови від нього.

> 🔧 **Навіщо це.** Звідси випливає різниця, яка збиває з пантелику при налагодженні служб: два процеси з однаковим `VSZ` можуть розгалужуватися за час, що відрізняється в сотні разів. Значення має не обсяг адресного простору, а обсяг **зачепленого** — приблизно те, що зветься `RSS`. Різницю між цими величинами й те, чому вони розходяться, розібрано в темі [як міряти пам'ять процесу](book:unix-linux/memory-accounting).

## Частина друга: рівно стільки сторінок, скільки зачеплено

Друга стаття витрат приходить після `fork` і розтягнута в часі. Щоб її спіймати, порахуємо двічі й двома незалежними способами: спершу очима ядра як обліковця подій, потім — очима ядра як обліковця пам'яті. Якщо обидва дадуть одне число, механізм справді той, на який ми думаємо.

Перший спосіб — `getrusage(RUSAGE_SELF, …)`, поле `ru_minflt`: число сторінкових винятків, обслужених без жодного вводу-виводу. Кожне копіювання при записі — саме такий виняток. Другий — приріст `Private_Dirty` у `/proc/self/smaps_rollup`, файлі, що подає підсумок по всіх ділянках процесу одним записом формату `smaps` (з'явився в ядрі 4.14). Про сам спосіб питати ядро файлами — тема [/proc: процеси як файли](book:unix-linux/proc-filesystem).

Тут ховається пастка, через яку вимір найчастіше й провалюють. Природно очікувати, що після `fork` у батька почне рости **`Rss`** — пам'ять же копіюється. Не росте. `Rss` не рухається взагалі: батько як відображав свої 512 МіБ, так і відображає, змінюються лише фізичні кадри під ними. Рухається інше — **чи є сторінка спільною**. Ядро вважає сторінку спільною, коли на неї вказує більш ніж один запис таблиці; одразу після `fork` спільними стають усі, а кожне копіювання при записі повертає одну сторінку в розряд власних.

![Три стовпці для ділянки на 512 МіБ: до fork уся власна, після fork уся спільна, після запису в 64 МіБ частина повернулась у власні; Rss однаковий скрізь](/reference/unix-linux/processes/fork-semantics/img/cow-smaps-states.svg)

*Ділянка та сама й `Rss` той самий у всіх трьох станах. Ціну копіювання видно як приріст `Private_Dirty` — і рівно вдвічі менший приріст `Pss`, бо друга половина сторінки лишається за дитиною.*

```
$ ./forkcost cow 512 64
  ділянка               : 512 МіБ, сторінка 4096 Б
  AnonHugePages         : 0 КіБ
  зачеплено сторінок    : 16384  (64 МіБ)
  ru_minflt приріст     : 16389
  Private_Dirty         : 132 → 65668 КіБ  (+65536)
  Shared_Dirty          : 524440 → 458904 КіБ  (-65536)
  Pss                   : 265120 → 297892 КіБ  (+32772)
  Rss                   : 528640 → 528652 КіБ  (+12)
  скопійовано сторінок  : 16384  (з приросту Private_Dirty)
```

Обидва способи дали те саме число, і воно те, яке передбачає арифметика:

```
зачеплено             =  64 МіБ / 4 КіБ    =  16384 сторінок
приріст Private_Dirty =  65536 КіБ         =  16384 · 4 КіБ
приріст Shared_Dirty  =  −65536 КіБ        =  рівно стільки ж пішло зі спільних
приріст Pss           =  32768 КіБ         =  16384 · 2 КіБ
приріст Rss           =  0
```

Зайвих винятків п'ять понад рівні 16384 — це власна дрібна робота програми: читання `/proc`, зростання стека, буфер `printf`. Дванадцять кілобайтів приросту `Rss` — звідти ж.

Приріст `Pss` удвічі менший за `Private_Dirty` — не збіг і не похибка. `Pss` ділить кожну спільну сторінку між тими, хто на неї вказує. До запису сторінка була спільною й давала батькові 2 КіБ; після — власною й дає всі 4 КіБ. Друга половина не зникла: вона перейшла до дитини, у якої стара сторінка теж стала власною. Саме тому **сумарна** пам'ять двох процесів росте на всі 4 КіБ за кожен зачеплений байт — і саме тому службі, що робить `fork` для знімка, тримають вільною пам'ять порядку половини її обсягу, а на записоємному навантаженні — і всі сто відсотків.

### Виняток без копіювання: чому дитина мусить бути жива

Режим `cow-dead` відрізняється однією дрібницею: канал закривають одразу, і дитину дочікуються **до** того, як батько починає писати.

```
$ ./forkcost cow-dead 512 64
  ділянка               : 512 МіБ, сторінка 4096 Б   [дитина вже завершилась]
  зачеплено сторінок    : 16384  (64 МіБ)
  ru_minflt приріст     : 16388
  Private_Dirty         : 524572 → 524572 КіБ  (+0)
  Shared_Dirty          : 0 → 0 КіБ  (+0)
  Pss                   : 526984 → 526988 КіБ  (+4)
  Rss                   : 528640 → 528652 КіБ  (+12)
  скопійовано сторінок  : 0  (з приросту Private_Dirty)
```

Винятків рівно стільки ж — а скопійовано нуль сторінок. Пояснення просте й важливе. Права запису знято з таблиці батька під час `fork`, і смерть дитини їх не повертає: ядро не ходить по чужих таблицях, щоб щось там полагодити. Тому виняток однаково стається на кожній сторінці. Але, увійшовши в обробник, ядро бачить, що на сторінку вказує вже тільки один — і копіювати нема сенсу: воно просто повертає право запису на місці.

Звідси два висновки, що прямо стосуються будь-якого виміру `fork`. По-перше, `ru_minflt` **сам по собі не є мірою скопійованого**: він рахує зупинки, а не копії. Зупинка теж не безкоштовна — вхід у ядро й вихід з нього, — але вона не тягне за собою ні пошуку вільного кадру, ні копіювання чотирьох кілобайтів, і фізичної пам'яті не додає ні на байт. Саме тому в парі з нею потрібен `Private_Dirty`. По-друге, значення має **час життя дитини**: та сама програма-батько заплатить зовсім різну ціну залежно від того, чи дитина ще жива, коли він пише. Тому в програмі дитина спить на читанні з [каналу](book:unix-linux/pipe-and-fifo) — це найдешевший спосіб тримати її живою, не даючи їй торкатися пам'яті.

## Пастки, через які числа брешуть

**Прозорі великі сторінки** псують обидві частини, і по-різному. У першій вони ділять кількість записів таблиці на 512 — нахил прямої падає до непомітного, і легко зробити хибний висновок «`fork` дешевий». У другій вони роблять протилежне: одиницею копіювання стає 2 МіБ, і **один змінений байт коштує двох мегабайтів**. Тому програма кличе `madvise(MADV_NOHUGEPAGE)` до першого дотику — після дотику це вже не подіє — і друкує `AnonHugePages`: якщо там не нуль, числа треба читати інакше. Загальний стан машини видно в `/sys/kernel/mm/transparent_hugepage/enabled`. Що таке ці сторінки й коли вони корисні — тема про [великі сторінки](book:unix-linux/huge-pages); тут важливо лише, що вони мовчки міняють одиницю обліку.

**Годинник.** `CLOCK_MONOTONIC` тут не примха: `CLOCK_REALTIME` службою часу підтягують і зсувають, і на мілісекундному вимірі зсув видно як від'ємну затримку. Різницю двох шкал розібрано в темі [монотонний час проти настінного](book:programming/monotonic-vs-wall-time). Другий бік — арифметика: перевести обидва моменти в мілісекунди й відняти означає втратити наносекунди в мантисі, бо `tv_sec` для монотонного годинника — це мільйони секунд від старту системи. Віднімати треба цілі поля.

**Мінімум, а не середнє.** Шум планувальника, переміщення між ядрами, зміна частоти процесора вміють тільки додати часу, ніколи не відняти. Тому з кількох повторів беруть найменше значення. Прив'язка до одного ядра (`taskset -c 3 ./forkcost lat`) прибирає ще частину розкиду.

**`_exit`, а не `exit`, у дитині.** Це не стильова дрібниця, а умова коректності виміру. `exit` запускає обробники `atexit` і спорожнює буфери стандартного вводу-виводу — тобто **пише в успадковану пам'ять**, а кожен такий запис у дитині сам породжує копіювання при записі й псує рахунок сторінок. До того ж дитина спорожнила б буфер, успадкований від батька, і вивід програми подвоївся б. `_exit` не робить нічого з цього: він одразу йде в ядро.

**Своп і надлишкові зобов'язання.** Режим `lat` із великим обсягом займає стільки пам'яті, скільки просить. Якщо вона не вміщується, вимір перетворюється на вимір диска. А при суворій політиці ядра `fork` великого процесу може взагалі отримати відмову — про це тема [overcommit і OOM-killer](book:unix-linux/overcommit-and-oom).

**Багатопотоковість.** `getrusage(RUSAGE_SELF)` підсумовує всі потоки процесу, тож у багатопотоковій програмі приріст `ru_minflt` міститиме й чужі винятки. Ця програма однопотокова навмисно.

## Що з цим робити далі

Найкорисніше — прогнати `lat` на тих машинах, де справді працюють твої служби, і записати нахил: він міняється між поколіннями процесорів і між гіпервізорами відчутно, а знаючи його, ціну розгалуження будь-якого процесу оцінюєш множенням у голові — за його `RSS`. Друга частина відповідає на питання іншого роду: коли процес після `fork` несподівано розпухає, вона показує, **скільки саме** сторінок він викуповує назад, — а далі лишається знайти, хто в них пише.
