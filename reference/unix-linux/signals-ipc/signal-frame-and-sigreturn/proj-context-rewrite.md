# 🛠️ Пережити збій, переписавши контекст

Обробник сигналу дістає третім аргументом вказівник на **справжній** збережений контекст на стеку, а не на його копію, — тож він може вирішити, куди програма повернеться. Нижче робоча програма мовою C, яка на цьому будує безпечне читання пам'яті: звертається за адресою, про яку невідомо, чи вона взагалі відображена, і замість смерти від `SIGSEGV` продовжує роботу з наперед призначеного місця.

## Задача

Хай діагностичний інструмент іде за ланцюжком покажчиків у структурі, яку вже зіпсувала чужа помилка. Половина покажчиків веде кудись, де сторінки немає. Перевіряти кожну адресу наперед незручно з двох причин. Розбирати `/proc/self/maps` дорого, та й відповідь застаріває одразу: у програмі з кількома потоками сусід може зняти відображення між перевіркою й читанням. Другий шлях — питати ядро системним викликом, який на поганій адресі повертає `EFAULT` замість того, щоб убити процес (класика — спробувати записати ту адресу в канал). Це надійно, але коштує повний перехід у ядро на кожне звернення, а переважна більшість звернень удалі.

Обидва підходи платять на успішному шляху за рідкісну невдачу. Ядро для себе цю задачу розв'язало інакше: коли воно копіює дані з простору користувача, воно **не перевіряє нічого** — просто читає. Якщо читання впало, обробник сторінкового збою шукає адресу аварійної інструкції в спеціальній таблиці винятків, знаходить там записану «латку» й переводить виконання туди. Успішний шлях не коштує жодної зайвої інструкції.

Ту саму таблицю — на один-єдиний запис — можна зробити в просторі користувача. Різниця лише в тому, що там, де ядро має свій обробник збою, у нас обробник `SIGSEGV`.

## Ідея: дві адреси, які треба знати

Щоб переставити покажчик інструкції, обробникові потрібні дві адреси: **де саме дозволено падати** й **куди приземлятися**. Жодної з них компілятор не оприлюднює. Але їх можна поставити самому — асемблерною вставкою з мітками, які асемблер перетворює на звичайні символи.

Той самий прийом розв'язує й другу проблему. Другий варіант відновлення — не тікати з дільниці, а вдати, ніби читання вдалося: пропустити аварійну інструкцію й підставити значення в регістр призначення. Для цього треба знати, **який** це регістр і **де закінчується** інструкція. На x86-64 інструкція має від одного до п'ятнадцяти байтів, тож «просто додати до RIP довжину» без повного розбору кодування неможливо; а який регістр обрав компілятор, обробник не бачить узагалі. Мітка одразу після інструкції дає перше, жорсткий вибір `%rax` — друге.

## Програма

```c
#define _GNU_SOURCE
#include <signal.h>
#include <ucontext.h>
#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/mman.h>

/* Мітки всередині ризикованої дільниці: асемблер робить із них звичайні
   символи, тож обробник знає точні адреси — і де дозволено впасти,
   і обидва місця, куди можна приземлитися. */
extern const char probe_fault[], probe_after[], probe_fail[];

enum { MODE_BAIL, MODE_SKIP };
static volatile sig_atomic_t mode = MODE_BAIL;
static void * volatile last_fault;        /* si_addr останнього перехоплення */

static void on_segv(int sig, siginfo_t *si, void *ctx)
{
    ucontext_t *uc = ctx;
    greg_t *r = uc->uc_mcontext.gregs;    /* НЕ копія — сам кадр на стеку */

    /* Єдина інструкція, збій на якій ми вміємо лагодити. Усе інше —
       справжня помилка: знімаємо обробник і даємо процесові чесно
       померти на тому самому місці. */
    if ((const char *)(uintptr_t)r[REG_RIP] != probe_fault) {
        signal(sig, SIG_DFL);
        return;
    }

    last_fault = si->si_addr;             /* адреса, до якої не вийшло дійти */

    if (mode == MODE_BAIL) {
        r[REG_RIP] = (greg_t)(uintptr_t)probe_fail;   /* вихід із дільниці */
    } else {
        r[REG_RAX] = 0;                               /* значення-заглушка */
        r[REG_RIP] = (greg_t)(uintptr_t)probe_after;  /* пропуск інструкції */
    }
}

/* noinline + noclone: вставка з глобальними мітками мусить потрапити
   в об'єктний файл РІВНО один раз, інакше буде подвійне визначення. */
__attribute__((noinline, noclone))
static int try_read64(const void *p, uint64_t *out)
{
    uint64_t v;
    int ok;
    __asm__ volatile(
        ".globl probe_fault, probe_after, probe_fail\n"
        "probe_fault: movq (%[p]), %%rax\n"
        "probe_after: movl $1, %[ok]\n"
        "             jmp  1f\n"
        "probe_fail:  movl $0, %[ok]\n"
        "1:\n"
        : "=a"(v), [ok] "=&r"(ok)
        : [p] "r"(p)
        : "cc", "memory");
    if (ok)
        *out = v;
    return ok;
}

static void install_handler(void)
{
    /* Розмір площадки — лише через sysconf: у glibc від 2.34 SIGSTKSZ
       уже не стала часу компіляції. */
    long sz = sysconf(_SC_SIGSTKSZ);
    if (sz <= 0)
        sz = 32 * 1024;

    stack_t ss;
    ss.ss_sp = malloc((size_t)sz);
    ss.ss_size = (size_t)sz;
    ss.ss_flags = 0;
    if (!ss.ss_sp || sigaltstack(&ss, NULL) != 0) {
        perror("sigaltstack");
        exit(1);
    }

    struct sigaction sa;
    memset(&sa, 0, sizeof sa);
    sa.sa_sigaction = on_segv;
    sa.sa_flags = SA_SIGINFO | SA_ONSTACK;   /* розширений обробник, на площадці */
    sigfillset(&sa.sa_mask);
    if (sigaction(SIGSEGV, &sa, NULL) != 0) {
        perror("sigaction");
        exit(1);
    }
}

int main(void)
{
    install_handler();

    uint64_t good = 0x1122334455667788ULL, v = 0xdead;
    char *closed = mmap(NULL, 4096, PROT_NONE,
                        MAP_PRIVATE | MAP_ANONYMOUS, -1, 0);
    if (closed == MAP_FAILED)
        return 1;

    int ok = try_read64(&good, &v);
    printf("добра адреса      : ok=%d v=0x%llx\n", ok, (unsigned long long)v);

    mode = MODE_BAIL;
    ok = try_read64(closed, &v);
    printf("закрита, «вихід»  : ok=%d v=0x%llx si_addr=%p\n",
           ok, (unsigned long long)v, last_fault);

    mode = MODE_SKIP;
    ok = try_read64(closed, &v);
    printf("закрита, «пропуск»: ok=%d v=0x%llx si_addr=%p\n",
           ok, (unsigned long long)v, last_fault);

    puts("програма жива");
    return 0;
}
```

**Збірка й запуск:** `gcc -O2 -Wall -o probe probe.c && ./probe`

```
добра адреса      : ok=1 v=0x1122334455667788
закрита, «вихід»  : ok=0 v=0xdead si_addr=0x7f3c8a1d2000
закрита, «пропуск»: ok=1 v=0x0 si_addr=0x7f3c8a1d2000
програма жива
```

Третій рядок — найпоказовіший. Читання за адресою, якої не існує, повернуло нуль і прапорець успіху: процесор ніколи не виконав інструкції, що мала покласти значення в `%rax`, — значення туди поклав обробник, дописавши його в кадр на стеку.

> 🔧 **Навіщо це.** Успішне читання коштує рівно одну інструкцію `movq` — стільки ж, скільки звичайне розіменування покажчика. Ні перевірки, ні системного виклику, ні гілки. Уся ціна перенесена на невдачу, яка трапляється рідко. Саме за це середовища виконання платять складністю: перевірка на нульовий покажчик у [компіляторах «на льоту»](book:programming/jit-compilation) — це часто взагалі не порівняння, а звичайне звернення до пам'яті, і збій на ньому ловить такий самий обробник.

## Пастки

**Не кожен збій — ваш.** Обробник, який лагодить будь-який `SIGSEGV`, перетворює справжні помилки на тихе псування даних або на вічний цикл. Тому перша дія — звірити `REG_RIP` з адресою єдиної дозволеної інструкції, і якщо не збіглося, повернути початкову долю сигналу. Це різниця між таблицею винятків і ковдрою, накинутою на всі проблеми відразу. `si_addr` для цієї звірки не годиться: там адреса **даних**, до яких не вийшло дійти, а не адреса коду.

**Повернення без правки — це нескінченний цикл.** Синхронний збій ловиться [сторінковим механізмом](book:unix-linux/page-fault) на конкретній інструкції, і збережений покажчик показує саме на неї. Якщо обробник просто повернеться, процесор виконає ту саму інструкцію, отримає той самий збій і покличе той самий обробник. Виходів рівно два: усунути причину (наприклад, відобразити сторінку — тоді правити контекст не треба) або перевести виконання деінде.

**Сигнал, що застав системний виклик.** Це найпідступніше. Коли сигнал приходить на перерваному виклику, ядро вирішує долю повтору **до того**, як побудує кадр. Якщо повтор буде, обробник бачить контекст, уже до нього підготований: покажчик інструкції відкочений на два байти назад, на саму інструкцію `syscall`, а в `RAX` лежить не результат, а номер виклику. Якщо повтору не буде, покажчик лишається за викликом, а в `RAX` — код `EINTR`. Одне й те саме поле означає різні речі, і необережна правка або скасує повтор, або пустить виклик із чужим номером. Що саме перезапускається й за яких умов — у [перезапуску перерваних викликів](book:unix-linux/eintr-and-restart): коротко, при `SA_RESTART` частина «повільних» викликів повторюється сама, решта повертає `EINTR`.

*Одне й те саме поле кадру означає різне залежно від того, звідки прийшов сигнал.*

Звідси правило: правити контекст безпечно лише для сигналів, породжених самою інструкцією, — `SIGSEGV`, `SIGBUS`, `SIGILL`, `SIGFPE`. Для сигналів, що приходять ззовні, потрібна геть інша обережність.

**Полів `gregs` не існує ніде, крім x86.** Це не абстракція, а знімок конкретного набору регістрів. На aarch64 замість масиву з іменованими індексами — окремі поля `uc_mcontext.pc`, `uc_mcontext.sp`, `uc_mcontext.pstate` й масив `regs[31]`. На RISC-V — `uc_mcontext.__gregs[]` зі своєю нумерацією. Переносного коду тут не буває взагалі; є код, розгалужений за архітектурою, і кожну гілку треба перевіряти окремо.

**Розширений стан лежить не в `uc_mcontext`.** Регістри співпроцесора й векторні регістри займають забагато місця, тож у машинному контексті на них лише вказівник (`uc_mcontext.fpregs` на x86-64). Хто спробує підставити значення у векторний регістр, змінивши щось у самому `uc_mcontext`, не змінить нічого. Правити треба в тій області за вказівником — і там ще й є заголовок із бітовою маскою, яка каже, які саме шматки стану в ній справді збережено.

**Площадка своя в кожного потоку, а її розмір — не константа.** `sigaltstack` призначає стек **потокові**, а не процесові; прапорець `SA_ONSTACK` у сигнальній дії спільний на весь процес, тож потік, який площадки собі не завів, побіжить із нею на своєму звичайному стеку. У програмі з кількома [потоками](book:unix-linux/threads-as-tasks) площадку доводиться заводити в кожному. А сталі `SIGSTKSZ` і `MINSIGSTKSZ` у glibc від версії 2.34 при `_GNU_SOURCE` розгорнуто в `sysconf(_SC_SIGSTKSZ)` і `sysconf(_SC_MINSIGSTKSZ)` — саме тому, що розширення процесорів ростуть і стала перестала бути правдою. Старий код виду `static char stack[SIGSTKSZ];` після цієї зміни просто не збирається.

**Усередині обробника ви майже нічого не можете.** У прикладі обробник виконує кілька присвоєнь і один виклик `signal` — і це не випадковість. Обробник міг перервати `malloc` посеред перебудови купи чи `printf` посеред запису в буфер, тож дозволено лише [асинхронно-безпечні виклики](book:unix-linux/async-signal-safety): ті, що не тримають спільного стану й повторний вхід у які нічого не ламає. Друк усередині `on_segv` перетворив би надійне відновлення на випадкове зависання.

**Оптимізатор не знає, що сюди можна прийти.** Компілятор бачить асемблерну вставку як один непрозорий шматок і про стрибок посеред неї не здогадується. Поки і збій, і обидва приземлення лежать усередині однієї вставки, це не має значення. Але спроба поставити приземлення в довільному місці функції на C розсипається: [оптимізатор](book:programming/compiler-optimizations) вважає, що керування туди приходить лише звичним шляхом, і має право тримати значення в регістрах, які до вашого стрибка ніхто не заповнив.

## Де прийом доречний

Він придатний рівно там, де ви точно знаєте, **що саме** лагодите: одну інструкцію, яку самі написали, з наперед відомим наслідком збою. Тоді це не хитрощі, а звичайна таблиця винятків — те, з чого зроблені перевірки на нульовий покажчик у віртуальних машинах, бар'єри збирачів сміття й відкат оптимізованого коду.

Коли ж потрібно не пережити збій, а справді обслуговувати відсутні сторінки — підвантажувати їх, рахувати звернення, будувати розподілену спільну пам'ять, — правка контексту стає крихкою: ви лагодите наслідок замість причини, і кожна нова архітектура коштує окремої гілки коду. Для цього в Linux є [обробка сторінкових збоїв у просторі користувача](book:unix-linux/userfaultfd): ядро повідомляє про збій через дескриптор, окремий потік вирішує, чим заповнити сторінку, а винний потік просто чекає — без обробників, без кадру й без жодного регістра, який довелося б підмінювати вручну.
