# ⚙️ Дослід: чи діє ionice саме на цій машині

Ця вставка пояснює ⚙️ дослід: чи діє ionice саме на цій машині та дозволяє зрозуміти її детальніше. Це стенд із однієї програми на два режими й скрипта, що їх поганяє, — він відповідає на єдине питання: чи змінює клас `ionice` бодай щось на тому носії, що стоїть у вас. Питання не риторичне — виклик, який виставляє пріоритет, повертає нуль однаково успішно і тоді, коли число читає планувальник, і тоді, коли воно не доїжджає нікуди. Успішний код повернення тут не доказ. Доказом може бути тільки затримка чужого читання.

## Що саме мірятимемо

Два читачі одного носія. Перший — **навантажувач**: удає нічне копіювання, тримає чергу повною, працює повз кеш. Другий — **вимірювач**: удає людину, читає по 4 КіБ з випадкових місць і записує тривалість кожного окремого читання.

Відповідь стенду — одне число: 99-й процентиль вимірювача під навантаженням. Медіана каже, як буває зазвичай, а людина відчуває хвіст: завмирання редактора під нічним копіюванням — це не середня затримка, це найгірший відсоток спроб.

Прогін перебирає дев'ять клітинок: три планувальники (`none`, `mq-deadline`, BFQ) на три стани навантажувача (вимкнений, клас `-c 2`, клас `-c 3`). Клітинка з вимкненим навантажувачем — контрольна: вона показує підлогу, тобто скільки затримки дає сам носій, коли черга порожня. Без цього рядка порівнювати нема з чим.

Навантажувач доводиться писати самому, а не брати `tar`. Копіювання витрачає час на стиснення й обхід каталогів — у затримці вимірювача переплелися б черга й процесор, і розділити їх було б нічим. Потрібне навантаження чистої черги: читання, які не роблять більше нічого.

## Вимірювач і навантажувач

Мова тут не питання смаку: потрібні `O_DIRECT` з його вимогами до вирівнювання, системний виклик без бібліотечної обгортки і замір кожного запиту з наносекундною роздільністю. Це C.

```c
/* iotest.c — стенд «чи діє ionice», два режими в одній програмі.
 *   cc -O2 -pthread -o iotest iotest.c
 *   ./iotest meter <файл> <секунд>
 *   ./iotest load  <файл> <клас> <рівень> <потоків> <секунд>
 */
#define _GNU_SOURCE
#include <fcntl.h>
#include <pthread.h>
#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/syscall.h>
#include <time.h>
#include <unistd.h>

#define MBLK  4096u            /* читання вимірювача   */
#define LBLK  (64u * 1024u)    /* читання навантажувача */
#define BUCKS 2000             /* корзина 10 мкс, стеля 20 мс */

#define IOPRIO_WHO_PROCESS 1
#define IOPRIO_VALUE(cls, lvl) (((cls) << 13) | (lvl))

static int ioprio_set(int which, int who, int prio)
{
    return (int)syscall(SYS_ioprio_set, which, who, prio);
}

static uint64_t now_ns(void)
{
    struct timespec t;
    clock_gettime(CLOCK_MONOTONIC, &t);
    return (uint64_t)t.tv_sec * 1000000000ull + (uint64_t)t.tv_nsec;
}

/* власний генератор: без блокувань і без звертань до ядра */
static uint64_t xorshift(uint64_t *s)
{
    *s ^= *s << 13; *s ^= *s >> 7; *s ^= *s << 17;
    return *s;
}

static int open_direct(const char *path, unsigned blk, uint64_t *blocks)
{
    int fd = open(path, O_RDONLY | O_DIRECT);
    if (fd < 0) { perror("open O_DIRECT"); exit(1); }
    off_t sz = lseek(fd, 0, SEEK_END);
    if (sz < (off_t)blk * 1024) { fprintf(stderr, "файл замалий\n"); exit(1); }
    *blocks = (uint64_t)sz / blk;
    return fd;
}

static double pct(const uint32_t *hist, uint64_t n, double p)
{
    uint64_t need = (uint64_t)(n * p), acc = 0;
    for (int i = 0; i <= BUCKS; i++) {
        acc += hist[i];
        if (acc >= need) return i * 10.0 + 5.0;   /* середина корзини */
    }
    return -1.0;
}

static void meter(const char *path, int seconds)
{
    uint64_t blocks;
    int fd = open_direct(path, MBLK, &blocks);
    void *buf;
    if (posix_memalign(&buf, MBLK, MBLK)) { perror("posix_memalign"); exit(1); }

    uint32_t *hist = calloc(BUCKS + 1, sizeof *hist);
    uint64_t seed = now_ns() | 1, n = 0;
    uint64_t stop = now_ns() + (uint64_t)seconds * 1000000000ull;

    while (now_ns() < stop) {
        off_t off = (off_t)((xorshift(&seed) % blocks) * MBLK);
        uint64_t t0 = now_ns();
        if (pread(fd, buf, MBLK, off) != (ssize_t)MBLK) { perror("pread"); exit(1); }
        uint64_t bucket = (now_ns() - t0) / 1000 / 10;
        hist[bucket < BUCKS ? bucket : BUCKS]++;
        n++;
    }
    if (n == 0) { fprintf(stderr, "жодної проби\n"); exit(1); }

    printf("проб %-8llu медіана %6.0f мкс   p99 %7.0f мкс   за стелею %.2f%%\n",
           (unsigned long long)n, pct(hist, n, 0.50), pct(hist, n, 0.99),
           100.0 * hist[BUCKS] / (double)n);
    close(fd); free(buf); free(hist);
}

struct job { const char *path; uint64_t stop; int prio; int id; };

static void *worker(void *arg)
{
    struct job *j = arg;
    /* пріоритет — атрибут ЗАДАЧІ, тож кожен потік виставляє його собі сам */
    if (ioprio_set(IOPRIO_WHO_PROCESS, 0, j->prio) < 0) { perror("ioprio_set"); exit(1); }

    uint64_t blocks;
    int fd = open_direct(j->path, LBLK, &blocks);
    void *buf;
    if (posix_memalign(&buf, MBLK, LBLK)) { perror("posix_memalign"); exit(1); }
    uint64_t seed = now_ns() + 0x9E3779B97F4A7C15ull * (uint64_t)(j->id + 1);

    while (now_ns() < j->stop) {
        off_t off = (off_t)((xorshift(&seed) % blocks) * LBLK);
        if (pread(fd, buf, LBLK, off) < 0) { perror("pread"); exit(1); }
    }
    close(fd); free(buf);
    return NULL;
}

static void loader(const char *path, int cls, int lvl, int nthr, int seconds)
{
    int prio = IOPRIO_VALUE(cls, lvl);
    uint64_t stop = now_ns() + (uint64_t)seconds * 1000000000ull;
    if (ioprio_set(IOPRIO_WHO_PROCESS, 0, prio) < 0) { perror("ioprio_set"); exit(1); }

    struct job *jobs = calloc(nthr, sizeof *jobs);
    pthread_t *th = calloc(nthr, sizeof *th);
    for (int i = 0; i < nthr; i++) {
        jobs[i] = (struct job){ path, stop, prio, i };
        pthread_create(&th[i], NULL, worker, &jobs[i]);
    }
    for (int i = 0; i < nthr; i++) pthread_join(th[i], NULL);
    free(th); free(jobs);
}

int main(int argc, char **argv)
{
    if (argc == 4 && !strcmp(argv[1], "meter"))
        meter(argv[2], atoi(argv[3]));
    else if (argc == 7 && !strcmp(argv[1], "load"))
        loader(argv[2], atoi(argv[3]), atoi(argv[4]), atoi(argv[5]), atoi(argv[6]));
    else {
        fprintf(stderr, "%s meter <файл> <секунд>\n"
                        "%s load  <файл> <клас> <рівень> <потоків> <секунд>\n",
                argv[0], argv[0]);
        return 2;
    }
    return 0;
}
```

Три рішення в цьому коді варті пояснення.

`O_DIRECT` — не оздоба, а сама умова досліду: [прямий ввід-вивід](book:unix-linux/buffered-and-direct-io) веде кожне читання повз кеш сторінок просто до носія, тож вимірювач справді чекає на пристрій. Ціна — вирівнювання: і буфер у пам'яті, і зсув у файлі, і довжина мають бути кратні логічному блокові пристрою (звідси `posix_memalign` і зсуви, кратні `MBLK`).

Гістограма замість масиву проб. За двадцять секунд вимірювач робить сотні тисяч читань; складати їх усі, а потім сортувати — це виділення пам'яті й робота процесора просто посеред заміру. Лічильники в корзинах по 10 мкс живуть у кеші процесора й не заважають нікому. Остання корзина збирає все, що довше за 20 мс, — програма друкує її частку окремо, і якщо на вашому носії туди потрапляє більш ніж відсоток проб, стелю треба піднімати, інакше p99 упреться в межу гістограми.

Клас виставляє собі **кожен потік навантажувача**. Пріоритет вводу-виводу — атрибут задачі, а не процесу: потоки одного процесу можуть мати різні числа. Виставити один раз у `main` і сподіватися, що новостворені потоки успадкують, — саме той спосіб отримати нуль різниці й вирішити, що `ionice` не працює.

## Скрипт, який зводить таблицю

```sh
#!/bin/sh
# run.sh — звід «чи діє ionice»: планувальник × клас навантажувача.
#   sudo ./run.sh nvme0n1 /mnt/test/big.bin
set -eu
DEV=$1; FILE=$2
SEC=20; THREADS=16

echo "пристрій: $DEV   файл: $FILE"
for sched in none mq-deadline bfq; do
    if ! echo "$sched" > "/sys/block/$DEV/queue/scheduler" 2>/dev/null; then
        echo "$sched — недоступний (modprobe bfq?)"
        continue
    fi
    for cls in off 2 3; do
        sync
        echo 3 > /proc/sys/vm/drop_caches
        pid=
        if [ "$cls" != off ]; then
            ./iotest load "$FILE" "$cls" 0 "$THREADS" $((SEC + 5)) &
            pid=$!
            sleep 2                     # дати черзі наповнитися
        fi
        printf '%-12s  навантажувач %-3s  ' "$sched" "$cls"
        ./iotest meter "$FILE" "$SEC"
        if [ -n "$pid" ]; then
            kill "$pid" 2>/dev/null || true
            wait "$pid" 2>/dev/null || true
        fi
        sleep 2                         # дати носієві заспокоїтися
    done
done
```

Дві секунди сну перед виміром — не забобон: перші запити застають чергу порожньою, і якби вимірювач стартував одночасно з навантажувачем, у гістограму потрапив би шматок незатореного носія. `sync` виносить брудні сторінки, щоб посеред досліду не прокинувся зворотний запис; `drop_caches` — страховка на випадок, коли `O_DIRECT` десь тихо виродився в буферизоване читання ([кеш і зворотний запис](book:unix-linux/page-cache-durability)).

Окремої уваги вартий сам файл. Спокуса створити його миттєво — `fallocate -l 8G big.bin` — псує дослід дощенту: файлова система лише закріплює простір за файлом, не записавши туди жодного байта, і читання з такої [діри](book:unix-linux/sparse-files) повертає нулі просто з ядра, не звертаючись до носія взагалі. Вимірювач тоді міряє, як швидко ядро заповнює буфер нулями. Дані мають бути справжні й нестисливі — інакше том зі стисненням збереже замість гігабайтів кілька блоків:

```
$ dd if=/dev/urandom of=/mnt/test/big.bin bs=1M count=8192 oflag=direct
$ sync
```

Кілька гігабайтів потрібні для того, щоб випадкові зсуви розходилися по носію, а не били в одне місце.

## Як читати звід

Абсолютні числа у вас будуть свої — вони залежать від носія. Значення має форма таблиці: порівняння рядків усередині одного планувальника.

| планувальник | навантажувач | медіана | p99 |
| --- | --- | --- | --- |
| `none` | вимкнено | 120 мкс | 210 мкс |
| `none` | `-c 2` | 940 мкс | 4100 мкс |
| `none` | `-c 3` | 930 мкс | 4050 мкс |
| `mq-deadline` | вимкнено | 120 мкс | 215 мкс |
| `mq-deadline` | `-c 2` | 900 мкс | 3900 мкс |
| `mq-deadline` | `-c 3` | 260 мкс | 640 мкс |
| BFQ | вимкнено | 130 мкс | 240 мкс |
| BFQ | `-c 2` | 700 мкс | 2600 мкс |
| BFQ | `-c 3` | 190 мкс | 430 мкс |

Читати треба трійками. Рядок «вимкнено» — підлога. Рядок `-c 2` показує, наскільки навантажувач узагалі здатен зіпсувати життя: якщо він недалеко відійшов від підлоги, затору немає й дослід нічого не доводить — додайте потоків навантажувачеві. Рядок `-c 3` і є відповідь: наскільки клас `idle` повернув вимірювача до підлоги.

У рядках `none` числа `-c 2` і `-c 3` збіглися. Це не поламаний дослід і не помилка в коді — це правильна відповідь: `none` означає, що [планувальника черги](book:unix-linux/io-schedulers) немає взагалі, запити йдуть до апаратних черг [NVMe](book:unix-linux/nvme-in-linux) у порядку надходження, і число пріоритету немає кому прочитати. Саме заради цієї клітинки стенд і будували.

> 🔧 **Навіщо це.** Дивитися треба на p99, а не на медіану, бо `idle` рятує саме хвіст. Планувальник не робить чуже читання швидшим — він лише не дає йому застрягти за сотнею чужих. Медіана може зрушити на десятки відсотків там, де 99-й процентиль падає вшестеро; людина за екраном помічає друге, а не перше.

Варто прогнати ще одну пару клітинок: `-c 2 -n 0` проти `-c 2 -n 7`. На `mq-deadline` вони збіжаться (він розрізняє класи, не рівні), на BFQ розійдуться. Це найдешевший спосіб побачити, що «однакова команда» на двох планувальниках означає різне.

## Пастки, через які дослід бреше

**Читання без `O_DIRECT`.** Найпоширеніша й найтихіша: приберіть прапорець — і вимірювач після першого проходу читатиме з пам'яті. Медіана впаде до одиниць мікросекунд, різниця між класами зникне, і висновок буде «ionice не діє». Насправді не діяв вимірювач: до черги він просто не дійшов.

**Буферизований запис як навантаження.** `dd if=/dev/zero of=big bs=1M` без `oflag=direct` не навантажує чергу в своєму контексті взагалі: `write()` бруднить сторінки в пам'яті й негайно повертається, а запити до носія подасть потік зворотного запису — зі своїм власним пріоритетом. Скільки б не чіпляли `ionice` до такого `dd`, у таблиці нічого не зміниться.

**Порожня черга.** Пріоритет упорядковує тих, хто чекає. Коли черга не заповнена, чекати нема кому, і жоден планувальник не покаже різниці. Шістнадцять потоків — орієнтир для SATA-SSD; на NVMe їх треба помітно більше, на обертовому диску вистачить і чотирьох. Здогадуватися тут не треба — глибину видно прямо:

```
$ cat /sys/block/nvme0n1/inflight     # читання  запис — скільки в роботі просто зараз
      31        0
```

Якщо тут або в колонці `aqu-sz` виводу `iostat -x 1` під навантаженням стоїть одиниця, то навантажувача фактично немає, і всі дев'ять клітинок будуть однакові з тієї самої причини, з якої однаковий порожній рядок.

**Посередник між файлом і пристроєм.** Перевірте через `lsblk`, що файл лежить прямо на розділі того пристрою, чий планувальник ви перемикаєте. Якщо між ними [device-mapper](book:unix-linux/device-mapper), програмний RAID або [шифрований том](book:unix-linux/dm-crypt), запит перекладає потік ядра, і початковий пріоритет далі не їде. У `/sys/block` лежать цілі пристрої, не розділи: `sda`, а не `sda1`.

**Планувальник, який не перемкнувся.** Після запису в `scheduler` перечитайте файл: BFQ може бути окремим модулем, і без `modprobe bfq` у переліку його просто не буде. Скрипт таку клітинку пропускає з поясненням, але мовчазне «нічого не змінилося» варте перевірки очима.

**Число, якого не виставили.** Поки не переконалися — не вірте. `ionice -p <TID>` для кожного потоку зі списку `/proc/<pid>/task/` покаже, що `ioprio_set` справді доїхав до всіх, а не лише до головної задачі.

І остання, найважливіша: **однократний прогін нічого не доводить**. Проженіть звід тричі й порівняйте прогони між собою — фоновий журнал, оновлення пакетів чи тепловий режим накопичувача дають розкид, на тлі якого «різниця» в десять відсотків не означає нічого. Ефект `idle` там, де він є, видно в рази, а не у відсотках.
