# ⚙️ Власний символьний драйвер із командами ioctl

Зберемо від початку до кінця пристрій, у якого є те, чого не буває в потоці байтів: період і заповнення імпульсу. Три файли — спільний заголовок, модуль ядра, програма — і чотири пастки, від яких код захищено навмисно й напоказ, бо жодну з них ядро за вас не закриє.

Пристрій — генератор імпульсів `/dev/pulsegen`. Заліза за ним немає: він лише тримає налаштування й лічильник виданих імпульсів. Саме тому він і зручний як зразок — уся форма справжнього драйвера на місці, а того, що відволікає, немає жодного рядка.

Команд буде три, і вони навмисно різних напрямів. Задати налаштування — програма передає структуру вниз (`_IOW`). Прочитати чинні — ядро заповнює структуру в пам'яті програми (`_IOR`). Видати пачку імпульсів — програма каже, скільки хоче, і тим самим викликом отримує назад лічильник (`_IOWR`). Третя команда й пояснює, навіщо тут узагалі ioctl: замовлення й відповідь мусять статися однією неподільною дією над цим самим відкритим дескриптором. Двох записів у [sysfs](topic:unix-linux/sysfs-device-model) так не склеїш — це два різні файли й два різні моменти часу.

## Спільний заголовок

Ключове слово — спільний. Номер команди й розкладка структури мусять збігтися по обидва боки межі до біта. Якщо заголовок продублювати «щоб не морочитися з установленням», розбіжність з'явиться в перший же тиждень і виявиться не помилкою складання, а мовчазним `−ENOTTY`. Тому файл один: у дереві ядра він живе в `include/uapi/`, у нашій вправі — просто поруч із двома іншими.

```c
/* pulsegen.h — цей файл читають обидві сторони */
#ifndef _UAPI_PULSEGEN_H
#define _UAPI_PULSEGEN_H

#include <linux/types.h>
#include <linux/ioctl.h>

struct pulse_cfg {
        __u32 period_us;      /* період імпульсу, мкс: 10 … 1000000 */
        __u32 duty_permille;  /* заповнення, ‰: 1 … 999          */
};

struct pulse_burst {
        __u32 want;           /* → скільки імпульсів замовляємо        */
        __u32 __pad;          /*   мусить бути 0                       */
        __u64 issued;         /* ← усього видано від завантаження      */
};

#define PULSE_IOC_TYPE 'B'

#define PULSE_SET_CFG  _IOW(PULSE_IOC_TYPE, 0x01, struct pulse_cfg)
#define PULSE_GET_CFG  _IOR(PULSE_IOC_TYPE, 0x02, struct pulse_cfg)
#define PULSE_BURST   _IOWR(PULSE_IOC_TYPE, 0x03, struct pulse_burst)

#endif
```

Три речі тут неочевидні, і всі три — наслідки того, що ядро про зміст аргументу не знає нічого.

**Типи фіксованої ширини.** `long` має різну ширину на різних ABI, і навіть `int` формально не прибитий цвяхами. Заголовок описує не змінні мови, а байти, що йдуть крізь межу, — тож у ньому лише `__u32` і `__u64`. Якби серед полів був покажчик, його теж довелося б оголосити як `__u64` і розгортати в ядрі через `u64_to_user_ptr()`.

**Явне поле `__pad`.** Компілятор і без нього вставив би чотири байти перед `issued` — але вставив би не скрізь. Це поле прибирає розбіжність розкладки між архітектурами ще до того, як вона з'явиться; чому саме так — за кілька абзаців.

**Літера типу.** `'B'` — підпис підсистеми, за яким команди різних драйверів розходяться. Для драйвера, що піде в дерево ядра, літеру й діапазон номерів треба застовпити в `Documentation/userspace-api/ioctl/ioctl-number.rst`, інакше ваші числа рано чи пізно збіжаться з чужими — і команда «видати пачку» прийде драйверові, у якого під тим самим числом записано щось зовсім інше.

## Модуль

```c
/* pulsegen.c — модуль ядра */
#include <linux/module.h>
#include <linux/miscdevice.h>
#include <linux/fs.h>
#include <linux/mutex.h>
#include <linux/string.h>
#include <linux/uaccess.h>
#include <linux/overflow.h>
#include "pulsegen.h"

#define PULSE_MAX_BURST    4096u
#define PULSE_MAX_SPAN_US  10000000u   /* стеля тривалості пачки: 10 с */

static DEFINE_MUTEX(pulse_lock);       /* накриває обидві змінні нижче */
static struct pulse_cfg cur_cfg = { .period_us = 1000, .duty_permille = 500 };
static u64 issued_total;

static long do_set_cfg(void __user *uarg)
{
        struct pulse_cfg c;

        if (copy_from_user(&c, uarg, sizeof(c)))
                return -EFAULT;
        if (c.period_us < 10 || c.period_us > 1000000)
                return -EINVAL;
        if (c.duty_permille == 0 || c.duty_permille > 999)
                return -EINVAL;

        mutex_lock(&pulse_lock);
        cur_cfg = c;
        mutex_unlock(&pulse_lock);
        return 0;
}

static long do_get_cfg(void __user *uarg)
{
        struct pulse_cfg c;

        memset(&c, 0, sizeof(c));          /* спершу нулі, потім поля */
        mutex_lock(&pulse_lock);
        c.period_us     = cur_cfg.period_us;
        c.duty_permille = cur_cfg.duty_permille;
        mutex_unlock(&pulse_lock);

        return copy_to_user(uarg, &c, sizeof(c)) ? -EFAULT : 0;
}

static long do_burst(void __user *uarg)
{
        struct pulse_burst b;
        u32 want, span_us;
        u64 total;

        if (copy_from_user(&b, uarg, sizeof(b)))
                return -EFAULT;
        if (b.__pad != 0 || b.issued != 0)  /* поля ядра мусять прийти нулями */
                return -EINVAL;

        want = b.want;
        if (want == 0 || want > PULSE_MAX_BURST)
                return -EINVAL;

        mutex_lock(&pulse_lock);
        if (check_mul_overflow(want, cur_cfg.period_us, &span_us) ||
            span_us > PULSE_MAX_SPAN_US) {
                mutex_unlock(&pulse_lock);
                return -EINVAL;
        }
        issued_total += want;              /* читання-зміна-запис під замком */
        total = issued_total;
        mutex_unlock(&pulse_lock);

        memset(&b, 0, sizeof(b));          /* доповнення теж піде нулем */
        b.want   = want;
        b.issued = total;

        return copy_to_user(uarg, &b, sizeof(b)) ? -EFAULT : 0;
}

static long pulse_ioctl(struct file *f, unsigned int cmd, unsigned long arg)
{
        void __user *uarg = (void __user *)arg;

        switch (cmd) {
        case PULSE_SET_CFG: return do_set_cfg(uarg);
        case PULSE_GET_CFG: return do_get_cfg(uarg);
        case PULSE_BURST:   return do_burst(uarg);
        default:            return -ENOTTY;
        }
}

static const struct file_operations pulse_fops = {
        .owner          = THIS_MODULE,
        .unlocked_ioctl = pulse_ioctl,
        .compat_ioctl   = compat_ptr_ioctl,
};

static struct miscdevice pulse_misc = {
        .minor = MISC_DYNAMIC_MINOR,
        .name  = "pulsegen",
        .fops  = &pulse_fops,
        .mode  = 0600,
};
module_misc_device(pulse_misc);

MODULE_DESCRIPTION("Навчальний генератор імпульсів із керуванням через ioctl");
MODULE_LICENSE("GPL");
```

Символьний пристрій тут заведено найдешевшим способом, який дає ядро. Замість того щоб просити власний старший номер і самому створювати вузол, [модуль](topic:unix-linux/kernel-modules) реєструється в підсистемі *misc*: вона роздає молодші номери під своїм спільним старшим — десятим, — а вузол `/dev/pulsegen` udev створює сам за іменем із структури. Макрос `module_misc_device()` розгортається у звичайну пару `misc_register()` під час завантаження і `misc_deregister()` під час вивантаження. Для драйвера, у якого один пристрій і жодних претензій на [власний старший номер](topic:unix-linux/major-minor-numbers), більшого й не треба.

## Програма

```c
/* pulsectl.c — простір користувача */
#include <errno.h>
#include <fcntl.h>
#include <stdio.h>
#include <string.h>
#include <unistd.h>
#include <sys/ioctl.h>
#include "pulsegen.h"

int main(void)
{
        int fd = open("/dev/pulsegen", O_RDWR);
        if (fd < 0) { perror("open /dev/pulsegen"); return 1; }

        struct pulse_cfg c = { .period_us = 2500, .duty_permille = 300 };
        if (ioctl(fd, PULSE_SET_CFG, &c) < 0) { perror("SET_CFG"); return 1; }

        memset(&c, 0, sizeof(c));
        if (ioctl(fd, PULSE_GET_CFG, &c) < 0) { perror("GET_CFG"); return 1; }
        printf("період %u мкс, заповнення %u‰\n", c.period_us, c.duty_permille);

        struct pulse_burst b = { .want = 64 };   /* решта полів — нулі */
        if (ioctl(fd, PULSE_BURST, &b) < 0) { perror("BURST"); return 1; }
        printf("видано %u, усього %llu\n", b.want, (unsigned long long)b.issued);

        if (ioctl(fd, _IO('Z', 0), 0) < 0)       /* команда чужої підсистеми */
                printf("чужа команда: %s\n", strerror(errno));

        close(fd);
        return 0;
}
```

Складання й запуск:

```sh
$ make -C /lib/modules/$(uname -r)/build M=$PWD modules   # obj-m := pulsegen.o
$ sudo insmod pulsegen.ko
$ ls -l /dev/pulsegen
crw------- 1 root root 10, 124 сер  8 21:03 /dev/pulsegen
$ cc -o pulsectl pulsectl.c && sudo ./pulsectl
період 2500 мкс, заповнення 300‰
видано 64, усього 64
чужа команда: Inappropriate ioctl for device
```

Останній рядок — це `default` з `switch`. Ніякого іншого способу сказати «такої операції я не вмію» в системи немає, і саме тому `−ENOTTY` не можна підміняти на `−EINVAL`: за `ENOTTY` програми з'ясовують наявність команди, а `EINVAL` означає «команду знаю, аргумент поганий».

## Чотири пастки, вже вбудовані в цей код

**Замок ніхто не візьме за вас.** Назва `unlocked_ioctl` читається буквально: ядро входить у ваш обробник, нічого перед тим не заблокувавши. `PULSE_BURST` робить читання-зміну-запис лічильника — три дії, між які легко втрутитися. Якщо дві програми одночасно замовлять по сто імпульсів, без замка лічильник виросте на сто замість двохсот: обидві прочитають те саме старе значення. Замок мусить накривати всю трійку, а не кожну дію окремо, і мусить накривати ще й читання `cur_cfg` у перевірці переповнення — інакше налаштування встигнуть змінитися між перевіркою й використанням. А от копіювання з простору користувача винесено за межі замка навмисно: `copy_from_user` має право заснути, бо потрібна сторінка може бути вивантажена. [М'ютекс це пережив би, спінлок — ні](topic:unix-linux/kernel-locking), тож звичка копіювати поза замком не зламається того дня, коли замок доведеться змінити.

**Розкладка структури залежить від ABI.** Ось чому в `pulse_burst` виписано `__pad`.

![Дві частини. Угорі структура без явного доповнення: на 32-бітному x86 поля want і issued лежать упритул, sizeof 12 і номер команди 0xC00C4203; на x86-64 між ними чотирибайтова дірка вирівнювання, sizeof 16 і номер 0xC0104203. Унизу та сама структура з явним полем __pad: розкладка однакова на обох ABI, sizeof 16 і один номер 0xC0104203](img/burst-layout.svg)

*Компілятор [вирівнює](topic:programming/memory-alignment) вісімбайтове поле по-різному залежно від архітектури: на x86-64 обов'язково по вісім, на 32-бітному x86 достатньо чотирьох. Явне поле доповнення робить те, що інакше зробив би компілятор, — але однаково скрізь.*

Ланцюжок наслідків тут короткий і нещадний: `sizeof` кладе макрос `_IOWR` у поле розміру самого номера команди, тож інша розкладка дає **інше число**, і `switch` у драйвері його не впізнає. 32-бітна програма отримає `−ENOTTY` від драйвера, з чиїм заголовком її щойно зібрали. З фіксованою розкладкою це не станеться, і тоді від [шару сумісності](topic:unix-linux/compat-32-on-64) лишається одна дрібниця: адреса від 32-бітної програми теж тридцятидвобітна, і покажчиком ядра сама собою вона не стає. Саме це й робить `compat_ptr_ioctl` у полі `.compat_ioctl`: перетворює адресу й кличе той самий `unlocked_ioctl`. Готовий помічник годиться рівно тоді, коли **всі** команди драйвера беруть покажчик на структуру однакової розкладки, — наш випадок саме такий. Драйвер, у якого хоч одна команда передає число в аргументі або структуру з `long`, мусить писати власний `compat_ioctl` і перекладати поля руками.

**Занулити перед тим, як віддавати.** `struct pulse_burst b` у `do_burst` лежить на стеку ядра, а стек ядра — не чиста дошка: там залишки того, що робила попередня функція, зокрема адреси ядерних об'єктів. `copy_to_user` копіює всі `sizeof(b)` байтів поспіль, разом із доповненням, про яке ви не думали. Заповнити лише поля й віддати структуру означає долити в неї сміття з ядра — це витік, і в історії ядра таких латок сотні. Тому `memset` стоїть **перед** заповненням полів, а не замість нього. І з тієї ж причини в `do_get_cfg` поля переписуються поодинці, а не присвоєнням структури цілком: присвоєння має право скопіювати й доповнення джерела.

> 🔧 **Навіщо це.** Правило зводиться до однієї звички: усе, що йде нагору через `copy_to_user`, спершу занулюється цілком, потім заповнюється по полю. Ця звичка переживає рефакторинг — коли через рік у структуру додадуть поле й забудуть присвоїти його в одній із гілок, воно поїде нулем, а не змістом стека.

**Межі й переповнення перевіряти нема кому.** Кожна команда ioctl — приватний системний виклик, який автор драйвера спроєктував сам; загального механізму, що перевірив би аргументи за нього, не існує, бо ядро не знає їхнього формату. Звідси три перевірки в `do_burst`. Стеля `want` — щоб замовлення не перетворилося на нескінченну роботу в ядрі. Далі рахуємо тривалість пачки, і ось тут потрібен `check_mul_overflow()`, хоч на перший погляд і не видно навіщо:

```
найбільший дозволений добуток   4096 × 1000000 = 4.096·10⁹
стеля u32                       2³² − 1        = 4.295·10⁹
запас                                          ≈ 4.6 %
```

Тобто зараз добуток влазить — але з такою вузькою смугою, що безпека множення тримається не на коді, а на двох константах, які хтось колись підніме, не перечитавши цього рядка. Хай `PULSE_MAX_BURST` стане 65536: замовлення на 65536 імпульсів по 65536 мкс дає рівно 2³², [беззнакове переповнення](topic:programming/overflow-wraparound) обертає це на нуль — і перевірка «не довше десяти секунд» радо пропускає роботу на сімдесят хвилин. Перевірка переповнення й потрібна для того, щоб такий день закінчився `−EINVAL`, а не мовчазним прольотом. Третя перевірка — вимога, щоб `__pad` та `issued` прийшли нулями. Вона виглядає причіпкою, а насправді це єдиний спосіб залишити ці байти вільними: поки драйвер відкидає ненульові, жодна програма не почне випадково класти туди сміття, і колись у це місце можна буде додати справжнє поле, не ламаючи нікого.

## Побачити свої команди у strace

Найкоротший шлях від «пристрій поводиться дивно» до «драйвер відповів мінус чимось» — [подивитися на самі виклики](topic:unix-linux/syscall-tracing):

```
$ sudo strace -e trace=ioctl ./pulsectl
ioctl(3, _IOC(_IOC_WRITE, 0x42, 0x1, 0x8), 0x7ffc1d4a2b30) = 0
ioctl(3, _IOC(_IOC_READ, 0x42, 0x2, 0x8), 0x7ffc1d4a2b30)  = 0
ioctl(3, _IOC(_IOC_READ|_IOC_WRITE, 0x42, 0x3, 0x10), 0x7ffc1d4a2b20) = 0
ioctl(3, _IOC(0, 0x5a, 0, 0), 0) = -1 ENOTTY (Inappropriate ioctl for device)
```

Наших назв тут немає й не буде: `strace` знає лише ті команди, які вписано в його власні таблиці. Зате він розібрав число на поля — і з голого номера читається все, крім сенсу. `0x42` — це `'B'`, наша літера. `0x3` — третя команда. `0x10` — шістнадцять байтів аргументу, тобто `struct pulse_burst`, і якби програму зібрали зі старим заголовком, тут стояло б інше число, видиме неозброєним оком. Останній рядок — `_IO('Z', 0)`: напрям нульовий, розміру немає, `0x5a` — це `'Z'`.

Саме тому налагодження чужого пристрою починається з пошуку заголовка з номерами: без нього видно розмову, але не мову. З ним же `strace` перетворюється на найточніший опис того, що насправді сталося між програмою й [ядром](topic:unix-linux/kernel-and-userspace), — без жодного посередника у вигляді бібліотеки, яка могла щось додати від себе.
