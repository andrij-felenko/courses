# ⚙️ Полічити злиття власноруч: чотири проходи й лічильники /sys/block

Твердження «черга зшиває сусідні запити, і тому суцільний потік дешевший за розсипаний» звучить переконливо — і саме тому його варто не запам'ятати, а перевірити. Ядро рахує злиття явно й показує лічильник кожному охочому, тож перевірка коштує сотню рядків: подати той самий обсяг даних кількома способами й після кожного зняти різницю лічильників. Заразом перевіримо друге, ще популярніше твердження — «невирівняний запис коштує дорожче». Найцікавіше знайдеться там, де ці два твердження зустрічаються.

## Що саме рахує ядро

`/sys/block/<пристрій>/stat` — один рядок із сімнадцяти чисел, [ще одне вікно у стан заліза через sysfs](book:unix-linux/sysfs-device-model). Це лічильники від завантаження, тож сенс має не значення, а різниця до й після. Нас цікавлять чотири поля:

| № | що це |
|---|---|
| 5 | запити на запис, які дійшли до драйвера |
| 6 | `bio`, що злилися з уже наявним у черзі запитом |
| 7 | записані сектори |
| 8 | сумарне очікування записів, мс |

Ключ до всього досліду — у першій парі. П'яте поле рахує **запити, які пристрій справді побачив**, шосте — **скільки подань уникло власного запиту**. Їхня сума дорівнює тому, що подала програма. Отже, шосте поле — це не абстрактна метрика, а прямо кількість зекономлених поїздок до носія.

Сьоме поле має пастку в означенні: сектори тут завжди **юніксові по 512 байтів**, незалежно від того, якими блоками адресується конкретний носій. На диску з логічним блоком 4096 байтів це число все одно ділиться на два, щоб дати кілобайти, — а хто про це не знає, отримує восьмикратну помилку й довго шукає, де подівся трафік.

## Задум: один чинник за раз

Спокуса написати два проходи — «добре» й «погано» — і порівняти. Але «погано» тут складається з двох різних гріхів: запис не в тому місці й запис не того розміру. З двох цифр не витягти, хто винен, тому чинники розводимо й міняємо по одному.

Об'єм у всіх проходах однаковий до байта: 65 536 записів по 4 КіБ у ту саму область на 256 МіБ. Різняться проходи двома незалежними речами.

**Зсув.** Потік починається або з нуля, або з 512 байтів. У другому разі кожен запис на 4 КіБ лягає верхи на два сусідні фізичні блоки замість одного.

**Порядок.** Або слот за слотом, або вроздріб. Розсипати треба так, щоб покрити ті самі слоти рівно по разу — інакше зміниться обсяг. Це робить проста арифметика: `слот = (i · 4097) mod 65536`. Множник непарний, а отже взаємно простий із степенем двійки, тож перебір зачепить кожен слот один раз; сусідні в часі записи при цьому опиняються за 16 МіБ один від одного й злитися не мають шансу.

Пишемо повз [сторінковий кеш, із `O_DIRECT`](book:unix-linux/buffered-and-direct-io) — інакше між програмою й чергою стане кеш, склеїть сусідні сторінки сам і зробить дослід беззмістовним.

## Програма

```c
/* merges.c — скільки запитів насправді доїхало до носія.
   збірка: cc -O2 -Wall -o merges merges.c
   запуск: sudo ./merges /var/tmp/scratch.bin sda            */

#define _GNU_SOURCE
#include <errno.h>
#include <fcntl.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <time.h>
#include <unistd.h>

#define REC   4096ull                 /* один запис */
#define NREC  65536ull                /* 65536 × 4 КіБ = 256 МіБ на прохід */
#define SPAN  (REC * NREC)
#define SKIP  4097ull                 /* непарний → обходить усі слоти рівно раз */

struct dstat { unsigned long long ios, merges, sectors, ticks; };

/* /sys/block/<dev>/stat: 17 полів; поля 5–8 — це запис:
   запити, злиття, сектори (завжди по 512 Б), очікування в мс */
static int read_stat(const char *dev, struct dstat *s)
{
    char path[256];
    unsigned long long v[17] = { 0 };
    int n = 0;

    snprintf(path, sizeof path, "/sys/block/%s/stat", dev);
    FILE *f = fopen(path, "r");
    if (!f)
        return -1;
    while (n < 17 && fscanf(f, "%llu", &v[n]) == 1)
        n++;
    fclose(f);
    if (n < 8)
        return -1;
    s->ios = v[4]; s->merges = v[5]; s->sectors = v[6]; s->ticks = v[7];
    return 0;
}

/* один прохід: NREC записів по REC байтів у ту саму область.
   base — зсув усього потоку (0 або 512);
   skip — крок у слотах: 1 = слот за слотом, SKIP = вроздріб */
static void one_pass(int fd, const void *buf,
                     unsigned long long base, unsigned long long skip)
{
    for (unsigned long long i = 0; i < NREC; i++) {
        unsigned long long slot = (i * skip) % NREC;
        off_t off = (off_t)(base + slot * REC);

        if (pwrite(fd, buf, REC, off) != (ssize_t)REC) {
            fprintf(stderr, "pwrite на зсуві %lld: %s\n",
                    (long long)off, strerror(errno));
            exit(1);
        }
    }
}

/* printf міряє ширину в байтах, а назви проходів — кирилицею */
static void put_label(const char *s)
{
    int chars = 0;

    for (const unsigned char *q = (const unsigned char *)s; *q; q++)
        if ((*q & 0xC0) != 0x80)
            chars++;
    fputs(s, stdout);
    while (chars++ < 22)
        putchar(' ');
}

static void measure(const char *label, int fd, const void *buf,
                    const char *dev, unsigned long long base,
                    unsigned long long skip)
{
    struct dstat a, b;
    struct timespec t0, t1;

    sync();                                    /* хай чуже доїде до носія */
    if (read_stat(dev, &a)) { perror("stat"); exit(1); }
    clock_gettime(CLOCK_MONOTONIC, &t0);

    one_pass(fd, buf, base, skip);
    if (fsync(fd)) { perror("fsync"); exit(1); }

    clock_gettime(CLOCK_MONOTONIC, &t1);
    if (read_stat(dev, &b)) { perror("stat"); exit(1); }

    unsigned long long ios  = b.ios     - a.ios;
    unsigned long long mrg  = b.merges  - a.merges;
    unsigned long long sect = b.sectors - a.sectors;
    double sec = (t1.tv_sec - t0.tv_sec) + (t1.tv_nsec - t0.tv_nsec) / 1e9;

    put_label(label);
    printf("%8llu %9llu %9.1f %10.1f %9.2f\n",
           ios, mrg,
           sect / 2048.0,                       /* сектори по 512 Б → МіБ */
           ios ? sect / 2.0 / ios : 0.0,        /* середній запит, КіБ */
           sec);
}

int main(int argc, char **argv)
{
    if (argc != 3) {
        fprintf(stderr, "вжиток: %s <порожній файл або пристрій> <ім'я в /sys/block>\n",
                argv[0]);
        return 2;
    }

    void *buf;
    if (posix_memalign(&buf, 4096, REC)) {      /* O_DIRECT хоче вирівняний буфер */
        fprintf(stderr, "posix_memalign не дав пам'яті\n");
        return 1;
    }
    memset(buf, 0xA5, REC);

    int fd = open(argv[1], O_WRONLY | O_CREAT | O_DIRECT, 0600);
    if (fd < 0) { perror(argv[1]); return 1; }
    if (fallocate(fd, 0, 0, (off_t)(SPAN + REC)))
        perror("fallocate (для блокового пристрою — очікувано)");

    one_pass(fd, buf, 0, 1);                    /* вхолосту: розмітити ділянки */
    fsync(fd);

    put_label("прохід");
    puts(" запитів    злитих       МіБ  запит,КіБ    секунд");
    measure("вирівняно, суцільно", fd, buf, argv[2],   0, 1);
    measure("зсув 512, суцільно",  fd, buf, argv[2], 512, 1);
    measure("вирівняно, вроздріб", fd, buf, argv[2],   0, SKIP);
    measure("зсув 512, вроздріб",  fd, buf, argv[2], 512, SKIP);

    close(fd);
    free(buf);
    return 0;
}
```

## Що показали лічильники

Твердотільний диск на SATA, логічний блок 512 байтів, фізичний 4096, упорядкувальник `mq-deadline`, стеля розміру запиту `max_sectors_kb` — 1280:

```
прохід                 запитів    злитих       МіБ  запит,КіБ    секунд
вирівняно, суцільно        205     65331     256.0     1278.8      0.42
зсув 512, суцільно         206     65330     256.0     1272.5      0.44
вирівняно, вроздріб      65536         0     256.0        4.0      3.91
зсув 512, вроздріб       65536         0     256.0        4.0      6.83
```

**Скільки поїздок з'їла черга.**

```
подано bio               = 65536
доїхало запитів          = 205
злито                    = 65331   = 65536 − 205
частка злитих            = 65331 / 65536 = 99.7 %
середній запит           = 524288 сект. × 512 Б / 205 = 1.25 МіБ
```

Черга зшила все, що фізично могла: 205 — це рівно стільки разів, скільки 1280 КіБ вміщується в 256 МіБ. Дев'яносто дев'ять і сім десятих відсотка подань не стали окремим звертанням, і саме на цьому тримається різниця 0.42 проти 3.91 секунди між першим і третім проходом. Дані ті самі, розмір записів той самий; різниця — тільки в порядку подання, тобто в тому, чи мала черга з чого зшивати.

А тепер несподіванка. Зсув на 512 байтів у суцільному потоці не коштував майже нічого: 0.44 замість 0.42, п'ять відсотків, на межі шуму. Той самий зсув у розсипаному потоці коштував сімдесят п'ять відсотків: 6.83 замість 3.91. Одна й та сама вада — а ціна відрізняється в п'ятнадцять разів.

![Угорі: злитий запит на 320 блоків, у якого часткові лише два крайні блоки. Унизу: одиничний запит на 4 КіБ, у якого часткові обидва зачеплені блоки](/reference/unix-linux/devices/block-device-model/img/merge-seams.svg)

*Зсув псує тільки кінці запиту, а не його середину: у довгого запиту зіпсовані кінці губляться серед цілих блоків, у чотирикілобайтового вони і є весь запит.*

Причина видна, щойно спитати, **де саме** носій платить за зсув. Він платить на шві — на фізичному блоці, зачепленому частково: такий блок доводиться прочитати, підмінити в ньому шматок і записати назад. У запиту рівно два кінці, а отже щонайбільше два шви — скільки б блоків не лежало між ними. Злитий запит на 1280 КіБ несе 321 фізичний блок, із них часткові два: зайвої роботи менше відсотка. Незлитий запит на 4 КіБ несе два фізичні блоки, і часткові обидва: на кожні 4 КіБ корисних даних носій читає 8 КіБ і записує 8 КіБ.

> 🔧 **Навіщо це.** Порада «вирівняйте початок розділу» ходить по мережі без жодного числа, і через це її або виконують як обряд, або відмахуються. Дослід показує, що вона умовна: там, де навантаження суцільне й черга зшиває, вирівнювання коштує одиниці відсотків; там, де навантаження дрібне й розсипане — база даних, журнал, каталог із мільйоном файликів, — та сама вада подвоює роботу носія. Тобто вирівнювання рятує саме тих, кому черга допомогти вже не може.

## Пастки

**`O_DIRECT` не пробачає невирівняності.** Вимог три: адреса буфера, зсув у файлі й довжина мусять бути кратні логічному блокові пристрою. Звідси `posix_memalign` замість `malloc` — сторінкове вирівнювання завжди підходить. Порушення дає `EINVAL` без пояснень. Головне: на носії з логічним блоком 4096 байтів (таких дедалі більше) проходи зі зсувом 512 просто не запустяться. Справжню вимогу віддає `/sys/block/<пристрій>/queue/logical_block_size`, а з Linux 6.1 — [розширений `statx`](book:unix-linux/statx-extended-stat) із прапорцем `STATX_DIOALIGN`: поля `stx_dio_mem_align` і `stx_dio_offset_align` кажуть окремо про пам'ять і окремо про зсув.

**Аргумент — на порожнє місце.** Програма пише 256 МіБ і нічого не питає. Помилка в першому аргументі — і замість файлика під `/var/tmp` вона рівним шаром закатає початок робочого диска. Або окремий файл, або пристрій, на якому свідомо немає нічого.

**Лічильники спільні на весь пристрій.** Різниця «до й після» вбирає все, що робили в ту саму мить інші процеси. Міряти треба на тихій машині, `sync()` перед зняттям обов'язковий, а прохід варто повторити тричі й дивитися на розкид, а не на одну цифру.

**Диск чи розділ.** `/sys/block/sda/stat` — увесь диск, `/sys/block/sda/sda1/stat` — лише розділ; у NVMe це `nvme0n1` та `nvme0n1p1`. Робота на розділі потрапляє в обидва лічильники, тож помилитися важко, але якщо файл лежить на іншому розділі того самого диска, різниця по розділу покаже нуль при цілком живому трафіку.

**Перший прохід дорожчий за решту.** `fallocate` віддає ділянки нерозміченими, і перший запис у кожну з них тягне за собою зміну метаданих файлової системи — зайві запити, яких немає в подальших проходах. Тому перший прохід іде вхолосту, а міряються вже наступні.

**Планувальник не вимикає злиття.** [Вибір планувальника](book:unix-linux/io-schedulers) на результат впливає менше, ніж очікуєш: на NVMe з `none` злиття все одно видно в лічильнику, бо перша спроба зшити відбувається ще в заглушці подавача, до всякого впорядкувальника. `none` лише звужує вікно, у якому запити встигають зустрітися, — тому запити виходять коротші, а не поодинокі. Хто вимикає злиття насправді — це `/sys/block/<пристрій>/queue/nomerges`: нуль дозволяє все, одиниця лишає тільки просту спробу зшити з попереднім запитом, двійка забороняє будь-яку. Найкраща перевірка самого досліду — записати туди `2` й повторити перший прохід: шосте поле мусить упасти в нуль, а кількість запитів — підскочити до 65 536. Якщо не впало, ви міряєте не те, що думаєте. Після досліду поверніть `0`.

**`O_DIRECT` — це не довговічність.** Обхід сторінкового кеша не означає, що дані вже на носії: попереду ще кеш запису самого пристрою. Саме тому в кінці кожного проходу стоїть `fsync`, і саме він видно в шістнадцятому полі — лічильнику операцій скидання. Про те, чим підтверджений запис відрізняється від збереженого, є [окрема розмова](book:unix-linux/page-cache-durability).

## Те саме очима iostat

Писати програму, щоб побачити злиття, треба один раз — заради розуміння, звідки беруться числа. Далі ті самі поля показує `iostat` із пакета sysstat, який просто читає ті самі лічильники двічі й ділить різницю на час:

```
$ iostat -dxm 1 /dev/sda

Device    w/s    wMB/s   wrqm/s  %wrqm  w_await  wareq-sz
sda     488.1    609.5 155550.0  99.69     1.31    1278.8      ← суцільно
sda   16760.0     65.5      0.0   0.00     0.62       4.0      ← вроздріб
```

`wrqm/s` — це «злитих запитів на запис за секунду», тобто шосте поле, поділене на час; `%wrqm` — його частка від усього поданого, та сама, що ми рахували руками. `wareq-sz` — середній розмір запиту, який дійшов до пристрою, і саме він найшвидше видає характер навантаження: кілобайт-два означають, що черзі нема з чого зшивати, і жодне налаштування носія цього не виправить — виправляти треба того, хто подає.

Різниця лише в тому, що `iostat` показує пристрій цілком і не питає, чий це трафік. Власна програма відповідає на точніше питання — скільки запитів породила **ця** сотня рядків, — а `iostat` придатний там, де програму вже не переписати: чужа база даних, чужий сервер, чужий контейнер.
