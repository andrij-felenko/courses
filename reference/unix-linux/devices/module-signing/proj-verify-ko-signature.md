# ⚙️ Читаємо підпис `.ko` своїми руками

Ядро на невдалу перевірку відповідає одним рядком — `Key was rejected by service`, — і той самий рядок однаково означає «підпису немає», «ключ невідомий» і «підпис не збігся». Причини не видно: усередині ядра нема кому її показати. Зате всю перевірку можна повторити в просторі користувача: півтори сотні рядків на C проходять рівно ті самі чотири кроки, що й `mod_verify_sig`, але з друком на кожному. Заразом стає видно, що «ядерного» в цій процедурі рівно одне місце — те, звідки береться сертифікат.

## Що саме повторюємо

Ядро робить чотири речі поспіль. Дивиться, чи закінчується образ маркером `~Module signature appended~\n` — двадцять вісім байтів. Читає перед маркером дванадцятибайтовий опис `struct module_signature`, звідки бере одне число: довжину підпису. Відрізає весь хвіст і вважає рештою тіло, по якому рахуватиметься хеш. І віддає підпис та тіло розбирачеві PKCS#7, який дістає з підпису ім'я підписанта, знаходить за цим іменем сертифікат і звіряє.

Три з чотирьох кроків — чиста арифметика над байтами; повторити їх може будь-яка програма. Четвертий теж повторюваний, окрім однієї деталі: ядро шукає сертифікат у [своїх кільцях ключів](book:unix-linux/kernel-keyrings) — іменованих сховищах усередині ядра, куди ключі потрапляють при збиранні образу або через строго обмежені двері. Ми туди дотягтися не можемо й не мусимо: сертифікат подамо аргументом командного рядка. Саме через це наша програма відповідає на трохи інше питання, ніж ядро, — не «чи довіряє система цьому підписантові», а «чи справді підписав саме цей». Через це програма цілком може сказати «збігся» там, де ядро каже «ні», — і саме така розбіжність найкорисніша.

Файл відображаємо в пам'ять через [`mmap`](book:unix-linux/mmap-model) — механізм, що робить вміст файлу шматком адресного простору процесу без копіювання в буфер. Причина не в швидкості: підпис лежить у кінці, тіло — на початку, і обидва потрібні як суцільні діапазони байтів. З відображенням код перетворюється на арифметику над одним вказівником — точнісінько як у ядра, яке працює з уже завантаженим образом.

## Програма

```c
/* verify-ko.c — повторення ядрової перевірки підпису модуля
 * збирання: cc -O2 -Wall -o verify-ko verify-ko.c -lcrypto        */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>
#include <fcntl.h>
#include <unistd.h>
#include <arpa/inet.h>          /* ntohl */
#include <sys/mman.h>
#include <sys/stat.h>

#include <openssl/bio.h>
#include <openssl/bn.h>
#include <openssl/cms.h>
#include <openssl/err.h>
#include <openssl/objects.h>
#include <openssl/pem.h>
#include <openssl/x509.h>

#define MARKER      "~Module signature appended~\n"
#define MARKER_LEN  (sizeof(MARKER) - 1)        /* 28 */

/* дослівно з include/uapi/linux/module_signature.h */
struct module_signature {
    uint8_t  algo;          /* завжди 0 */
    uint8_t  hash;          /* завжди 0 */
    uint8_t  id_type;       /* 2 = MODULE_SIGNATURE_TYPE_PKCS7 */
    uint8_t  signer_len;    /* завжди 0 */
    uint8_t  key_id_len;    /* завжди 0 */
    uint8_t  pad[3];        /* у ядрі зветься __pad, завжди нулі */
    uint32_t sig_len;       /* мережевий порядок байтів! */
};
_Static_assert(sizeof(struct module_signature) == 12, "опис мусить бути 12 Б");

static void die(const char *msg)
{
    fprintf(stderr, "%s\n", msg);
    ERR_print_errors_fp(stderr);
    exit(1);
}

/* Ім'я підписанта й алгоритм хешування. Читається з самого повідомлення,
 * без жодного сертифіката.                                               */
static void print_signer(CMS_ContentInfo *cms)
{
    STACK_OF(CMS_SignerInfo) *sis = CMS_get0_SignerInfos(cms);
    if (!sis || sk_CMS_SignerInfo_num(sis) < 1)
        die("у повідомленні немає жодного підписанта");

    for (int i = 0; i < sk_CMS_SignerInfo_num(sis); i++) {
        CMS_SignerInfo  *si    = sk_CMS_SignerInfo_value(sis, i);
        ASN1_OCTET_STRING *kid = NULL;
        X509_NAME      *issuer = NULL;
        ASN1_INTEGER      *sno = NULL;

        if (!CMS_SignerInfo_get0_signer_id(si, &kid, &issuer, &sno))
            die("не читається ідентифікатор підписанта");

        if (issuer) {
            char   *name = X509_NAME_oneline(issuer, NULL, 0);
            BIGNUM *bn   = ASN1_INTEGER_to_BN(sno, NULL);
            char   *hex  = BN_bn2hex(bn);
            printf("підписант: видавець %s\n           серійний %s\n", name, hex);
            OPENSSL_free(hex); BN_free(bn); OPENSSL_free(name);
        } else if (kid) {
            const unsigned char *p = ASN1_STRING_get0_data(kid);
            int n = ASN1_STRING_length(kid);
            printf("підписант: ідентифікатор ключа ");
            for (int j = 0; j < n; j++) printf("%02x", p[j]);
            putchar('\n');
        }

        /* яким хешем підписано — це те, чого ядро може не вміти */
        X509_ALGOR *dig = NULL;
        CMS_SignerInfo_get0_algs(si, NULL, NULL, &dig, NULL);
        if (dig) {
            const ASN1_OBJECT *oid = NULL;
            char buf[80];
            X509_ALGOR_get0(&oid, NULL, NULL, dig);
            OBJ_obj2txt(buf, sizeof(buf), oid, 0);
            printf("хеш:       %s\n", buf);
        }
    }
}

static X509 *load_cert(const char *path)
{
    BIO *b = BIO_new_file(path, "rb");
    if (!b) die("сертифікат не відкривається");

    X509 *c = PEM_read_bio_X509(b, NULL, NULL, NULL);
    if (!c) {                       /* не PEM — пробуємо як сирий DER */
        ERR_clear_error();
        BIO_reset(b);
        c = d2i_X509_bio(b, NULL);
    }
    BIO_free(b);
    if (!c) die("сертифікат не читається ні як PEM, ні як DER");
    return c;
}

int main(int argc, char **argv)
{
    if (argc < 2 || argc > 3) {
        fprintf(stderr, "вжиток: %s <модуль.ko> [сертифікат.pem|.der]\n", argv[0]);
        return 2;
    }

    int fd = open(argv[1], O_RDONLY);
    if (fd < 0) { perror(argv[1]); return 1; }

    struct stat st;
    if (fstat(fd, &st) < 0) { perror("fstat"); return 1; }
    size_t len = (size_t)st.st_size;

    if (len <= MARKER_LEN + sizeof(struct module_signature)) {
        printf("файл закороткий, щоб узагалі мати підпис\n");
        return 1;
    }

    const unsigned char *map = mmap(NULL, len, PROT_READ, MAP_PRIVATE, fd, 0);
    if (map == MAP_FAILED) { perror("mmap"); return 1; }
    close(fd);                      /* відображення живе й без дескриптора */

    /* 1. маркер у самому кінці */
    if (memcmp(map + len - MARKER_LEN, MARKER, MARKER_LEN) != 0) {
        printf("маркера немає — для ядра це непідписаний файл (-ENODATA)\n");
        return 1;
    }

    /* 2. опис підпису — рівно перед маркером */
    struct module_signature ms;
    memcpy(&ms, map + len - MARKER_LEN - sizeof(ms), sizeof(ms));

    if (ms.id_type != 2) {
        printf("опис каже не про PKCS#7 (id_type=%u) — ядро дало б -ENOPKG\n",
               ms.id_type);
        return 1;
    }
    if (ms.algo || ms.hash || ms.signer_len || ms.key_id_len ||
        ms.pad[0] || ms.pad[1] || ms.pad[2]) {
        printf("в описі ненульові поля — ядро дало б -EBADMSG\n");
        return 1;
    }

    size_t sig_len = ntohl(ms.sig_len);          /* ← без цього все розсиплеться */
    size_t tail    = MARKER_LEN + sizeof(ms) + sig_len;
    if (tail >= len) {
        printf("довжина підпису більша за сам файл — -EBADMSG\n");
        return 1;
    }
    size_t body = len - tail;

    printf("розкладка: тіло %zu + підпис %zu + опис %zu + маркер %zu = %zu\n",
           body, sig_len, sizeof(ms), MARKER_LEN, len);

    /* 3. розбір повідомлення CMS */
    BIO *sig_bio = BIO_new_mem_buf(map + body, (int)sig_len);
    CMS_ContentInfo *cms = d2i_CMS_bio(sig_bio, NULL);
    if (!cms) die("повідомлення CMS не розбирається");

    print_signer(cms);

    int rc = 0;
    if (argc == 3) {
        /* 4. звірка підпису з поданим сертифікатом */
        STACK_OF(X509) *certs = sk_X509_new_null();
        sk_X509_push(certs, load_cert(argv[2]));

        BIO *body_bio = BIO_new_mem_buf(map, (int)body);

        int ok = CMS_verify(cms, certs, NULL, body_bio, NULL,
                            CMS_NO_SIGNER_CERT_VERIFY | CMS_BINARY | CMS_NOINTERN);
        if (ok == 1) {
            printf("ПІДПИС ЗБІГСЯ\n");
        } else {
            printf("ПІДПИС НЕ ЗБІГСЯ\n");
            ERR_print_errors_fp(stderr);
            rc = 1;
        }
        BIO_free(body_bio);
        sk_X509_pop_free(certs, X509_free);
    } else {
        printf("сертифіката не подано — звіряти нема з чим\n");
    }

    CMS_ContentInfo_free(cms);
    BIO_free(sig_bio);
    munmap((void *)map, len);
    return rc;
}
```

Три прапорці у виклику `CMS_verify` — не прикраса, кожен закриває свою діру. `CMS_NOINTERN` забороняє шукати сертифікат підписанта всередині самого повідомлення: `sign-file` кладе туди лише ім'я, а не сертифікат, тож шукати там нема чого, і дозвіл шукати означав би, що зловмисник міг би вкласти власний сертифікат і сам себе «підтвердити». `CMS_NO_SIGNER_CERT_VERIFY` вимикає перевірку ланцюга довіри — без нього OpenSSL зажадав би сховища кореневих центрів, а ми навмисно питаємо лише «чи підписав саме цей ключ». `CMS_BINARY` забороняє бібліотеці чіпати переведення рядків: тіло модуля — двійкові дані, і будь-яка «нормалізація» зіпсувала б хеш. Через вимкнену перевірку ланцюга сховище `X509_STORE` можна не створювати взагалі й передати `NULL`.

Сам підпис — [відокремлений](book:math/hash-and-digital-signature): у повідомленні лежить перетворений хеш, а дані, по яких його рахували, подають окремо. Саме тому тіло їде третім аргументом, у параметрі `dcont`.

## Збирання й запуск

Дистрибутиви давно тримають модулі стиснутими, і стискають їх **після** підписування — отже, розпакований файл несе підпис у цілості:

```sh
$ cc -O2 -Wall -o verify-ko verify-ko.c -lcrypto
$ zstd -dc /lib/modules/$(uname -r)/kernel/fs/xfs/xfs.ko.zst > /tmp/xfs.ko
$ ./verify-ko /tmp/xfs.ko
розкладка: тіло 3512281 + підпис 471 + опис 12 + маркер 28 = 3512792
підписант: видавець /CN=Build time autogenerated kernel key
           серійний 6C1F0A3D8B7E4C2159A0
хеш:       sha256
сертифіката не подано — звіряти нема з чим
```

Ім'я `Build time autogenerated kernel key` — типовий рядок із `certs/default_x509.genkey` у дереві ядра; воно з'являється там, де ядро зібрано з ключем, згенерованим на час збирання. Далі показувати нема чого: приватний ключ знищено після збирання, а публічний сертифікат вкомпільовано в образ ядра й окремим файлом на диску не лежить. Тому доведення кола замкнемо власним ключем:

```sh
$ openssl req -new -x509 -newkey rsa:4096 -nodes -days 3650 \
      -keyout my.key -out my.crt -subj '/CN=Мій ключ для модулів'
$ cp /tmp/xfs.ko /tmp/copy.ko
$ /usr/src/linux-headers-$(uname -r)/scripts/sign-file sha256 my.key my.crt /tmp/copy.ko
$ ./verify-ko /tmp/copy.ko my.crt
розкладка: тіло 3513303 + підпис 758 + опис 12 + маркер 28 = 3514101
підписант: видавець /CN=Мій ключ для модулів
           серійний 5B0FE31C7A94D2680B
хеш:       sha256
ПІДПИС ЗБІГСЯ
```

(У Fedora `sign-file` лежить у `/usr/src/kernels/$(uname -r)/scripts/`.) Тіло тут довше за початковий файл — і це не помилка: старий хвіст ніхто не знімав, він просто став частиною тіла під новим підписом. Підписи вкладаються один в одного, а читають завжди лише найзовнішній.

Тепер зіпсуємо один байт у тілі:

```sh
$ printf '\x00' | dd of=/tmp/copy.ko bs=1 seek=1000 count=1 conv=notrunc
$ ./verify-ko /tmp/copy.ko my.crt
ПІДПИС НЕ ЗБІГСЯ
```

## Що з цим робити далі

Головна користь програми не в слові «збігся», а в тому, що вона розводить чотири різні біди, які ядро зливає в одну відповідь.

Немає маркера — підпис не накладали взагалі або зняли пізнішою обробкою файлу; шукати треба в збиранні, а не в ключах. Опис із чужим `id_type` чи ненульовими полями — файл підписано не тим форматом, який це ядро розуміє. Рядок «хеш» називає алгоритм: якщо там щось, чого в цьому ядрі не зібрано, дістанете `-ENOPKG` — окрема відмова, яку легко сплутати з проблемою ключа. І, нарешті, найкорисніший випадок: програма каже **ПІДПИС ЗБІГСЯ**, а ядро той самий модуль однаково не бере.

Це означає, що з криптографією все гаразд, а проблема суто в довірі: сертифіката підписанта немає в тому кільці, куди дивиться перевірка модулів. Перевіряється це вже не програмою, а поглядом у самі кільця:

```sh
$ sudo keyctl list %:.builtin_trusted_keys
1 key in keyring:
1015069133: ---lswrv     0     0 asymmetric: Build time autogenerated kernel key: 6c1f0a...
$ sudo keyctl list %:.machine
keyring is empty
```

Порівнювати треба обережно. У списку стоїть **суб'єкт** ключа й **ідентифікатор ключа**, а наша програма друкує **видавця** й **серійний номер** — це різні поля, і збіг шістнадцяткових рядків буде лише тоді, коли модуль підписано з `sign-file -k`, тобто за ідентифікатором ключа. Надійно звіряється ім'я: коли в підписі стоїть `/CN=Мій ключ для модулів`, а в кільцях такого суб'єкта немає ніде, відповідь знайдено — і жоден новий підпис справи не змінить, доки сертифікат не опиниться в кільці, яке перевірка модулів справді питає.

## Ціна й пастки

Роботи тут рівно один прохід по файлу: `mmap` не копіює нічого, розбір CMS торкається кількох сотень байтів, а весь час з'їдає хешування тіла — лінійне від розміру модуля, десятки мегабайтів за секунду на будь-якому сучасному процесорі. Ядро платить рівно стільки ж, і саме тому перевірка при завантаженні непомітна.

**Забутий `ntohl`.** Поле `sig_len` записане в мережевому порядку байтів — [старшим байтом уперед](book:programming/bits-bytes-endianness), тоді як процесор x86 чи ARM читає число навпаки. Без перетворення 471 обернеться на 3 942 383 616, довжина хвоста перевищить файл — і найкраще, що станеться, це чесна помилка. Найгірше — програма мовчки візьме за тіло не той діапазон.

**Ненульові поля опису.** Перевірку решти семи байтів легко полічити зайвою, але вона змістовна: колись ті поля справді щось означали, і ненульове значення сьогодні каже, що файл підписано форматом, якого ця перевірка не розуміє. Промовчати про це — те саме, що видати чужий формат за перевірений.

**Сертифікат обов'язково ззовні.** Пропустити `CMS_NOINTERN` спокусливо: без нього код «просто працює» на повідомленнях із вкладеним сертифікатом. Але тоді перевірка перестає бути перевіркою — підписант сам приносить документ, яким його впізнають.

**Відсутні підписані атрибути.** `sign-file` кладе підпис прямо на байти модуля, без проміжного набору полів. Через це виникає спокуса зробити все руками: порахувати `sha256` тіла й порівняти з розшифрованим числом. У цьому випадку вийде — а на повідомленні з атрибутами не вийде, бо там підпис накриває не дані, а їхній DER-опис. `CMS_verify` розрізняє обидва випадки сам, і це достатня причина не писати цю арифметику власноруч. Розбір DER, до речі, і є той `d2i` в назві `d2i_CMS_bio`: «DER to internal», з байтового запису [ASN.1](book:communications/asn1-per) у структури бібліотеки.

**Файл, який хтось чіпав після підписання.** Стиснення підпису не шкодить, а от `strip` чи `objcopy` — вбивають: хвіст або відпадає разом із кінцем файлу, або перестає відповідати новому тілу. Такий модуль наша програма покаже як «маркера немає» або «підпис не збігся» — залежно від того, що саме зробив інструмент.

**Дрібниця, яка колись стрельне.** `BIO_new_mem_buf` бере довжину як `int`. Для модулів це не межа — найбільші не дотягують і до сотні мегабайтів, — але код, скопійований звідси для файлів іншого роду, на двох гігабайтах поводитиметься дивно.
