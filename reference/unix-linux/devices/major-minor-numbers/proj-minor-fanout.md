# ⚙️ Один драйвер на чотири пристрої: розведення за молодшим номером

Найкоротший спосіб побачити, чим старший номер відрізняється від молодшого, — написати драйвер, який бере в ядра один старший на всіх і сам розводить виклики за молодшим. Нижче — робочий [модуль ядра](book:unix-linux/kernel-modules) на сотню рядків, у якому кожен крок цього поділу перевіряється командою в оболонці.

## Задача

Поставити в системі чотири символьні пристрої — `/dev/fanout0` … `/dev/fanout3`, — за якими стоїть **один** шматок коду й **одна** таблиця операцій. Кожен вузол мусить поводитися як окремий маленький буфер: що записали в нульовий, читається лише з нульового, і сусіди про це не знають.

Мову тут не обирають: код виконується в ядрі, збирається його заголовками й має відповідати його ABI. Це C.

## Ідея: попросити відтинок і розвести всередині

Драйвер робить три дії, і кожна лягає рівно на одну частину пари.

**Просить номери.** `alloc_chrdev_region(&first, 0, 4, "fanout")` каже ядру: «дай будь-який вільний старший і закріпи за мною чотири молодші, починаючи з нульового». Назад приїжджає одне число типу `dev_t` — упакована пара «виданий старший, молодший 0».

**Реєструє один код на весь відтинок.** `cdev_add()` кладе в мапу ядра запис: «пари, починаючи від цієї, кількістю чотири, — мої, ось таблиця операцій». Таблиця одна; вона нічого не знає про те, що примірників чотири.

**Розводить в `open`.** Ядро довело виклик до драйвера й далі не втручається. Молодший номер лежить в іноді недоторканим, драйвер дістає його через `MINOR(inode->i_rdev)`, обирає свій примірник і кладе вказівник на нього в `file->private_data`. Після цього номер більше нікому не потрібен: `read` і `write` беруть готовий вказівник.

Розведення, отже, коштує один вибір з масиву й одне присвоєння — і трапляється воно **раз на відкриття**, а не на кожен виклик.

## Модуль

```c
/* fanout.c — один драйвер, чотири незалежні примірники */
#include <linux/cdev.h>
#include <linux/device.h>
#include <linux/fs.h>
#include <linux/init.h>
#include <linux/module.h>
#include <linux/mutex.h>
#include <linux/uaccess.h>

#define FANOUT_COUNT 4
#define FANOUT_SIZE  512

struct fanout_slot {
    struct mutex lock;
    char   buf[FANOUT_SIZE];
    size_t len;
};

static struct fanout_slot slots[FANOUT_COUNT];
static dev_t          first;          /* пара «виданий старший, молодший 0» */
static struct cdev    fanout_cdev;
static struct class  *fanout_class;

static int fanout_open(struct inode *inode, struct file *file)
{
    unsigned int idx = MINOR(inode->i_rdev);   /* те саме, що iminor(inode) */

    if (idx >= FANOUT_COUNT)
        return -ENXIO;
    file->private_data = &slots[idx];          /* далі номер не потрібен */
    return 0;
}

static ssize_t fanout_read(struct file *file, char __user *ubuf,
                           size_t count, loff_t *ppos)
{
    struct fanout_slot *s = file->private_data;
    ssize_t n;

    if (mutex_lock_interruptible(&s->lock))
        return -ERESTARTSYS;
    n = simple_read_from_buffer(ubuf, count, ppos, s->buf, s->len);
    mutex_unlock(&s->lock);
    return n;
}

static ssize_t fanout_write(struct file *file, const char __user *ubuf,
                            size_t count, loff_t *ppos)
{
    struct fanout_slot *s = file->private_data;
    ssize_t n;

    if (mutex_lock_interruptible(&s->lock))
        return -ERESTARTSYS;
    n = simple_write_to_buffer(s->buf, FANOUT_SIZE, ppos, ubuf, count);
    if (n > 0)
        s->len = *ppos;      /* довжина = де скінчився останній запис */
    mutex_unlock(&s->lock);
    return n;
}

static const struct file_operations fanout_fops = {
    .owner  = THIS_MODULE,
    .open   = fanout_open,
    .read   = fanout_read,
    .write  = fanout_write,
    .llseek = default_llseek,
};

static int __init fanout_init(void)
{
    int i, err;

    err = alloc_chrdev_region(&first, 0, FANOUT_COUNT, "fanout");
    if (err)
        return err;

    for (i = 0; i < FANOUT_COUNT; i++)
        mutex_init(&slots[i].lock);

    cdev_init(&fanout_cdev, &fanout_fops);
    fanout_cdev.owner = THIS_MODULE;
    err = cdev_add(&fanout_cdev, first, FANOUT_COUNT);   /* ВЕСЬ відтинок */
    if (err)
        goto err_region;

    fanout_class = class_create("fanout");               /* ядро 6.4 і новіші */
    if (IS_ERR(fanout_class)) {
        err = PTR_ERR(fanout_class);
        goto err_cdev;
    }

    for (i = 0; i < FANOUT_COUNT; i++) {
        struct device *d = device_create(fanout_class, NULL,
                                         MKDEV(MAJOR(first), i), NULL,
                                         "fanout%d", i);
        if (IS_ERR(d)) {
            err = PTR_ERR(d);
            goto err_devices;
        }
    }

    pr_info("fanout: старший %d, молодші %d…%d\n", MAJOR(first),
            MINOR(first), MINOR(first) + FANOUT_COUNT - 1);
    return 0;

err_devices:
    while (--i >= 0)
        device_destroy(fanout_class, MKDEV(MAJOR(first), i));
    class_destroy(fanout_class);
err_cdev:
    cdev_del(&fanout_cdev);
err_region:
    unregister_chrdev_region(first, FANOUT_COUNT);
    return err;
}

static void __exit fanout_exit(void)
{
    int i;

    for (i = 0; i < FANOUT_COUNT; i++)
        device_destroy(fanout_class, MKDEV(MAJOR(first), i));
    class_destroy(fanout_class);
    cdev_del(&fanout_cdev);
    unregister_chrdev_region(first, FANOUT_COUNT);
}

module_init(fanout_init);
module_exit(fanout_exit);
MODULE_LICENSE("GPL");
MODULE_DESCRIPTION("Один драйвер на чотири примірники, розведені молодшим номером");
```

> 🔧 **Навіщо це.** `MINOR(inode->i_rdev)` править тут за індекс масиву лише тому, що ми попросили відтинок **від нульового** молодшого. Попросіть `alloc_chrdev_region(&first, 8, 4, "fanout")` — і ядро закріпить пари з молодшими 8…11, а `MINOR()` віддасть вісімку там, де код чекає нуля. Чесний індекс — завжди різниця `MINOR(inode->i_rdev) - MINOR(first)`, і саме тому справжні драйвери тримають базовий молодший номер у своїй структурі поряд зі старшим. Перевірка `idx >= FANOUT_COUNT` від цієї помилки не рятує: вона ловить лише ті випадки, коли зміщене число випадково вийшло завеликим, — а відтинок від восьмого довжиною шістнадцять дасть індекс 12, який перевірку пройде й полізе в чужу пам'ять.

Ще один рядок залежить від версії ядра, і на ньому спотикається кожен, хто збирає чужий приклад. До версії 6.3 включно `class_create()` була макросом із двома доводами — `class_create(THIS_MODULE, "fanout")`; у 6.4 довід із модулем викинули, і лишилося `class_create(const char *name)`. Це той самий вид ламання, про який попереджає [нестійкість внутрішнього ABI ядра](book:unix-linux/kernel-abi-stability): підпис функції всередині ядра можуть змінити між випусками, і модуль просто не збереться.

Збирається все системою збирання самого ядра — вона знає прапорці й заголовки саме тієї версії, у яку модуль потім вантажать ([звідки береться це дерево](book:unix-linux/kernel-config-and-build)):

```makefile
obj-m := fanout.o
KDIR  ?= /lib/modules/$(shell uname -r)/build

all:
	$(MAKE) -C $(KDIR) M=$(CURDIR) modules
```

## Номер, якого ви не обирали

```sh
$ make && sudo insmod fanout.ko
$ dmesg | tail -1
fanout: старший 511, молодші 0…3
$ grep fanout /proc/devices
511 fanout
```

Число буде іншим на іншій машині — і навіть на цій після перезавантаження. Ядро видає старший із відтинка 234–254, а коли той повний (як тут), іде згори вниз від 511. Дізнатися його ззовні можна рівно з двох місць: `/proc/devices` і атрибут `dev` у [дереві sysfs](book:unix-linux/sysfs-device-model).

```sh
$ cat /sys/class/fanout/fanout2/dev
511:2
```

Зверніть увагу: `/proc/devices` — перелік **реєстрацій, а не пристроїв**. Там стоїть один рядок «511 fanout», а не чотири. Ядро показує в ньому саме те, що бачить, — хто взяв старший номер. Скільки під ним примірників і що означають їхні молодші, воно не знає й показати не може.

Вузлів у `/dev` ми руками не робили, а вони з'явилися. Це робота `device_create()`: вона заводить пристрій у моделі пристроїв ядра, ядро розсилає про нього подію, і `devtmpfs` тут-таки, всередині ядра, творить вузол із правильними числами. [udev](book:unix-linux/udev-rules) далі лише довішує права, власника й сталі символьні посилання.

Зробити те саме вручну можна — і корисно зробити раз, щоб побачити, чому так не роблять. Приберімо `class_create`/`device_create` з коду: модуль завантажиться, номери будуть закріплені, а в `/dev` не з'явиться нічого.

```sh
$ major=$(awk '$2 == "fanout" { print $1 }' /proc/devices)
$ for i in 0 1 2 3; do sudo mknod /dev/fanout$i c "$major" "$i"; done
$ sudo chmod 666 /dev/fanout*
```

Придивіться до першого рядка: номер доводиться **вичитувати з `/proc/devices` після завантаження**, бо до завантаження його не існує. Скрипт установки не може створити вузли наперед; він мусить бігти щоразу після `insmod` і щоразу перероблювати `/dev`. Саме цю безвихідь і закриває пара `class_create`/`device_create` — вузол творить той, хто перший дізнався пару. Заразом `mknod` вимагає [можливості](book:unix-linux/capabilities) `CAP_MKNOD`, тобто цей шлях недоступний нікому, крім привілейованого коду.

## Перевірка розведення

```sh
$ echo 'у нульовому' > /dev/fanout0
$ echo 'у третьому'  > /dev/fanout3
$ cat /dev/fanout0
у нульовому
$ cat /dev/fanout3
у третьому
$ cat /dev/fanout1        # порожньо: сюди ніхто не писав
$
```

Чотири вузли, чотири буфери, один `fanout_write` на всіх. А тепер найгостріший дослід — вузол під чужим іменем і в чужій файловій системі:

```sh
$ sudo mknod /tmp/inshe-imya c 511 3
$ cat /tmp/inshe-imya
у третьому
```

Ім'я не значить нічого. Ядро дійшло до `fanout_open` за старшим номером, а `fanout_open` вибрав третій примірник за молодшим — і жодну з цих дій ім'я файла не проходило.

Ціна такого розведення близька до нуля, і корисно побачити чому. Пошук у мапі ядро робить один раз, під час `open`: воно ділить пару за старшим номером на кошик і йде коротким списком зареєстрованих у ньому відтинків. Наш драйвер додав туди рівно один запис — на всі чотири пристрої. Далі не лишається нічого: `read` бере таблицю операцій з відкритого файла, а примірник — з `private_data`, обидва вказівники вже лежать готові. Тобто розведення на сотню примірників коштує стільки ж, скільки на чотири, і жодна з цих сотні не сповільнює решту. Саме тому драйверові послідовних портів однаково дешево тримати один порт і тридцять два. Ця байдужість до кількості й пояснює, чому пара чисел збудована саме так, а не як плаский перелік пристроїв.

## Пастки

### Відтинок номерів, що лишився за нікого

`alloc_chrdev_region()` записує номери за модулем у ядерному реєстрі, і цей запис **не прив'язаний до життя модуля**. Якщо `cdev_add()` не вдався, а ви просто повернули помилку, `insmod` чесно відмовить — а номери лишаться закріпленими. `/proc/devices` і далі показуватиме «511 fanout», хоча коду за цим рядком уже немає, а звільнити відтинок може лише сам ядерний виклик `unregister_chrdev_region()` — з оболонки до нього не дотягнутися, тож на практиці номери зайняті до перезавантаження. Наступний `insmod` дістане інший старший номер, ваші вузли в `/dev` тихо вкажуть у порожнечу, а лічильник вільних старших зменшиться на один назавжди.

Тому в `fanout_init()` сходинки `goto` йдуть точно у зворотному порядку до захоплень. Те саме стосується `fanout_exit()`: забути там `unregister_chrdev_region()` — це та сама діра, лише виявляється вона після `rmmod`, коли все ніби прибрано.

### Вивантаження модуля з відкритими дескрипторами

Тут ядро підстеляє солому, але лише якщо ви попросили. Поле `.owner = THIS_MODULE` у таблиці операцій і `fanout_cdev.owner` — не оздоба: коли VFS відкриває символьний вузол, він бере посилання на модуль, і лічильник користувачів росте.

```sh
$ exec 3<> /dev/fanout2          # тримаємо дескриптор відкритим
$ sudo rmmod fanout
rmmod: ERROR: Module fanout is in use
$ exec 3>&-                      # закрили
$ sudo rmmod fanout              # тепер проходить
```

Приберіть `.owner` — і `rmmod` пройде. Вузол лишиться, дескриптор лишиться, а код, на який показує `file->f_op`, зникне разом із пам'яттю модуля. Наступне читання стрибне за вказівником у звільнену область, і система впаде — не в мить `rmmod`, а колись потім, у нічим не пов'язаному місці. Це найгірший рід помилки: наслідок відірваний від причини і в часі, і за стеком.

### `cdev_add` на один номер і на відтинок

Останній довід `cdev_add()` — це **скільки послідовних пар покриває ця реєстрація**. Замініть четвірку на одиницю, лишивши все інше без змін:

![Дві панелі під спільною верхньою смугою. Верхня смуга: виклик alloc_chrdev_region з чотирма номерами — в обох випадках однаково, пари 511:0 … 511:3 записано за модулем. Ліва панель, заголовок cdev_add із доводом чотири: чотири зелені комірки 511:0, 511:1, 511:2, 511:3, під ними суцільна зелена смуга «один cdev, одна таблиця операцій», від якої вгору до кожної комірки йде стрілка; підпис — open будь-якого з чотирьох вузлів доходить до драйвера, а той обирає примірник за молодшим номером. Права панель, заголовок cdev_add із доводом один: комірка 511:0 зелена, комірки 511:1, 511:2, 511:3 червоні; під нульовою вузька зелена смуга cdev зі стрілкою вгору, під рештою — червона смуга «у мапі порожньо»; підпис — open /dev/fanout1 віддає ENXIO, хоча вузол і пара чисел бездоганні](/reference/unix-linux/devices/major-minor-numbers/img/cdev-span.svg)

*Резервування номерів і реєстрація коду — дві різні дії над двома різними таблицями; збігатися вони мусять з вашої ласки, а не самі собою.*

```sh
$ cat /dev/fanout0
у нульовому
$ cat /dev/fanout1
cat: /dev/fanout1: No such device or address
```

`ENXIO` при бездоганному вузлі спантеличує рівно доти, доки не розділити дві дії. `alloc_chrdev_region()` пише в реєстр номерів — це облік, «чиї числа». `cdev_add()` пише в мапу диспетчеризації — це вже адреса коду, і саме її переглядає ядро в мить `open`. Числа зарезервовано за вами, а коду за трьома з них немає, тож ядро чесно каже «такого пристрою немає». Плутанина тут поширена через стару однокрокову помічницю `register_chrdev()`, яка робила обидві дії разом і завжди на всі 256 молодших — звідки й звичка вважати, що реєстрація одна.

Є й другий спосіб розвести примірники, і різниця між ними — не про смак. Замість одного `cdev` на відтинок можна дати кожному примірникові власний і додати їх поодинці:

```c
struct fanout_slot {
    struct cdev  cdev;       /* свій запис у мапі на кожен примірник */
    struct mutex lock;
    char   buf[FANOUT_SIZE];
    size_t len;
};

/* в init, для кожного i: */
cdev_init(&slots[i].cdev, &fanout_fops);
slots[i].cdev.owner = THIS_MODULE;
cdev_add(&slots[i].cdev, MKDEV(MAJOR(first), i), 1);

/* і тоді в open молодший номер узагалі не потрібен: */
static int fanout_open(struct inode *inode, struct file *file)
{
    file->private_data = container_of(inode->i_cdev, struct fanout_slot, cdev);
    return 0;
}
```

Ядро само знайшло потрібний `cdev` у мапі й поклало його в `inode->i_cdev`, а `container_of()` перерахував цей вказівник у вказівник на структуру, всередині якої той `cdev` лежить. Ані `MINOR`, ані масиву, ані перевірки меж.

Обирати між двома підходами треба за тим, чи приходять і йдуть примірники поодинці. Один `cdev` на відтинок дешевший і природний, коли примірники однакові, з'являються разом і зникають разом. Окремий `cdev` на примірник дорожчий на один запис у мапі, зате дозволяє зняти третій пристрій, не торкнувшись інших, — а це рівно те, що потрібно драйверові заліза, яке [приєднують і від'єднують на ходу](book:unix-linux/usb-in-linux).
