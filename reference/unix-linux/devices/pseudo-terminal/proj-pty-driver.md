# ⚙️ Драйвер псевдотермінала на сотню рядків

Це повна робоча програма мовою C, яка створює пару псевдотермінала, запускає в ній оболонку й переганяє байти між нею та вашим справжнім терміналом — тобто найменший можливий емулятор термінала, той самий кістяк, що всередині `script`, `ssh -t`, `tmux` і `expect`. Писати її варто хоча б раз, бо саме тут стає видно, скільки роботи виконує ядро й скільки лишається програмі.

## Задача

Оболонка має бути певна, що працює за терміналом, і при цьому лишатися повністю в наших руках: кожен байт, що йде до неї, записуємо ми, кожен байт, що йде від неї, читаємо теж ми. Ззовні не має змінитися нічого — те саме запрошення, те саме редагування рядка, той самий Ctrl+C, `vim` на весь екран і правильна ширина після того, як ви потягли за ріг вікна.

Наївна версія («створити пару, породити оболонку, ганяти байти в циклі») збирається з першого разу й одразу поводиться дивно: кожен символ двоїться, Ctrl+C убиває не те, `top` малює 24 рядки в будь-якому вікні, а після виходу термінал лишається німим. Усі ці поломки мають одну спільну причину.

## Ідея: терміналів насправді два

Перший — справжній, той, у якому ви запустили програму. Другий — щойно створений псевдотермінал, у якому сидить оболонка. Обидва мають власну лінійну дисципліну, і обидва за замовчуванням увімкнені на повну: обидва малюють відлуння, обидва тримають рядок до Enter, обидва перетворюють `0x03` на сигнал.

А працювати має рівно один — той, під яким живе оболонка. Наш власний термінал мусить перестати думати й стати голим проводом. Звідси й уся програма: загасити обробку в себе, перекачувати байти в обидва боки, тримати розмір вікна однаковим — і на виході повернути свій [`termios`](book:unix-linux/tty-and-termios) таким, яким він був.

![Схема з двох рядів. Верхній ряд зліва направо: рамка «наш термінал і наш процес» із приміткою, що в сирому режимі вимкнено ECHO, ICANON та ISIG, а процес робить read з дескриптора 0 і write у ведучий кінець; стрілка веде до рамки «ядро pty», де ECHO повертає байт назад, а ICANON тримає рядок до Enter; далі стрілка з написом «після Enter» веде до рамки «оболонка читає підлеглий кінець». Від рамки ядра вниз іде зелена стрілка з написом, що це відлуння й той самий байт з’явився на ведучому кінці. Нижній ряд справа наліво: рамка «наш процес: read з ведучого кінця, write у дескриптор 1», стрілка до рамки «наш екран, символ з’являється один раз». Унизу червоний напис, що коли лишити ECHO в себе, на екрані буде подвоєний символ, бо намалювали б і ми, і ядро pty](/reference/unix-linux/devices/pseudo-terminal/img/two-terminals.svg)

*Символ на екрані малює ядро другого термінала, а наш процес лише переносить його. Свою власну обробку доводиться гасити саме тому, що інакше та сама робота відбудеться двічі.*

## Код

```c
#define _XOPEN_SOURCE 600
#define _DEFAULT_SOURCE
#include <errno.h>
#include <fcntl.h>
#include <poll.h>
#include <signal.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/ioctl.h>
#include <sys/wait.h>
#include <termios.h>
#include <unistd.h>

static struct termios saved;      /* режим НАШОГО термінала до втручання */
static int master = -1;
static int winch_pipe[2];         /* самотруба: обробник SIGWINCH → цикл */

static void restore(void) { tcsetattr(STDIN_FILENO, TCSAFLUSH, &saved); }

static void on_winch(int sig) {
    int e = errno;                /* обробник не сміє псувати чужий errno */
    (void)sig;
    write(winch_pipe[1], "!", 1);
    errno = e;
}

static void push_winsize(void) {
    struct winsize ws;
    if (ioctl(STDIN_FILENO, TIOCGWINSZ, &ws) == 0)
        ioctl(master, TIOCSWINSZ, &ws);   /* SIGWINCH дитині надішле ядро само */
}

static int write_all(int fd, const char *p, size_t n) {
    while (n > 0) {
        ssize_t w = write(fd, p, n);
        if (w < 0) { if (errno == EINTR) continue; return -1; }
        p += w; n -= (size_t)w;
    }
    return 0;
}

int main(int argc, char **argv) {
    const char *shell = (argc > 1) ? argv[1] : "/bin/sh";

    if (tcgetattr(STDIN_FILENO, &saved) < 0) {
        fprintf(stderr, "стандартний вхід — не термінал\n");
        return 1;
    }

    master = posix_openpt(O_RDWR | O_NOCTTY);
    if (master < 0 || grantpt(master) < 0 || unlockpt(master) < 0) {
        perror("псевдотермінал"); return 1;
    }
    int slave = open(ptsname(master), O_RDWR);
    if (slave < 0) { perror("підлеглий кінець"); return 1; }

    tcsetattr(master, TCSANOW, &saved);   /* дитині — наші erase, kill, ISIG */
    push_winsize();                       /* і наш розмір ще до старту */

    pid_t pid = fork();
    if (pid < 0) { perror("fork"); return 1; }
    if (pid == 0) {
        close(master);
        setsid();                         /* власний сеанс без керуючого термінала */
        ioctl(slave, TIOCSCTTY, 0);       /* керуючим стає підлеглий кінець */
        dup2(slave, STDIN_FILENO);
        dup2(slave, STDOUT_FILENO);
        dup2(slave, STDERR_FILENO);
        if (slave > STDERR_FILENO) close(slave);
        execl(shell, shell, (char *)NULL);
        _exit(127);
    }
    close(slave);                         /* інакше EIO не прийде НІКОЛИ */

    struct termios raw = saved;
    cfmakeraw(&raw);
    tcsetattr(STDIN_FILENO, TCSAFLUSH, &raw);
    atexit(restore);

    if (pipe(winch_pipe) < 0) { perror("pipe"); return 1; }
    struct sigaction sa;
    memset(&sa, 0, sizeof sa);
    sa.sa_handler = on_winch;
    sigemptyset(&sa.sa_mask);
    sa.sa_flags = SA_RESTART;
    sigaction(SIGWINCH, &sa, NULL);

    struct pollfd p[3] = {
        { STDIN_FILENO,  POLLIN, 0 },
        { master,        POLLIN, 0 },
        { winch_pipe[0], POLLIN, 0 },
    };
    char buf[4096];

    for (;;) {
        if (poll(p, 3, -1) < 0) {
            if (errno == EINTR) continue;
            break;
        }
        if (p[2].revents & POLLIN) {              /* вікно змінили */
            char drain[64];
            read(winch_pipe[0], drain, sizeof drain);
            push_winsize();
        }
        if (p[0].revents & POLLIN) {              /* клавіші → ведучий кінець */
            ssize_t n = read(STDIN_FILENO, buf, sizeof buf);
            if (n < 0 && errno == EINTR) continue;
            if (n <= 0) break;
            if (write_all(master, buf, (size_t)n) < 0) break;
        }
        if (p[1].revents & (POLLIN | POLLHUP)) {  /* вивід дитини → наш екран */
            ssize_t n = read(master, buf, sizeof buf);
            if (n < 0 && errno == EINTR) continue;
            if (n <= 0) break;                    /* 0 або EIO — розмова скінчилася */
            if (write_all(STDOUT_FILENO, buf, (size_t)n) < 0) break;
        }
    }

    int status = 0;
    while (waitpid(pid, &status, 0) < 0 && errno == EINTR)
        ;
    restore();
    return WIFEXITED(status) ? WEXITSTATUS(status) : 1;
}
```

Збирається без бібліотек: `cc -o ptyrun ptyrun.c`, запускається як `./ptyrun /bin/bash`.

`cfmakeraw` прийшла з BSD й у POSIX не входить, хоча є в glibc, musl і в самих BSD-системах. Вона гасить рівно те, що заважає: `ECHO`, `ICANON` та `ISIG` у `c_lflag`, `ICRNL` та `IXON` у `c_iflag` і `OPOST` у `c_oflag`. Останнє важливе окремо: вихідну обробку вже зробило ядро псевдотермінала, тож наш вихід має пройти байт у байт.

Усе, що в програмі стоїть до `fork`, бібліотека вміє зробити одним викликом. `openpty` створює пару й повертає обидва дескриптори, `forkpty` додає до цього `fork`, `setsid`, `TIOCSCTTY` і три `dup2`. Обидві прийшли з BSD й лежать у glibc в окремій бібліотеці `libutil`. У готовій програмі ними варто користуватися; тут вони не годяться саме тому, що ховають рівно ті кроки, заради яких таку програму й пишуть.

## Як перевірити кожну частину окремо

Програма влаштована так, що кожен її шматок має власний видимий симптом. Перевірки займають хвилину:

```sh
$ ./ptyrun /bin/bash
bash-5.2$ tty
/dev/pts/7          # dup2 спрацював: оболонка бачить підлеглий кінець
bash-5.2$ stty size
33 118              # TIOCSWINSZ до fork спрацював, інакше було б «0 0»
bash-5.2$ sleep 100
^C                  # TIOCSCTTY спрацював: SIGINT дійшов саме до sleep
bash-5.2$ stty size
41 96               # потягнули за ріг: SIGWINCH → самотруба → TIOCSWINSZ
```

Якщо забути `setsid`, оболонка при старті поскаржиться `no job control in this shell`, а Ctrl+C уб'є не `sleep`, а вашу власну програму разом із ним — обидві лишилися в одній групі переднього плану. Якщо забути `cfmakeraw`, кожна набрана літера з'явиться двічі. Якщо забути `close(slave)` у батька, вихід з оболонки просто нічого не змінить: запрошення зникне, а програма висітиме.

## Що тут неочевидне

**`close(slave)` у батька.** Поки батько тримає підлеглий кінець відкритим, для ядра на тому боці ще хтось є. Оболонка завершиться, а читання з ведучого кінця й далі спокійно чекатиме — програма зависне назавжди, і жоден із трьох дескрипторів у `poll` ніколи не прокинеться.

**Налаштування до `fork`, а не після.** `tcsetattr` на ведучому кінці править `termios` усієї пари, тож оболонка стартує з тим самим символом забою й тими самими керуючими символами, які має ваш термінал. Розмір теж треба покласти заздалегідь: невстановлений розмір дорівнює нулю, і повноекранні програми падають до припущення 80×24.

**Самотруба замість прапорця.** Обробник сигналу не може викликати `ioctl`: дозволені лише [асинхронно-сигнально-безпечні](book:unix-linux/async-signal-safety) виклики, а `write` серед них є. Тому обробник пише один байт у [трубу](book:unix-linux/pipes-and-fifo), а справжню роботу робить головний цикл. Простий `volatile sig_atomic_t` тут теж майже працює: [`poll`](book:unix-linux/select-poll-epoll) у Linux ніколи не перезапускається після обробника й повертає [`EINTR`](book:unix-linux/eintr-and-restart) навіть із `SA_RESTART`, тож зміну ми б помітили. Але лишається вузьке вікно: сигнал може прийти після перевірки прапорця й до того, як `poll` засне. Тоді розмір оновиться аж після наступного натискання клавіші. Байт у трубі цього вікна не має — він уже лежить, і `poll` повертається негайно.

**Немає обробника `SIGCHLD` — і це навмисно.** Умова виходу тут одна: читання з ведучого кінця перестало давати дані. Смерть дитини цією умовою не є, бо після неї в буфері може ще лежати виведення.

**Змінна `TERM` не встановлюється ніде — і працює.** Дитина успадковує наше довкілля, а в ньому `TERM` уже описує той самий екран, куди ми зрештою все й виводимо. Це не збіг, а знову та сама прозорість: малює той самий емулятор. Але щойно програму запускають звідти, де термінала не було — з демона, з правила `cron`, з агента збірки, — `TERM` порожня, і `vim` усередині відмовиться стартувати. У такому разі значення треба покласти самому, і чесне значення тут `dumb`, доки ви не знаєте, хто на тому боці читає.

**Код завершення переноситься теж.** `waitpid` дає статус дитини, і `WIFEXITED`/`WEXITSTATUS` перетворюють його на власний код повернення. Без цього обгортка з'їдала б результат, а всі конвеєри та збірки, що дивляться на код виходу, вважали б будь-яку невдалу команду вдалою.

## Чому сотні рядків вистачає

Програма не розуміє жодної керуючої послідовності. Вона не знає, що `\x1b[2J` очищає екран, а `\x1b[31m` робить текст червоним, — і знати не мусить: обидва термінали говорять однією мовою, тож достатньо переносити байти незміненими. Звідси й приємний наслідок: `read` спокійно може розрізати навпіл багатобайтовий символ UTF-8 чи саму послідовність, бо склеїть їх той, хто малює.

Уся термінальна семантика поділена між двома чужими для нас сторонами. Ядро тримає редагування рядка, сигнали й розмір. Справжній емулятор на іншому кінці перетворює потік байтів на пікселі. Наш посередник — прозорий провід між ними, і саме тому він короткий. Довгим він стає тоді, коли перестає бути проводом: `tmux` показує кілька псевдотерміналів в одному вікні, а отже мусить тримати власну модель екрана для кожної панелі, розбирати весь вхідний потік послідовностей і синтезувати новий. Це вже інша задача, і кістяк із цієї сотні рядків у ній — найменша частина.

## Пастки

**Подвійне відлуння.** Найперша поломка й найпростіша: не загасили `ECHO` в себе — і кожен символ малюють двоє, ви й ядро другого термінала. Симптом упізнаваний: набране `ls -l` перетворюється на `llss  --ll`, бо два джерела чергуються посимвольно.

**Утрачений хвіст виведення.** Спокуса вийти, щойно `poll` повернув `POLLHUP` на ведучому кінці, коштує останнього екрана тексту. `POLLHUP` каже «більше не надійде», а не «нічого не лишилося»: дані, які оболонка встигла записати перед смертю, досі чекають у буфері. Тому в коді `POLLHUP` не є приводом вийти — приводом є `read`, що повернув нуль або `EIO`. З тієї ж причини `waitpid` стоїть **після** циклу, а не всередині.

**Невідновлений `termios`.** Вийшли аварійно — і термінал лишається сирим: ні відлуння, ні Enter, ні Ctrl+C. `atexit` рятує від звичайних шляхів виходу, але не від `SIGKILL` і не від падіння за сигналом із типовою дією. Варто знати й ціну `TCSAFLUSH` у наших двох викликах: на вході в сирий режим він викидає все, що вже набрано, і це саме те, що треба — рядок, набраний за старими правилами, за новими прочитався б неправильно. На виході та сама дія викине клавіші, натиснуті в останні мілісекунди сеансу; кому вони дорогі, той відновлює режим через `TCSADRAIN`. Надійніше — ще й перехоплювати `SIGTERM` та `SIGHUP`, а користувачеві, який уже потрапив у німу оболонку, допоможе набране наосліп `stty sane` або `reset`.

**Часткові записи й тупик.** `write` має право записати менше, ніж просили, — звідси `write_all`. Але й повний цикл не рятує від гіршого: дескриптори тут [блокувальні](book:unix-linux/blocking-and-nonblocking), і якщо вхідна черга псевдотермінала переповниться (у канонічному режимі вона обривається на 4095 байтах), ми заснемо всередині `write_all`. Тим часом дитина заповнить власну чергу виведення й засне у своєму `write`, бо читати з ведучого кінця нікому — ми ж спимо. Обидва стоять назавжди. Ловиться це вставленням у програму великого шматка тексту й лікується тим, що роблять справжні емулятори: обидва дескриптори переводять у неблокуючий режим, заводять окрему чергу на кожен напрямок і додають у `poll` подію `POLLOUT`, коли ту чергу є куди відвантажувати. Це подвоює довжину програми — і саме тому такого тут немає.
