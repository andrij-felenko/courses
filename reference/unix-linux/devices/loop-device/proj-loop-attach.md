# ⚙️ Приєднати образ власноруч: loop-пристрій без losetup

Усе, що робить `losetup`, зводиться до трьох звертань до ядра: узяти вільний номер, налаштувати на ньому файл, відпустити. Написані власноруч, ці три звертання показують контракт драйвера зблизька — разом із півдесятком місць, де він поводиться не так, як здається з опису.

## Що має вміти програма

Вона приймає шлях до образу й, за бажанням, зсув у байтах. Знаходить вільний loop-пристрій, приєднує до нього образ із указаним зсувом, розміром блока й проханням працювати повз кеш, читає з готового пристрою нульовий сектор і друкує його початок, після чого відчіпляє все за собою. Ключова вимога — остання: пристрій не має лишитися висіти ні при успіху, ні при помилці на півдорозі.

## Три звертання

Пристроями керує [`ioctl`](book:unix-linux/ioctl-interface) — виклик, який передає драйверові команду й структуру даних через звичайний відкритий файл. Файлів тут два різні, і плутати їхні ролі не варто.

`/dev/loop-control` — не пристрій, а окреме віконце для запитів про пристрої. Єдине, чого від нього треба, — `LOOP_CTL_GET_FREE`: він повертає номер пристрою, який зараз ні до чого не прив'язаний, а якщо всі наявні зайняті — створює новий і повертає його. Другий файл, `/dev/loopN`, — уже сам пристрій, і прив'язка робиться на ньому: `LOOP_CONFIGURE` приймає структуру `loop_config` з дескриптором образу, розміром блока й вкладеною `loop_info64`, де лежать зсув, обмеження розміру й прапорці. Розчеплює `LOOP_CLR_FD` — без аргументів узагалі.

![Послідовність викликів від вільного номера до відчеплення, з гілками EBUSY і очікування вузла](/reference/unix-linux/devices/loop-device/img/attach-lifecycle.svg)

*Дві гілки, повз які не пройти: вузол у `/dev` з'являється не миттєво, а номер, виданий як вільний, ніхто за нами не закріплює.*

## Код

```c
// loop-attach.c — те, що робить losetup, власними руками.
// збірка:  cc -O2 -Wall -Wextra -o loop-attach loop-attach.c
#define _GNU_SOURCE
#include <errno.h>
#include <fcntl.h>
#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <time.h>
#include <unistd.h>
#include <sys/ioctl.h>
#include <linux/loop.h>

#define TRIES     16          /* стільки ж спроб робить losetup */
#define PAUSE_NS  25000000L   /* 25 мс між ними */

struct attach_req {
    int         fd_img;      /* уже відкритий образ */
    const char *img_path;    /* лише напис у lo_file_name, ядро ним не користується */
    uint64_t    offset;      /* байтів від початку файлу до сектора 0 */
    uint64_t    sizelimit;   /* 0 — до кінця файлу */
    uint32_t    block_size;  /* 0 — типовий; інакше степінь двійки, 512…4096 */
    int         direct_io;   /* прохання працювати повз кеш сторінок господаря */
};

static void pause_a_bit(void)
{
    struct timespec ts = { 0, PAUSE_NS };
    nanosleep(&ts, NULL);
}

/* Вузол у /dev створює devtmpfs, права доводить udev — між поверненням
   номера й придатним вузлом є вікно. O_RDWR обов'язковий: пристрій,
   відкритий лише на читання, ядро зробить захищеним від запису. */
static int open_loop_node(int n)
{
    char path[32];
    snprintf(path, sizeof path, "/dev/loop%d", n);
    for (int i = 0; i < TRIES; i++) {
        int fd = open(path, O_RDWR | O_CLOEXEC);
        if (fd >= 0)
            return fd;
        if (errno != ENOENT && errno != EACCES)
            return -1;
        pause_a_bit();
    }
    errno = ETIMEDOUT;
    return -1;
}

/* Повертає дескриптор налаштованого пристрою, номер кладе в *out_n. */
static int loop_attach(int fd_ctl, const struct attach_req *r, int *out_n)
{
    for (int attempt = 0; attempt < TRIES; attempt++) {
        int n = ioctl(fd_ctl, LOOP_CTL_GET_FREE);
        if (n < 0)
            return -1;

        int fd_dev = open_loop_node(n);
        if (fd_dev < 0)
            return -1;

        struct loop_config cfg;
        memset(&cfg, 0, sizeof cfg);
        cfg.fd                = (uint32_t) r->fd_img;
        cfg.block_size        = r->block_size;
        cfg.info.lo_offset    = r->offset;
        cfg.info.lo_sizelimit = r->sizelimit;
        if (r->direct_io)
            cfg.info.lo_flags |= LO_FLAGS_DIRECT_IO;
        snprintf((char *) cfg.info.lo_file_name, LO_NAME_SIZE, "%s", r->img_path);

        if (ioctl(fd_dev, LOOP_CONFIGURE, &cfg) == 0) {
            *out_n = n;
            return fd_dev;
        }

        int err = errno;
        close(fd_dev);
        if (err != EBUSY) {   /* EBUSY — номер перехопили; решта помилок фатальні */
            errno = err;
            return -1;
        }
    }
    errno = EBUSY;
    return -1;
}

int main(int argc, char **argv)
{
    if (argc < 2) {
        fprintf(stderr, "вжиток: %s образ [зсув-у-байтах]\n", argv[0]);
        return 2;
    }
    struct attach_req r = {
        .img_path   = argv[1],
        .offset     = (argc > 2) ? strtoull(argv[2], NULL, 0) : 0,
        .sizelimit  = 0,
        .block_size = 512,
        .direct_io  = 1,
    };

    int fd_ctl = open("/dev/loop-control", O_RDWR | O_CLOEXEC);
    if (fd_ctl < 0) { perror("/dev/loop-control"); return 1; }

    r.fd_img = open(r.img_path, O_RDWR | O_CLOEXEC);
    if (r.fd_img < 0) { perror(r.img_path); return 1; }

    int n = -1;
    int fd_dev = loop_attach(fd_ctl, &r, &n);
    if (fd_dev < 0) { perror("LOOP_CONFIGURE"); return 1; }

    /* образ нам більше не потрібен: ядро тримає власне посилання на файл */
    close(r.fd_img);
    printf("/dev/loop%d ← %s, зсув %llu\n",
           n, r.img_path, (unsigned long long) r.offset);

    struct loop_info64 info;
    if (ioctl(fd_dev, LOOP_GET_STATUS64, &info) == 0)
        printf("прямий ввід-вивід: %s\n",
               (info.lo_flags & LO_FLAGS_DIRECT_IO) ? "увімкнено" : "ядро його зняло");

    unsigned char sec[512];
    if (pread(fd_dev, sec, sizeof sec, 0) == (ssize_t) sizeof sec) {
        for (int i = 0; i < 32; i++)
            printf("%02x%s", sec[i], (i % 16 == 15) ? "\n" : " ");
        printf("підпис 0x55aa наприкінці сектора: %s\n",
               (sec[510] == 0x55 && sec[511] == 0xaa) ? "є" : "немає");
    } else {
        perror("pread");
    }

    if (ioctl(fd_dev, LOOP_CLR_FD) < 0)
        perror("LOOP_CLR_FD");
    close(fd_dev);   /* прив'язка розпадається саме тут */
    close(fd_ctl);
    return 0;
}
```

Кожен крок коштує один виклик у ядро, і жоден з них не залежить від розміру образу: приєднання гігабайтного файлу й гігантського однаково швидкі, бо нічого не копіюється. Увесь відчутний час програми — це очікування вузла в `/dev`; читання сектора коштує стільки ж, скільки читання з файлу, плюс один прохід блоковим шаром.

## Чому ядру дають дескриптор, а не шлях

`cfg.fd` — це число, [дескриптор](book:unix-linux/file-descriptor) уже відкритого файлу. Ім'я образу теж їде в структурі, але як напис: ядро його не розбирає й нікуди за ним не ходить.

Причина в тому, що шлях — не річ, а маршрут, і маршрут залежить від того, хто ним іде. У кожного процесу може бути свій корінь, свій набір монтувань, свої символьні посилання на дорозі; «той самий» рядок у двох процесів приводить до різних файлів. Ядро, отримавши рядок, мусило б відкрити файл ще раз — від свого імені й трохи пізніше, ніж робилася перевірка прав. Це класична гонитва: між перевіркою й ужитком підмінюють саме те, що перевіряли. Дескриптор такої щілини не лишає — файл уже відкритий саме тим, хто просить, і саме з тими правами, які він мав у момент відкриття.

Звідси й решта поведінки. Право на запис пристрій успадковує від режиму відкриття, а не від прав на файл: відкрили образ на `O_RDONLY` — пристрій вийде тільки для читання, і `mount` скаржитиметься саме на це. Причому те саме стосується й другого дескриптора: якщо `/dev/loopN` відкрити на читання, ядро позначить пристрій захищеним від запису, хоч би яким писабельним був образ. Тому в коді обидва `open` — `O_RDWR`.

І навпаки, після `LOOP_CONFIGURE` наш `fd_img` уже не потрібен — ядро тримає власне посилання, тож у програмі він одразу закривається. Перейменування чи видалення образу прив'язку не розірвуть; місце звільниться лише разом із пристроєм.

## Вузол, якого ще немає

`LOOP_CTL_GET_FREE` повернув `3` — це ще не означає, що `/dev/loop3` можна відкрити. Пристрій щойно народився в ядрі; вузол у `/dev` створює `devtmpfs`, а власника, групу й режим доводить до ладу [`udev`](book:unix-linux/udev-rules) — окремим процесом, який лише прокидається на подію. У вікні між цими подіями `open` дає `ENOENT` (вузла ще немає) або `EACCES` (вузол є, але типові права ще не замінено на `root:disk`, режим 0660).

Тому в `open_loop_node` не одна спроба, а цикл із паузою — рівно те, що робить `losetup`: шістнадцять спроб по 25 мілісекунд, разом до 400 мс. Просте «поспати сто мілісекунд» гірше з обох боків: на завантаженій машині цього замало, а на порожній це чверть секунди, викинута на порожньому місці. Точніший, хоч і громіздкіший, шлях — [стежити за подіями в каталозі](book:unix-linux/inotify-and-fanotify) замість опитування.

## EBUSY: вільний номер ніхто не закріплює

`LOOP_CTL_GET_FREE` знімає стару гонитву — ту, де програма сама перебирала пристрої й питала стан кожного. Але він робить менше, ніж хочеться прочитати в назві: ядро проходить свій перелік, знаходить перший непричеплений пристрій і віддає його номер — **не резервуючи**. Між поверненням номера й нашим `LOOP_CONFIGURE` той самий номер може отримати й зайняти інший процес.

Програш видно одразу: `LOOP_CONFIGURE` повертає `EBUSY`, бо пристрій уже прив'язаний. Правильна реакція — не помилка, а нова спроба з початку: закрити дескриптор, попросити наступний вільний номер, налаштувати. Тому в `loop_attach` увесь блок загорнутий у цикл, а `EBUSY` — єдина помилка, що не перериває його.

## Відчеплення, якого не сталося

Найнесподіваніше в цій програмі — рядок після `LOOP_CLR_FD`.

Виклик повертає нуль, але пристрій, найімовірніше, ще прив'язаний. Ядро в цьому місці нічого не розбирає: воно ставить прапорець `LO_FLAGS_AUTOCLEAR` — «прибрати на останньому закритті» — і виходить. Справжнє відчеплення станеться тоді, коли зникне останній, хто тримає пристрій відкритим. У нашій програмі це `close(fd_dev)` наступним рядком; у житті це може бути змонтована файлова система, чужий `blkid` або той самий `udev`, який саме опитує щойно створений пристрій.

Інакше й не могло бути. Відчепити пристрій, на якому стоїть монтування, — означає висмикнути підлогу з-під працюючої файлової системи; відповідати помилкою теж погано, бо тоді будь-яка стороння проба перетворювала б `losetup -d` на лотерею. Відкладення — єдиний варіант, який не бреше.

Практичних наслідків два. По-перше, «повернуло 0» не можна читати як «файл уже вільний»: якщо після цього одразу видалити образ і перевірити місце, воно ще не звільнилося. Чесна перевірка — `/sys/block/loopN/loop/backing_file` [в sysfs](book:unix-linux/sysfs-device-model): цей файл зникає разом із прив'язкою. По-друге, той самий прапорець можна поставити наперед, прямо в `cfg.info.lo_flags`, — і тоді пристрій помре разом із програмою навіть при аварійному завершенні, бо дескриптори закриє ядро. Саме на цьому тримається `mount -o loop`: прив'язка живе рівно доти, доки на пристрої стоїть монтування, і зникає сама після `umount`.

## Права: не там, де їх шукають

Поширене формулювання «loop потребує `CAP_SYS_ADMIN`» правдиве за наслідком і хибне за механізмом. У самому драйвері на шляху `LOOP_CTL_GET_FREE` → `LOOP_CONFIGURE` перевірок [прав-можливостей](book:unix-linux/capabilities) немає взагалі. Воротами служать права на самі вузли: `/dev/loop-control` і `/dev/loop*` належать `root:disk` із режимом 0660, тож пройти може root або член групи `disk` — а це, до речі, [не пом'якшення, а те саме право](book:unix-linux/device-access-groups) іншими словами, бо `disk` дає прямий доступ до всіх дисків машини.

`CAP_SYS_ADMIN` у драйвері таки згадується, але в несподіваній ролі — як **заміна** відкриттю на запис: `LOOP_SET_STATUS64`, `LOOP_SET_BLOCK_SIZE`, `LOOP_SET_CAPACITY` і `LOOP_SET_DIRECT_IO` проходять або тому, що дескриптор пристрою відкрито на запис, або тому, що є ця можливість. А справжня стіна стоїть за межами драйвера: щоб змонтувати готовий пристрій, `CAP_SYS_ADMIN` потрібен обов'язково — і саме тому непривілейований користувач не приєднає образ так, щоб із цього вийшла користь.

## Прямий ввід-вивід — прохання, а не гарантія

`LO_FLAGS_DIRECT_IO` у конфігурації просить драйвер читати й писати образ повз [кеш сторінок](book:unix-linux/buffered-and-direct-io) господаря, щоб ті самі байти не лежали в пам'яті двічі. Прохання виконується не завжди: якщо файлова система під образом не вміє прямого вводу-виводу або оголошений розмір блока пристрою менший за той, якого вимагає носій, ядро мовчки зніме прапорець і працюватиме як звичайно. Помилки не буде — саме тому в програмі одразу після налаштування стоїть `LOOP_GET_STATUS64`, який показує, що з проханням сталося насправді.

Плутати не варто й напрямки. Прапорець стосується того, як драйвер ходить у файл-образ. Наш власний `pread` із `/dev/loopN` іде через кеш блокового пристрою в будь-якому разі, тож вирівнювати його не треба — на відміну від читання, відкритого з `O_DIRECT`.

## Запасний шлях для старих ядер

`LOOP_CONFIGURE` з'явився в ядрі 5.8 (2020). До того прив'язку складали з кількох викликів:

```c
/* Ядра до 5.8: те саме, але не за один раз. */
static int loop_attach_legacy(int fd_dev, const struct attach_req *r)
{
    if (ioctl(fd_dev, LOOP_SET_FD, r->fd_img) < 0)
        return -1;              /* пристрій уже живий — і вже неправильний */

    struct loop_info64 info;
    memset(&info, 0, sizeof info);
    info.lo_offset    = r->offset;
    info.lo_sizelimit = r->sizelimit;
    snprintf((char *) info.lo_file_name, LO_NAME_SIZE, "%s", r->img_path);

    int err = 0;
    if (r->block_size &&
        ioctl(fd_dev, LOOP_SET_BLOCK_SIZE, (unsigned long) r->block_size) < 0)
        err = errno;
    else if (ioctl(fd_dev, LOOP_SET_STATUS64, &info) < 0)
        err = errno;
    else if (r->direct_io && ioctl(fd_dev, LOOP_SET_DIRECT_IO, 1UL) < 0)
        err = errno;

    if (err) {                  /* прибираємо за собою недоналаштований пристрій */
        ioctl(fd_dev, LOOP_CLR_FD);
        errno = err;
        return -1;
    }
    return 0;
}
```

Чим це гірше, видно з першого ж рядка. `LOOP_SET_FD` створює **робочий** пристрій із типовими параметрами: зсув нуль, розмір — весь файл, блок 512 байтів. Пристрій одразу з'являється в системі, `udev` прокидається на подію й береться читати з нього суперблок — тобто не з початку розділу, а з початку образу, і те, що він там знайде, він же й запам'ятає. Наступний виклик міняє геометрію під ногами того, хто вже читає. Вікно маленьке, але цілком реальне: класичний симптом — `blkid` показує не ту файлову систему, а від'єднати пристрій одразу не вдається, бо його ще хтось тримає.

Друга біда — кожен крок є окремою нагодою впасти, і після падіння лишається наполовину налаштований пристрій, який мусить прибрати за собою сам виклик. Саме тому `LOOP_CONFIGURE` і з'явився: він або робить усе, або не робить нічого.

Перемикатися між шляхами доводиться навпомацки. Ядро, яке не знає нової команди, відповідає `EINVAL` — тим самим кодом, яким нове ядро відповідає на негодящий розмір блока. Розрізнити їх зсередини неможливо, тому `losetup` просто пробує старий шлях на будь-якому `EINVAL`, `ENOTTY` чи `ENOSYS` — і на новому ядрі з поганими параметрами марно ходить туди двічі. Дрібниця, але добре показує ціну того, що інтерфейс колись зробили з кількох кроків замість одного.
