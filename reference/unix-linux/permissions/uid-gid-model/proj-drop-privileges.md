# ⚙️ Скинути root назавжди — лабораторна на C

Служба, що стартує суперкористувачем заради одного-єдиного привілею, а далі мусить перестати ним бути, — найпоширеніша конструкція в системі й найчастіше місце, де зниження прав виявляється удаваним. Нижче — одна невелика програма мовою C, яка чотири рази поспіль намагається позбутися `root`: двічі наївно, один раз як належить і один раз бездоганно, але приречено. Після кожної спроби вона не вірить собі на слово, а йде дорогою назад і друкує, чи пустило ядро.

## Що будуємо

Служба прив'язується до `127.0.0.1:80`. Порт нижче 1024 вимагає можливості `CAP_NET_BIND_SERVICE` ([можливості замість всесильного root](topic:unix-linux/capabilities) — дрібні права, на які розібрано колишню всесильність нуля), і саме заради цього рядка потрібен привілейований старт. Одразу після `bind` привілей стає непотрібним: перевірку зроблено один раз, а далі роботу веде вже сам дескриптор ([сокет як дескриптор](topic:unix-linux/socket-as-descriptor) — прив'язаний сокет живе звичайним номером у таблиці процесу), і повторно його ніхто не перевіряє.

Далі — чотири сцени. Перша знижується рядком `setuid(getuid())`, не дивлячись на код повернення. Друга міняє тільки **діючі** числа — `setegid`/`seteuid`. Третя проходить залізну послідовність `setgroups → setresgid → setresuid → звірка читанням назад`. Четверта робить усе правильно, дістає успіх на всіх викликах — і однаково гине, бо в цільового користувача вичерпано ліміт процесів.

Кожна сцена живе в окремій дитині ([fork](topic:unix-linux/fork-semantics) — дитина дістає копію стану батька й нічого йому не повертає). Причина не в охайності: зниження прав безповоротне, тож дві сцени в одному процесі просто не помістяться. Батько тільки тримає сокет і чекає.

## Код

```c
#define _GNU_SOURCE
#include <stdio.h>
#include <string.h>
#include <errno.h>
#include <signal.h>
#include <unistd.h>
#include <fcntl.h>
#include <grp.h>
#include <pwd.h>
#include <arpa/inet.h>
#include <netinet/in.h>
#include <sys/socket.h>
#include <sys/resource.h>
#include <sys/wait.h>

static uid_t g_uid = 65534;          /* nobody, якщо в базі його не знайдемо */
static gid_t g_gid = 65534;

/* Прив'язатися до порту нижче 1024 — єдине, заради чого тут потрібен root. */
static int bind_low(int port)
{
    int fd = socket(AF_INET, SOCK_STREAM, 0);
    if (fd < 0) return -1;
    int one = 1;
    setsockopt(fd, SOL_SOCKET, SO_REUSEADDR, &one, sizeof one);

    struct sockaddr_in a;
    memset(&a, 0, sizeof a);
    a.sin_family = AF_INET;
    a.sin_port   = htons(port);
    a.sin_addr.s_addr = htonl(INADDR_LOOPBACK);

    if (bind(fd, (struct sockaddr *)&a, sizeof a) != 0 || listen(fd, 16) != 0) {
        int e = errno; close(fd); errno = e; return -1;
    }
    return fd;
}

/* Ким процес є ЗАРАЗ — не за пам'яттю програми, а прочитавши числа в ядра. */
static void show(const char *label)
{
    uid_t r, e, s;  gid_t rg, eg, sg;  gid_t list[64];
    getresuid(&r, &e, &s);
    getresgid(&rg, &eg, &sg);
    int n = getgroups(64, list);
    printf("    %s uid=%u/%u/%u  gid=%u/%u/%u  групи=",
           label, (unsigned)r, (unsigned)e, (unsigned)s,
           (unsigned)rg, (unsigned)eg, (unsigned)sg);
    for (int i = 0; i < n; i++) printf("%s%u", i ? "," : "", (unsigned)list[i]);
    puts(n > 0 ? "" : "(немає)");
}

/* Чи лишилася дорога назад? Не питаємо — йдемо нею. */
static void probe_return(void)
{
    if (seteuid(0) != 0)
        printf("    seteuid(0)        -> %s  ← дороги назад немає\n", strerror(errno));
    else
        printf("    seteuid(0)        -> ПОВЕРНУЛИСЬ, euid=%u\n", (unsigned)geteuid());

    int fd = open("/etc/shadow", O_RDONLY);
    printf("    open(/etc/shadow) -> %s\n", fd >= 0 ? "ВІДКРИТО" : strerror(errno));
    if (fd >= 0) close(fd);
}

/* Те саме, але очима ядра. */
static void show_status(void)
{
    FILE *f = fopen("/proc/self/status", "r");
    if (!f) return;
    char line[256];
    while (fgets(line, sizeof line, f))
        if (!strncmp(line, "Uid:", 4) || !strncmp(line, "Gid:", 4) ||
            !strncmp(line, "Groups:", 7) || !strncmp(line, "CapPrm:", 7) ||
            !strncmp(line, "CapEff:", 7))
            printf("    status  %s", line);
    fclose(f);
}

/* (1) Класика: «знижуємось до того, хто нас запустив». */
static void scene_setuid_getuid(int lfd)
{
    (void)lfd;
    puts("[1] Наївно: setuid(getuid()), код повернення нікого не цікавить");
    show("до зниження:");
    setuid(getuid());
    show("після зниження:");
    probe_return();
}

/* (2) Наївно інакше: міняємо лише ДІЮЧІ числа. */
static void scene_effective_only(int lfd)
{
    (void)lfd;
    puts("[2] Наївно: setegid/seteuid — міняємо лише діючі числа");
    if (setegid(g_gid) != 0 || seteuid(g_uid) != 0) { perror("    seteuid"); _exit(1); }
    show("після зниження:");
    probe_return();
}

/* Незворотний перехід. 0 — вдалося, -1 — ні; служба мусить померти на -1. */
static int drop_forever(uid_t uid, gid_t gid)
{
    if (setgroups(1, &gid)       != 0) return -1;   /* групи — першими */
    if (setresgid(gid, gid, gid) != 0) return -1;   /* усі три групові числа */
    if (setresuid(uid, uid, uid) != 0) return -1;   /* аж тепер користувач */

    uid_t r, e, s;  gid_t rg, eg, sg;               /* звірка читанням назад */
    if (getresuid(&r, &e, &s) != 0 || getresgid(&rg, &eg, &sg) != 0) return -1;
    if (r  != uid || e  != uid || s  != uid) { errno = EPERM; return -1; }
    if (rg != gid || eg != gid || sg != gid) { errno = EPERM; return -1; }

    gid_t list[2];
    int n = getgroups(2, list);
    if (n < 0 || n > 1 || (n == 1 && list[0] != gid)) { errno = EPERM; return -1; }
    return 0;
}

/* (3) Як належить — і перевірка, що назад справді нема куди. */
static void scene_correct(int lfd)
{
    puts("[3] Як треба: setgroups -> setresgid -> setresuid -> звірка");
    if (drop_forever(g_uid, g_gid) != 0) {
        printf("    зниження не вдалося: %s — тут служба мусить померти\n", strerror(errno));
        _exit(1);
    }
    show("після зниження:");
    probe_return();
    show_status();

    struct sockaddr_in sa;  socklen_t sl = sizeof sa;
    if (getsockname(lfd, (struct sockaddr *)&sa, &sl) == 0)
        printf("    успадкований сокет: порт %u — досі наш\n", ntohs(sa.sin_port));
    int fd2 = bind_low(81);
    printf("    bind(127.0.0.1:81) -> %s\n", fd2 >= 0 ? "успіх (!)" : strerror(errno));
    if (fd2 >= 0) close(fd2);
}

/* (4) Усі виклики вдалися. Служба однаково приречена. */
static void scene_nproc(int lfd)
{
    (void)lfd;
    puts("[4] Вичерпаний RLIMIT_NPROC: зниження вдається, а exec — ні");

    int p[2];                                  /* тримач місця: щоб у цільового */
    if (pipe(p) != 0) { perror("    pipe"); _exit(1); }
    pid_t helper = fork();                     /* користувача був хоч один процес */
    if (helper == 0) {
        close(p[0]);
        if (setresuid(g_uid, g_uid, g_uid) != 0) _exit(1);
        char c = 'x';
        if (write(p[1], &c, 1) != 1) _exit(1);
        pause();
        _exit(0);
    }
    close(p[1]);
    char c;
    if (read(p[0], &c, 1) != 1) { puts("    помічник не піднявся"); _exit(1); }

    struct rlimit rl = { 0, 0 };               /* «жодного процесу цьому uid» */
    if (setrlimit(RLIMIT_NPROC, &rl) != 0) { perror("    setrlimit"); _exit(1); }

    if (setresuid(g_uid, g_uid, g_uid) != 0)
        printf("    setresuid -> %s  (так відповідали ядра до 3.1)\n", strerror(errno));
    else
        puts("    setresuid -> успіх; звірка чисел теж пройде — усе ніби гаразд");

    execl("/bin/true", "true", (char *)NULL);
    printf("    execve    -> %s  ← ліміт наздогнав аж тут\n", strerror(errno));
    kill(helper, SIGKILL);
    waitpid(helper, NULL, 0);
}

static void run(void (*scene)(int), int lfd)
{
    pid_t pid = fork();
    if (pid == 0) { scene(lfd); _exit(0); }
    if (pid  > 0) waitpid(pid, NULL, 0);
    putchar('\n');
}

int main(void)
{
    struct passwd *pw = getpwnam("nobody");
    if (pw) { g_uid = pw->pw_uid; g_gid = pw->pw_gid; }

    if (geteuid() != 0) { puts("Потрібен root: усе цікаве починається з привілею."); return 1; }

    int lfd = bind_low(80);
    printf("bind(127.0.0.1:80) -> %s\n\n", lfd >= 0 ? "порт наш" : strerror(errno));
    if (lfd < 0) return 1;

    run(scene_setuid_getuid, lfd);
    run(scene_effective_only, lfd);
    run(scene_correct,        lfd);
    run(scene_nproc,          lfd);

    close(lfd);
    return 0;
}
```

Перша сцена не помилилася — вона взагалі нічого не зробила. Під `sudo` реальний UID процесу вже нульовий, тож `getuid()` повертає 0, і `setuid(0)` слухняно ставить нуль там, де нуль і був. Числа не змінилися, помилки не сталося, перевіряти нема чого — а служба відтоді вважає себе непривілейованою. Корінь непорозуміння в тому, що `getuid()` не пам'ятає, хто набрав команду: це число самого процесу, і `sudo` його перезаписав, бо для того й існує. Того, хто набрав команду, `sudo` передає окремо — змінною середовища `SUDO_UID`.

Друга сцена змінила рівно одне число з трьох, і саме те, на яке ядро дивиться в перевірках. Тому зниження виглядає справжнім: усі перевірки доступу тепер ідуть за `nobody`, і `geteuid()` чесно повертає 65534. Але збережений UID лишився нульовим, і `seteuid(0)` бере його назад без жодного питання — саме для повернення це число й придумано. Помітьте ще й групу: після «зниження» процес досі в групі 0, бо додаткового списку ніхто не чіпав.

## Чому третя сцена справді незворотна

Числа самі по собі нічого не доводять: непривілейований процес теж має три числа, просто інші. Доказ дає рядок `CapPrm`.

Правило ядра таке: якщо хоч один із трьох UID був нульовим, а після зміни всі три стали ненульовими, ядро вичищає **дозволений**, діючий і оточуючий набори можливостей повністю. Не «забирає нуль» — витирає самі повноваження, з яких колись і складалася всесильність. Через це `seteuid(0)` більше нема на чому виконати: піднімати діючий UID дозволено тому, хто має `CAP_SETUID`, а її вже нема в дозволеному наборі, звідки її беруть.

У другій сцені цього не сталося: реальний і збережений лишалися нулями, тож умова «усі три ненульові» не виконалася, і дозволений набір не чіпали — можливості просто пішли з діючого набору й терпляче чекали. Виняток із правила існує один, свідомий: `PR_SET_KEEPCAPS` (він же біт `SECBIT_KEEP_CAPS`) велить зберегти дозволений набір крізь зниження — саме так служба лишає собі `CAP_NET_BIND_SERVICE`, щоб перевідкривати низький порт після переналаштування. Прапорець скидається на кожному [exec](topic:unix-linux/exec-semantics).

Отже, `Uid:`/`Gid:` у [/proc/self/status](topic:unix-linux/proc-filesystem) кажуть, ким процес назвався, а `CapPrm:`/`CapEff:` — що він насправді може. Нулі в обох рядках — і є та перевірка, яку варто вставляти в тести служби.

## Пастка, що пережила своє виправлення

Четверта сцена — про виклик, який удався, і про службу, яка від цього не врятувалася.

Історія почалася не з лімітів, а зі спостереження, що коду повернення `setuid()` систематично ніхто не читає. Канонічний випадок — `sendmail` 2000 року: вада в тодішній реалізації можливостей дала змогу місцевому користувачеві відібрати в нього право міняти UID, зниження повернуло −1, ніхто на це не глянув — і пошта далі оброблялася від імені `root`.

Далі стало гірше. Виявилося, що [ліміт на кількість процесів](topic:unix-linux/resource-limits) обходиться безкоштовно: знизився до чужого користувача — і рахунок пішов уже його. Ліміт почали перевіряти прямо в `setuid()`, і зміна прав дістала право провалитися з `EAGAIN`. Тим самим неуважні програми отримали не просто ваду, а кнопку для зловмисника: забий чужий ліміт процесами — і зниження в чужій службі провалиться саме тоді, коли ти цього захочеш. У січні 2011-го на цьому підірвався `Exim` (CVE-2011-0017): у ньому `open_log()` не дивився на код повернення `setuid`/`setgid`, і місцевий користувач міг змусити `root` дописувати текст у довільний файл через символьне посилання. Вада стосувалася тільки Linux — саме тому, що тільки там зниження вміло провалюватися.

У ядрі 3.1 (2011) перевірку з `set*uid()` прибрали — з міркування, що хто вже має право міняти користувача, той має право й на цей ліміт; пропозицію подав Василь Куліков з Openwall, а спосіб не втратити ліміт зовсім — прапорець `PF_NPROC_EXCEEDED` — запропонував Нейл Браун. Тепер ядро на зниженні лише ставить цей прапорець, а `execve()` на самому початку, ще до відкривання файлу, дивиться на нього й повертає `EAGAIN`, якщо ліміт і досі перевищено. Для звичайного вигляду служби — `fork` → зниження → `exec` — ліміт лишився чинним, а рядок зі зниженням перестав бути місцем непомітної поразки.

Тому четверта сцена показує саме те, що показує: `setresuid` повертає нуль, звірка читанням назад проходить бездоганно, `/proc/self/status` показує чесного `nobody` — і наступний же `exec` падає з «Resource temporarily unavailable». Урок не в тому, що перевіряти код повернення `setresuid` не треба (на ядрах до 3.1 і на інших системах він падає й досі, і сторінка настанови зве пропуск перевірки грубою помилкою безпеки), а в тому, що **`exec` теж уміє відмовити** — і служба, яка ставиться до цього як до неможливого, у найгіршу мить просто мовчки не запустить свого робітника.

## На чому легко спіткнутися

**Порядок жорсткий.** `setgroups` мусить іти першим: прибирати додаткові групи дозволено лише привілейованому, і після `setresuid` цього права вже не буде. У бойовому коді замість `setgroups(1, &gid)` зазвичай викликають `initgroups(user, gid)` — він збирає для користувача саме ті групи, що записані в базі.

**Звірка — не перестраховка.** Спосіб дізнатися, ким процес став насправді, є лише один: прочитати числа назад. Перша сцена — приклад коду, який пройшов би будь-яку перевірку «чи повернув виклик нуль», бо він таки повернув нуль.

**Провал зниження — привід померти.** Немає безпечного «попрацюємо як є»: єдина правильна реакція на `-1` — завершити процес. Служба, яка після невдалого зниження обслуговує запити, робить це від імені `root`.

**Зниження мовчки вимикає ще дві речі.** Щойно процес змінює діючий UID, ядро скидає його ознаку «dumpable» до значення з `/proc/sys/fs/suid_dumpable` — типово нуль. Наслідки видно лише в біді. Служба перестає лишати образ пам'яті після падіння: сигнал, який мав би породити core, тепер просто вбиває процес, і аварію нема з чого розбирати. А каталог `/proc/PID` такого процесу переходить у власність `root:root` — тож знижена служба не прочитає навіть власних `/proc/self/environ` чи `/proc/self/mem` (загальнодоступний `status` із третьої сцени читається й далі). Кому образи пам'яті потрібні, вмикає ознаку назад — `prctl(PR_SET_DUMPABLE, 1)` після зниження, свідомо погоджуючись, що каталог знову стане видимим кожному з тим самим UID.

**Порт 80 не всюди привілейований.** Від ядра 4.11 межа задається окремо на мережевий простір імен — `net.ipv4.ip_unprivileged_port_start` (типово 1024, нуль знімає обмеження зовсім). У багатьох контейнерах її вже посунуто, і тоді вся зав'язка досліду відпадає: `bind(80)` пройде й без привілею.

**І головне, чого дослід не доводить.** Знижений UID не закриває успадкованих дескрипторів — третя сцена сама це й демонструє портом 80, який лишився при ній. Не звужує обмежувального набору можливостей, не заважає читати чужі `/proc/PID`, не рятує від вад у власному розборі протоколу. Зниження прав — це стеля збитку від однієї помилки, а не стіна; стіну будують окремим маленьким довіреним процесом ([розділення привілеїв](topic:unix-linux/privilege-separation)) і фільтром викликів поверх нього.
