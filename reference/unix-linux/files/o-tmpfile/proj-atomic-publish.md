# 🛠 Публікація готового файлу одним рухом

Служба читає свій індекс за сталим шляхом `/var/lib/app/index.db`, і час від часу цей індекс треба перебудувати. Перебудова — це кілька мегабайтів запису, і поки вона триває, за сталим шляхом лежить половина файлу. Хто відкриє його саме тоді, прочитає половину. Домовитися про блокування можна лише з тими, хто про домовленість знає, — а `cp`, `rsync` і нічне резервне копіювання не знають.

Отже, ціль така: за кінцевим шляхом або немає нічого, або лежить повністю готовий файл, і жодного стану між цими двома не існує. Спосіб — не давати назви, доки файл не готовий: наповнити безіменний inode, виставити на ньому все, що треба виставити, і назвати останнім викликом, який спрацьовує цілком або не спрацьовує зовсім.

## Програма

Функція нижче створює файл із заданим умістом. Наявне ім'я вона **не** заміняє — це «створити нове», а не «оновити старе». Гілка `publish_via_rename()` вмикається тоді, коли файлова система не вміє `O_TMPFILE`.

```c
#define _GNU_SOURCE
#include <errno.h>
#include <fcntl.h>
#include <libgen.h>
#include <limits.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/stat.h>
#include <sys/xattr.h>
#include <unistd.h>

/* Один write не зобов'язаний узяти весь буфер. */
static int write_all(int fd, const void *buf, size_t len)
{
    const char *p = buf;
    while (len > 0) {
        ssize_t n = write(fd, p, len);
        if (n < 0) {
            if (errno == EINTR) continue;   /* сигнал, а не помилка */
            return -1;
        }
        p += n;
        len -= (size_t) n;
    }
    return 0;
}

/* Довговічність імені: сам каталог теж треба скинути на носій. */
static int fsync_dir(const char *dir)
{
    int dfd = open(dir, O_RDONLY | O_DIRECTORY);
    if (dfd < 0)
        return -1;
    if (fsync(dfd) < 0) {
        int e = errno;
        close(dfd);
        errno = e;
        return -1;
    }
    return close(dfd);
}

/* Спільна частина обох гілок: наповнити відкритий дескриптор і довести до пуття.
   uid/gid == -1 означає «не чіпати»; тоді CAP_CHOWN не потрібен. */
static int fill(int fd, const void *buf, size_t len,
                mode_t mode, uid_t uid, gid_t gid)
{
    if (write_all(fd, buf, len) < 0)
        return -1;
    if (fchown(fd, uid, gid) < 0)        /* спершу власник: він гасить біти set-uid/set-gid */
        return -1;
    if (fchmod(fd, mode) < 0)            /* аж потім права, і то точні: без umask */
        return -1;
    if (fsetxattr(fd, "user.app.generation", "7", 1, 0) < 0
        && errno != EOPNOTSUPP)          /* xattr вимкнено — не привід падати */
        return -1;
    return fsync(fd);                    /* уміст на носії — ДО того, як з'явиться ім'я */
}

/* Запасний шлях: файлова система не вміє O_TMPFILE. */
static int publish_via_rename(const char *dir, const char *target,
                              const void *buf, size_t len,
                              mode_t mode, uid_t uid, gid_t gid)
{
    char tmp[PATH_MAX];
    if (snprintf(tmp, sizeof tmp, "%s/.pub-XXXXXX", dir) >= (int) sizeof tmp) {
        errno = ENAMETOOLONG;
        return -1;
    }

    int fd = mkstemp(tmp);               /* із цієї миті ім'я видно в каталозі */
    if (fd < 0)
        return -1;

    int e;
    if (fill(fd, buf, len, mode, uid, gid) < 0) { e = errno; close(fd); goto fail; }
    if (close(fd) < 0)                          { e = errno;            goto fail; }

    /* RENAME_NOREPLACE тримає ту саму семантику, що linkat: зайняте ім'я — EEXIST. */
    if (renameat2(AT_FDCWD, tmp, AT_FDCWD, target, RENAME_NOREPLACE) < 0) {
        e = errno;
        if (e != EINVAL && e != ENOSYS)  /* ФС не знає прапорця / ядро не знає виклику */
            goto fail;
        if (rename(tmp, target) < 0)     /* тут семантика вже інша: мовчазна заміна */
            { e = errno; goto fail; }
    }
    return fsync_dir(dir);

fail:
    unlink(tmp);                         /* не лишати сміття під випадковим іменем */
    errno = e;
    return -1;
}

int publish(const char *target, const void *buf, size_t len,
            mode_t mode, uid_t uid, gid_t gid)
{
    char dirbuf[PATH_MAX];
    if (snprintf(dirbuf, sizeof dirbuf, "%s", target) >= (int) sizeof dirbuf) {
        errno = ENAMETOOLONG;
        return -1;
    }
    /* Каталог беремо з самої цілі: так inode напевно виникне на тій самій
       файловій системі, де йому потім давати ім'я. dirname() псує аргумент. */
    const char *dir = dirname(dirbuf);

    int fd = open(dir, O_TMPFILE | O_RDWR, mode);
    if (fd < 0) {
        if (errno == EOPNOTSUPP || errno == EISDIR)
            return publish_via_rename(dir, target, buf, len, mode, uid, gid);
        return -1;
    }

    int e;
    if (fill(fd, buf, len, mode, uid, gid) < 0) { e = errno; goto fail; }

    char proc[64];
    snprintf(proc, sizeof proc, "/proc/self/fd/%d", fd);
    if (linkat(AT_FDCWD, proc, AT_FDCWD, target, AT_SYMLINK_FOLLOW) < 0)
        { e = errno; goto fail; }        /* EEXIST — ім'я вже зайняте */

    if (fsync_dir(dir) < 0)              /* ім'я вже є: помилка тут — про довговічність */
        { e = errno; goto fail; }
    return close(fd);

fail:
    close(fd);                           /* безіменний inode зникає сам */
    errno = e;
    return -1;
}
```

`renameat2()` має обгортку в glibc від версії 2.28; на старішій її кличуть як `syscall(SYS_renameat2, …)`.

## Чому саме такий порядок

Синхронізації тут не для годиться, і місце кожної визначене однозначно. `linkat()` чіпає каталог, тобто метадані; про те, чи доїхали до носія байти вмісту, він нічого не знає й нічого не обіцяє. Тож `fsync(fd)` мусить статися **до** появи імені: інакше збій живлення може лишити на носії ім'я, під яким записано не те, що ви писали, — а це рівно та біда, від якої вся конструкція й рятує. Другий `fsync`, уже на каталозі, мусить статися **після**: сам запис у каталозі теж спершу живе тільки в пам'яті, і без нього успішний `linkat()` після збою може виявитися таким, якого не було.

*Дві синхронізації обіймають появу імені з двох боків, і кожна закриває свій різновид напівзробленого стану. Порядок не переставляється: між ними лежить єдина мить, коли файл переходить із небуття в готовність.*

Про те, чому успішний `write` іще нічого не гарантує і чому помилку `fsync` не можна проковтнути мовчки, є окрема розмова: [кеш сторінок, fsync і довговічність запису](book:unix-linux/page-cache-durability). Коротко: ядро повідомляє про помилку запису рівно один раз, тому пропущене значення `fsync` — це загублена звістка про втрачені дані, а не дрібниця.

## Що встигає статися до імені

Усередині `fill()` немає жодного виклику, що працює зі шляхом, — самі лише `fchown`, `fchmod`, `fsetxattr`, `fsync` над дескриптором. Це не стилістичний вибір: шляху просто нема, а отже, немає й способу зробити щось не так. Файл спорядили повністю, поки він був невидимий.

Різниця добре видно на службі, яка перебудовує індекс від імені `root`, а віддавати його мусить користувачеві `app` із правами `0640`. У класичному способі файл з'являється в каталозі одразу після `mkstemp()` — з правами `0600` і власником `root`, — і живе так рівно доти, доки триває спорядження. Це не «майже готовий файл»: це файл із чужими правами й чужим власником, доступний усім, хто читає каталог. Із `O_TMPFILE` того проміжку немає взагалі, бо немає імені, під яким його можна було б застати.

Порядок усередині теж не довільний. `fchown()` іде **перед** `fchmod()`, бо зміна власника гасить біти `set-user-ID` і `set-group-ID`: ядро вважає, що разом із власником змінюється й той, чиї повноваження успадкував би виконуваний файл. З ядра 2.2.13 це стосується й `root` нарівні з усіма. Тож права треба ставити після власника, інакше `fchown()` тихо зніме те, що ви щойно виставили. Сама зміна власника потребує повноваження `CAP_CHOWN`; значення `-1` в обох полях означає «не чіпати» і повноважень не вимагає. Розширені атрибути тут ідуть останніми серед метаданих — про те, що вони таке й де живуть, є окремо: [розширені права й атрибути](book:unix-linux/acl-and-xattr).

## Дві форми linkat

Документація показує коротшу форму — порожній шлях і прапорець `AT_EMPTY_PATH`, що означає «дій над самим об'єктом, який відкриває дескриптор»:

```c
linkat(fd, "", AT_FDCWD, target, AT_EMPTY_PATH);
```

На звичайному процесі вона не працює. Ядро вимагає для `AT_EMPTY_PATH` у `linkat()` [можливості](book:unix-linux/capabilities) `CAP_DAC_READ_SEARCH` — окремого повноваження, що дозволяє обійти перевірку прав на читання каталогів. Причина в тому, що дескриптор можна отримати від когось іншого через сокет, а порожній шлях узагалі не проходить розбором шляху: без замка кожен, кому передали дескриптор, міг би пришити чужому файлові своє постійне ім'я у своєму каталозі. Замок цей старший за `O_TMPFILE` — його поставили ще у 2011 році, разом із самим `AT_EMPTY_PATH`, задовго до того, як з'явився безіменний файл, якому ім'я треба давати законно.

Гірше те, як саме виглядає відмова. Чинна сторінка `linkat(2)` називає код прямо: `ENOENT` — той самий, що й «нема такого шляху». Тобто виклик, у якому шлях порожній навмисно, відмовляє так, ніби ви помилилися зі шляхом, і з коду помилки причину не видно.

Робоча форма — через `/proc`, і саме її ввів Ал Віро окремою правкою влітку 2013 року, разом із `O_TMPFILE`:

```c
snprintf(proc, sizeof proc, "/proc/self/fd/%d", fd);
linkat(AT_FDCWD, proc, AT_FDCWD, target, AT_SYMLINK_FOLLOW);
```

`AT_SYMLINK_FOLLOW` тут обов'язковий: `/proc/self/fd/3` — символьне посилання, а `linkat()` за умовчанням посилань не розкриває й зробив би жорстке посилання на саме посилання. Ця форма проходить звичайним розбором шляху зі звичайними перевірками, тож окремого повноваження не потребує. Ціна — залежність від змонтованого `/proc`; у вирізаному контейнері чи під `chroot` без нього ця гілка мовчки перестає працювати. Про те, звідки взагалі взялося сімейство викликів із `AT_FDCWD` і чим шлях від дескриптора кращий за звичайний, — [сімейство *at](book:unix-linux/at-family-syscalls).

## Коли файлова система не вміє

`EOPNOTSUPP` від `open()` — не аварія, а звичайна відповідь: прапорця не підтримує саме ця файлова система. `EISDIR` означає інше — ядро таке старе, що про `O_TMPFILE` не чуло, і від прапорця лишився самий `O_DIRECTORY`, тобто спроба відкрити каталог на запис. Обидва коди ведуть в одну гілку, і в ній чесніше визнати: та гілка робить **не те саме**.

Різниця не в надійності, а в семантиці останнього кроку. `linkat()` не заміняє нічого й ніколи: зайняте ім'я — це `EEXIST`. `rename()` заміняє мовчки: старий файл під цим іменем просто зникає, і саме на цій властивості стоїть [атомарна заміна файлу](book:programming/atomic-file-replace). Якщо підставити другий виклик замість першого не думаючи, «створити, якщо ще немає» тихо перетвориться на «затерти те, що є», — і виявиться це вже після того, як чужі дані зникнуть.

Тому в запасній гілці спершу йде `renameat2()` з прапорцем `RENAME_NOREPLACE`: він повертає `EEXIST` так само, як `linkat()`, тобто зберігає семантику. Але й він не всюди є: файлова система, яка не знає прапорця, відповідає `EINVAL`, а ядро, старше за 3.15, — `ENOSYS`. Аж тоді лишається звичайний `rename()`, і ось із цієї миті поведінка справді розходиться з головною гілкою. Це варто не приховувати, а винести в опис функції: за таких-от умов виклик заміняє наявний файл.

І ще одна відмінність, яку видно неозброєним оком у самому коді: у запасній гілці між `mkstemp()` і перейменуванням у каталозі живе видиме ім'я `.pub-XXXXXX`. Якщо процес там помре, воно лишиться — тому `unlink(tmp)` на шляху помилки не косметика, а єдиний спосіб прибрати за собою. У головній гілці такого рядка немає й бути не може: там прибирати нічого.

## Пастки

**Пропущений `mode`.** `open()` — виклик зі змінним числом аргументів, тож третій параметр нікуди не зникає з мови: без нього ядро візьме за права доступу випадкові байти. Захист є, але частковий — glibc під `_FORTIFY_SOURCE` ловить `O_TMPFILE` без третього аргументу так само, як `O_CREAT`, і зупиняє складання з виразною помилкою. Без цієї збірки перевірки немає, а файл може вийти зі спорядженням, якого ви йому не давали. Що саме `open()` робить із переданим `mode`, поки він дійде до inode, — [umask і типові права](book:unix-linux/umask-and-defaults); саме тому остаточні права в коді ставить `fchmod()`, а не самий лише `open()`.

**Недоречний `O_EXCL`.** У парі з `O_TMPFILE` цей прапорець означає не те, що зазвичай: він назавжди позбавляє inode права дістати ім'я. Для чернетки, яка не має витекти назовні, це доречний замок — але у програмі, що публікує, він знищує самий задум. Відмовляє `linkat()` при цьому знову ж таки кодом `ENOENT`, тим самим, що й брак повноваження, тож дві зовсім різні причини виглядають однаково.

**`EXDEV`.** Жорстке посилання не перетинає межу монтування, а `linkat()` — це жорстке посилання. Якщо каталог для `O_TMPFILE` узяти зі звички (`/tmp`), а публікувати в `/var/lib/app`, виклик відмовить, бо це різні файлові системи. У програмі вище пастки просто немає: каталог обчислюється з кінцевого шляху, тож inode від початку виникає там, де йому й лежати. Це той випадок, коли помилку не ловлять, а роблять неможливою.

**Мовчазні повернення.** `fsync()`, `close()`, `rename()` повертають значення, і жодне з них не декоративне. `close()` на мережевій файловій системі — остання нагода почути про помилку відкладеного запису; після нього дескриптора вже немає й спитати нема кого.
