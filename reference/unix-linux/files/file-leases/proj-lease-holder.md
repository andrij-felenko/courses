# ⚙️ Орендар на C: віддати оренду за мілісекунди й виміряти це

Оренда дає програмі сорок п'ять секунд на реакцію — і весь інженерний зміст механізму в тому, скільки з них програма справді витрачає. Нижче — робоча програма-орендар, вимірювальний зонд до неї й дослід у двох терміналах, у якому видно різницю між орендарем, що відповів, і орендарем, що промовчав. Мову диктує предмет: усе спілкування тут іде системними викликами, тож C.

## Задум: обробник не робить нічого

Звістка приходить сигналом, а сигнал перериває програму в довільній точці — посеред `malloc`, під узятим замком, усередині `printf`. [Обробникові сигналу](book:unix-linux/async-signal-safety) дозволено лише вузький перелік викликів, і саме те, заради чого оренду брали — скинути буфери, дописати хвости, поговорити по мережі, — робити там не можна.

Розв'язка стара й проста: обробник не робить роботи, а лише **переносить факт у дескриптор**, чекати на який уміє [головний цикл на `poll`](book:unix-linux/select-poll-epoll). Один `write` у [канал](book:unix-linux/pipe-and-fifo), обидва кінці якого належать самій програмі, — звідси й назва «самотруба».

Чому труба, а не [eventfd](book:unix-linux/eventfd-and-futex)? eventfd — лічильник: він додає числа й чесно відповідає на питання «скільки подій сталося», але не на питання «яка саме». А `siginfo_t` приносить `si_fd` — номер дескриптора, чию оренду рвуть; для програми, що тримає оренди на сотнях файлів, це вся суть звістки. Труба переносить байт зі значенням. Коли орендований файл один, eventfd справді достатньо.

![Дві зони орендаря: ліворуч обробник сигналу — прилетів SIGRTMIN із si_fd, зберегти errno, write у трубу, повернути errno, і заборона на fsync, malloc, замки, мережу, printf; посередині самотруба «1 байт = 1 подія»; праворуч головний цикл — спить у poll, прокидається, доводить файл до ладу, питає F_GETLEASE і залежно від відповіді знижує оренду до F_RDLCK або віддає її через F_UNLCK](/reference/unix-linux/files/file-leases/img/holder-inside.svg)

*Байт у трубі — не оптимізація, а спосіб перенести подію туди, де на неї є право реагувати.*

## Орендар

```c
/* leaser.c — тримає оренду на запис і встигає відповісти вчасно.
 * gcc -O2 -Wall -o leaser leaser.c
 * ./leaser /tmp/leased.dat [--silent] [--work=МС]                    */
#define _GNU_SOURCE
#include <errno.h>
#include <fcntl.h>
#include <poll.h>
#include <signal.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <time.h>
#include <unistd.h>

static int wake[2];        /* самотруба: [0] читає цикл, [1] пише обробник */

static void on_break(int sig, siginfo_t *si, void *uc)
{
    (void)sig; (void)uc;
    int saved = errno;                      /* write поставить свій errno   */
    unsigned char fd = (unsigned char)si->si_fd;
    ssize_t n = write(wake[1], &fd, 1);     /* єдина дозволена тут дія      */
    (void)n;
    errno = saved;
}

static double now_ms(void)
{
    struct timespec t;
    clock_gettime(CLOCK_MONOTONIC, &t);
    return t.tv_sec * 1e3 + t.tv_nsec / 1e6;
}

int main(int argc, char **argv)
{
    const char *path = (argc > 1) ? argv[1] : "/tmp/leased.dat";
    int silent = 0, work_ms = 0;
    for (int i = 2; i < argc; i++) {
        if (!strcmp(argv[i], "--silent"))        silent = 1;
        else if (!strncmp(argv[i], "--work=", 7)) work_ms = atoi(argv[i] + 7);
    }

    /* O_RDONLY — не недогляд, а умова майбутнього зниження (див. нижче). */
    int fd = open(path, O_RDONLY);
    if (fd < 0) { perror("open"); return 1; }

    if (pipe2(wake, O_CLOEXEC | O_NONBLOCK) < 0) { perror("pipe2"); return 1; }

    struct sigaction sa;
    memset(&sa, 0, sizeof sa);
    sa.sa_sigaction = on_break;
    sa.sa_flags = SA_SIGINFO | SA_RESTART;
    sigaction(SIGRTMIN, &sa, NULL);

    if (fcntl(fd, F_SETSIG, SIGRTMIN) < 0) { perror("F_SETSIG"); return 1; }
    if (fcntl(fd, F_SETOWN, getpid())  < 0) { perror("F_SETOWN"); return 1; }
    if (fcntl(fd, F_SETLEASE, F_WRLCK) < 0) {
        /* EINVAL — не звичайний файл, ФС без оренд або leases-enable = 0;
           EAGAIN — файл уже хтось тримає відкритим;
           EACCES — файл чужий, а CAP_LEASE у нас немає.                */
        perror("F_SETLEASE(F_WRLCK)");
        return 1;
    }
    printf("оренда взята на %s, чекаємо\n", path);
    fflush(stdout);

    struct pollfd pf = { .fd = wake[0], .events = POLLIN, .revents = 0 };
    for (;;) {
        if (poll(&pf, 1, -1) < 0) {
            if (errno == EINTR) continue;   /* poll не рестартується й із SA_RESTART */
            perror("poll"); return 1;
        }
        unsigned char b;
        while (read(wake[0], &b, 1) == 1) {
            double t0 = now_ms();
            if (silent) {
                printf("перерву помічено на fd=%u — навмисно мовчимо\n", b);
                fflush(stdout);
                continue;
            }
            if (work_ms) {                  /* модель «скидання кешу»      */
                struct timespec ts = { work_ms / 1000,
                                       (work_ms % 1000) * 1000000L };
                nanosleep(&ts, NULL);
            }
            /* під час перерви F_GETLEASE віддає не теперішній стан оренди,
               а ТОЙ, ЯКОГО ядро від нас чекає.                            */
            int want = fcntl(fd, F_GETLEASE);
            if (want == F_RDLCK && fcntl(fd, F_SETLEASE, F_RDLCK) == 0)
                printf("знижено до F_RDLCK за %.2f мс\n", now_ms() - t0);
            else if (fcntl(fd, F_SETLEASE, F_UNLCK) == 0)
                printf("віддано зовсім за %.2f мс\n", now_ms() - t0);
            else
                perror("F_SETLEASE");
            fflush(stdout);
        }
    }
}
```

Три місця тут неочевидні.

**`open(path, O_RDONLY)`.** Оренду на запис ядро видає й на дескриптор, відкритий лише для читання: `F_WRLCK` вимагає не права писати, а відсутності інших відкриттів. А от **знизити** оренду до `F_RDLCK` ядро погодиться тільки тоді, коли на inode взагалі немає відкриттів на запис, — і власне відкриття орендаря рахується теж. Тому орендар, що відкрив файл `O_RDWR` заради зручності, назавжди втрачає право на зниження: `F_SETLEASE(F_RDLCK)` повертатиме `EAGAIN` через його ж дескриптор. Хто мусить писати — пише, але відповідати на перерву вміє лише одним способом: `F_UNLCK`.

> 🔧 **Навіщо це.** Режим відкриття тут — не деталь реалізації, а рішення про поведінку програми, і ухвалити його треба до першого рядка коду. `O_RDONLY` купує право пережити чужого читача, не втративши оренди: чужий `cat` промайне, а орендар лишиться з обіцянкою «файл ніхто не пише» й далі триматиме свій кеш чинним. `O_RDWR` це право обмінює на можливість писати самому — і кожен чужий читач тоді коштує повної втрати оренди, після якої її треба брати наново, а взяти `F_WRLCK` вдасться лише тоді, коли той читач нарешті закриє файл. Для файлового сервера різниця буквально в тому, скільки разів на день клієнтові доведеться викидати кеш.

**`F_GETLEASE` посеред перерви — це не питання «що я маю», а питання «чого від мене хочуть».** Сигнал про наміри відкривача не каже нічого: у `siginfo_t` є номер дескриптора й тільки він. Зате ядро, позначаючи оренду як таку, що рветься, одразу записує **цільовий** стан — `F_UNLCK`, якщо відкривач іде на запис, і `F_RDLCK`, якщо той лише читатиме. `F_GETLEASE` під час перерви віддає саме цю цільову позначку, а не теперішній тип оренди. Один дешевий виклик, і невідомість зникає.

Статус доказу тут варто назвати: поведінка однозначно випливає з коду ядра (`fcntl_getlease` віддає `target_leasetype` у `fs/locks.c`), але сторінка `fcntl(2)` її не описує. Тому програма не має права спиратися лише на неї — і не спирається: якщо зниження все-таки не пройде, наступним рядком іде `F_UNLCK`. Друга підказка приходить сама собою, з `EAGAIN`: лічильник письменників inode ядро збільшує **раніше**, ніж перевіряє оренди, тож на перерві від письменника зниження впирається в те саме правило «читацька оренда — лише поки файл ніхто не пише» й не проходить.

Ця обережність не зайва, бо хибне зниження коштує дорого й мовчки. Перехід на `F_RDLCK` знімає з оренди позначку «слід знизитися», але не знімає позначки «слід віддати»: письменник, що спить в `open`, прокинеться, побачить, що читацька оренда з ним однаково конфліктує, і ляже спати далі — до кінця тих самих сорока п'яти секунд. Помилки не буде ніде: ні орендар, ні відкривач не дізнаються, що півхвилини згоріло намарне.

**Збережений `errno`** ловить рідкісні й нез'ясовні збої. `write` в обробнику виставить свій `errno` поверх того, який щойно поставив перерваний виклик, — і головний код побачить чуже значення там, де перевіряє своє.

## Зонд для вимірювання

```c
/* opener.c — скільки коштує відкрити орендований файл.
 * gcc -O2 -Wall -o opener opener.c
 * ./opener /tmp/leased.dat [w] [n]   w — на запис, n — з O_NONBLOCK    */
#include <errno.h>
#include <fcntl.h>
#include <stdio.h>
#include <string.h>
#include <time.h>
#include <unistd.h>

int main(int argc, char **argv)
{
    int flags = O_RDONLY;
    for (int i = 2; i < argc; i++) {
        if (!strcmp(argv[i], "w")) flags = (flags & ~O_ACCMODE) | O_WRONLY;
        if (!strcmp(argv[i], "n")) flags |= O_NONBLOCK;
    }
    struct timespec a, b;
    clock_gettime(CLOCK_MONOTONIC, &a);
    int fd = open(argv[1], flags);
    int e = errno;
    clock_gettime(CLOCK_MONOTONIC, &b);
    printf("open: %s, %.1f мс\n", fd < 0 ? strerror(e) : "успіх",
           (b.tv_sec - a.tv_sec) * 1e3 + (b.tv_nsec - a.tv_nsec) / 1e6);
    if (fd >= 0) close(fd);
    return 0;
}
```

## Дослід у двох терміналах

Перший термінал тримає оренду, другий заходить збоку. Орендаря перезапускаємо перед кожним дослідом: після відповіді оренда на файлі вже інша.

```
термінал 1                            термінал 2
──────────────────────────────────    ──────────────────────────────────
$ dd if=/dev/zero of=/tmp/leased.dat bs=1M count=8 status=none
$ ./leaser /tmp/leased.dat
оренда взята, чекаємо
                                      $ ./opener /tmp/leased.dat
знижено до F_RDLCK за 0.04 мс         open: успіх, 0.6 мс

$ ./leaser /tmp/leased.dat            $ ./opener /tmp/leased.dat w
віддано зовсім за 0.05 мс             open: успіх, 0.7 мс

$ ./leaser /tmp/leased.dat            $ ./opener /tmp/leased.dat n
знижено до F_RDLCK за 0.04 мс         open: Resource temporarily
                                            unavailable, 0.1 мс

$ ./leaser /tmp/leased.dat --silent   $ time cat /tmp/leased.dat >/dev/null
перерву помічено на fd=3 —            real    0m45.003s
навмисно мовчимо
```

Чотири рядки — чотири різні механізми. Читач і письменник проходять однаково швидко, але **різними шляхами**: першому вистачило зниження, другому довелося віддавати оренду цілком — і різницю орендар побачив ще до того, як щось зробив. Виклик з `O_NONBLOCK` не спить зовсім, а повертає `EWOULDBLOCK` одразу; перерву він при цьому таки запустив, тож орендар устиг знизитися, і повторна спроба за мить пройде без затримки. А мовчазний орендар перетворив звичайний `cat` на три чверті хвилини очікування, і `cat` не показав ані помилки, ані натяку на причину.

Ті самі сорок п'ять секунд читаються з `/proc/sys/fs/lease-break-time`; поставте туди `5` — і останній рядок стане `0m5.00s`.

## Подивитися збоку

Поки триває дослід із мовчазним орендарем, стан оренди видно в [псевдофайлі](book:unix-linux/proc-filesystem) `/proc/locks`:

```
$ grep LEASE /proc/locks                    # орендар чекає
3: LEASE  ACTIVE    WRITE 5312 08:02:1441802 0 EOF

$ grep LEASE /proc/locks                    # поки чужий open спить
3: LEASE  BREAKING  UNLCK 5312 08:02:1441802 0 EOF
```

Третя колонка — стан оренди, четверта — **не** її теперішній тип, а той самий цільовий: `UNLCK` означає «від цього орендаря чекають повної відмови», `READ` — «досить зниження». Ту саму річ, якої немає в сигналі, ядро віддає двома шляхами, і `/proc/locks` — той із них, що годиться, коли треба подивитися на чужий процес, а не на власний. Слово `DELEG` замість `LEASE` у першій колонці позначає делегування, видане сервером NFS зсередини ядра: той самий механізм у руках іншого споживача.

## Пастки

**Труба мусить бути неблокуючою.** Повна труба присипляла б обробник — а спати в обробнику означає витрачати той самий строк, задля якого все й будувалося. З `O_NONBLOCK` переповнення дає `EAGAIN`, і втрачений байт нічого не ламає: у трубі вже лежать нерозібрані події, цикл однаково прокинеться.

**Строк рахує ядро, а не ваш цикл.** Відлік іде від чужого `open`, тож у сорок п'ять секунд має вкластися все: і час, поки цикл дійде до труби, і сама робота. `./leaser file --work=60000` показує це найгрубіше — ядро забере оренду раніше, ніж програма договорить.

Дослід із зондом дає мілісекунди тому, що орендар тут не робить нічого. Справжній бюджет виглядає інакше — ось він для файлового сервера, який мусить сходити по дані до мережевого клієнта:

**Скільки з 45 секунд лишається на роботу**

```
найгірша затримка головного циклу        =  2.0 с   (важкий запит у тому ж циклі)
дорога до клієнта і назад                =  0.2 с   (0.1 с в один бік)
клієнт скидає свій кеш на сервер         =  8.0 с   (64 МБ через 64 Мбіт/с)
запас на планувальник і завантаження     =  5.0 с
                                           ───────
витрачається                             = 15.2 с
lease-break-time                         = 45.0 с
                                           ───────
вільний залишок                          = 29.8 с
```

Запас здоровий — але тільки поки перше число лишається двома секундами. Виросте воно до тридцяти (а це всього-на-всього один синхронний запит до бази в тому самому циклі) — і бюджет зійдеться в нуль, причому не в лабораторії, а під навантаженням, коли черга й так довга. Тому оренду беруть там, де є подієвий цикл із передбачуваною затримкою, а не там, де просто є цікавість до файлу.

**Власний повторний `open` рве власну оренду.** Виняток «свій відкривач не рахується» звичайним орендам не належить: другий потік, що відкрив той самий шлях, розірве вашу ж обіцянку, і сигнал прилетить вам самим. Шлях відкривають один раз; `dup` і `fork` безпечні, бо оренди не множать, а лише додають посилання на ту саму.

**Перевірка придатності — на старті, один раз.** `EINVAL` від `F_SETLEASE` означає щонайменше три різні речі: файл не звичайний, файлова система оренд не вміє, оренди вимкнено глобально. Програма, що спирається на оренди, мусить помітити це відразу й перейти на запасний шлях, а не з'ясовувати посеред роботи, що обіцянки не було ніколи.

**Сигнал реального часу може загубитися.** Черга таких сигналів обмежена [лімітом ресурсів](book:unix-linux/resource-limits) `RLIMIT_SIGPENDING`; при переповненні ядро повертається до `SIGIO`, а той не несе `si_fd`. Тому обробник на `SIGIO` теж потрібен, і означає він рівно одне: «перевір усі свої оренди через `F_GETLEASE`», бо який саме файл — уже не дізнатися.
