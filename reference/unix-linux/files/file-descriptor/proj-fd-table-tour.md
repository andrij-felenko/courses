# ⚙️ Прогулянка власною таблицею дескрипторів

Це програма на C для Linux, яка дивиться на себе: друкує свою таблицю дескрипторів, а тоді по черзі робить із нею все, про що зазвичай доводиться вірити на слово — забирає найменший вільний номер, розводить два рівні прапорців, перенаправляє вивід без оболонки, витікає до `EMFILE` і пише в номер, який уже віддано іншому файлу. Кожне твердження тут перевіряється очима за хвилину, а самі прийоми — не навчальні: рівно так шукають витік дескрипторів у справжньому сервері.

## Як зібрати й запустити

Програм дві. Головна робить усі досліди, крихітна друга потрібна лише в одному з них — щоб було кого запустити через `exec` і подивитися, що дісталося нащадкові.

```sh
gcc -Wall -Wextra -O2 -o fdtour  fdtour.c
gcc -Wall -Wextra -O2 -o fdchild fdchild.c
mkdir -p /tmp/fdtour && cd /tmp/fdtour
/шлях/до/fdtour /шлях/до/fdchild
```

Запускати краще в порожньому каталозі: програма створює там кілька файлів із назвами на `fdtour-`. Усе, що вона показує, читається з `/proc` ([/proc: процеси як файли](topic:unix-linux/proc-filesystem)), тому потрібен саме Linux — на macOS чи FreeBSD більша частина коду скомпілюється, але дивитися буде нікуди.

## Каркас: чим узагалі дивитися

Ядро не має виклику «покажи мою таблицю». Натомість воно виставляє її як каталог `/proc/self/fd`, де кожен відкритий номер лежить символьним посиланням, а ціль посилання каже, що за ним стоїть: шлях до файлу, `socket:[12345]`, `pipe:[67890]`.

У цьому обході є дві тонкощі, і обидві варті того, щоб їх побачити в коді. Перша: щоб прочитати каталог, треба його відкрити — тобто витратити дескриптор, який негайно з'явиться у власному ж переліку. Друга: поки ми ходимо по каталогу й розпитуємо номери, таблиця жива, і кожне наше `readlink` чи `fopen` її ворушить. Тому номери спершу збираємо в масив одним проходом, і аж потім розпитуємо кожен окремо.

```c
/* fdtour.c — прогулянка власною таблицею дескрипторів.
 *
 *   gcc -Wall -Wextra -O2 -o fdtour fdtour.c
 *
 * Потрібен Linux: усе, що ми розглядаємо, видно через /proc.
 */
#define _GNU_SOURCE
#include <dirent.h>
#include <errno.h>
#include <fcntl.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/resource.h>
#include <sys/stat.h>
#include <sys/wait.h>
#include <unistd.h>

static void die(const char *what)
{
    fprintf(stderr, "!! %s: %s\n", what, strerror(errno));
    exit(EXIT_FAILURE);
}

static void banner(const char *title)
{
    printf("\n──── %s ────\n", title);
}

/* Збирає номери відкритих дескрипторів у масив. Повертає їхню кількість
 * або -1, якщо не вдалося навіть відкрити каталог (таке буває — див. дослід
 * із EMFILE). Номер, що його зайняв сам обхід, у перелік не потрапляє. */
static int list_fds(int *out, int max)
{
    DIR *d = opendir("/proc/self/fd");
    if (d == NULL)
        return -1;

    int own = dirfd(d);            /* номер, під яким відкрито сам каталог */
    if (own < 0) die("dirfd");

    int n = 0;
    for (;;) {
        errno = 0;
        struct dirent *e = readdir(d);
        if (e == NULL) {
            if (errno != 0) die("readdir /proc/self/fd");
            break;
        }
        if (e->d_name[0] == '.') continue;          /* "." і ".." */

        int fd = (int)strtol(e->d_name, NULL, 10);
        if (fd == own) continue;
        if (n < max) out[n++] = fd;
    }
    closedir(d);
    return n;
}
```

Далі — рядок про один номер. Тут важлива саме пара запитів: `F_GETFD` питає **комірку таблиці** й уміє відповісти рівно одну річ — чи стоїть на ній `FD_CLOEXEC`; `F_GETFL` питає **опис відкритого файлу**, за яким ця комірка закріплена, і повертає режим доступу разом зі статусними прапорцями ([опис відкритого файлу](topic:unix-linux/open-file-description)). Два виклики поруч у коді — найкоротший спосіб побачити, що це два різні рівні, а не два способи спитати те саме.

```c
/* Режим доступу зашитий у молодші біти прапорців опису. */
static const char *access_mode(int fl)
{
    switch (fl & O_ACCMODE) {
    case O_RDONLY: return "r-";
    case O_WRONLY: return "-w";
    case O_RDWR:   return "rw";
    default:       return "??";
    }
}

/* Один рядок про один номер: куди веде і які прапорці на якому рівні. */
static void show_fd(int fd)
{
    char path[64];
    char target[512];

    snprintf(path, sizeof path, "/proc/self/fd/%d", fd);
    ssize_t len = readlink(path, target, sizeof target - 1);
    if (len < 0) len = 0;          /* номер міг зникнути між двома проходами */
    target[len] = '\0';

    int fdfl = fcntl(fd, F_GETFD); /* рівень ДЕСКРИПТОРА: лише FD_CLOEXEC */
    int flfl = fcntl(fd, F_GETFL); /* рівень ОПИСУ: режим і статусні прапорці */
    if (fdfl < 0 || flfl < 0) {
        printf("  %3d  (уже закритий)\n", fd);
        return;
    }

    printf("  %3d  %-28s  %s%s%s%s\n",
           fd, target[0] ? target : "?",
           access_mode(flfl),
           (flfl & O_APPEND)    ? " O_APPEND"    : "",
           (flfl & O_NONBLOCK)  ? " O_NONBLOCK"  : "",
           (fdfl & FD_CLOEXEC)  ? " +FD_CLOEXEC" : "");
}

/* Скільки номерів зайнято просто зараз (-1, якщо подивитися нічим). */
static int count_fds(void)
{
    int fds[1024];
    return list_fds(fds, 1024);
}

static void dump_fds(const char *label)
{
    int fds[1024];
    int n = list_fds(fds, 1024);
    if (n < 0) {
        printf("%s — подивитися нічим: %s\n", label, strerror(errno));
        return;
    }
    printf("%s — записів: %d\n", label, n);
    for (int i = 0; i < n; i++)
        show_fd(fds[i]);
}

/* Друкує невеликий файл із відступом — щоб одразу бачити результат досліду. */
static void show_file(const char *path)
{
    int fd = open(path, O_RDONLY);
    if (fd < 0) die("open для читання");

    char buf[1024];
    ssize_t n;
    int at_line_start = 1;
    while ((n = read(fd, buf, sizeof buf)) > 0) {
        for (ssize_t i = 0; i < n; i++) {
            if (at_line_start) { fputs("  | ", stdout); at_line_start = 0; }
            putchar(buf[i]);
            if (buf[i] == '\n') at_line_start = 1;
        }
    }
    if (n < 0) die("read");
    if (close(fd) < 0) die("close");
}
```

## Дослід перший: найменший вільний

Правило видно на трьох `open` і одному `close`. Відкриваємо три рази, закриваємо середній — і питаємо ще раз. Якби номери роздавав лічильник, ми дістали б наступне число; натомість дірку латають одразу.

```c
static void demo_lowest(void)
{
    banner("Найменший вільний номер");
    dump_fds("на старті");

    int a = open("/dev/null", O_RDONLY);
    if (a < 0) die("open /dev/null");
    int b = open("/dev/null", O_RDONLY);
    if (b < 0) die("open /dev/null");
    int c = open("/dev/null", O_RDONLY);
    if (c < 0) die("open /dev/null");
    printf("три open      -> %d, %d, %d\n", a, b, c);

    if (close(b) < 0) die("close");
    printf("close(%d)      -> у таблиці дірка\n", b);

    int again = open("/dev/null", O_RDONLY);
    if (again < 0) die("open /dev/null");
    printf("ще один open  -> %d  (дірку залатано, %d не видали)\n", again, c + 1);

    if (close(a) < 0 || close(again) < 0 || close(c) < 0) die("close");
}
```

```
──── Найменший вільний номер ────
на старті — записів: 3
    0  /dev/pts/3                    rw
    1  /dev/pts/3                    rw
    2  /dev/pts/3                    rw
три open      -> 3, 4, 5
close(4)      -> у таблиці дірка
ще один open  -> 4  (дірку залатано, 6 не видали)
```

Три однакові рядки на старті — це і є вся конвенція стандартних потоків: три записи, які поставила оболонка перед запуском, усі три на той самий термінал ([три потоки](topic:unix-linux/standard-streams)).

## Дослід другий: два рівні прапорців на одному номері

Найкоротший спосіб довести, що дескриптор і опис — різні речі: узяти два номери на один опис і подивитися, яка правка «перетікає» з одного на другий, а яка ні.

```c
static void demo_two_levels(void)
{
    banner("F_GETFD і F_GETFL: два рівні на одному номері");

    int fd = open("/dev/null", O_WRONLY | O_APPEND | O_NONBLOCK);
    if (fd < 0) die("open /dev/null");
    int twin = dup(fd);                 /* інший номер, ТОЙ САМИЙ опис */
    if (twin < 0) die("dup");
    printf("одразу після dup:\n");
    show_fd(fd);
    show_fd(twin);

    /* F_SETFD править комірку таблиці — сусідній номер не чіпає */
    if (fcntl(fd, F_SETFD, FD_CLOEXEC) < 0) die("F_SETFD");
    printf("ставимо FD_CLOEXEC на номер %d:\n", fd);
    show_fd(fd);
    show_fd(twin);

    /* F_SETFL править опис — а він у них спільний */
    int fl = fcntl(fd, F_GETFL);
    if (fl < 0) die("F_GETFL");
    if (fcntl(fd, F_SETFL, fl & ~O_NONBLOCK) < 0) die("F_SETFL");
    printf("знімаємо O_NONBLOCK з номера %d:\n", fd);
    show_fd(fd);
    show_fd(twin);

    /* dup прапорця дескриптора не копіює: новий номер завжди успадковуваний */
    int third = dup(fd);
    if (third < 0) die("dup");
    printf("копія номера %d, на якому стоїть FD_CLOEXEC:\n", fd);
    show_fd(third);

    if (close(fd) < 0 || close(twin) < 0 || close(third) < 0) die("close");
}
```

```
──── F_GETFD і F_GETFL: два рівні на одному номері ────
одразу після dup:
    3  /dev/null                     -w O_APPEND O_NONBLOCK
    4  /dev/null                     -w O_APPEND O_NONBLOCK
ставимо FD_CLOEXEC на номер 3:
    3  /dev/null                     -w O_APPEND O_NONBLOCK +FD_CLOEXEC
    4  /dev/null                     -w O_APPEND O_NONBLOCK
знімаємо O_NONBLOCK з номера 3:
    3  /dev/null                     -w O_APPEND +FD_CLOEXEC
    4  /dev/null                     -w O_APPEND
копія номера 3, на якому стоїть FD_CLOEXEC:
    5  /dev/null                     -w O_APPEND
```

Три правки — три різні наслідки, і кожен випливає просто з того, де саме лежить прапорець. `FD_CLOEXEC` змінив один рядок із двох: він живе в комірці таблиці, а комірки різні. `O_NONBLOCK` змінив обидва рядки, хоч чіпали ми один номер: він живе в описі, а опис один ([блокуючий і неблокуючий режим](topic:unix-linux/blocking-and-nonblocking)). Третій `dup` дав номер без `FD_CLOEXEC`, хоч копіювали з номера, на якому той стояв: `dup` заводить **новий запис таблиці**, а прапорець належить запису, і копіювати його ніхто не обіцяв.

Остання деталь виглядає як дрібниця, поки не побачиш її наслідок: саме завдяки їй звичний рецепт перенаправлення працює навіть тоді, коли файл відкривали з `O_CLOEXEC`. Дублікат, покладений у номер 1, переживає `exec` — бо він уже інший запис.

## Дослід третій: перенаправлення без жодного спеціального виклику

`close(1)` плюс `open` дають готове перенаправлення, і жодна частина програми про це не знає. Робимо це в окремому нащадку — інакше зіпсуємо собі власний вивід і не побачимо решти дослідів ([fork: розмноження процесу](topic:unix-linux/fork-semantics)).

```c
static void demo_manual_redirect(void)
{
    banner("close(1) + open: перенаправлення без спеціального виклику");

    fflush(NULL);                  /* буфери батька не повинні поїхати в дитину */
    pid_t pid = fork();
    if (pid < 0) die("fork");

    if (pid == 0) {
        if (close(1) < 0) _exit(10);
        /* найменший вільний тепер 1 — його й забере наступний open */
        int fd = open("fdtour-stdout.txt", O_WRONLY | O_CREAT | O_TRUNC, 0644);
        if (fd < 0)  _exit(11);
        if (fd != 1) _exit(12);
        printf("цей рядок надруковано звичайним printf\n");
        fflush(stdout);            /* _exit не скидає буферів stdio */
        _exit(0);
    }

    int st;
    if (waitpid(pid, &st, 0) < 0) die("waitpid");
    if (!WIFEXITED(st) || WEXITSTATUS(st) != 0) {
        printf("дитина завершилася з кодом %d\n",
               WIFEXITED(st) ? WEXITSTATUS(st) : -1);
        return;
    }
    printf("а у файлі fdtour-stdout.txt лежить:\n");
    show_file("fdtour-stdout.txt");
}
```

```
──── close(1) + open: перенаправлення без спеціального виклику ────
а у файлі fdtour-stdout.txt лежить:
  | цей рядок надруковано звичайним printf
```

Два рядки коду тут — не про дескриптори, а про буфери, і без них дослід ламається. `fflush(NULL)` перед `fork` потрібен тому, що `fork` копіює й пам'ять теж, разом із недописаним буфером `stdout`; без нього все, що батько встиг накопичити й не вивести, надрукується двічі — по разу з кожного процесу. Другий `fflush` потрібен тому, що `_exit` іде в ядро напряму й нічого не дописує: без нього рядок так і згине в буфері, а файл лишиться порожнім ([буферизація stdio](topic:unix-linux/stdio-buffering)).

## Дослід четвертий: fork + dup2 + exec і що переживає заміну образу

Тепер те саме, але як його справді робить оболонка — і з нащадком, який уміє розповісти, що йому дісталося. Батько відкриває три речі: файл під майбутній `stdout`, звичайний дескриптор і такий самий, але з `O_CLOEXEC`. Номери двох останніх передаються нащадкові рядками в `argv`, щоб той міг про них спитати.

```c
static void demo_exec(const char *child)
{
    banner("fork + dup2 + exec: перенаправлення руками");

    int out = open("fdtour-exec.txt", O_WRONLY | O_CREAT | O_TRUNC, 0644);
    if (out < 0) die("open fdtour-exec.txt");
    int plain = open("/dev/null", O_RDONLY);               /* має пережити exec */
    if (plain < 0) die("open /dev/null");
    int sealed = open("/dev/null", O_RDONLY | O_CLOEXEC);  /* не має пережити */
    if (sealed < 0) die("open /dev/null");
    printf("готуємо: out=%d  plain=%d  sealed=%d\n", out, plain, sealed);

    char a[16], b[16];
    snprintf(a, sizeof a, "%d", plain);
    snprintf(b, sizeof b, "%d", sealed);

    fflush(NULL);
    pid_t pid = fork();
    if (pid < 0) die("fork");
    if (pid == 0) {
        if (dup2(out, 1) < 0) _exit(20);  /* номер 1 → той самий опис, що й out */
        if (close(out) < 0)   _exit(21);  /* саме число out більше не потрібне */
        execl(child, child, a, b, (char *)NULL);
        fprintf(stderr, "!! execl %s: %s\n", child, strerror(errno));
        _exit(127);                       /* сюди доходимо лише при невдачі exec */
    }

    int st;
    if (waitpid(pid, &st, 0) < 0) die("waitpid");
    printf("у fdtour-exec.txt (тобто в stdout нащадка):\n");
    show_file("fdtour-exec.txt");

    /* out теж треба закрити: копія в нащадкові вже своя, а наш номер
     * інакше витече й зсуне всі номери в наступних дослідах. */
    if (close(out) < 0 || close(plain) < 0 || close(sealed) < 0) die("close");
}
```

Нащадок — окремий файл, і навмисно найпростіший, щоб було видно: жодної домовленості з батьком у ньому немає. Свій звіт він пише в номер 2, бо номер 1 йому перенаправили, і все, що піде туди, читатиме вже не людина.

```c
/* fdchild.c — програма-нащадок: друкує свою таблицю дескрипторів
 * і перевіряє номери, названі в аргументах.
 *
 *   gcc -Wall -Wextra -O2 -o fdchild fdchild.c
 */
#define _GNU_SOURCE
#include <dirent.h>
#include <errno.h>
#include <fcntl.h>
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>

int main(int argc, char **argv)
{
    DIR *d = opendir("/proc/self/fd");
    if (d == NULL) { perror("opendir"); return 1; }
    int own = dirfd(d);

    int fds[256], n = 0;
    struct dirent *e;
    while ((e = readdir(d)) != NULL && n < 256) {
        if (e->d_name[0] == '.') continue;
        int fd = (int)strtol(e->d_name, NULL, 10);
        if (fd != own) fds[n++] = fd;
    }
    closedir(d);

    for (int i = 0; i < n; i++) {
        char path[64], target[512];
        snprintf(path, sizeof path, "/proc/self/fd/%d", fds[i]);
        ssize_t len = readlink(path, target, sizeof target - 1);
        if (len < 0) len = 0;
        target[len] = '\0';
        fprintf(stderr, "[нащадок] %d -> %s\n", fds[i], target);
    }

    for (int i = 1; i < argc; i++) {
        int fd = (int)strtol(argv[i], NULL, 10);
        errno = 0;
        int alive = fcntl(fd, F_GETFD) >= 0;
        fprintf(stderr, "[нащадок] переданий номер %s: %s\n",
                argv[i], alive ? "живий" : strerror(errno));
    }

    printf("[нащадок] цей рядок пішов у номер 1\n");
    fflush(stdout);
    return 0;
}
```

```
готуємо: out=3  plain=4  sealed=5
[нащадок] 0 -> /dev/pts/3
[нащадок] 1 -> /tmp/fdtour/fdtour-exec.txt
[нащадок] 2 -> /dev/pts/3
[нащадок] 4 -> /dev/null
[нащадок] переданий номер 4: живий
[нащадок] переданий номер 5: Bad file descriptor
у fdtour-exec.txt (тобто в stdout нащадка):
  | [нащадок] цей рядок пішов у номер 1
```

Тут варто прочитати кожен рядок окремо. Номер 1 нащадка веде у файл, хоч сам нащадок про файл нічого не знає й `open` не викликав: таблицю йому налаштували до запуску ([exec: заміна образу](topic:unix-linux/exec-semantics)). Номер 4 живий — звичайний дескриптор пережив повну заміну коду, даних і стека. Номер 5 відповідає `EBADF`: `O_CLOEXEC` спрацював рівно в мить `exec`, і в новому образі цього запису вже немає. А номера 3 у переліку немає взагалі — його звільнив `close(out)` ще до `exec`, і в таблиці лишилася дірка. Дірки — це нормально: номери не зобов'язані йти підряд, і код, що перебирає їх від нуля до першого невдалого, помиляється саме тут.

Прикметне не те, що це працює, а те, чого в коді немає. Немає жодного рядка про програму, яку ми запускаємо, немає прапорця командного рядка, немає домовленості. Уся команда `ls > out.txt` — це рівно ці чотири виклики над таблицею майбутнього процесу ([перенаправлення як робота з дескрипторами](topic:unix-linux/redirection-model)).

## Дослід п'ятий: номер, який уже не ваш

Найтихіша з усіх помилок із дескрипторами: запис у номер, що його ви закрили, а хтось інший уже зайняв. Помилки не буде — байти просто підуть не туди.

```c
static void demo_reuse(void)
{
    banner("Номер після close уже не ваш");

    int a = open("fdtour-a.txt", O_WRONLY | O_CREAT | O_TRUNC, 0644);
    if (a < 0) die("open fdtour-a.txt");
    printf("fdtour-a.txt дістав номер %d\n", a);
    if (close(a) < 0) die("close");
    /* змінна a досі тримає число — саме це нас і погубить */

    int b = open("fdtour-b.txt", O_WRONLY | O_CREAT | O_TRUNC, 0644);
    if (b < 0) die("open fdtour-b.txt");
    printf("fdtour-b.txt дістав номер %d%s\n",
           b, (a == b) ? "  — те саме число" : "");

    static const char msg[] = "рядок, призначений файлові A\n";
    ssize_t k = write(a, msg, sizeof msg - 1);
    printf("write(%d, ...) повернув %zd — жодної помилки\n", a, k);

    if (close(b) < 0) die("close");
    printf("а лежить він у fdtour-b.txt:\n");
    show_file("fdtour-b.txt");

    /* Як не наступити: число знищують тієї ж миті, що й об'єкт. */
    int c = open("/dev/null", O_RDONLY);
    if (c < 0) die("open /dev/null");
    if (close(c) < 0) die("close");
    c = -1;
    errno = 0;
    if (write(c, msg, 1) < 0)
        printf("write(-1, ...): %s — помилка на місці, а не за три модулі звідси\n",
               strerror(errno));
}
```

```
──── Номер після close уже не ваш ────
fdtour-a.txt дістав номер 3
fdtour-b.txt дістав номер 3  — те саме число
write(3, ...) повернув 52 — жодної помилки
а лежить він у fdtour-b.txt:
  | рядок, призначений файлові A
write(-1, ...): Bad file descriptor — помилка на місці, а не за три модулі звідси
```

`write` повернув повну кількість байтів, `errno` не знадобився, жодна перевірка коду повернення такого не спіймає. Знайти це можна тільки за наслідком — чужим рядком у чужому файлі, за півсистеми звідси. Ліки одні й прості: `close` і привласнення `-1` мають стояти поруч, як одна операція; у мовах із детермінованим руйнуванням це доручають обгортці, яка закриває номер рівно один раз ([RAII: ресурс прив'язаний до часу життя](topic:programming/raii)).

## Дослід шостий: витік до EMFILE

Щоб побачити стелю, не треба відкривати тисячу файлів — досить тимчасово опустити свою м'яку межу. Піднімати її назад до жорсткої дозволено будь-кому, тож дослід нічого по собі не лишає ([ліміти ресурсів](topic:unix-linux/resource-limits)).

```c
static void demo_leak(void)
{
    banner("Витік до EMFILE");

    struct rlimit saved;
    if (getrlimit(RLIMIT_NOFILE, &saved) < 0) die("getrlimit");
    printf("RLIMIT_NOFILE: м'який %llu, жорсткий %llu\n",
           (unsigned long long)saved.rlim_cur,
           (unsigned long long)saved.rlim_max);

    struct rlimit low = saved;
    low.rlim_cur = (saved.rlim_max < 64) ? saved.rlim_max : 64;
    if (setrlimit(RLIMIT_NOFILE, &low) < 0) die("setrlimit");
    printf("тимчасово опустили м'яку межу до %llu\n",
           (unsigned long long)low.rlim_cur);

    int leaked[256];
    int n = 0, last = -1, err = 0;
    while (n < (int)(sizeof leaked / sizeof leaked[0])) {
        int fd = open("/dev/null", O_RDONLY);
        if (fd < 0) { err = errno; break; }
        last = fd;
        leaked[n++] = fd;
    }
    printf("успішних open: %d, останній номер: %d\n", n, last);
    if (err != 0)
        printf("наступний open: %s\n", strerror(err));

    /* Подивитися в /proc/self/fd теж треба чимось. А нічим. */
    printf("подивитися в /proc/self/fd: %d\n", count_fds());

    /* Ось навіщо серверові «запасний» номер: щоб на EMFILE було чим
     * зробити хоч щось — прийняти й одразу закрити з'єднання, написати в лог. */
    if (n > 0 && close(leaked[--n]) < 0) die("close");
    printf("звільнили один номер — записів у таблиці: %d\n", count_fds());

    while (n > 0)
        if (close(leaked[--n]) < 0) die("close");
    if (setrlimit(RLIMIT_NOFILE, &saved) < 0) die("setrlimit назад");
}
```

```
──── Витік до EMFILE ────
RLIMIT_NOFILE: м'який 1024, жорсткий 1048576
тимчасово опустили м'яку межу до 64
успішних open: 61, останній номер: 63
наступний open: Too many open files
подивитися в /proc/self/fd: -1
звільнили один номер — записів у таблиці: 63
```

Арифметика сходиться до одиниці, і це варто перевірити самому:

```
м'яка межа RLIMIT_NOFILE      = 64
зайнято на старті: 0, 1, 2    = 3
вільних номерів               = 64 − 3 = 61
останній виданий номер        = 63
```

Але найважливіший рядок тут третій з кінця: `подивитися в /proc/self/fd: -1`. Щоб зазирнути у власну таблицю, потрібен ще один дескриптор — а їх немає, тому `opendir` теж падає з `EMFILE`. У цю саму стіну впирається сервер: він не може ні прийняти з'єднання, ні відкрити файл журналу, щоб про це поскаржитися. Тому серверам і роздають «запасний» номер — дескриптор, відкритий на старті просто так, щоб на `EMFILE` було що закрити й чим зробити хоч один необхідний виклик.

Ця відмова взагалі не схожа на відмову. Процес не падає, нічого не пише, усі вже прийняті з'єднання обслуговуються бездоганно — просто нові більше не приймаються. Ззовні це виглядає як загадкове «підвисання під навантаженням».

> 🔧 **Навіщо це.** Той самий дослід навпаки — і є готовий рецепт пошуку витоку в чужій програмі. Коли лічильник у `/proc/<pid>/fd` уже показав, що номери течуть, лишається питання «який саме `close` забули» — і відповідають на нього цілі посилань: `socket:[…]` вкаже на шлях обробки з'єднання, `pipe:[…]` — на запуск помічника, а шлях із позначкою `(deleted)` — на файл, що його вже видалили, але хтось досі тримає відкритим, а тому й місце на диску не звільнилося.

## Головна програма

Лишається порядок дослідів. Один рядок на початку `main` не декоративний: якщо вивід іде в конвеєр чи файл, `stdout` стає поблоковим, а нащадки пишуть у номер 2 напряму — і звіт нащадка вилізе на екран раніше, ніж рядок батька, надрукований до `fork`. Порядкова буферизація повертає рядкам той порядок, у якому справді відбувалися події.

```c
int main(int argc, char **argv)
{
    setvbuf(stdout, NULL, _IOLBF, 0);   /* порядок рядків = порядок подій */

    const char *child = (argc > 1) ? argv[1] : "./fdchild";

    demo_lowest();
    demo_two_levels();
    demo_manual_redirect();
    demo_exec(child);
    demo_reuse();
    demo_leak();
    return 0;
}
```

## Скільки це коштує

Перевірка дескриптора на вході в системний виклик — це порівняння з розміром таблиці, одне звертання в масив і перевірка, чи комірка не порожня. Видача нового номера дорожча, але не набагато: Linux тримає поряд із масивом бітову мапу зайнятих номерів (`open_fds`) і підказку `next_fd`, тож пошук найменшого вільного — це сканування кількох 64-бітних слів, де кожне слово покриває 64 номери одразу. Підказку зсувають донизу при кожному `close`, який звільнив менший номер, — саме тому правило найменшого вільного не коштує повного обходу таблиці навіть тоді, коли відкритих файлів десятки тисяч.

Сама таблиця росте подвоєнням, і місце під неї видно неозброєним оком. Вказівник на опис — 8 байтів, плюс кілька бітів на номер під мапи зайнятих і `close-on-exec`:

```
межу піднято до            = 1 048 576 номерів
масив вказівників          = 1 048 576 · 8 Б ≈ 8.4 МБ
дві бітові мапи            = 1 048 576 · 2 біт ≈ 0.26 МБ
```

Тому «підняти ліміт до мільйона» — рішення не безкоштовне, хоч і не страшне: пам'ять виділяється в міру росту таблиці, а не одразу.

А от читання `/proc/self/fd` дешевим не є взагалі. Це один `openat`, кілька `getdents`, а далі **три системні виклики на кожен номер** — `readlink` і два `fcntl`. Для процесу зі ста тисячами відкритих дескрипторів це триста тисяч переходів у ядро й назад ([системний виклик](topic:unix-linux/syscall-mechanics)). Для періодичного нагляду це задорого; загальносистемний лічильник у `/proc/sys/fs/file-nr` коштує одне читання й показує, скільки описів зайнято в усій системі, — цього досить, щоб помітити біду, а вже потім шукати винуватця обходом.

## Пастки

Усе нижче — граблі, на які наступають, поки код не написано власноруч.

- **Дивитися на таблицю — теж витрачати номер.** `opendir("/proc/self/fd")` займає дескриптор, який негайно з'являється у власному ж переліку, а на межі `EMFILE` просто не відкривається. Звідси й ідея запасного номера.
- **Перелік — не миттєвий знімок.** Поки ви ходите по каталогу й розпитуєте номери, таблиця живе. Спершу зберіть номери, потім розпитуйте; і не дивуйтеся `EBADF` на номер, який щойно був у переліку.
- **`/proc/<pid>/fd/N` — не спосіб продублювати дескриптор.** Відкриття такого шляху робить **нове відкриття** об'єкта з новим описом і власною позицією, до того ж із повною перевіркою прав. Для сокетів воно взагалі не спрацює: ядро віддає їм `no_open`, який завжди повертає `ENXIO` — те саме стосується `eventfd`, `epoll` і `timerfd`. Канали, до речі, відкриваються нормально. Єдиний законний спосіб узяти чужий дескриптор — щоб власник сам віддав його через сокет домену Unix ([сокети домену Unix і передача дескрипторів](topic:unix-linux/unix-domain-sockets)).
- **Буфер stdio живе окремо від дескриптора.** `fflush(NULL)` перед `fork` — інакше недописане надрукується двічі. `fflush` перед `_exit` — інакше не надрукується взагалі. `close(1)` не скидає буфер `stdout` і не дізнається, що номер тепер веде в інше місце.
- **`dup2(fd, fd)` — окремий випадок.** Якщо `fd` дійсний, `dup2` нічого не робить і повертає його ж, не чіпаючи прапорців. А `dup3` із такими самими аргументами навмисно повертає `EINVAL` — бо в нього є третій аргумент, який у цьому разі не мав би куди подітися.
- **Порядок при перенаправленні має значення.** `dup2(fd, 1)` спершу закриває те, що стояло в номері 1. Якщо в тому ж наборі правок номер 1 комусь ще потрібен як джерело, спершу зробіть із нього копію. Це та сама відмінність, через яку `2>&1 1>file` і `1>file 2>&1` в оболонці дають різний результат.
- **`FD_CLOEXEC` не копіюється при `dup`.** Іноді це рятує (перенаправлення переживає `exec`), іноді ранить: копія приватного дескриптора, зроблена «просто щоб було», витече в кожного запущеного помічника.
- **Дірки в таблиці — норма.** Перебирати номери від нуля до першого невдалого не можна: `close(3)` посеред роботи створює саме таку дірку. Щоб закрити все зайве перед `exec`, є `close_range(3, ~0U, 0)` (Linux 5.9), а з 5.11 той самий виклик уміє замість закривання просто розставити `FD_CLOEXEC`.
- **`EMFILE` і `ENFILE` — різні діагнози.** Перше означає «цей процес вичерпав свій ліміт» і лікується правкою межі або пошуком витоку в одній програмі. Друге — «вичерпалася система», і винуватець може бути де завгодно.
- **`select` ламається на номерах від 1024.** `fd_set` — це бітова мапа фіксованого розміру `FD_SETSIZE`, і `FD_SET(2000, &s)` тихо псує пам'ять поруч. Піднявши `RLIMIT_NOFILE`, ви отримуєте великі номери — і мусите брати `poll` або `epoll` ([select, poll, epoll](topic:unix-linux/select-poll-epoll)).
- **`close` не повторюють після `EINTR`.** Linux звільняє номер на самому початку операції, тож повтор закриє вже чужий об'єкт. Якщо дані важливі — `fsync` треба робити **до** `close`, бо помилка, повернута самим `close`, приходить тоді, коли врятувати вже нічого не можна ([EINTR: перервані виклики](topic:unix-linux/eintr-and-restart)).
