# ⚙️ Лабораторія розподільника: чотири способи записати ті самі 32 МіБ

Це стенд, на якому за хвилину видно, від чого насправді залежить роздробленість файлу: одна програма записує той самий обсяг чотирма різними способами, а потім питає ядро, на скільки шматків кожен файл розпався на носії. Числа виходять несподівані — самі собою дрібні записи майже не винні.

## Майданчик

Міряти на робочій файловій системі немає сенсу: результат розкаже переважно про те, що на ній діялося останні пів року. Потрібна власна — порожня, невелика й нікому більше не потрібна. Найдешевше зробити її з файлу: [петльовий пристрій](book:unix-linux/loop-device) віддає ядру звичайний файл як блоковий носій, і всередині цього носія живе повноцінна ext4.

```sh
dd if=/dev/zero of=/var/tmp/lab.img bs=1M count=1024
mkfs.ext4 -q -F /var/tmp/lab.img     # -F: ціль не блоковий пристрій, інакше mke2fs перепитає
sudo mkdir -p /mnt/lab
sudo mount -o loop /var/tmp/lab.img /mnt/lab
sudo chown "$USER" /mnt/lab
# після дослідів:  sudo umount /mnt/lab && rm /var/tmp/lab.img
```

Гігабайта досить: усі чотири досліди разом заберуть менш як п'яту частину, і розподільникові весь час буде з чого вибирати. Що закладають ключі `mkfs` і як вони описують геометрію носія — у [довідці зі створення й монтування](book:unix-linux/filesystem-families/api-mkfs-and-mount.md).

Одразу обмовка про те, чого цей майданчик **не** міряє. Образ лежить на файловій системі господаря і кешується нею, тож кожен `fsync` усередині проходить крізь два шари [кешу сторінок](book:unix-linux/page-cache-durability), а час виконання тут не означає нічого. Стенд міряє розкладку — і саме вона від подвійного кешування не залежить.

## Що міняється між дослідами

Досліди дібрано так, щоб між сусідніми мінявся рівно один чинник.

| дослід | порція | `fsync` | `fallocate` | письменників |
|---|---|---|---|---|
| `4k-fsync` | 4 КіБ | після кожної порції | ні | 1 |
| `stream` | 1 МіБ | лише наприкінці | ні | 1 |
| `duet` | 4 КіБ | після кожної порції | ні | 2 |
| `duet-falloc` | 4 КіБ | після кожної порції | так | 2 |

Перша пара питає, чи справді дрібний запис із негайною синхронізацією руйнує розкладку: `stream` віддає всю свободу відкладеному виділенню, `4k-fsync` не лишає файловій системі жодної можливості відкласти вибір. Третій дослід міняє щодо першого одну обставину — поруч з'являється сусід, який пише так само. Четвертий міняє щодо третього один системний виклик.

## Програма

```c
/* alloclab.c — чотири способи записати той самий обсяг і карта, що з них вийшла.
   cc -O2 -Wall -o alloclab alloclab.c                                          */
#define _GNU_SOURCE
#include <errno.h>
#include <fcntl.h>
#include <linux/fiemap.h>
#include <linux/fs.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/ioctl.h>
#include <sys/stat.h>
#include <sys/wait.h>
#include <unistd.h>

#define BATCH 512                       /* екстентів за один ioctl */
#define BLKSZ 4096ULL

struct plan {                           /* що саме різнить досліди */
    const char *name;
    size_t chunk;                       /* розмір однієї порції запису   */
    int    sync_each;                   /* fsync після кожної порції     */
    int    prealloc;                    /* сказати розмір наперед        */
    int    writers;                     /* скільки процесів пишуть разом */
};

static const struct plan PLANS[] = {
    { "4k-fsync",    4096,     1, 0, 1 },
    { "stream",      1u << 20, 0, 0, 1 },
    { "duet",        4096,     1, 0, 2 },
    { "duet-falloc", 4096,     1, 1, 2 },
};

struct map { unsigned long long extents, longest, meta; };

static int write_one(const char *path, const struct plan *pl, unsigned long long bytes)
{
    char *buf = malloc(pl->chunk);
    int fd = open(path, O_WRONLY | O_CREAT | O_TRUNC, 0644);

    if (fd < 0 || !buf) { perror(path); return -1; }
    memset(buf, 'x', pl->chunk);

    if (pl->prealloc && fallocate(fd, 0, 0, (off_t)bytes) < 0) {
        perror("fallocate");            /* EOPNOTSUPP тут теж поразка досліду */
        return -1;
    }
    for (unsigned long long off = 0; off < bytes; ) {
        size_t want = pl->chunk, done = 0;
        if (off + want > bytes) want = (size_t)(bytes - off);
        while (done < want) {
            ssize_t n = write(fd, buf + done, want - done);
            if (n < 0) { if (errno == EINTR) continue; perror("write"); return -1; }
            done += (size_t)n;
        }
        off += want;
        if (pl->sync_each && fsync(fd) < 0) { perror("fsync"); return -1; }
    }
    if (fsync(fd) < 0) { perror("fsync"); return -1; }
    free(buf);
    return close(fd);
}

static int map_file(const char *path, unsigned long long bytes, struct map *out)
{
    union {                             /* union, а не char[]: потрібне вирівнювання */
        struct fiemap fm;
        char raw[sizeof(struct fiemap) + BATCH * sizeof(struct fiemap_extent)];
    } q;
    unsigned long long next = 0;
    struct stat st;
    int fd, last = 0;

    memset(out, 0, sizeof(*out));
    if ((fd = open(path, O_RDONLY)) < 0) { perror(path); return -1; }

    while (!last) {
        memset(&q, 0, sizeof(q));
        q.fm.fm_start        = next;
        q.fm.fm_length       = FIEMAP_MAX_OFFSET;
        q.fm.fm_flags        = FIEMAP_FLAG_SYNC;   /* інакше побачимо намір */
        q.fm.fm_extent_count = BATCH;
        if (ioctl(fd, FS_IOC_FIEMAP, &q) < 0) { perror("FIEMAP"); close(fd); return -1; }
        if (q.fm.fm_mapped_extents == 0) break;

        for (unsigned i = 0; i < q.fm.fm_mapped_extents; i++) {
            struct fiemap_extent *e = &q.fm.fm_extents[i];
            out->extents++;
            if (e->fe_length > out->longest) out->longest = e->fe_length;
            next = e->fe_logical + e->fe_length;
            if (e->fe_flags & FIEMAP_EXTENT_LAST) last = 1;
        }
    }
    /* Дерево екстентів теж займає блоки, і ext4 враховує їх у st_blocks:
       усе, що обліковано понад округлені до блока дані, — це карта.      */
    if (fstat(fd, &st) == 0) {
        unsigned long long data = (bytes + BLKSZ - 1) / BLKSZ * BLKSZ;
        unsigned long long held = (unsigned long long)st.st_blocks * 512;
        out->meta = held > data ? held - data : 0;
    }
    close(fd);
    return 0;
}

int main(int argc, char **argv)
{
    unsigned long long bytes = 32ULL << 20;
    const char *dir = argc > 1 ? argv[1] : ".";
    char path[512];

    if (argc > 2) bytes = strtoull(argv[2], NULL, 10) << 20;
    printf("%-14s %10s %12s %10s\n", "strategy", "extents", "longest KiB", "map B");

    for (unsigned s = 0; s < sizeof PLANS / sizeof PLANS[0]; s++) {
        const struct plan *pl = &PLANS[s];
        pid_t kids[4];
        struct map m;

        for (int w = 0; w < pl->writers; w++) {
            snprintf(path, sizeof path, "%s/%s-%d.dat", dir, pl->name, w);
            if ((kids[w] = fork()) == 0)
                _exit(write_one(path, pl, bytes) == 0 ? 0 : 1);
        }
        for (int w = 0; w < pl->writers; w++) {
            int status;
            waitpid(kids[w], &status, 0);
            if (!WIFEXITED(status) || WEXITSTATUS(status)) {
                fprintf(stderr, "письменник упав на досліді %s\n", pl->name);
                return 1;
            }
        }
        snprintf(path, sizeof path, "%s/%s-0.dat", dir, pl->name);
        if (map_file(path, bytes, &m) < 0) return 1;
        printf("%-14s %10llu %12llu %10llu\n",
               pl->name, m.extents, m.longest >> 10, m.meta);
    }
    return 0;
}
```

Два місця тут варті окремої уваги. Перше — `FIEMAP` віддає екстенти порціями: у заголовку ми оголошуємо, скільки їх готові прийняти, а ядро відповідає полем `fm_mapped_extents`. Тому виклик стоїть у циклі, який щоразу починає з логічного зсуву за останнім отриманим екстентом і зупиняється на ознаці `FIEMAP_EXTENT_LAST`. Повний розбір формату — у [власному filefrag](book:unix-linux/filesystem-families/proj-fiemap-layout.md); тут із нього взято мінімум.

Друге — стовпчик `map B`. Дерево екстентів ext4 тримає в самому inode лише перші чотири записи; далі воно виходить на носій окремими блоками, і ці блоки враховані в `st_blocks`. Отже, різниця між обліком і округленим до блока розміром файлу і є розміром карти. Це та сама причина, через яку `du` роздробленого файлу перевищує його довжину.

## Що виходить

```
strategy          extents  longest KiB      map B
4k-fsync                2        30720          0
stream                  1        32768          0
duet                 2593           88      36864
duet-falloc             1        32768          0
```

*Порядок величин на щойно створеному образі, типове ядро 6.x і ext4 з усталеними налаштуваннями; точні числа у вас будуть інші.*

Перший рядок — головна несподіванка стенда. Вісім тисяч записів по 4 КіБ, кожен із негайною синхронізацією, тобто жодної можливості відкласти рішення, — а файл вийшов практично суцільним. Отже, дрібний запис сам собою розкладку не псує: розподільник бере блоки із запасу, а запас видає смугами, і поки з тієї смуги ніхто більше не бере, файл росте суцільно.

Другий рядок — ідеал, і обсяг досліду дібрано так, щоб цей ідеал взагалі був досяжний. Запис карти ext4 має стелю довжини: 32768 блоків, тобто 128 МіБ при блоці в 4 КіБ. Тридцять два мебібайти в неї вміщаються цілком, тож «один екстент» тут справді означає одну смугу. Візьміть гігабайт — і найкращий можливий результат стане вісім екстентів, покладених стик у стик, а стовпчик перестане читатися очима. Друга межа ідеалу — політика виштовхування: щойно брудних сторінок назбирується забагато, ядро починає виносити їх, не чекаючи кінця запису, і відкладене виділення бачить уже не весь файл, а стільки, скільки встигло накопичитися.

Третій рядок показує, чого бракувало, — сусіда. Той самий спосіб запису, але тепер їх двоє, і файл розсипався на дві з половиною тисячі шматків по 88 КіБ у найкращому разі. Карта, яка в першому досліді вміщалася в inode, тепер займає дев'ять блоків носія.

## Чому двоє дають гребінку

Розгадка — у тому, звідки береться запас. Ext4 має два різні джерела, і вибір між ними робиться за розміром запиту. Коли просять щонайменше `mb_stream_req` блоків (усталено 16, тобто 64 КіБ), запас відкладається **на inode**: він належить одному файлові й ділити його ні з ким не треба. Дрібніші запити обслуговуються зі **спільного запасу групи локальності** — свого на кожне ядро процесора, але спільного для всіх файлів, що пишуть на цьому ядрі.

Запис по 4 КіБ — це один блок, глибоко під порогом. Тому обидва письменники беруть блоки з тієї самої смуги, по черзі, у ритмі, який задають їхні `fsync`. Кожен захід — один блок, наступний блок тієї смуги дістається сусідові, і у файлі відкривається новий екстент. Коли смуга (усталено `mb_group_prealloc` = 512 блоків, 2 МіБ) вичерпується, береться наступна — і чергування починається спочатку.

Звідси видно й точну умову, за якої біда не настає: досить, щоб порція перевищила 64 КіБ. Служба, яка накопичує рядки й скидає їх мегабайтом, отримає запас на inode і не помітить сусіда взагалі.

## Чому `fallocate` прибирає її повністю

Четвертий дослід пише так само дрібно й так само вдвох — а виходить один екстент. `fallocate` на всі 32 МіБ — це запит на 8192 блоки поспіль, тобто далеко за порогом: файл дістає власне місце ще до першого байта. Далі відбувається головне: наступні `write` **уже нічого не виділяють**. Вони лише перетворюють незаписаний екстент на записаний у межах уже закріпленої смуги, а розподільника не турбують зовсім — тож і сусід їм байдужий.

Тобто гребінка з'явилася не тому, що записи були дрібні, а тому, що дрібними були **рішення** розподільника, і вони чергувалися з чужими. `fallocate` не зменшує кількості записів — він зводить вісім тисяч рішень до одного. Режими виклику й вимоги до вирівнювання — у [контракті розріджених операцій](book:unix-linux/sparse-files/api-sparse-syscalls.md).

## Пастки

**Без `FIEMAP_FLAG_SYNC` міряється намір.** Поки сторінки брудні, блоків під ними немає, і [ioctl](book:unix-linux/ioctl-interface) віддає один довгий екстент з ознакою `FIEMAP_EXTENT_DELALLOC` і нульовою фізичною адресою. Зняти прапорець — і `stream` та `duet` покажуть однакову ідилію.

![Дві відповіді FIEMAP на той самий файл. Угорі виклик без прапорця: один суцільний екстент з ознакою delalloc і нульовою фізичною адресою — виміряно намір програми, а не розкладку на носії. Унизу виклик із прапорцем FIEMAP_FLAG_SYNC: той самий файл після виштовхування розпадається на дванадцять шматків із фізичними адресами](img/fiemap-sync.svg)

*Один і той самий файл дає два різні числа, і лише друге стосується носія.*

**Стенд варто хоч раз звірити зі стороннім приладом.** `filefrag -e -s /mnt/lab/duet-0.dat` покаже той самий перелік у людському вигляді, і ключ `-s` відповідає тут тому самому `FIEMAP_FLAG_SYNC`. Розбіжність між утилітою й програмою означала б помилку в обході карти, а не відкриття про розподільника.

**Двоє письменників мають бути на одному ядрі.** Групи локальності — окремі на кожен процесор, тож розведені по різних ядрах письменники братимуть із різних смуг, і гребінка вийде куди слабшою або зникне зовсім. Запускайте стенд як `taskset -c 0 ./alloclab /mnt/lab` — [прив'язка](book:unix-linux/cpu-affinity) успадковується нащадками `fork`.

**На XFS `du` бреше кілька хвилин.** Ця файлова система лишає за кінцем зростаючого файлу спекулятивний запас, і одразу після запису облік показує більше, ніж у файлі байтів. Число саме зменшиться, коли запас підріже фоновий обхідник. Кількості екстентів це не стосується — лише стовпчика `map B`, який на XFS узагалі не має того сенсу, що на ext4.

**Числа залежать від стану майданчика.** На свіжому образі вони найкращі з можливих; на образі, який спершу заповнили дрібними файлами й наполовину спорожнили, ті самі досліди дадуть гірші результати всюди. Проста перевірка, чи не поплив майданчик під час прогону: переставте досліди у зворотному порядку. Якщо числа помітно змінилися — образові забракло вільного місця, і треба брати більший.
