# ⚙️ Реалізація виколювання та вставляння стирань у декодері Вітербі

Виколювання (puncturing) перетворює вихідний потік материнського кодера у потік з вищою кодовою швидкістю, а приймач виконує девіколювання (depuncturing), відновлюючи довжину потоку шляхом вставляння стирань (`LLR = 0`). У даному проекті показано практичну реалізацію модуля виколювання та девіколювання для материнського коду `R₀ = 1/2` та маски виколювання `R = 3/4`.

## Архітектура перетворювача швидкості

Програму розбито на два автономних програмних блоки:
1. **Кодер-виколювач (Puncturing Engine):** Приймає послідовність вихідних бітів материнського коду `R₀ = 1/2`, накладає двовимірну маску `P = [1 1 0; 1 0 1]` періоду `p = 3` та формує стиснутий масив переданих бітів.
2. **Девіколювач приймача (Depuncturer / Erasure Inserter):** Приймає м'які відліки каналу LLR, розгортає їх назад у потік довжини материнського коду та підставляє `LLR = 0.0f` на місця виколотих бітів перед передачею в [декодер Вітербі](root:com-modulation/viterbi-decoder).

Оскільки періодичність маски становить `p = 3` тактів вхідних даних (що відповідає 6 бітам материнського кодера та 4 переданим бітам у канал), індексація фази виконується за модулем 3 (`i % 3`). На кожному такті функція перевіряє значення маски `P[0][phase]` та `P[1][phase]` для двох вихідних гілок кодера.

Нижче наведено робочий код на мовах C та C++, де кожна реалізація є повністю ідіоматичною для своєї системи.

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

// Матриця виколювання для R = 3/4 (період p = 3, 2 виходи кодера)
// P[0] = {1, 1, 0}, P[1] = {1, 0, 1}
static const int PUNC_MASK_3_4[2][3] = {
    {1, 1, 0},
    {1, 0, 1}
};

// Застосування виколювання R = 3/4 до материнського потоку R = 1/2
// in_bits: масив пар бітів [v1_0, v2_0, v1_1, v2_1, ...] завдовжки 2 * n_input_bits
// out_bits: вихідний масив для передачі у канал (завдовжки (4/3) * n_input_bits)
size_t puncture_encode_3_4(const int *in_bits, size_t n_input_bits, int *out_bits) {
    size_t out_idx = 0;
    for (size_t i = 0; i < n_input_bits; i++) {
        int phase = (int)(i % 3);
        int v1 = in_bits[2 * i + 0];
        int v2 = in_bits[2 * i + 1];

        if (PUNC_MASK_3_4[0][phase]) {
            out_bits[out_idx++] = v1;
        }
        if (PUNC_MASK_3_4[1][phase]) {
            out_bits[out_idx++] = v2;
        }
    }
    return out_idx;
}

// Девіколювання на приймачі: вставляння LLR = 0.0f у місця виколотих бітів
// rx_llr: м'які вибірки з каналу (довжина n_rx_llr)
// out_llr: відновлений потік для декодера Вітербі (довжина n_input_bits * 2)
void depuncture_decode_3_4(const float *rx_llr, size_t n_input_bits, float *out_llr) {
    size_t rx_idx = 0;
    for (size_t i = 0; i < n_input_bits; i++) {
        int phase = (int)(i % 3);

        // Відновлення v1
        if (PUNC_MASK_3_4[0][phase]) {
            out_llr[2 * i + 0] = rx_llr[rx_idx++];
        } else {
            out_llr[2 * i + 0] = 0.0f; // Стирання (erasure)
        }

        // Відновлення v2
        if (PUNC_MASK_3_4[1][phase]) {
            out_llr[2 * i + 1] = rx_llr[rx_idx++];
        } else {
            out_llr[2 * i + 1] = 0.0f; // Стирання (erasure)
        }
    }
}

int main(void) {
    // 3 інформаційні біти -> 6 бітів материнського коду R=1/2 -> 4 виколоті біти R=3/4
    int mother_bits[6] = {1, 1,  1, 0,  0, 1}; // 3 пари (v1, v2)
    int punctured_bits[4];

    size_t n_punc = puncture_encode_3_4(mother_bits, 3, punctured_bits);
    printf("Виколотий потік (4 біти замість 6): ");
    for (size_t i = 0; i < n_punc; i++) {
        printf("%d ", punctured_bits[i]);
    }
    printf("\n");

    // Моделювання каналу AWGN (прості LLR значення)
    float rx_channel_llr[4] = {2.5f, 1.8f, 2.1f, -1.9f};
    float restored_viterbi_llr[6];

    depuncture_decode_3_4(rx_channel_llr, 3, restored_viterbi_llr);

    printf("Відновлений LLR потік для декодера Вітербі:\n");
    for (size_t i = 0; i < 3; i++) {
        printf("Такт %zu: v1 = %+.1f, v2 = %+.1f\n",
               i + 1, restored_viterbi_llr[2 * i], restored_viterbi_llr[2 * i + 1]);
    }

    return 0;
}
```

```cpp
#include <iostream>
#include <vector>
#include <span>
#include <array>

class RateAdapter {
public:
    // Період p = 3, швидкість R = 3/4 для материнського коду R0 = 1/2
    static constexpr std::array<std::array<bool, 3>, 2> PUNC_MASK_3_4 = {{
        {true, true, false},
        {true, false, true}
    }};

    // Кодувальник виколювання R = 3/4
    [[nodiscard]] static std::vector<int> puncture(std::span<const int> mother_bits) {
        const size_t n_pairs = mother_bits.size() / 2;
        std::vector<int> result;
        result.reserve((n_pairs * 4) / 3);

        for (size_t i = 0; i < n_pairs; ++i) {
            const size_t phase = i % 3;
            const int v1 = mother_bits[2 * i + 0];
            const int v2 = mother_bits[2 * i + 1];

            if (PUNC_MASK_3_4[0][phase]) {
                result.push_back(v1);
            }
            if (PUNC_MASK_3_4[1][phase]) {
                result.push_back(v2);
            }
        }
        return result;
    }

    // Девіколювач на приймачі (вставляння LLR = 0.0f)
    [[nodiscard]] static std::vector<float> depuncture(std::span<const float> channel_llr, size_t n_pairs) {
        std::vector<float> restored_llr(n_pairs * 2, 0.0f);
        size_t rx_idx = 0;

        for (size_t i = 0; i < n_pairs; ++i) {
            const size_t phase = i % 3;

            if (PUNC_MASK_3_4[0][phase]) {
                restored_llr[2 * i + 0] = channel_llr[rx_idx++];
            } // інакше залишається 0.0f (стирання)

            if (PUNC_MASK_3_4[1][phase]) {
                restored_llr[2 * i + 1] = channel_llr[rx_idx++];
            } // інакше залишається 0.0f (стирання)
        }
        return restored_llr;
    }
};

int main() {
    const std::vector<int> mother_stream = {1, 1,  1, 0,  0, 1};
    const auto punctured = RateAdapter::puncture(mother_stream);

    std::cout << "Виколотий потік (C++): ";
    for (int b : punctured) {
        std::cout << b << " ";
    }
    std::cout << "\n";

    const std::vector<float> channel_llr = {2.5f, 1.8f, 2.1f, -1.9f};
    const auto restored = RateAdapter::depuncture(channel_llr, 3);

    std::cout << "Відновлений LLR потік для декодера:\n";
    for (size_t i = 0; i < restored.size() / 2; ++i) {
        std::cout << "Такт " << (i + 1) << ": v1 = " << restored[2 * i]
                  << ", v2 = " << restored[2 * i + 1] << "\n";
    }

    return 0;
}
```

## Детальний аналіз реалізації та покроковий розбір коду

Розгляньмо крок за кроком, як працюють функції кодування та девіколювання у наведених реалізаціях.

### 1. Функція кодувальника виколювання (`puncture_encode_3_4` / `RateAdapter::puncture`)

Функція приймає на вхід масив бітів материнського коду `mother_bits`, де біти організовані парами `(v1, v2)` для кожного такту входу.
- Змінна `phase = i % 3` визначає поточну колонку у двовимірній матриці виколювання `PUNC_MASK_3_4`.
- Для першої вихідної гілки кодера перевіряється умова `PUNC_MASK_3_4[0][phase]`. Якщо прапорець істинний, біт `v1` зберігається у вихідному масиві `out_bits`.
- Для другої вихідної гілки перевіряється `PUNC_MASK_3_4[1][phase]`. Якщо прапорець істинний, біт `v2` додається у вихідний масив.
- За один повний період з 3 тактів (`i = 0, 1, 2`) з 6 вхідних бітів материнського кодера зберігається рівно 4 біти. У результаті розмір переданого потоку скорочується у 1.5 раза.

### 2. Функція девіколювача приймача (`depuncture_decode_3_4` / `RateAdapter::depuncture`)

Функція девіколювання приймає з каналу масив м'яких відліків `rx_llr` завдовжки `N_rx` та відновлює повну довжину масиву `2 * n_input_bits` для подачі на декодер Вітербі.
- Для кожної з двох гілок кодера випробовується відповідний елемент маски виколювання.
- Якщо маска містить `1`, чергове значення `rx_llr[rx_idx++]` зчитується з канального масиву й записується на відповідну позицію відновленого вектора `out_llr`.
- Якщо маска містить `0`, у відновлений масив записується плаваюче значення `0.0f` (стирання). Індекс канального масиву `rx_idx` при цьому не інкрементується.

### 3. Детальний розрахунок буферів пам'яті та вирівнювання

Під час розробки високопродуктивних програмних і залізобетонних модулів узгодження швидкості слід дотримуватися кількох критичних алгоритмічних та архітектурних вимог:

- **Розрахунок розміру бувера пам'яті:** Для потоку з `N` вхідних інформаційних бітів материнський кодер створює `2N` бітів (`R₀ = 1/2`). При виколюванні за маскою `R = 3/4` розмір вихідного масиву передавача становить точні `⌈4N / 3⌉` бітів. На приймачі девіколювач повинен виділити пам'ять під повні `2N` елементів типу `float`, щоб передати їх у масив метрик декодера Вітербі.
- **Стирання LLR = 0.0f та динамічний діапазон:** У нульові позиції девіколювач записує плаваюче значення `0.0f` (або `0` при квантуванні LLR у 8-бітні цілі числа `int8_t`). Оскільки метрика накопиченого шляху Вітербі обчислюється як сума `D_new = D_old + ∑ (-1)^v̂_j · LLR_j`, нульовий доданок не зміщує відносну перевагу жодного з кандидатів у ґратці.
- **Оптимізація SIMD (AVX2 / NEON):** На сучасних процесорах розгортання LLR масиву та вставляння нулів виконується через векторні інструкції зсуву та маскування (`_mm256_blendv_ps` або `vbslq_f32`). Це дозволяє обробляти до 8 або 16 м'яких відліків за один такт процесора, виключаючи розгалуження `if/else` у гарячому циклі приймача.
- **Хвостові біти (Tail bits):** При скиданні стану кодера в 0 наприкінці кадри (tail bits) виколювання зазвичай вимикається (передаються всі хвостові біти парності), щоб гарантувати надійне завершення простеження шляху назад (traceback) у нульовий стан.
- **Кільцевий буфер узгодження швидкостей (Circular Buffer Rate Matching):** У сучасних 4G LTE та 5G NR трансиверах вихід кодера складається у кільцевий буфер пам'яті, звідки зчитується потрібна кількість бітів залежно від обраного розміру транспортного блоку (TBS). Виколювання зводиться до того, що зчитувальна головка зчитує лише потрібний безперервний шматок бітів з кільця.

### 4. Інтеграція з драйверами ядра та мережевими стеками

У реальних бездротових мостах та драйверах ядра Linux (зокрема у стеку `mac80211` та драйверах Wi-Fi `ath9k`/`iwlwifi`) таблиця маск виколювання не обчислюється в кожному циклі знову. Драйвер формує таблицю зсувів індексів (lookup table) на етапі ініціалізації пристрою.

Під час обробки переривання від ДМА-контролера приймача драйвер викликає векторну функцію девіколювання, яка розгортає пачку прийнятих пакетів безпосередньо в Ring-буфер сигнального процесора (DSP). Завдяки відсутності динамічного виділення пам'яті (`malloc`/`new`) у режимі ядра вдається досягти нульових затримок обробки кадрів.
