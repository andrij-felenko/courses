# ⚙️ Забрати консоль під себе й віддати назад: VT_PROCESS у робочому коді

Це програма мовою C, яка знаходить вільну віртуальну консоль, перемикається на неї, вимикає малювання тексту й бере перемикання на себе — а потім чесно віддає екран, коли людина тисне Alt+F1. Половина роботи тут припадає не на «забрати», а на «повернути»: помилка саме в цій половині лишає систему з екраном, на якому нічого немає й нічого не перемикається.

**Задача.** Дійти від «я звичайний процес» до «я розпоряджаюся екраном» і повернутися назад, не лишивши по собі стану, який довелося б розгрібати перезавантаженням.

**Ідея.** Ядро в цій схемі нічого не вирішує само: воно надсилає сигнал і чекає на відповідь рівно доти, доки відповідь прийде. Отже вся програма — це цикл із двома подіями («віддай» і «тримай назад») та двома відповідями на них. Уся ж складність — у тому, що між сигналом і відповіддю не можна ні впасти, ні заснути назавжди, а обробник сигналу — найгірше в програмі місце, щоб робити в ньому хоч щось.

## Порядок дій має значення, і він не симетричний

Вільний номер консолі підказує саме ядро: `VT_OPENQRY` на дескрипторі `/dev/tty0` повертає перший невживаний номер (або −1, якщо вільних не лишилося). Далі — `VT_ACTIVATE` на цей номер і `VT_WAITACTIVE`, щоб дочекатися, поки перемикання справді відбудеться, а не просто буде замовлене.

А тоді йдуть дві дії, і їхній порядок — не смак. Спершу `VT_SETMODE` з `VT_PROCESS`: домовляємося про рукостискання. І лише потім `KDSETMODE` з `KD_GRAPHICS`: вимикаємо текст. Якщо зробити навпаки й на цьому обірватися, консоль опиниться в стані, з якого немає виходу. У ядрі, у `change_console()`, стоїть перевірка: `if (vc->vc_mode == KD_GRAPHICS) return;` — перемикання з консолі, яка в графічному режимі, але **не** під керуванням процесу, ядро мовчки викидає. Ні сигналу, ні помилки, ні перемикання: клавіші просто перестають діяти.

Розбирати конструкцію треба точно у зворотному порядку: спершу `KD_TEXT`, потім `VT_AUTO`. Не навпаки — інакше на мить (а якщо програма впаде між двома викликами, то назавжди) виникає та сама пара «графіка без господаря».

```c
/* vtgrab.c — забрати вільну віртуальну консоль і коректно її віддати.
   Збірка: cc -O2 -Wall -o vtgrab vtgrab.c
   Запуск: sudo ./vtgrab   — краще по мережі, бо екран консолі буде порожній */
#define _GNU_SOURCE
#include <errno.h>
#include <fcntl.h>
#include <signal.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/ioctl.h>
#include <linux/kd.h>
#include <linux/vt.h>

static volatile sig_atomic_t got_rel, got_acq, got_quit;

static int vtfd = -1, vtnum = -1, prev_vt = -1;
static int have_mode, have_gfx;

static void handler(int sig)
{
    if (sig == SIGRTMIN)          got_rel = 1;   /* «віддай екран» */
    else if (sig == SIGRTMIN + 1) got_acq = 1;   /* «екран знову твій» */
    else                          got_quit = 1;  /* SIGINT або SIGTERM */
}

/* Викликається з atexit(), тобто й на аварійному виході через die(). */
static void restore(void)
{
    if (vtfd < 0)
        return;
    if (have_gfx)                                /* спершу повертаємо текст… */
        ioctl(vtfd, KDSETMODE, KD_TEXT);
    if (have_mode) {                             /* …і тільки потім автомат */
        struct vt_mode m = { .mode = VT_AUTO };
        ioctl(vtfd, VT_SETMODE, &m);
    }
    if (prev_vt > 0)
        ioctl(vtfd, VT_ACTIVATE, prev_vt);
    close(vtfd);
    vtfd = -1;
}

static void die(const char *what)
{
    fprintf(stderr, "%s: %s\n", what, strerror(errno));
    exit(1);                                     /* atexit() поверне консоль */
}

int main(void)
{
    char path[32];
    struct vt_stat st;
    struct vt_mode vm;
    struct sigaction sa;
    sigset_t block, prev;
    int ctl;

    ctl = open("/dev/tty0", O_RDWR | O_NOCTTY);  /* «та консоль, що на екрані» */
    if (ctl < 0) die("open /dev/tty0");
    if (ioctl(ctl, VT_GETSTATE, &st) < 0) die("VT_GETSTATE");
    prev_vt = st.v_active;                       /* куди повертатися на виході */
    if (ioctl(ctl, VT_OPENQRY, &vtnum) < 0 || vtnum < 0) die("VT_OPENQRY");
    close(ctl);

    snprintf(path, sizeof path, "/dev/tty%d", vtnum);
    vtfd = open(path, O_RDWR | O_NOCTTY);
    if (vtfd < 0) die(path);
    atexit(restore);

    /* Блокуємо сигнали ДО того, як вони можуть прийти: прапорець і перевірку
       прапорця не можна лишати без захисту, інакше сигнал прослизне між ними. */
    sigemptyset(&block);
    sigaddset(&block, SIGRTMIN);
    sigaddset(&block, SIGRTMIN + 1);
    sigaddset(&block, SIGINT);
    sigaddset(&block, SIGTERM);
    if (sigprocmask(SIG_BLOCK, &block, &prev) < 0) die("sigprocmask");

    memset(&sa, 0, sizeof sa);
    sa.sa_handler = handler;
    sa.sa_mask = block;              /* обробника не перерве інший обробник */
    sigaction(SIGRTMIN, &sa, NULL);
    sigaction(SIGRTMIN + 1, &sa, NULL);
    sigaction(SIGINT, &sa, NULL);
    sigaction(SIGTERM, &sa, NULL);

    if (ioctl(vtfd, VT_ACTIVATE, vtnum) < 0) die("VT_ACTIVATE");
    if (ioctl(vtfd, VT_WAITACTIVE, vtnum) < 0) die("VT_WAITACTIVE");

    memset(&vm, 0, sizeof vm);
    vm.mode = VT_PROCESS;
    vm.waitv = 0;
    vm.relsig = SIGRTMIN;
    vm.acqsig = SIGRTMIN + 1;
    if (ioctl(vtfd, VT_SETMODE, &vm) < 0) die("VT_SETMODE");
    have_mode = 1;

    if (ioctl(vtfd, KDSETMODE, KD_GRAPHICS) < 0) die("KDSETMODE");
    have_gfx = 1;
    fprintf(stderr, "тримаю /dev/tty%d, Alt+F%d поверне назад\n", vtnum, prev_vt);

    for (;;) {
        while (!got_rel && !got_acq && !got_quit)
            sigsuspend(&prev);       /* сигнали розблоковано лише всередині */

        if (got_rel) {
            got_rel = 0;
            fprintf(stderr, "просять віддати\n");
            /* Тут — справжня робота: зняти з себе право розпоряджатися
               графічним пристроєм і дочекатися, поки залізо справді вільне. */
            if (ioctl(vtfd, VT_RELDISP, 1) < 0)          /* 1 — «віддаю» */
                perror("VT_RELDISP");
        }
        if (got_acq) {
            got_acq = 0;
            fprintf(stderr, "екран повернули\n");
            /* Забрати пристрій назад, поставити свій відеорежим, перемалювати. */
            if (ioctl(vtfd, VT_RELDISP, VT_ACKACQ) < 0)
                perror("VT_RELDISP ack");
        }
        if (got_quit)
            break;
    }
    return 0;                        /* restore() з atexit() зробить решту */
}
```

## Обробник лише зводить прапорець

Обробник тут — три присвоєння, і це не мінімалізм заради краси. Сигнал приходить у довільній точці програми: посеред `malloc()`, посеред `fprintf()`, посеред перерозподілу внутрішніх структур бібліотеки. Викликати з обробника ті самі функції означає зайти в них удруге, поки вони не завершили першого разу, — і отримати або зіпсовану пам'ять, або взаємне блокування на замку, який утримує перервана половина програми. Дозволений із обробника набір — короткий і перелічений у стандарті: [безпечні в обробнику виклики](root:sys-unix/async-signal-safety) — це переважно системні виклики, і `printf` до них не належить.

Звідси й `volatile sig_atomic_t`: тип, запис у який відбувається однією неподільною дією, і слово `volatile`, щоб компілятор не «оптимізував» повторне читання прапорця в циклі, вирішивши, що змінити його нікому.

Друга половина захисту — блокування. Якби сигнали були розблоковані, він міг би прийти в проміжку між перевіркою `while (!got_rel …)` і викликом сну: прапорець уже зведений, а програма щойно вирішила заснути й прокинеться лише від наступного сигналу, якого не буде. Тому сигнали заблоковані весь час, а `sigsuspend()` розблоковує їх лише на час свого власного сну — неподільно, як одна дія.

> 🔧 **Навіщо це.** Ціна помилки тут не «програма впаде», а «система стане». Ядро тримає перемикання й чекає на `VT_RELDISP` від процесу, який заблокувався сам на собі в обробнику сигналу. З погляду людини за клавіатурою це чорний екран, який не реагує ні на що.

## Відповісти можна й «ні»

Одиниця в `VT_RELDISP` — не просто «я почув». Це відповідь «так», і в неї є пара: нуль означає «не віддаю». Ядро на нього не ображається — воно просто забуває про замовлене перемикання (`vc->vt_newvt = -1`) і лишає екран там, де він був. Для людини за клавіатурою це виглядає як натиснута й проігнорована клавіша: жодного повідомлення, жодної помилки.

Це право вето, а не відстрочка: сказати «пізніше» протокол не вміє, і сказати «зачекай три секунди» теж. Або віддаю зараз, або не віддаю зовсім і людина тисне ще раз. Так само не вийде перенаправити перемикання: номер консолі-призначення ядро записує в себе **до** того, як надіслати сигнал (інакше між сигналом і відповіддю виникала б гонитва), тож на момент, коли ви прокинулися, вибір уже зроблено — вам лишається його схвалити або скасувати.

## Відновлення на виході — і чому його не можна гарантувати

`atexit(restore)` покриває звичайне завершення й будь-який `exit()` з помилки. `SIGINT` і `SIGTERM` заведені в той самий прапорцевий механізм: обробник ставить `got_quit`, головний цикл виходить, `restore()` спрацьовує дорогою назовні.

Чого цим не покрити — `SIGKILL` і падіння з `SIGSEGV` посеред роботи з графікою. Зсередини процесу гарантії тут не існує в принципі, і ядро це знає. Коли надходить час перемикати консоль, `change_console()` спершу надсилає relsig і дивиться на результат: якщо `kill_pid()` каже, що адресата немає, ядро викликає `reset_vc()` — повертає консоль у `KD_TEXT` і `VT_AUTO`, скидає палітру й перемикається саме. У коментарі до цього місця так і написано: інакше «доводиться мучитися, коли X-сервер помер, а екран лишився погашеним через KD_GRAPHICS».

Сітка є, але з двома дірками. Перша: вона спрацьовує не в мить смерті процесу, а на **наступній** спробі перемкнутися. Якщо процес помер уже після того, як сигнал успішно доставлено, то поточне перемикання просто загубиться — ядро чекає на відповідь від того, кого вже немає. Коментар у `change_console()` описує це прямо: перемикання «губиться» в очікуванні відповіді, і надія на те, що людина натисне ще раз. Вона натискає — і другого разу все розплутується.

Друга діра серйозніша: сітка натягнута лише під гілкою `VT_PROCESS`. Консоль, яка лишилася в `KD_GRAPHICS` із режимом `VT_AUTO`, у цю гілку не потрапляє взагалі — ядро просто виходить із `change_console()`, нікого не питаючи й нікуди не перемикаючись. Такої консолі не зрушить ні гаряча клавіша, ні `chvt` по мережі: обидва шляхи ведуть у ту саму перевірку. Лікується це лише поверненням `KD_TEXT` на цю консоль — тобто трирядковою програмою з правами адміністратора, запущеною ззовні.

## Навмисне зависання

Щоб побачити першу половину протоколу без сітки, приберіть із гілки `got_rel` виклик `VT_RELDISP` — лишіть повідомлення й нічого більше. Зберіть, запустіть по ssh, натисніть Alt+F1.

Не станеться нічого. У журналі на ssh з'явиться «просять віддати», екран консолі лишиться порожнім, клавіші перемикання перестануть діяти. Ядро вже записало номер консолі, на яку треба перейти, вже надіслало сигнал і тепер чекає. Часу очікування тут немає навмисно: увірвати очікування означало б висмикнути екран з-під процесу, який, можливо, саме цієї миті пише в графічний пристрій, — а це той самий одночасний доступ двох сторін до відеопам'яті, від якого система зависає намертво.

Виходів два. По мережі: вбити процес — і натиснути Alt+F1 **ще раз**, бо перше натискання вже витрачене, а розпізнати смерть господаря ядро зможе лише на наступній спробі. Без мережі — [магічний SysRq](root:sys-unix/magic-sysrq): Alt+SysRq+E розсилає `SIGTERM` усім процесам, крім першого. Він працює навіть тоді, коли клавіатура переведена в режим `K_OFF` чи сирий, бо обробник SysRq зареєстрований у підсистемі вводу як фільтр — він бачить події раніше за розкладку й раніше за tty.

## Пастки

**Права.** Дозвіл на ці виклики ядро рахує так: `perm = 1`, якщо термінал за дескриптором — це керівний термінал процесу, **або** якщо в процесу є [можливість](root:sys-unix/capabilities) `CAP_SYS_TTY_CONFIG` (окреме право з розібраного набору повноважень адміністратора, яке дає керувати терміналами). Звідси практичний висновок: `chvt` із власної текстової консолі працює й від звичайного користувача, бо це його ж термінал. А от чужа вільна консоль керівним терміналом не є — і `VT_ACTIVATE`, `VT_SETMODE`, `KDSETMODE` на ній вимагають `CAP_SYS_TTY_CONFIG`. Виняток — `VT_OPENQRY` і `VT_GETMODE`: їх ядро віддає без перевірки.

**Господар не один, а останній.** `VT_SETMODE` нічого не арбітрує: він просто перезаписує в консолі номер процесу-господаря. Дві програми, що поставили `VT_PROCESS` на одну консоль, не отримають ні помилки, ні попередження — перша й далі вважатиме, що керує екраном, але relsig піде другій. Це головна причина, чому сьогодні цим має розпоряджатися один арбітр — [logind](root:sys-unix/logind-sessions-seats), служба, що веде облік сеансів і робочих місць: він тримає консоль сеансу в `VT_PROCESS` із тими самими `SIGRTMIN` і `SIGRTMIN+1`, ставить `KD_GRAPHICS` і `K_OFF`, а на завершенні сеансу повертає все назад.

**Рукостискання не забирає залізо.** Протокол консолі домовляється лише про екран як поняття; [графічний пристрій](root:sys-unix/drm-kms-object-model) — той, через який справді малюють пікселі, — він не чіпає. Програма, яка відповіла `VT_RELDISP(1)`, але не зняла з себе право керувати пристроєм, віддала екран лише на словах. Тому logind роздає сеансам дескриптори пристроїв сам і вміє забрати право на них незалежно від того, відповів сеанс на сигнал чи ні.

**Друга відповідь — формальність.** У Linux `VT_RELDISP(VT_ACKACQ)` не робить нічого: перемикання на вашу консоль уже відбулося до того, як ви отримали acqsig, і ядро лише перевіряє, що аргумент саме такий, а інакше повертає `EINVAL`. Надсилати його варто (протокол симетричний, і на інших системах симетрія може бути справжньою), але будувати логіку на тому, що ядро вас десь чекає, — не можна. Так само варто знати, що поле `frsig` зі структури `vt_mode` ядро мовчки обнуляє: третього сигналу, «звільни консоль», у Linux не існує.
