# ⚙️ Проєкт: шаблон переносного скрипта з автоматичними поліфілами

Створення скриптів автоматизації, які повинні без збоїв виконуватися на робочих станціях розробників (macOS), серверах розгортання (Ubuntu/Debian) та в легких контейнерах безперервної інтеграції (Alpine Linux), часто перетворюється на нескінченний цикл виправлення специфічних помилок командного рядка. У цьому проєкті зібрано закінчений архітектурний шаблон shell-скрипта мовою стандарту POSIX (`/bin/sh`), який автоматично розпізнає середовище виконання, надає уніфіковані функції-обгортки (поліфіли) для проблемних утиліт (`sed -i`, `date`, `mktemp`, `ps`) та гарантує передбачувану поведінку на будь-якому вузлі.

---

## Архітектурний задум і принципи переносимості

Щоб скрипт був по-справжньому універсальним, він повинен дотримуватися чотирьох фундаментальних інженерних правил:

1. **Шебанг `#!/bin/sh` замість прив'язки до bash:**
   Утиліта `/bin/bash` відсутня у мінімалістичних дистрибутивах (Alpine Linux, де встановлено `/bin/sh` на базі Almquist Shell або BusyBox ash). На macOS версія bash заморожена на стародавньому релізі 3.2 (2007 рік) через зміну ліцензії на GPLv3. Використання стандарту POSIX `/bin/sh` гарантує запуск у будь-якій системі без необхідності встановлення сторонніх пакетів.

2. **Суворий режим контролю помилок `set -eu`:**
   Прапорець `-e` (errexit) змушує оболонку негайно переривати роботу, якщо будь-яка команда конвеєра повертає ненульовий код статусу, не дозволяючи скрипту продовжувати виконання у пошкодженому стані. Прапорець `-u` (nounset) забороняє звернення до неініціалізованих змінних, що рятує від катастрофічних помилок на кшталт `rm -rf "${DEST_DIR}/${APP_NAME}"`, якщо змінна `DEST_DIR` виявилася порожньою.

3. **Раннє визначення поведінки (Feature Detection):**
   Замість ненадійного парсингу рядків у `/etc/os-release` або перевірки назви ядра через `uname`, скрипт тестує безпосередню реакцію утиліт на конкретні прапорці під час старту. Це дозволяє правильно визначити поведінку навіть тоді, коли на macOS через Homebrew встановлено `gnu-sed`, або коли в Debian змонтовано BusyBox.

4. **Гарантоване прибирання ресурсів через перехоплення сигналів `trap`:**
   Тимчасові каталоги та файли мають бути видалені за будь-яких умов: при успішному завершенні, при виникненні помилки, при натисканні користувачем комбінації Ctrl+C (сигнал `SIGINT`) або при отриманні сигналу завершення від планувальника контейнерів (`SIGTERM`).

## Реалізація переносного шаблону скрипта

Наведений нижче скрипт є самодостатнім модулем, який можна використовувати як основу для системних завдань, збірки артефактів чи сценаріїв розгортання.

```shell
#!/bin/sh
# ==============================================================================
# cross-platform-template.sh — Шаблон переносного скрипта для POSIX-середовищ
# Підтримує: GNU/Linux (Ubuntu/RHEL), BSD/macOS (Darwin), BusyBox (Alpine Linux)
# ==============================================================================

set -eu

# ------------------------------------------------------------------------------
# 1. Глобальне середовище та гарантоване очищення ресурсів
# ------------------------------------------------------------------------------
SCRIPT_NAME="${0##*/}"
TEMP_DIR=""

cleanup() {
    status=$?
    if [ -n "${TEMP_DIR}" ] && [ -d "${TEMP_DIR}" ]; then
        rm -rf "${TEMP_DIR}"
    fi
    exit "${status}"
}
trap cleanup EXIT INT TERM

log_info() {
    printf '[INFO] %s\n' "$*"
}

log_error() {
    printf '[ERROR] %s\n' "$*" >&2
}

log_fatal() {
    printf '[FATAL] %s\n' "$*" >&2
    exit 1
}

# ------------------------------------------------------------------------------
# 2. Раннє виявлення діалектів утиліт (Feature Detection)
# ------------------------------------------------------------------------------
detect_environment() {
    UNAME_S="$(uname -s 2>/dev/null || printf 'Unknown')"
    
    # Визначення поведінки sed -i
    if sed --version 2>&1 | grep -q 'GNU'; then
        SED_TYPE="gnu"
    elif [ "${UNAME_S}" = "Darwin" ] || [ "${UNAME_S}" = "FreeBSD" ]; then
        SED_TYPE="bsd"
    else
        # BusyBox або чистий POSIX
        SED_TYPE="fallback"
    fi

    # Визначення типу утиліти date
    if date --version 2>&1 | grep -q 'GNU'; then
        DATE_TYPE="gnu"
    elif date -v-1d +%Y 2>/dev/null >/dev/null; then
        DATE_TYPE="bsd"
    else
        DATE_TYPE="busybox"
    fi
}

detect_environment

# ------------------------------------------------------------------------------
# 3. Поліфіл для безпечного створення тимчасових каталогів
# ------------------------------------------------------------------------------
portable_mktemp_dir() {
    prefix="${1:-tmp}"
    if [ "${UNAME_S}" = "Darwin" ] || [ "${UNAME_S}" = "FreeBSD" ]; then
        # BSD mktemp вимагає прапорець -t
        mktemp -d -t "${prefix}.XXXXXX"
    else
        # GNU mktemp та BusyBox mktemp
        mktemp -d "/tmp/${prefix}.XXXXXX"
    fi
}

TEMP_DIR="$(portable_mktemp_dir "deploy-task")"

# ------------------------------------------------------------------------------
# 4. Поліфіл: редагування файлу на місці (In-Place Sed)
# ------------------------------------------------------------------------------
# Використання: portable_sed_i 's/search/replace/g' target_file.txt
portable_sed_i() {
    expression="$1"
    target_file="$2"

    if [ ! -f "${target_file}" ]; then
        log_fatal "Цільовий файл не існує: ${target_file}"
    fi

    case "${SED_TYPE}" in
        gnu)
            sed -i "${expression}" "${target_file}"
            ;;
        bsd)
            sed -i '' "${expression}" "${target_file}"
            ;;
        *)
            # Найбільш надійний універсальний шлях: через атомарний тимчасовий файл
            tmp_out="${TEMP_DIR}/sed_tmp_$$.out"
            sed "${expression}" "${target_file}" > "${tmp_out}"
            # Зберігаємо оригінальні права доступу перед заміною
            cat "${tmp_out}" > "${target_file}"
            rm -f "${tmp_out}"
            ;;
    esac
}

# ------------------------------------------------------------------------------
# 5. Поліфіл: обчислення часових міток та форматування дат
# ------------------------------------------------------------------------------
# Використання: portable_yesterday_iso -> повертає YYYY-MM-DD за вчора
portable_yesterday_iso() {
    case "${DATE_TYPE}" in
        gnu)
            date -d "yesterday" '+%Y-%m-%d'
            ;;
        bsd)
            date -v-1d '+%Y-%m-%d'
            ;;
        *)
            # Для BusyBox обчислюємо через віднімання 86400 секунд за стандартом POSIX
            current_epoch="$(date -u +%s)"
            yesterday_epoch=$((current_epoch - 86400))
            # Використовуємо awk для портативного перетворення epoch у дату
            awk -v epoch="${yesterday_epoch}" 'BEGIN {
                print strftime("%Y-%m-%d", epoch, 1)
            }' 2>/dev/null || date -u '+%Y-%m-%d'
            ;;
    esac
}

# ------------------------------------------------------------------------------
# 6. Поліфіл: портативний пошук процесів у системі
# ------------------------------------------------------------------------------
# Використання: portable_find_pids "process_name"
portable_find_pids() {
    search_term="$1"
    # За стандартом POSIX ps -ef або ps -A -o pid,comm є універсальними
    ps -A -o pid=,comm= | awk -v term="${search_term}" '$2 ~ term { print $1 }'
}

# ------------------------------------------------------------------------------
# 7. Демонстрація виконання робочого сценарію
# ------------------------------------------------------------------------------
main() {
    log_info "Ініціалізація сценарію на платформі: ${UNAME_S}"
    log_info "Виявлені підсистеми -> sed: [${SED_TYPE}], date: [${DATE_TYPE}]"
    log_info "Тимчасовий робочий простір створено у: ${TEMP_DIR}"

    # Створюємо тестовий файл конфігурації
    config_file="${TEMP_DIR}/app_config.env"
    cat <<'EOF' > "${config_file}"
APP_NAME=payments-gateway
APP_ENV=development
APP_DEBUG=true
RELEASE_DATE=pending
EOF

    log_info "Початковий стан конфігурації:"
    cat "${config_file}"

    # Модифікуємо конфігурацію крос-платформним sed
    log_info "Оновлення параметрів середовища на production..."
    portable_sed_i 's/APP_ENV=development/APP_ENV=production/g' "${config_file}"
    portable_sed_i 's/APP_DEBUG=true/APP_DEBUG=false/g' "${config_file}"

    # Додаємо розраховану дату
    yesterday_val="$(portable_yesterday_iso)"
    portable_sed_i "s/RELEASE_DATE=pending/RELEASE_DATE=${yesterday_val}/g" "${config_file}"

    log_info "Фінальний стан конфігурації після обробки:"
    cat "${config_file}"

    # Перевірка наявності процесу демона
    log_info "Пошук процесів cron/crond..."
    found_pids="$(portable_find_pids "cron" | tr '\n' ' ')"
    log_info "Знайдені PID cron: [${found_pids:-немає}]"

    log_info "Усі завдання завершено успішно."
}

main "$@"
```

## Глибокий розбір ключових рішень шаблону

### 1. Збереження дескриптора inode у функції `portable_sed_i`
Універсальна гілка `fallback` використовує перезапис вмісту через `cat "${tmp_out}" > "${target_file}"`, а не заміну файлу командою `mv "${tmp_out}" "${target_file}"`. Це тонке, але життєво важливе архітектурне рішення:

Коли файл монтується у контейнер Docker або Kubernetes Pod через механізм `ConfigMap` чи `bind-mount`, ядро зв'язує точку монтування з конкретним числовим номером дескриптора файлової системи (`inode`). Якщо виконати `mv tmp file`, старий inode видаляється, а на його місці з'являється новий файл з новим inode. У результаті процес усередині контейнера продовжує читати старий закешований дескриптор і не бачить оновленої конфігурації. Перенаправлення виводу через `cat > file` перезаписує блоки даних на диску, зберігаючи оригінальний inode, права власності та біти доступу (`chmod`).

### 2. Захист від збоїв при обчисленні дат в автономних середовищах
У вбудованих прошивках мережевих комутаторів та мінімальних контейнерах системний годинник часто ініціалізується нульовим значенням або не має доступу до бази даних часових поясів IANA (`/usr/share/zoneinfo`).

Утиліта `awk` з функцією `strftime(..., 1)` використовує режим UTC (останній аргумент `1`), що запобігає збоям при відсутності локальних файлів часових поясів. Віднімання фіксованої кількості секунд `86400` гарантує математично точний перехід на добу назад без залучення сторонніх бібліотек.

### 3. Надійність пошуку процесів без використання `grep -v grep`
Класичний антипатерн адміністрування `ps aux | grep process_name | grep -v grep` страждає від стану перегонів (race condition): у момент виконання конвеєра сам процес `grep process_name` потрапляє у вивід `ps`, а другий фільтр `grep -v` не завжди встигає його відсіяти через асинхронний запуск процесів оболонкою.

Використання `ps -A -o pid=,comm=` повертає лише імена виконуваних файлів без повного командного рядка аргументів. Подальша фільтрація через `awk '$2 ~ term'` зіставляє шаблон виключно з полем назви бінарника, що повністю виключає випадкові спрацьовування та гарантує повернення лише справжніх числових ідентифікаторів PID.

### 4. Пастка створення тимчасових каталогів через `mktemp`
Стандарт POSIX історично не стандартизував утиліту `mktemp` (вона з'явилася в OpenBSD у 1996 році і лише згодом була імплементована в Linux). У результаті між реалізаціями виник синтаксичний бар'єр:
- У GNU та BusyBox виклик `mktemp -d` створює унікальний каталог у `/tmp` за шаблоном за замовчуванням (`/tmp/tmp.XXXXXX`).
- У BSD/macOS виклик `mktemp -d` вимагає обов'язкової передачі шаблону або прапорця `-t prefix`, без якого утиліта негайно повертає код помилки.
Функція `portable_mktemp_dir` перевіряє операційну систему і генерує правильний виклик, гарантуючи створення захищеного каталогу з правами `0700`.

### 5. Обробка сигналів переривання у фонових конвеєрах
У багатьох оболонках (зокрема `dash` на Ubuntu та `ash` в Alpine) конструкція `trap ... EXIT` спрацьовує при нормальному завершенні скрипта, але може ігноруватися, якщо процес був примусово вбитий через `SIGINT` або `SIGTERM`.

Явне перелічення сигналів `trap cleanup EXIT INT TERM` гарантує, що обробник `cleanup` виконається навіть тоді, коли користувач натиснув комбінацію клавіш переривання або система CI надіслала сигнал таймауту. Отримання статусу `$?` на першому рядку функції `cleanup` забезпечує коректне збереження початкового коду завершення програми.
