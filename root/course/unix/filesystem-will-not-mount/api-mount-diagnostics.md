# 📋 Довідник діагностики монтування: системні виклики, коди помилок, магічні числа та прапорці

Цей довідник містить точні сигнатури системних викликів, маски бітових прапорців, дискові структури суперблоків та діагностичні інтерфейси ядра Linux, необхідні для програмного та системного розбору аварій монтування файлових систем.

## Еволюція системних викликів: від монолітного `mount(2)` до компонованого Mount API

Підсистема VFS ядра Linux історично використовувала єдиний монолітний системний виклик `mount(2)` для виконання абсолютно всіх операцій: від монтування фізичного блокового пристрою до зміни прапорців на льоту (*remount*), створення зв'язаних дерев (*bind mount*) та керування просторами імен.

У монолітному інтерфейсі конфігурація файлової системи (перевірка параметрів, виділення пам'яті, зчитування суперблоку) була жорстко зв'язана в одну неподільну транзакцію з прикріпленням результату до дерева каталогів VFS. Це створювало фундаментальні проблеми: неможливість безпечно налаштувати параметри файлової системи до її активації, складність передачі великої кількості параметрів через неструктурований рядок `data` та вразливості стану гонки в ізольованих контейнерах.

Починаючи з версії ядра Linux 5.2, підсистема VFS отримала новий компонований Mount API, який розділяє процес на три незалежні етапи:
1. Створення анонімного контексту файлової системи (`fsopen`);
2. Поетапне передавання параметрів конфігурації у форматі ключ-значення через дескриптор (`fsconfig`);
3. Матеріалізація суперблоку та отримання дескриптора готового дерева монтування (`fsmount`), яке згодом атомарно прикріплюється до каталогу (`move_mount`).

### Сигнатури та виклик класичного `mount(2)`

Класичний інтерфейс приймає п'ять параметрів, що визначають джерело, ціль, тип драйвера, бітову маску поведінки та специфічні параметри драйвера.

:::tabs
```c
#include <stdio.h>
#include <string.h>
#include <errno.h>
#include <sys/mount.h>

/* Виклик класичного mount(2) у стилі C */
int mount_filesystem_classic(const char *source, const char *target, const char *fstype) {
    unsigned long flags = MS_RDONLY | MS_NODEV | MS_NOSUID | MS_NOEXEC;
    const char *data_opts = "errors=remount-ro,data=ordered";

    if (mount(source, target, fstype, flags, data_opts) < 0) {
        fprintf(stderr, "Помилка mount(2) для %s на %s: %s (errno=%d)\n",
                source, target, strerror(errno), errno);
        return -1;
    }
    return 0;
}
```
```cpp
#include <iostream>
#include <string_view>
#include <expected>
#include <system_error>
#include <format>
#include <sys/mount.h>

/* Ідіоматична обгортка mount(2) на C++23 */
enum class MountError {
    PermissionDenied,
    DeviceBusy,
    InvalidSuperblock,
    UnknownFilesystem,
    TargetNotFound,
    SystemFailure
};

[[nodiscard]] std::expected<void, std::pair<MountError, std::string>>
mountFilesystem(std::string_view source, std::string_view target, std::string_view fstype) {
    constexpr unsigned long flags = MS_RDONLY | MS_NODEV | MS_NOSUID | MS_NOEXEC;
    constexpr std::string_view dataOpts = "errors=remount-ro,data=ordered";

    if (::mount(source.data(), target.data(), fstype.data(), flags, dataOpts.data()) < 0) {
        const int err = errno;
        MountError category = MountError::SystemFailure;

        switch (err) {
            case EPERM:
            case EACCES: category = MountError::PermissionDenied; break;
            case EBUSY:  category = MountError::DeviceBusy; break;
            case EINVAL: category = MountError::InvalidSuperblock; break;
            case ENODEV: category = MountError::UnknownFilesystem; break;
            case ENOENT: category = MountError::TargetNotFound; break;
            default:     category = MountError::SystemFailure; break;
        }

        return std::unexpected(std::make_pair(
            category,
            std::format("Збій монтування '{}' на '{}': {} (errno={})",
                        source, target, std::generic_category().message(err), err)
        ));
    }
    return {};
}
```

### Програмний інтерфейс сучасного Mount API (Linux 5.2+)

Сучасний Mount API базується на роботі з анонімними дескрипторами контексту файлової системи. Це дозволяє поетапно конфігурувати параметри драйвера, ізолюючи процес налаштування від глобального дерева VFS.

:::tabs
```c
#define _GNU_SOURCE
#include <stdio.h>
#include <unistd.h>
#include <sys/syscall.h>
#include <fcntl.h>

/* Визначення номерів системних викликів, якщо вони відсутні в старих версіях libc */
#ifndef __NR_fsopen
#define __NR_fsopen 430
#define __NR_fsconfig 431
#define __NR_fsmount 432
#define __NR_move_mount 429
#endif

#define FSCONFIG_SET_STRING 1
#define FSCONFIG_SET_FLAG   0
#define FSCONFIG_CMD_CREATE 6

int mount_via_new_api(const char *source, const char *target, const char *fstype) {
    /* 1. Створення контексту файлової системи */
    int fs_fd = syscall(__NR_fsopen, fstype, 0);
    if (fs_fd < 0) {
        perror("fsopen");
        return -1;
    }

    /* 2. Поетапне передавання параметрів конфігурації */
    syscall(__NR_fsconfig, fs_fd, FSCONFIG_SET_STRING, "source", source, 0);
    syscall(__NR_fsconfig, fs_fd, FSCONFIG_SET_FLAG, "ro", NULL, 0);
    syscall(__NR_fsconfig, fs_fd, FSCONFIG_SET_STRING, "errors", "remount-ro", 0);

    /* 3. Команда ядру зчитати суперблок та ініціалізувати інстанс ФС */
    if (syscall(__NR_fsconfig, fs_fd, FSCONFIG_CMD_CREATE, NULL, NULL, 0) < 0) {
        perror("fsconfig(CREATE)");
        close(fs_fd);
        return -1;
    }

    /* 4. Матеріалізація суперблоку та отримання дескриптора дерева */
    int mnt_fd = syscall(__NR_fsmount, fs_fd, 0, 0);
    close(fs_fd);
    if (mnt_fd < 0) {
        perror("fsmount");
        return -1;
    }

    /* 5. Атомарне прикріплення дерева монтування до цільового каталогу */
    if (syscall(__NR_move_mount, mnt_fd, "", AT_FDCWD, target, 0x00000004 /* MOVE_MOUNT_F_EMPTY_PATH */) < 0) {
        perror("move_mount");
        close(mnt_fd);
        return -1;
    }

    close(mnt_fd);
    return 0;
}
```
```cpp
#include <iostream>
#include <string_view>
#include <expected>
#include <format>
#include <system_error>
#include <unistd.h>
#include <sys/syscall.h>
#include <fcntl.h>

#ifndef __NR_fsopen
#define __NR_fsopen 430
#define __NR_fsconfig 431
#define __NR_fsmount 432
#define __NR_move_mount 429
#endif

class ScopedDescriptor {
    int fd_ = -1;
public:
    explicit ScopedDescriptor(int fd) noexcept : fd_(fd) {}
    ~ScopedDescriptor() noexcept { if (fd_ >= 0) ::close(fd_); }
    ScopedDescriptor(const ScopedDescriptor&) = delete;
    ScopedDescriptor& operator=(const ScopedDescriptor&) = delete;
    ScopedDescriptor(ScopedDescriptor&& other) noexcept : fd_(other.fd_) { other.fd_ = -1; }
    ScopedDescriptor& operator=(ScopedDescriptor&& other) noexcept {
        if (this != &other) {
            if (fd_ >= 0) ::close(fd_);
            fd_ = other.fd_;
            other.fd_ = -1;
        }
        return *this;
    }
    [[nodiscard]] int get() const noexcept { return fd_; }
    [[nodiscard]] bool valid() const noexcept { return fd_ >= 0; }
    int release() noexcept { int tmp = fd_; fd_ = -1; return tmp; }
};

[[nodiscard]] std::expected<void, std::string>
mountViaModernApi(std::string_view source, std::string_view target, std::string_view fstype) {
    /* 1. fsopen */
    ScopedDescriptor fsContext(static_cast<int>(::syscall(__NR_fsopen, fstype.data(), 0)));
    if (!fsContext.valid()) {
        return std::unexpected(std::format("Збій fsopen('{}'): {}", fstype, std::generic_category().message(errno)));
    }

    /* 2. fsconfig налаштування джерела */
    if (::syscall(__NR_fsconfig, fsContext.get(), 1 /* FSCONFIG_SET_STRING */, "source", source.data(), 0) < 0) {
        return std::unexpected(std::format("Збій конфігурації source: {}", std::generic_category().message(errno)));
    }

    /* 3. fsconfig валідація суперблоку */
    if (::syscall(__NR_fsconfig, fsContext.get(), 6 /* FSCONFIG_CMD_CREATE */, nullptr, nullptr, 0) < 0) {
        return std::unexpected(std::format("Збій створення суперблоку: {}", std::generic_category().message(errno)));
    }

    /* 4. fsmount */
    ScopedDescriptor mntTree(static_cast<int>(::syscall(__NR_fsmount, fsContext.get(), 0, 0)));
    if (!mntTree.valid()) {
        return std::unexpected(std::format("Збій fsmount: {}", std::generic_category().message(errno)));
    }

    /* 5. move_mount */
    if (::syscall(__NR_move_mount, mntTree.get(), "", AT_FDCWD, target.data(), 0x00000004) < 0) {
        return std::unexpected(std::format("Збій move_mount на '{}': {}", target, std::generic_category().message(errno)));
    }

    return {};
}
```

## Повна матриця системних помилок ядра (`errno`)

Коли виклик `mount(2)` або етап `fsconfig(FSCONFIG_CMD_CREATE)` повертає помилку, ядро записує точний числовий код у змінну `errno`. Розуміння точного внутрішнього джерела помилки в підсистемі VFS дозволяє системному інженеру розрізняти відмови обладнання, помилки конфігурації та блокування просторів імен.

| Код помилки | Число | Внутрішнє джерело в ядрі (VFS та драйвер) | Типовий вивід консолі mount(8) |
|---|---|---|---|
| `EINVAL` | `22` | Некоректне або спотворене магічне число суперблоку, невідома або взаємовиключна опція монтування, непідтримуваний прапорець `s_feature_incompat`, або зсув геометрії LBA. | `wrong fs type, bad option, bad superblock` |
| `EBUSY` | `16` | Блоковий пристрій уже захоплено ексклюзивно (`bd_open_exclusive` via LVM, LUKS, RAID), або цільовий каталог є активною точкою монтування, або каталог утримується відкритими fd. | `target is busy` або `device or resource busy` |
| `ENODEV` | `19` | Запитуваний тип файлової системи відсутній у внутрішній таблиці `file_systems` ядра (драйвер не зібрано в моноліт ядра, а відповідний модуль відсутній на диску або в initramfs). | `unknown filesystem type` |
| `EACCES` | `13` | Відсутні права доступу на читання або виконання одного з батьківських каталогів у шляху до точки монтування. | `permission denied` |
| `EPERM` | `1` | Процес не має адміністративного привілею `CAP_SYS_ADMIN` у поточному просторі імен користувача (*User Namespace*). | `operation not permitted` |
| `ENOENT` | `2` | Вказаний блоковий пристрій джерела (`source`) або цільовий каталог точки монтування (`target`) фізично не існує в дереві файлової системи. | `no such file or directory` |
| `ENOTBLK` | `15` | Параметр `source` вказує на регулярний файл або каталог замість очікуваного блокового пристрою (без використання прапорця `MS_BIND` або попереднього монтування loop-пристрою). | `not a block device` |
| `ENOTDIR` | `20` | Цільовий шлях точки монтування `target` не є каталогом (вказує на регулярний файл або символьне посилання на файл). | `not a directory` |
| `ENXIO` | `6` | Вказаний блоковий пристрій не відповідає жодному фізичному накопичувачу або запитаний діапазон розділу виходить за межі фізичної ємності носія. | `no such device or address` |
| `EROFS` | `30` | Спроба змонтувати пошкоджену файлову систему або том з незавершеним журналом у режимі читання-запису на носії, який апаратно заблокований лише для читання. | `read-only file system` |
| `EIO` | `5` | Апаратна помилка контролера при зчитуванні фізичного сектора суперблоку чи дескрипторів транзакційного журналу (Uncorrectable ECC Error). | `input/output error` |
| `ELOOP` | `40` | Перевищено максимальну глибину переходів за символьними посиланнями під час розбору шляху точки монтування. | `too many levels of symbolic links` |
| `EMFILE` | `24` | Процес вичерпав встановлений ліміт відкритих файлових дескрипторів під час створення внутрішніх структур контексту монтування. | `too many open files` |
| `ENAMETOOLONG` | `36` | Довжина рядка шляху джерела, точки монтування чи назви типу файлової системи перевищує ліміт `PATH_MAX` (4096) або `NAME_MAX` (255). | `file name too long` |
| `ENOMEM` | `12` | Ядру операційної системи забракло вільної фізичної оперативної пам'яті для виділення об'єктів `struct super_block`, кореневого `dentry` або буферів журналу. | `cannot allocate memory` |

## Магічні числа та внутрішня будова суперблоків

Драйвер кожної файлової системи починає перевірку диска з пошуку магічного числа за суворо фіксованим зміщенням від початку пристрою або групи виділення. Якщо магічне число пошкоджено, VFS негайно припиняє обробку і повертає `EINVAL`.

```
Структура суперблоків основних файлових систем Linux:
├── Ext2/3/4: Зміщення 1024 байти (0x400), Магічне число: 0xEF53 (Little Endian: 53 EF)
├── XFS:      Зміщення 0 байтів у кожній AG, Магічне число: 0x58465342 (ASCII 'XFSB')
├── Btrfs:    Зміщення 65 536 байтів (64 КіБ), Магічне число: 0x4D53535266425F5F (ASCII '_BHRfS_M')
├── F2FS:     Зміщення 1024 байти (0x400), Магічне число: 0xF2F52010 (Little Endian)
└── VFAT:     Зміщення 510 байтів (0x1FE), Сигнатура завантажувального сектора: 0xAA55
```

### Структура та критичні поля `ext4_super_block`

У файлових системах родини Ext2/3/4 перші 1024 байти розділу зарезервовані для завантажувача MBR/x86 Boot Record. Початковий суперблок розташований точно за зміщенням 1024 байти:

| Зміщення (Bytes) | Тип поля | Назва поля | Опис та критичні значення |
|---|---|---|---|
| `0x00` (`1024`) | `__le32` | `s_inodes_count` | Загальна кількість індексних дескрипторів (*inode*) у файловій системі. |
| `0x04` (`1028`) | `__le32` | `s_blocks_count_lo` | Загальна кількість блоків даних (молодші 32 біти). |
| `0x18` (`1048`) | `__le32` | `s_log_block_size` | Розмір блоку файлової системи: обчислюється як `1024 << s_log_block_size` (`0` = 1 КіБ, `1` = 2 КіБ, `2` = 4 КіБ). |
| `0x20` (`1056`) | `__le32` | `s_blocks_per_group` | Кількість блоків даних в одній Block Group (стандартно 32 768 для блоку 4 КіБ). |
| `0x38` (`1080`) | `__le16` | `s_magic` | **Магічне число Ext4:** значення обов'язково має дорівнювати `0xEF53`. Якщо байт відрізняється, ядро повертає `EINVAL`. |
| `0x3A` (`1082`) | `__le16` | `s_state` | Стан узгодженості ФС: `0x0001` (`EXT4_VALID_FS` — чисто), `0x0002` (`EXT4_ERROR_FS` — зафіксовано помилки). |
| `0x3C` (`1084`) | `__le16` | `s_errors` | Дія при виявленні помилки: `1` (продовжувати роботу), `2` (перевести в Read-Only), `3` (викликати Kernel Panic). |
| `0x60` (`1120`) | `__u8[16]` | `s_uuid` | Унікальний 128-бітний ідентифікатор тома (використовується в `/etc/fstab` як `UUID=`). |
| `0x64` (`1156`) | `__le32` | `s_feature_compat` | Маска сумісних розширень (старі драйвери можуть безпечно монтувати та писати). |
| `0x68` (`1160`) | `__le32` | `s_feature_incompat` | **Маска несумісних розширень:** якщо драйвер не знає бодай одного біта — монтування блокується з `EINVAL`. |
| `0x6C` (`1164`) | `__le32` | `s_feature_ro_compat` | Прапорці, несумісні для запису: дозволяють монтування старішим ядром лише в режимі `MS_RDONLY`. |
| `0xD0` (`1232`) | `__le32` | `s_journal_inum` | Номер inode транзакційного журналу JBD2 (стандартно inode 8). |
| `0x180` (`1408`) | `__le32` | `s_checksum` | Контрольна сума всього суперблоку (обчислюється алгоритмом CRC32C). |

#### Критичні біти прапорців `s_feature_incompat` Ext4

* `0x0001`: `EXT4_FEATURE_INCOMPAT_COMPRESSION` — підтримка стиснення блоків;
* `0x0002`: `EXT4_FEATURE_INCOMPAT_FILETYPE` — збереження типу файлу (каталог, сокет, регулярний) прямо в записах каталогу;
* `0x0004`: `EXT4_FEATURE_INCOMPAT_RECOVER` — **прапорець незавершеного відновлення журналу** (Journal Recovery Pending);
* `0x0008`: `EXT4_FEATURE_INCOMPAT_JOURNAL_DEV` — пристрій є виділеним зовнішнім накопичувачем журналу;
* `0x0010`: `EXT4_FEATURE_INCOMPAT_META_BG` — підтримка розширених груп метаданих для надвеликих файлових систем;
* `0x0040`: `EXT4_FEATURE_INCOMPAT_EXTENTS` — файли використовують дерево екстентів замість традиційних непрямих покажчиків блоків;
* `0x0080`: `EXT4_FEATURE_INCOMPAT_64BIT` — 64-бітна адресація блоків (дозволяє створювати томи понад 16 ТіБ);
* `0x0200`: `EXT4_FEATURE_INCOMPAT_FLEX_BG` — об'єднані групи метаданих (*flex block groups*) для прискорення читання;
* `0x0400`: `EXT4_FEATURE_INCOMPAT_EA_INODE` — розміщення розширених атрибутів великого розміру в окремих виділених inode;
* `0x1000`: `EXT4_FEATURE_INCOMPAT_CSUM_SEED` — використання унікального зерна контрольної суми для томів.

### Структура суперблоку XFS (`xfs_sb_t`)

Суперблок XFS розташований за зміщенням `0` на початку кожної групи виділення (*Allocation Group*, AG). Завдяки цьому XFS підтримує паралельну обробку метаданих кількома процесорними ядрами:

| Зміщення (Bytes) | Тип поля | Назва поля | Опис |
|---|---|---|---|
| `0x00` (`0`) | `__be32` | `sb_magicnum` | **Магічне число XFS:** `0x58465342` (рядок ASCII "XFSB" у форматі Big Endian). |
| `0x04` (`4`) | `__be32` | `sb_blocksize` | Розмір фундаментального блоку файлової системи в байтах (стандартно 4096). |
| `0x08` (`8`) | `__be64` | `sb_dblocks` | Загальна кількість блоків даних у всій файловій системі. |
| `0x20` (`32`) | `uuid_t` | `sb_uuid` | Унікальний 128-бітний ідентифікатор тома UUID. |
| `0x38` (`56`) | `__be32` | `sb_agblocks` | Розмір однієї групи виділення (AG) у блоках. |
| `0x3C` (`60`) | `__be32` | `sb_agcount` | Загальна кількість груп виділення (Allocation Groups) на диску. |
| `0x64` (`100`) | `__be16` | `sb_versionnum` | Номер версії суперблоку (сучасні ФС мають версію 5 з підтримкою контрольних сум метаданих). |
| `0xD8` (`216`) | `__be32` | `sb_crc` | Контрольна сума суперблоку CRC32C (для суперблоків версії v5). |

### Структури суперблоків Btrfs та F2FS

Файлова система Btrfs використовує зміщення 64 КіБ (65 536 байтів) для первинного суперблоку, залишаючи перші 64 КіБ під завантажувальні структури завантажувачів GRUB та системні таблиці розділів. Додаткові дзеркальні копії суперблоку Btrfs розташовані за зміщеннями 64 МіБ, 256 ГіБ та 1 ТиБ, що забезпечує автоматичне відновлення структури дерева B-tree навіть при масштабних збоях носія.

Файлова система F2FS (*Flash-Friendly File System*), оптимізована для накопичувачів NAND Flash, зберігає два суперблоки за фіксованими зміщеннями 1024 та 4096 байтів. Під час монтування драйвер F2FS перевіряє обидва суперблоки за магічним числом `0xF2F52010` та значенням лічильника версій, автоматично вибираючи блок із новішою валідною транзакцією.

## Бітові коди виходу консольних утиліт перевірки

Під час завантаження системи системний менеджер systemd перевіряє коди виходу утиліт `fsck`, `e2fsck` та `xfs_repair`.

### Бітова маска статусів `fsck` та `e2fsck`

Код повернення `e2fsck` формується як побітове об'єднання (OR) відповідних бітів стану:

```
Бітова маска статусів e2fsck:
├── 0  (0x00) — Помилок не знайдено, стан ФС повністю узгоджений.
├── 1  (0x01) — Помилки файлової системи виявлено та успішно виправлено в автоматичному режимі.
├── 2  (0x02) — Помилки виправлено, але система ВИМАГАЄ перезавантаження вузла (змінено системні структури).
├── 4  (0x04) — Помилки виявлено, але залишено НЕВИПРАВЛЕНИМИ (потрібне ручне втручання адміністратора).
├── 8  (0x08) — Операційна помилка під час виконання утиліти перевірки.
├── 16 (0x10) — Помилка синтаксису або некоректні аргументи командного рядка.
├── 32 (0x20) — Перевірку перервано за запитом користувача (отримано сигнал переривання).
└── 128 (0x80) — Помилка завантаження динамічної спільної бібліотеки.
```

Якщо підсумковий код виходу містить встановлений біт `4` або вище (наприклад, комбінований код `5` = `1 | 4`), завантажувач systemd вважає диск аварійним, зупиняє перехід до `local-fs.target` і перемикає консоль в аварійний режим `emergency.target`.

## Розширені параметри `/etc/fstab` для керування `systemd`

Генератор `systemd-fstab-generator` підтримує спеціальні директиви з префіксом `x-systemd.*` для точного керування таймаутами та залежностями:

| Параметр конфігурації | Призначення та вплив на завантаження системи |
|---|---|
| `nofail` | Перетворює залежність точки монтування з блокуючої `Requires=` у м'яку `Wants=`. Збій монтування не зупиняє завантаження ОС. |
| `_netdev` | Вказує, що диск розміщений у мережі (NFS, iSCSI, Ceph); додає очікування служб `network-online.target`. |
| `x-systemd.device-timeout=5s` | Задає максимальний час очікування появи блокового пристрою в udev (за замовчуванням 90 секунд). |
| `x-systemd.mount-timeout=10s` | Задає максимальний ліміт часу на виконання операції `mount(2)` ядром. |
| `x-systemd.automount` | Створює юніт автомонтування autofs: каталог монтується лише під час першого реального звернення до нього. |
| `x-systemd.idle-timeout=10m` | Автоматично відмонтовує файлову систему після 10 хвилин повної відсутності звернень процесів. |
| `x-systemd.requires=foo.service` | Додає сувору залежність точки монтування від успішного запуску вказаної системної служби. |

## Інтерфейси інспектування стану блокувань у `/sys` та `/proc`

Для точної локалізації ексклюзивних блокувань ядра використовуються псевдофайлові інтерфейси:

```bash
# Перегляд списку зареєстрованих VFS-типів ядра
cat /proc/filesystems

# Виявлення процесів, що блокують точку монтування
fuser -vm /mnt/point
lsof +f -- /mnt/point

# Перевірка холдерів (холдер утримує пристрій через bd_open_exclusive)
# Якщо каталог містить записи, пристрій зайнятий LVM, dm-crypt або md-масивом:
ls -l /sys/class/block/<dev_name>/holders/

# Перевірка підлеглих пристроїв (slaves):
ls -l /sys/class/block/<dev_name>/slaves/
```

Кожна структура в `/proc/self/mountinfo` містить вичерпний зліпок точки монтування:

```
Ідентифікатори рядка /proc/self/mountinfo:
36 1 8:2 / /data rw,relatime shared:1 - ext4 /dev/sda2 rw,data=ordered
 │ │  │  │   │       │          │     │   │      │          │
 │ │  │  │   │       │          │     │   │      │          └─ Опції конкретної ФС
 │ │  │  │   │       │          │     │   │      └─ Джерело монтування
 │ │  │  │   │       │          │     │   └─ Тип файлової системи
 │ │  │  │   │       │          │     └─ Роздільник полів (-)
 │ │  │  │   │       │          └─ Необов'язкові теги поширення (shared/master)
 │ │  │  │   │       └─ Прапорці точки VFS
 │ │  │  │   └─ Точка монтування в дереві процесу
 │ │  │  └─ Кореневий каталог монтування всередині ФС
 │ │  └─ Старший і молодший номери пристрою (Major:Minor)
 │ └─ ID батьківської точки монтування
 └─ Унікальний Mount ID
```

У контейнерних середовищах тег поширення монтування (`shared:N`, `master:N`, `unbindable`) визначає, чи будуть операції монтування всередині контейнера видимими на хості. Якщо кореневе монтування хоста налаштовано як `shared`, створення нового монтування всередині контейнера без прапорця приватності може призвести до блокування ресурсів у глобальному просторі імен і виникнення помилок `EBUSY` під час подальших спроб демонтажу.
